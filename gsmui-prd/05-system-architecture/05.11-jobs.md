# 05.11 - Jobs & Queue Architecture

**Project:** GsmUi
**Architecture:** Background Processing & Queue System
**Framework:** Laravel 13 + Horizon + Redis
**Version:** 1.0.0

---

# Purpose

The Jobs & Queue Architecture enables GsmUi to perform long-running, resource-intensive, or asynchronous tasks outside the HTTP request lifecycle.

By leveraging Laravel Queues and Horizon, the platform ensures:

* Faster user responses
* Improved scalability
* Reliable background processing
* Automatic retries
* Fault isolation
* Better observability

---

# Objectives

The queue system should:

* Process asynchronous work reliably
* Support multiple priority levels
* Handle retries and failures gracefully
* Scale horizontally
* Minimize HTTP response times
* Be observable through Horizon and Telescope

# Risk

| Risk | Likelihood | Impact | Mitigation |
|------|-----------|--------|------------|
| Queue backlog during traffic spikes | Medium | High | Auto-scaling workers, priority queues, backpressure alerts |
| Job failure causing data inconsistency | Medium | High | Idempotent jobs, transactional dispatch, dead-letter queue inspection |
| Redis outage halting all background processing | Low | Critical | Redis sentinel/cluster, graceful degradation, synchronous fallback for critical jobs |
| Worker memory leaks | Medium | Medium | Horizon memory limits, worker restart thresholds, monitoring |
| Stale jobs consuming resources | Medium | Medium | TTL on jobs, scheduled cleanup, max-age policies |
| Poison job blocking a queue | Low | High | Job-level timeout, retry limits, failed job alerting |
| Serialization bugs in job payloads | Low | High | DTO-based payloads, versioned serialization, payload validation tests |

# Job Implementation Patterns

## Media Processing Jobs

### OptimizeUploadedImage

```php
// app/Domains/Components/Jobs/OptimizeUploadedImage.php

declare(strict_types=1);

namespace App\Domains\Components\Jobs;

use App\Domains\Storage\Contracts\StorageServiceInterface;
use App\Domains\Components\Models\Component;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\{InteractsWithQueue, SerializesModels};
use Intervention\Image\ImageManagerStatic as Image;
use Illuminate\Support\Facades\Log;

class OptimizeUploadedImage implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $tries = 3;

    public int $timeout = 120;

    public array $backoff = [30, 60, 120];

    public function __construct(
        public readonly int $componentId,
        public readonly string $imagePath,
        public readonly array $options = [],
    ) {}

    public function handle(StorageServiceInterface $storage): void
    {
        $component = Component::findOrFail($this->componentId);

        // Download original image
        $originalContent = $storage->get($this->imagePath);

        // Optimize image
        $manager = new Image(['driver' => 'gd']);
        $image = $manager->make($originalContent);

        // Resize if needed
        $maxWidth = $this->options['max_width'] ?? 1200;
        if ($image->width() > $maxWidth) {
            $image->resize($maxWidth, null, fn($constraint) => $constraint->aspectRatio());
        }

        // Convert to WebP
        $webpPath = $this->getWebPPath($this->imagePath);
        $webpContent = $image->toWebp($this->options['quality'] ?? 85);

        // Store optimized image
        $storage->put($webpPath, $webpContent);

        // Generate thumbnails
        $this->generateThumbnails($image, $storage);

        // Update component record
        $component->update([
            'preview_image' => $webpPath,
        ]);

        // Delete original if different format
        if ($this->imagePath !== $webpPath) {
            $storage->delete($this->imagePath);
        }

        Log::info('Image optimized', [
            'component_id' => $this->componentId,
            'original' => $this->imagePath,
            'optimized' => $webpPath,
            'size' => strlen($webpContent),
        ]);
    }

    private function getWebPPath(string $path): string
    {
        $pathInfo = pathinfo($path);
        return $pathInfo['dirname'] . '/' . $pathInfo['filename'] . '.webp';
    }

    private function generateThumbnails($image, StorageServiceInterface $storage): void
    {
        $sizes = [
            'thumb' => ['width' => 150, 'height' => 150],
            'small' => ['width' => 400, 'height' => 300],
            'medium' => ['width' => 800, 'height' => 600],
        ];

        foreach ($sizes as $size => $dimensions) {
            $thumbnail = $image->fit($dimensions['width'], $dimensions['height']);
            $thumbPath = $this->getThumbnailPath($this->imagePath, $size);
            $storage->put($thumbPath, $thumbnail->toWebp(80));
        }
    }

    private function getThumbnailPath(string $path, string $size): string
    {
        $pathInfo = pathinfo($path);
        return $pathInfo['dirname'] . '/thumbs/' . $size . '/' . $pathInfo['filename'] . '.webp';
    }

    public function failed(\Throwable $exception): void
    {
        Log::error('Image optimization failed', [
            'component_id' => $this->componentId,
            'image_path' => $this->imagePath,
            'error' => $exception->getMessage(),
        ]);
    }
}
```

### GenerateImageVariants

```php
// app/Domains/Components/Jobs/GenerateImageVariants.php

declare(strict_types=1);

namespace App\Domains\Components\Jobs;

use App\Domains\Storage\Contracts\StorageServiceInterface;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;

class GenerateImageVariants implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public int $tries = 3;
    public int $timeout = 180;

    public function __construct(
        public readonly int $componentId,
        public readonly string $imagePath,
    ) {}

    public function handle(StorageServiceInterface $storage): void
    {
        $originalContent = $storage->get($this->imagePath);

        $variants = [
            'original' => $originalContent,
            'retina' => $this->generateRetina($originalContent),
            'compressed' => $this->generateCompressed($originalContent),
            'placeholder' => $this->generatePlaceholder($originalContent),
        ];

        foreach ($variants as $variant => $content) {
            $variantPath = $this->getVariantPath($this->imagePath, $variant);
            $storage->put($variantPath, $content);
        }
    }

    private function generateRetina(string $content): string
    {
        $manager = new Image(['driver' => 'gd']);
        $image = $manager->make($content);
        $image->resize(2400, null, fn($c) => $c->aspectRatio());
        return $image->toWebp(90);
    }

    private function generateCompressed(string $content): string
    {
        $manager = new Image(['driver' => 'gd']);
        $image = $manager->make($content);
        $image->resize(1200, null, fn($c) => $c->aspectRatio());
        return $image->toWebp(75);
    }

    private function generatePlaceholder(string $content): string
    {
        $manager = new Image(['driver' => 'gd']);
        $image = $manager->make($content);
        $image->resize(50, null, fn($c) => $c->aspectRatio());
        return $image->toWebp(40);
    }

    private function getVariantPath(string $path, string $variant): string
    {
        $pathInfo = pathinfo($path);
        return $pathInfo['dirname'] . '/variants/' . $variant . '/' . $pathInfo['filename'] . '.webp';
    }
}

## Documentation Jobs

### GenerateDocumentation

```php
// app/Domains/Documentation/Jobs/GenerateDocumentation.php

declare(strict_types=1);

namespace App\Domains\Documentation\Jobs;

use App\Domains\Documentation\Contracts\DocumentationServiceInterface;
use App\Domains\Components\Contracts\ComponentServiceInterface;
use App\Domains\Storage\Contracts\StorageServiceInterface;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;

class GenerateDocumentation implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public int $tries = 3;
    public int $timeout = 300;

    public function __construct(
        public readonly int $componentId,
    ) {}

    public function handle(
        DocumentationServiceInterface $docService,
        ComponentServiceInterface $componentService,
        StorageServiceInterface $storage,
    ): void {
        $component = $componentService->findById($this->componentId);

        // Generate API reference
        $apiReference = $this->generateApiReference($component);

        // Generate usage guide
        $usageGuide = $this->generateUsageGuide($component);

        // Generate code examples
        $examples = $this->generateExamples($component);

        // Store documentation
        $docService->store([
            'component_id' => $this->componentId,
            'api_reference' => $apiReference,
            'usage_guide' => $usageGuide,
            'examples' => $examples,
            'generated_at' => now(),
        ]);

        // Index in search
        $docService->indexDocumentation($this->componentId);
    }

    private function generateApiReference($component): string
    {
        $props = $component->props ?? [];
        $events = $component->events ?? [];
        $slots = $component->slots ?? [];

        $reference = "# {$component->name} API Reference\n\n";
        $reference .= "## Props\n\n";
        $reference .= "| Prop | Type | Default | Description |\n";
        $reference .= "|------|------|---------|-------------|\n";

        foreach ($props as $prop) {
            $reference .= "| {$prop['name']} | {$prop['type']} | {$prop['default'] ?? 'N/A'} | {$prop['description']} |\n";
        }

        $reference .= "\n## Events\n\n";
        foreach ($events as $event) {
            $reference .= "- **{$event['name']}**: {$event['description']}\n";
        }

        $reference .= "\n## Slots\n\n";
        foreach ($slots as $slot) {
            $reference .= "- **{$slot['name']}**: {$slot['description']}\n";
        }

        return $reference;
    }

    private function generateUsageGuide($component): string
    {
        return "# Using {$component->name}\n\n" .
               "## Installation\n\n" .
               "```bash\n" .
               "php artisan gsmui:install {$component->slug}\n" .
               "```\n\n" .
               "## Basic Usage\n\n" .
               "```blade\n" .
               "<x-gsmui::{$component->slug} />\n" .
               "```\n\n" .
               "## Advanced Usage\n\n" .
               "See the API Reference for all available props and events.\n";
    }

    private function generateExamples($component): array
    {
        $examples = [];

        foreach ($component->examples as $example) {
            $examples[] = [
                'title' => $example->title,
                'code' => $example->code,
                'description' => $example->description,
            ];
        }

        return $examples;
    }
}

## Search Jobs

### SyncSearchIndex

```php
// app/Domains/Search/Jobs/SyncSearchIndex.php

declare(strict_types=1);

namespace App\Domains\Search\Jobs;

use App\Domains\Search\Contracts\SearchServiceInterface;
use App\Domains\Components\Contracts\ComponentServiceInterface;
use App\Domains\Themes\Contracts\ThemeServiceInterface;
use App\Domains\Templates\Contracts\TemplateServiceInterface;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;

class SyncSearchIndex implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public int $tries = 3;
    public int $timeout = 600;

    public function __construct(
        public readonly ?string $type = null,
    ) {}

    public function handle(
        SearchServiceInterface $searchService,
        ComponentServiceInterface $componentService,
        ThemeServiceInterface $themeService,
        TemplateServiceInterface $templateService,
    ): void {
        match ($this->type) {
            'components' => $this->syncComponents($searchService, $componentService),
            'themes' => $this->syncThemes($searchService, $themeService),
            'templates' => $this->syncTemplates($searchService, $templateService),
            default => $this->syncAll($searchService, $componentService, $themeService, $templateService),
        };
    }

    private function syncComponents(SearchServiceInterface $search, ComponentServiceInterface $components): void
    {
        $search->clearIndex('components');

        $components->getPublished()->each(function ($component) use ($search) {
            $search->indexComponent($component);
        });
    }

    private function syncThemes(SearchServiceInterface $search, ThemeServiceInterface $themes): void
    {
        $search->clearIndex('themes');

        $themes->getPublished()->each(function ($theme) use ($search) {
            $search->indexTheme($theme);
        });
    }

    private function syncTemplates(SearchServiceInterface $search, TemplateServiceInterface $templates): void
    {
        $search->clearIndex('templates');

        $templates->getPublished()->each(function ($template) use ($search) {
            $search->indexTemplate($template);
        });
    }

    private function syncAll(
        SearchServiceInterface $search,
        ComponentServiceInterface $components,
        ThemeServiceInterface $themes,
        TemplateServiceInterface $templates,
    ): void {
        $search->clearAllIndexes();

        $this->syncComponents($search, $components);
        $this->syncThemes($search, $themes);
        $this->syncTemplates($search, $templates);
    }
}

// app/Domains/Billing/Jobs/GenerateInvoicePdf.php

declare(strict_types=1);

namespace App\Domains\Billing\Jobs;

use App\Domains\Billing\Contracts\BillingServiceInterface;
use App\Domains\Billing\Models\Invoice;
use App\Domains\Storage\Contracts\StorageServiceInterface;
use App\Domains\Billing\Notifications\InvoiceReadyNotification;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Support\Facades\Notification;

class GenerateInvoicePdf implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public int $tries = 3;
    public int $timeout = 60;

    public function __construct(
        public readonly int $invoiceId,
    ) {}

    public function handle(
        StorageServiceInterface $storage,
        BillingServiceInterface $billingService,
    ): void {
        $invoice = Invoice::findOrFail($this->invoiceId);

        // Generate PDF content
        $pdfContent = $this->generatePdf($invoice);

        // Store PDF
        $path = "invoices/{$invoice->invoice_number}.pdf";
        $storage->put($path, $pdfContent);

        // Update invoice record
        $invoice->update([
            'pdf_path' => $path,
            'pdf_generated_at' => now(),
        ]);

        // Notify user
        Notification::send(
            $invoice->user,
            new InvoiceReadyNotification($invoice)
        );
    }

    private function generatePdf(Invoice $invoice): string
    {
        // Use a PDF library like DomPDF or Snappy
        $html = view('emails.invoice', [
            'invoice' => $invoice,
            'items' => $invoice->items,
            'company' => config('gsmui.company'),
        ])->render();

        return \Barryvdh\DomPDF\PDF::loadHTML($html)
            ->setPaper('a4')
            ->setOrientation('portrait')
            ->setOptions(['isRemoteEnabled' => true])
            ->output();
    }
}

        public readonly array $payload,
        public readonly string $signature,
    ) {}

    public function handle(BillingServiceInterface $billingService): void
    {
        // Verify signature
        if (!$billingService->verifyWebhookSignature($this->provider, $this->payload, $this->signature)) {
            throw new \InvalidArgumentException('Invalid webhook signature');
        }

        // Process webhook
        $billingService->handleWebhook($this->provider, $this->payload);

        Log::info('Webhook processed', [
            'provider' => $this->provider,
            'event_type' => $this->payload['type'] ?? 'unknown',
        ]);
    }

    public function failed(\Throwable $exception): void
    {
        Log::error('Webhook processing failed', [
            'provider' => $this->provider,
            'error' => $exception->getMessage(),
            'payload' => $this->payload,
        ]);
    }

use App\Domains\Themes\Contracts\ThemeServiceInterface;
use App\Domains\Storage\Contracts\StorageServiceInterface;

use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Support\Facades\Notification;
use ZipArchive;

class ExportThemePackage implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public int $tries = 3;
    public int $timeout = 300;

    public function __construct(
        public readonly int $themeId,
        public readonly int $userId,
    ) {}

    public function handle(
        ThemeServiceInterface $themeService,
        StorageServiceInterface $storage,
    ): void {
        $theme = Theme::findOrFail($this->themeId);

        // Generate package files
        $files = $this->generatePackageFiles($theme);

        // Create ZIP archive
        $zipContent = $this->createZipArchive($theme->slug, $files);

        // Store package
        $path = "themes/{$theme->slug}/package/{$theme->slug}-v{$theme->version}.zip";
        $storage->put($path, $zipContent);

        // Update theme record
        $theme->update([
            'package_path' => $path,
            'package_generated_at' => now(),
        ]);

        // Notify user
        $user = \App\Domains\Users\Models\User::find($this->userId);
        Notification::send($user, new ThemeExportReadyNotification($theme));
    }

    private function generatePackageFiles(Theme $theme): array
    {
        $files = [];

        // CSS variables
        $files['css/variables.css'] = $this->generateCssVariables($theme);

        // Tailwind config
        $files['tailwind.config.js'] = $this->generateTailwindConfig($theme);

        // Preview images
        foreach ($theme->previews as $preview) {
            $files["previews/{$preview->filename}"] = file_get_contents($preview->path);
        }

        // README
        $files['README.md'] = $this->generateReadme($theme);

        return $files;
    }

    private function createZipArchive(string $name, array $files): string
    {
        $zip = new ZipArchive();
        $tempFile = tempnam(sys_get_temp_dir(), 'theme_export_');

        if ($zip->open($tempFile, ZipArchive::CREATE | ZipArchive::OVERWRITE) !== true) {
            throw new \RuntimeException('Failed to create ZIP archive');
        }

        foreach ($files as $path => $content) {
            $zip->addFromString($path, $content);
        }

        $zip->close();

        $content = file_get_contents($tempFile);
        unlink($tempFile);

        return $content;
    }

    private function generateCssVariables(Theme $theme): string
    {
        $css = ":root {\n";

        foreach ($theme->tokens as $category => $tokens) {
            $css .= "  /* {$category} */\n";
            foreach ($tokens as $name => $value) {
                $css .= "  --gsmui-{$name}: {$value};\n";
            }
            $css .= "\n";
        }

        $css .= "}\n";
        return $css;
    }
}

// app/Domains/System/Jobs/CleanupTempFiles.php

declare(strict_types=1);

namespace App\Domains\System\Jobs;

use App\Domains\Storage\Contracts\StorageServiceInterface;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Support\Facades\Log;

class CleanupTempFiles implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public int $tries = 3;
    public int $timeout = 300;

    public function __construct(
        public readonly int $olderThanHours = 24,
    ) {}

    public function handle(StorageServiceInterface $storage): void
    {
        $cutoff = now()->subHours($this->olderThanHours);

        // Clean temp uploads
        $tempFiles = $storage->files('temp');
        $deletedCount = 0;

        foreach ($tempFiles as $file) {
            $lastModified = $storage->lastModified($file);
            if ($lastModified < $cutoff->timestamp) {
                $storage->delete($file);
                $deletedCount++;
            }
        }

        // Clean orphaned thumbnails
        $thumbFiles = $storage->files('thumbnails');
        foreach ($thumbFiles as $file) {
            $lastModified = $storage->lastModified($file);
            if ($lastModified < $cutoff->timestamp) {
                $storage->delete($file);
                $deletedCount++;
            }
        }

        Log::info('Temp files cleaned', [
            'files_deleted' => $deletedCount,
            'cutoff' => $cutoff->toIso8601String(),
        ]);
    }
}

class JobRegistry
{

    public function register(string $jobClass, array $metadata): void
    {
        $this->jobs[$jobClass] = [
            'class' => $jobClass,
            'name' => class_basename($jobClass),
            'description' => $metadata['description'] ?? '',
            'queue' => $metadata['queue'] ?? 'default',

    {
        return $this->jobs;
    }

    {
        return array_filter($this->jobs, fn($job) => $job['queue'] === $queue);
    }
}
