```php
class ProfileManagement extends Component
{
    public string $name = '';
    public string $email = '';
    public ?string $avatarUrl = null;
    public string $timezone = 'UTC';
    public string $locale = 'en';
    public bool $isEditing = false;

    // Edit form properties
    public string $editName = '';
    public string $editEmail = '';
    public string $editTimezone = '';
    public string $editLocale = '';

    protected function getListeners(): array
    {
        return ['profile.updated' => 'refresh'];
    }

    public function mount(): void
    {
        $user = auth()->user();
        $this->name = $user->name;
        $this->email = $user->email;
        $this->avatarUrl = $user->avatar_url;
        $this->timezone = $user->timezone ?? 'UTC';
        $this->locale = $user->locale ?? 'en';
    }

    public function startEditing(): void
    {
        $this->isEditing = true;
        $this->editName = $this->name;
        $this->editEmail = $this->email;
        $this->editTimezone = $this->timezone;
        $this->editLocale = $this->locale;
    }

    public function saveProfile(): void
    {
        $this->validate([
            'editName' => 'required|string|max:255',
            'editEmail' => 'required|email|max:255|unique:users,email,' . auth()->id(),
            'editTimezone' => 'nullable|string|in:' . implode(',', DateTimeZone::listIdentifiers()),
            'editLocale' => 'nullable|string|in:en,fa,ar,de,fr,es,ja,zh',
        ]);

        $user = auth()->user();
        $user->update([
            'name' => $this->editName,
            'email' => $this->editEmail,
            'timezone' => $this->editTimezone,
            'locale' => $this->editLocale,
        ]);

        $this->name = $user->name;
        $this->email = $user->email;
        $this->timezone = $user->timezone;
        $this->locale = $user->locale;
        $this->isEditing = false;

        $this->dispatch('profile.updated');
        session()->flash('success', 'Profile updated successfully');
    }

    public function cancelEditing(): void
    {
        $this->isEditing = false;
        $this->resetValidation();
    }
}
```

### Security Settings Component

**Two-Factor Authentication Setup:**

```php
class TwoFactorSetup extends Component
{
    public bool $enabled = false;
    public ?string $qrCode = null;
    public ?string $recoveryCodes = null;
    public string $verificationCode = '';
    public string $password = '';
    public bool $showRecoveryCodes = false;

    public function mount(): void
    {
        $this->enabled = auth()->user()->two_factor_secret !== null;
    }

    public function enableTwoFactor(): void
    {
        $this->validate([
            'password' => 'required',
            'verificationCode' => 'required|string|size:6',
        ]);

        $user = auth()->user();

        if (!Hash::check($this->password, $user->password)) {
            $this->addError('password', 'Password is incorrect');
            return;
        }

        $secret = $this->generateSecret();
        $google2fa = app(Google2FA::class);

        if (!$google2fa->verifyKey($secret, $this->verificationCode)) {
            $this->addError('verificationCode', 'Invalid verification code');
            return;
        }

        $recoveryCodes = $this->generateRecoveryCodes();

        $user->update([
            'two_factor_secret' => encrypt($secret),
            'two_factor_recovery_codes' => encrypt(json_encode($recoveryCodes)),
        ]);

        $this->enabled = true;
        $this->recoveryCodes = json_encode($recoveryCodes, JSON_PRETTY_PRINT);
        $this->showRecoveryCodes = true;
    }

    public function disableTwoFactor(): void
    {
        $this->validate([
            'password' => 'required',
        ]);

        $user = auth()->user();

        if (!Hash::check($this->password, $user->password)) {
            $this->addError('password', 'Password is incorrect');
            return;
        }

        $user->update([
            'two_factor_secret' => null,
            'two_factor_recovery_codes' => null,
        ]);

        $this->enabled = false;
        $this->recoveryCodes = null;
        session()->flash('success', 'Two-factor authentication disabled');
    }

    public function regenerateRecoveryCodes(): void
    {
        $recoveryCodes = $this->generateRecoveryCodes();

        auth()->user()->update([
            'two_factor_recovery_codes' => encrypt(json_encode($recoveryCodes)),
        ]);

        $this->recoveryCodes = json_encode($recoveryCodes, JSON_PRETTY_PRINT);
        session()->flash('success', 'Recovery codes regenerated');
    }

    private function generateSecret(): string
    {
        return app(Google2FA::class)->generateSecretKey();
    }

    private function generateRecoveryCodes(): array
    {
        $codes = [];
        for ($i = 0; $i < 10; $i++) {
            $codes[] = strtoupper(bin2hex(random_bytes(5)));
        }
        return $codes;
    }
}
```

### Session Management Component

```php
class SessionManagement extends Component
{
    public Collection $sessions;
    public string $currentSessionId;

    public function mount(): void
    {
        $this->currentSessionId = session()->getId();
        $this->loadSessions();
    }

    public function loadSessions(): void
    {
        $this->sessions = auth()->user()->sessions()
            ->orderBy('last_activity', 'desc')
            ->get()
            ->map(function ($session) {
                $session->is_current = $session->id === $this->currentSessionId;
                $session->device_info = $this->parseUserAgent($session->user_agent);
                $session->last_active_human = $session->last_activity_at->diffForHumans();
                return $session;
            });
    }

    public function revokeSession(string $sessionId): void
    {
        if ($sessionId === $this->currentSessionId) {
            session()->flash('error', 'Cannot revoke current session');
            return;
        }

        auth()->user()->sessions()
            ->where('id', $sessionId)
            ->delete();

        $this->loadSessions();
        session()->flash('success', 'Session revoked');
    }

    public function revokeAllSessions(): void
    {
        auth()->user()->sessions()
            ->where('id', '!=', $this->currentSessionId)
            ->delete();

        $this->loadSessions();
        session()->flash('success', 'All other sessions revoked');
    }

    private function parseUserAgent(string $userAgent): array
    {
        $browser = 'Unknown';
        $os = 'Unknown';
        $device = 'Desktop';

        if (str_contains($userAgent, 'Windows')) $os = 'Windows';
        elseif (str_contains($userAgent, 'Mac')) $os = 'MacOS';
        elseif (str_contains($userAgent, 'Linux')) $os = 'Linux';
        elseif (str_contains($userAgent, 'Android')) $os = 'Android';
        elseif (str_contains($userAgent, 'iPhone') || str_contains($userAgent, 'iPad')) $os = 'iOS';

        if (str_contains($userAgent, 'Chrome')) $browser = 'Chrome';
        elseif (str_contains($userAgent, 'Firefox')) $browser = 'Firefox';
        elseif (str_contains($userAgent, 'Safari')) $browser = 'Safari';
        elseif (str_contains($userAgent, 'Edge')) $browser = 'Edge';

        if (str_contains($userAgent, 'Mobile') || str_contains($userAgent, 'iPhone')) $device = 'Mobile';
        elseif (str_contains($userAgent, 'iPad') || str_contains($userAgent, 'Tablet')) $device = 'Tablet';

        return [
            'browser' => $browser,
            'os' => $os,
            'device' => $device,
        ];
    }
}
```

### API Token Management Component

```php
class ApiTokenManagement extends Component
{
    public Collection $tokens;
    public string $newTokenName = '';
    public ?string $newTokenValue = null;
    public bool $showCreateForm = false;
    
    public array $availablePermissions = [
        'downloads' => 'Download purchased items',
        'licenses' => 'Manage license keys',
        'profile' => 'View and edit profile',
        'playground' => 'Access playground API',
        'marketplace' => 'Browse marketplace',
    ];
    
    public array $selectedPermissions = [];

    public function mount(): void
    {
        $this->loadTokens();
    }

    public function loadTokens(): void
    {
        $this->tokens = auth()->user()->tokens()
            ->orderBy('created_at', 'desc')
            ->get()
            ->map(function ($token) {
                $token->created_human = $token->created_at->diffForHumans();
                $token->last_used_human = $token->last_used_at?->diffForHumans() ?? 'Never';
                return $token;
            });
    }

    public function createToken(): void
    {
        $this->validate([
            'newTokenName' => 'required|string|max:255',
            'selectedPermissions' => 'required|array|min:1',
        ]);

        // Check token limit
        $existingTokens = auth()->user()->tokens()->count();
        if ($existingTokens >= 10) {
            session()->flash('error', 'Maximum token limit reached (10)');
            return;
        }

        $token = auth()->user()->createToken(
            $this->newTokenName,
            $this->selectedPermissions
        );

        $this->newTokenValue = $token->plainTextToken;
        $this->showCreateForm = false;
        $this->loadTokens();
        
        $this->reset(['newTokenName', 'selectedPermissions']);
    }

    public function revokeToken(int $tokenId): void
    {
        auth()->user()->tokens()
            ->where('id', $tokenId)
            ->delete();

        $this->loadTokens();
        session()->flash('success', 'Token revoked');
    }

    public function revokeAllTokens(): void
    {
        auth()->user()->tokens()->delete();
        $this->loadTokens();
        session()->flash('success', 'All tokens revoked');
    }
}
```

```php
        $this->currentPlanId = $subscription?->plan_id;

    public function selectPlan(string $planId): void
    {
        $this->selectedPlanId = $planId;
        $this->showUpgradeModal = true;
    }

    public function confirmUpgrade(): void
    {
        $plan = Plan::findOrFail($this->selectedPlanId);
        $user = auth()->user();

        // Check if upgrading or downgrading
        $currentPlan = $user->activeSubscription?->plan;
        $isUpgrade = !$currentPlan || $plan->price > $currentPlan->price;

        if ($isUpgrade) {
            // Process immediate upgrade
            $this->processUpgrade($user, $plan);
        } else {
            // Schedule downgrade for period end
            $this->scheduleDowngrade($user, $plan);
        }

        $this->showUpgradeModal = false;
        $this->selectedPlanId = null;
        
        $this->dispatch('subscription.updated');
    }

    private function processUpgrade(User $user, Plan $plan): void
    {
        $stripe = new StripeClient(config('services.stripe.secret'));
        
        $subscription = $user->activeSubscription;
        
        if ($subscription) {
            // Update existing subscription
            $stripe->subscriptions->update($subscription->stripe_id, [
                'items' => [
                    [
                        'id' => $subscription->stripe_item_id,
                        'price' => $plan->stripe_price_id,
                    ],
                ],
                'proration_behavior' => 'create_prorations',
            ]);
        } else {
            // Create new subscription
            $stripe->subscriptions->create([
                'customer' => $user->stripe_customer_id,
                'items' => [
                    ['price' => $plan->stripe_price_id],
                ],
                'payment_behavior' => 'default_incomplete',
                'expand' => ['latest_invoice.payment_intent'],
            ]);
        }

        // Update local subscription
        $user->updateOrCreateSubscription($plan);
    }

    private function scheduleDowngrade(User $user, Plan $plan): void
    {
        $subscription = $user->activeSubscription;
        
        if ($subscription) {
            $stripe = new StripeClient(config('services.stripe.secret'));
            
            // Get current period end
            $stripeSub = $stripe->subscriptions->retrieve($subscription->stripe_id);
            
            // Schedule update for period end
            $stripe->subscriptions->update($subscription->stripe_id, [
                'items' => [
                    [
                        'id' => $subscription->stripe_item_id,
                        'price' => $plan->stripe_price_id,
                    ],
                ],
                'proration_behavior' => 'none',
                'billing_cycle_anchor' => 'unchanged',
            ]);

            // Store pending change
            $subscription->update([
                'pending_plan_id' => $plan->id,
                'pending_change_date' => Carbon::createFromTimestamp($stripeSub->current_period_end),
            ]);

            session()->flash('success', "Downgrade scheduled for {$subscription->pending_change_date->format('M d, Y')}");
        }
    }
```

```php
    public function download(string $productId, ?string $versionId = null): ?string
    {
        $purchase = auth()->user()->purchases()
            ->where('product_id', $productId)
            ->firstOrFail();

        // Verify license
        if (!$purchase->license || $purchase->license->status !== 'active') {
            session()->flash('error', 'No active license for this product');
            return null;
        }

        // Check download rate limit
        $recentDownloads = Download::where('user_id', auth()->id())
            ->where('created_at', '>=', now()->subHour())
            ->count();

        if ($recentDownloads >= 20) {
            session()->flash('error', 'Download limit reached. Please try again later.');
            return null;
        }

        // Get version
        $version = $versionId
            ? $purchase->product->versions()->findOrFail($versionId)
            : $purchase->product->latestVersion;

        if (!$version) {
            session()->flash('error', 'No version available for download');
            return null;
        }

        // Generate signed URL
        $downloadUrl = URL::temporarySignedRoute(
            'downloads.file',
            now()->addHour(),
            [
                'token' => $this->generateDownloadToken($purchase, $version),
                'product' => $purchase->product->slug,
                'version' => $version->version,
            ]
        );

        // Log download
        Download::create([
            'user_id' => auth()->id(),
            'product_id' => $productId,
            'version_id' => $version->id,
            'license_id' => $purchase->license->id,
            'ip_address' => request()->ip(),
        ]);

        return $downloadUrl;
    }

        }

    }
```

```php
            ->orderBy('created_at', 'desc')
            ->get();

            'csv' => $this->exportCsv($activities),
            'json' => $this->exportJson($activities),
            default => null,
        };
    }

    private function exportCsv(Collection $activities): void
    {
        $headers = ['Date', 'Type', 'Description', 'IP Address', 'Device'];

        $callback = function () use ($headers, $activities) {
            $file = fopen('php://output', 'w');
            fputcsv($file, $headers);

            foreach ($activities as $activity) {
                fputcsv($file, [
                    $activity->created_at->format('Y-m-d H:i:s'),
                    $activity->type,
                    $activity->description,
                    $activity->ip_address,
                    $activity->user_agent,
                ]);
            }

            fclose($file);
        };

        $headers = [
            'Content-Type' => 'text/csv',
            'Content-Disposition' => 'attachment; filename="activity-' . now()->format('Y-m-d') . '.csv"',
        ];

        return response()->stream($callback, 200, $headers);
    }

    private function exportJson(Collection $activities): void
    {
        return response()->json([
            'user' => auth()->user()->only(['id', 'name', 'email']),
            'exported_at' => now()->toIso8601String(),
            'activities' => $activities->toArray(),
        ], 200, [], JSON_PRETTY_PRINT);
    }
```

```php
// app/Http/Middleware/DashboardMiddleware.php
class DashboardMiddleware
{
    public function handle(Request $request, Closure $next): Response
    {
        // Check if user is authenticated
        if (!auth()->check()) {
            if ($request->expectsJson()) {
                return response()->json(['error' => 'Unauthenticated'], 401);
            }
            return redirect()->route('login');
        }

        // Check if email is verified
        if (!auth()->user()->hasVerifiedEmail()) {
            if ($request->expectsJson()) {
                return response()->json(['error' => 'Email not verified'], 403);
            }
            return redirect()->route('verification.notice');
        }

        // Check if account is active
        if (auth()->user()->status !== 'active') {
            auth()->logout();
            if ($request->expectsJson()) {
                return response()->json(['error' => 'Account suspended'], 403);
            }
            return redirect()->route('login')->withErrors([
                'email' => 'Your account has been suspended.',
            ]);
        }

        // Set user timezone
        if (auth()->user()->timezone) {
            config(['app.timezone' => auth()->user()->timezone]);
            date_default_timezone_set(auth()->user()->timezone);
        }

        // Check session timeout
        if ($this->isSessionTimedOut()) {
            auth()->logout();
            if ($request->expectsJson()) {
                return response()->json(['error' => 'Session expired'], 401);
            }
            return redirect()->route('login')->withErrors([
                'email' => 'Your session has expired. Please log in again.',
            ]);
        }

        // Update last activity timestamp
        session(['last_activity' => now()->timestamp]);

        return $next($request);
    }

    private function isSessionTimedOut(): bool
    {
        $lastActivity = session('last_activity');
        $idleTimeout = config('dashboard.session.idle_timeout', 30) * 60;

        if (!$lastActivity) {
            return false;
        }

        return (now()->timestamp - $lastActivity) > $idleTimeout;
    }
}
```

```php
// app/Http/Middleware/DashboardRateLimit.php
class DashboardRateLimit
{
    public function handle(Request $request, Closure $next): Response
    {
        $key = 'dashboard:' . auth()->id();
        $maxAttempts = config('dashboard.rate_limit.max_attempts', 60);
        $decayMinutes = config('dashboard.rate_limit.decay_minutes', 1);

        if (RateLimiter::tooManyAttempts($key, $maxAttempts)) {
            $retryAfter = RateLimiter::availableIn($key);

            if ($request->expectsJson()) {
                return response()->json([
                    'error' => 'Too many requests',
                    'retry_after' => $retryAfter,
                ], 429);
            }

            return redirect()->back()
                ->withErrors(['rate_limit' => "Please try again in {$retryAfter} seconds."])
                ->withHeaders([
                    'Retry-After' => $retryAfter,
                    'X-RateLimit-Limit' => $maxAttempts,
                    'X-RateLimit-Remaining' => 0,
                ]);
        }

        RateLimiter::hit($key, $decayMinutes * 60);

        $response = $next($request);

        return $response->withHeaders([
            'X-RateLimit-Limit' => $maxAttempts,
            'X-RateLimit-Remaining' => RateLimiter::remaining($key, $maxAttempts),
        ]);
    }
}
```

```php
class DashboardCacheTest extends TestCase
{
    private DashboardCache $cache;

    protected function setUp(): void
    {
        parent::setUp();
        $this->cache = app(DashboardCache::class);
    }

    public function test_can_cache_profile(): void
    {
        $user = User::factory()->create();
        
        $profile = $this->cache->getProfile($user->id);
        
        $this->assertNotNull($profile);
        $this->assertEquals($user->name, $profile['name']);
    }

    public function test_can_invalidate_profile(): void
    {
        $user = User::factory()->create();
        $this->cache->getProfile($user->id);
        
        $this->cache->invalidateProfile($user->id);
        
        // Should fetch fresh data
        $profile = $this->cache->getProfile($user->id);
        $this->assertNotNull($profile);
    }
}
```

```php
// tests/Feature/Dashboard/AccountTest.php
class AccountTest extends TestCase
{
    public function test_user_can_view_account(): void
    {
        $user = User::factory()->create();
        
        $response = $this->actingAs($user)->get('/dashboard/account');
        
        $response->assertOk();
        $response->assertSee('Account');
    }

    public function test_user_can_update_profile(): void
    {
        $user = User::factory()->create();
        
        $response = $this->actingAs($user)->put('/dashboard/profile', [
            'name' => 'Updated Name',
            'email' => $user->email,
        ]);
        
        $response->assertRedirect();
        $this->assertEquals('Updated Name', $user->fresh()->name);
    }

    public function test_user_cannot_update_email_to_existing(): void
    {
        $user = User::factory()->create();
        $other = User::factory()->create();
        
        $response = $this->actingAs($user)->put('/dashboard/profile', [
            'name' => $user->name,
            'email' => $other->email,
        ]);
        
        $response->assertSessionHasErrors('email');
    }
}
```
