## Team Billing Flow

```php
class TeamBillingService
{
    public function createTeamSubscription(Team $team, Plan $plan): Subscription
    {
        $seats = $team->members()->count();
        $price = $plan->price_per_seat * $seats;

        $subscription = $this->cashier->createSubscription(
            user: $team->owner,
            plan: $plan->stripe_plan_id,
            quantity: $seats,
            metadata: [
                'team_id' => $team->id,
                'seat_count' => $seats,
            ]
        );

        $team->update([
            'subscription_id' => $subscription->id,
            'plan_id' => $plan->id,
            'seats_limit' => $plan->max_seats,
        ]);

        return $subscription;
    }

    public function addSeat(Team $team): void
    {
        $currentSeats = $team->members()->count();
        $newSeats = $currentSeats + 1;

        if ($newSeats > $team->seats_limit) {
            throw new SeatLimitExceededException();
        }

        $this->cashier->updateQuantity(
            subscription: $team->subscription,
            quantity: $newSeats
        );

        $this->seatHistory->record($team, 'added', $currentSeats, $newSeats);
    }

    public function removeSeat(Team $team, Member $member): void
    {
        $currentSeats = $team->members()->count();
        $newSeats = $currentSeats - 1;

        $member->delete();

        $this->cashier->updateQuantity(
            subscription: $team->subscription,
            quantity: $newSeats
        );

        $this->seatHistory->record($team, 'removed', $currentSeats, $newSeats);
    }
}
```

## Seat Tracking Service

```php
class SeatService
{
    public function getAvailableSeats(Team $team): int
    {
        $used = $team->members()->count();
        $reserved = $team->pendingInvitations()->count();

        return $team->seats_limit - $used - $reserved;
    }

    public function canAddSeat(Team $team): bool
    {
        return $this->getAvailableSeats($team) > 0;
    }

    public function getSeatUtilization(Team $team): float
    {
        $used = $team->members()->count();
        return $used / $team->seats_limit * 100;
    }

    public function notifyOverage(Team $team): void
    {
        $utilization = $this->getSeatUtilization($team);

        if ($utilization >= 100) {
            Notification::send($team->owner, new SeatLimitReachedNotification($team));
        } elseif ($utilization >= 80) {
            Notification::send($team->owner, new SeatUsageWarningNotification($team));
        }
    }
}
```

## Invitation Service

```php
class InvitationService
{
    public function invite(Team $team, string $email, string $role): Invitation
    {
        if (!$this->seatService->canAddSeat($team)) {
            throw new SeatLimitExceededException();
        }

        $existing = Invitation::where('team_id', $team->id)
            ->where('email', $email)
            ->where('status', 'pending')
            ->first();

        if ($existing) {
            return $existing->fresh(['created_at' => now()]);
        }

        $invitation = Invitation::create([
            'team_id' => $team->id,
            'email' => $email,
            'role' => $role,
            'token' => Str::random(64),
            'expires_at' => now()->addDays(7),
        ]);

        Mail::to($email)->send(new TeamInvitationMail($invitation));

        return $invitation;
    }

    public function accept(Invitation $invitation, User $user): Member
    {
        if ($invitation->expires_at->isPast()) {
            throw new InvitationExpiredException();
        }

        if ($invitation->email !== $user->email) {
            throw new InvitationEmailMismatchException();
        }

        $member = Member::create([
            'team_id' => $invitation->team_id,
            'user_id' => $user->id,
            'role' => $invitation->role,
            'joined_at' => now(),
        ]);

        $invitation->update(['status' => 'accepted']);

        $this->seatService->addSeat($invitation->team);

        return $member;
    }

    public function decline(Invitation $invitation): void
    {
        $invitation->update(['status' => 'declined']);
    }
}
```

## Role Management Service

```php
class RoleService
{
    private const PERMISSIONS = [
        'owner' => ['*'],
        'admin' => ['team.manage_members', 'team.view_billing', 'team.view_analytics'],
        'member' => ['team.use_features'],
        'guest' => ['team.view'],
    ];

    public function hasPermission(Member $member, string $permission): bool
    {
        $rolePermissions = self::PERMISSIONS[$member->role] ?? [];

        if (in_array('*', $rolePermissions)) {
            return true;
        }

        return in_array($permission, $rolePermissions);
    }

    public function canManageBilling(Member $member): bool
    {
        return $this->hasPermission($member, 'team.view_billing');
    }

    public function canManageMembers(Member $member): bool
    {
        return $this->hasPermission($member, 'team.manage_members');
    }
}
```

class TeamProrationCalculator
{
    public function calculateUpgrade(Team $team, Plan $newPlan): ProrationResult
    {
        $currentPlan = $team->plan;
        $seats = $team->members()->count();
        $daysRemaining = $this->getDaysRemainingInCycle($team);

        $currentDailyRate = ($currentPlan->price_per_seat * $seats) / 30;
        $newDailyRate = ($newPlan->price_per_seat * $seats) / 30;

        $credit = $currentDailyRate * $daysRemaining;
        $charge = $newDailyRate * $daysRemaining;

        return new ProrationResult(
            credit: $credit,
            charge: $charge,
            net: $charge - $credit,

## Member Service

```php
class MemberService
{
    public function inviteMember(Team $team, string $email, string $role): Invitation
    {
        if (!$this->roleService->canManageMembers($team->owner)) {
            throw new UnauthorizedException();
        }

        return $this->invitationService->invite($team, $email, $role);
    }

    public function updateRole(Team $team, Member $member, string $newRole): void
    {
        if (!$this->roleService->canManageMembers($team->owner)) {
            throw new UnauthorizedException();
        }

        if ($member->user_id === $team->owner_id) {
            throw new CannotChangeOwnerRoleException();
        }

        $member->update(['role' => $newRole]);
    }

    public function removeMember(Team $team, Member $member): void
    {
        if (!$this->roleService->canManageMembers($team->owner)) {
            throw new UnauthorizedException();
        }

        if ($member->user_id === $team->owner_id) {
            throw new CannotRemoveOwnerException();
        }

        $member->delete();
        $this->seatService->removeSeat($team, $member);
    }
}
```

    }

    {
        $team = Team::factory()->create(['seats_limit' => 10]);
        // Create 3 members
        for ($i = 1; $i <= 3; $i++) {
            $team->members()->create(['user_id' => $i, 'role' => 'member']);
        }

        $utilization = $this->seatService->getSeatUtilization($team);

        $this->assertEquals(30.0, $utilization);
    }
