# Purpose

The checkout system handles the purchase flow for both subscriptions and one-time purchases.

---

# Checkout Flow

## Subscription Checkout

1. User selects plan
2. Redirect to Stripe Checkout
3. Enter payment details
4. Confirm subscription
5. Redirect back with session ID
6. Verify webhook confirmation

## One-Time Purchase Checkout

1. User adds to cart
2. Review cart
3. Enter payment details
4. Confirm purchase
5. Generate license
6. Show download link

```php
// app/Models/CheckoutSession.php
class CheckoutSession extends Model
{
    use HasFactory;

    protected $fillable = [
        'user_id', 'type', 'status', 'stripe_session_id',
        'product_id', 'plan_id', 'coupon_code',
        'subtotal', 'tax_amount', 'discount_amount', 'total',
        'currency', 'metadata',
        'abandoned_at', 'completed_at',
    ];

    protected $casts = [
        'subtotal' => 'integer',
        'tax_amount' => 'integer',
        'discount_amount' => 'integer',
        'total' => 'integer',
        'metadata' => 'array',
        'abandoned_at' => 'datetime',
        'completed_at' => 'datetime',
    ];

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class);
    }

    public function product(): BelongsTo
    {
        return $this->belongsTo(Product::class);
    }

    public function plan(): BelongsTo
    {
        return $this->belongsTo(Plan::class);
    }

    public function events(): HasMany
    {
        return $this->hasMany(CheckoutEvent::class);
    }

    public function scopeActive(Builder $query): Builder
    {
        return $query->where('status', 'active')
            ->where('created_at', '>=', now()->subHours(24));
    }

    public function isExpired(): bool
    {
        return $this->created_at->diffInMinutes(now()) > 60;
    }

    public function getTotalFormattedAttribute(): string
    {
        return '$' . number_format($this->total / 100, 2);
    }
}
```

## Subscription Checkout

```php
// app/Services/SubscriptionCheckoutService.php
class SubscriptionCheckoutService
{
    public function __construct(
        private StripeClient $stripe,
        private TaxService $taxService,
        private CouponService $couponService,
    ) {}

    public function createSession(
        User $user,
        Plan $plan,
        ?string $couponCode = null,
        ?array $taxAddress = null,
    ): CheckoutSession {
        // Calculate pricing
        $subtotal = $plan->price;
        $discount = $couponCode ? $this->couponService->calculateDiscount($couponCode, $subtotal, $user) : 0;
        $tax = $taxAddress ? $this->taxService->calculateTax($subtotal - $discount, $plan->currency, $taxAddress) : null;
        $taxAmount = $tax?->amount ?? 0;
        $total = $subtotal - $discount + $taxAmount;

        // Create Stripe Checkout Session
        $stripeSession = $this->createStripeSession(
            user: $user,
            plan: $plan,
            subtotal: $subtotal,
            discount: $discount,
            taxAmount: $taxAmount,
            couponCode: $couponCode,
        );

        // Create local session
        $session = CheckoutSession::create([
            'user_id' => $user->id,
            'type' => 'subscription',
            'status' => 'active',
            'stripe_session_id' => $stripeSession->id,
            'plan_id' => $plan->id,
            'coupon_code' => $couponCode,
            'subtotal' => $subtotal,
            'tax_amount' => $taxAmount,
            'discount_amount' => $discount,
            'total' => $total,
            'currency' => $plan->currency,
            'metadata' => [
                'plan_name' => $plan->name,
                'interval' => $plan->interval,
                'trial_days' => $plan->trial_days,
            ],
        ]);

        // Record event
        $session->events()->create([
            'event_type' => 'created',
            'payload' => ['plan_id' => $plan->id, 'total' => $total],
        ]);

        event(new CheckoutCreated($session));

        return $session;
    }

    private function createStripeSession(
        User $user,
        Plan $plan,
        int $subtotal,
        int $discount,
        int $taxAmount,
        ?string $couponCode,
    ): \Stripe\Checkout\Session {
        $params = [
            'payment_method_types' => ['card'],
            'mode' => 'subscription',
            'customer' => $user->stripe_id,
            'line_items' => [
                [
                    'price_data' => [
                        'currency' => strtolower($plan->currency),
                        'product_data' => [
                            'name' => "{$plan->name} Plan",
                            'description' => $plan->description,
                        ],
                        'unit_amount' => $plan->price,
                        'recurring' => [
                            'interval' => $plan->interval === 'yearly' ? 'year' : 'month',
                        ],
                    ],
                    'quantity' => 1,
                ],
            ],
            'success_url' => route('checkout.success', [
                'session_id' => '{CHECKOUT_SESSION_ID}',
                'type' => 'subscription',
            ]),
            'cancel_url' => route('checkout.cancel', [
                'session_id' => '{CHECKOUT_SESSION_ID}',
            ]),
            'metadata' => [
                'user_id' => $user->id,
                'plan_id' => $plan->id,
                'type' => 'subscription',
            ],
            'subscription_data' => [
                'trial_period_days' => $plan->trial_days > 0 ? $plan->trial_days : null,
                'metadata' => [
                    'user_id' => $user->id,
                    'plan_slug' => $plan->slug,
                ],
            ],
        ];

        // Apply coupon
        if ($couponCode) {
            $coupon = $this->couponService->getStripeCoupon($couponCode);
            if ($coupon) {
                $params['discounts'] = [['coupon' => $coupon->id]];
            }
        }

        // Apply tax collection
        if ($taxAmount > 0) {
            $params['taxes'] = [['amount' => 'auto']];
        }

        return $this->stripe->checkout->sessions->create($params);
    }
}
```

```php
class OneTimeCheckoutService
{
    public function createSession(
        User $user,
        Product $product,
        ?string $couponCode = null,
        ?array $taxAddress = null,
    ): CheckoutSession {
        $subtotal = $product->price;
        $discount = $couponCode ? $this->couponService->calculateDiscount($couponCode, $subtotal, $user) : 0;
        $tax = $taxAddress ? $this->taxService->calculateTax($subtotal - $discount, $product->currency ?? 'usd', $taxAddress) : null;
        $taxAmount = $tax?->amount ?? 0;
        $total = $subtotal - $discount + $taxAmount;

        $stripeSession = $this->stripe->checkout->sessions->create([
            'payment_method_types' => ['card'],
            'mode' => 'payment',
            'customer' => $user->stripe_id,
            'line_items' => [
                [
                    'price_data' => [
                        'currency' => $product->currency ?? 'usd',
                        'product_data' => [
                            'name' => $product->name,
                            'description' => $product->short_description,
                            'images' => array_slice($product->preview_images ?? [], 0, 1),
                        ],
                        'unit_amount' => $product->price,
                    ],
                    'quantity' => 1,
                ],
            ],
            'success_url' => route('checkout.success', [
                'session_id' => '{CHECKOUT_SESSION_ID}',
                'type' => 'one_time',
            ]),
            'cancel_url' => route('checkout.cancel', [
                'session_id' => '{CHECKOUT_SESSION_ID}',
            ]),
            'metadata' => [
                'user_id' => $user->id,
                'product_id' => $product->id,
                'type' => 'one_time',
            ],
        ]);

        $session = CheckoutSession::create([
            'user_id' => $user->id,
            'type' => 'one_time',
            'status' => 'active',
            'stripe_session_id' => $stripeSession->id,
            'product_id' => $product->id,
            'coupon_code' => $couponCode,
            'subtotal' => $subtotal,
            'tax_amount' => $taxAmount,
            'discount_amount' => $discount,
            'total' => $total,
            'currency' => $product->currency ?? 'usd',
        ]);

        $session->events()->create([
            'event_type' => 'created',
            'payload' => ['product_id' => $product->id, 'total' => $total],
        ]);

        event(new CheckoutCreated($session));

        return $session;
    }
}
```

```php
// app/Livewire/Checkout/CheckoutPage.php
class CheckoutPage extends Component
{
    public string $type = 'subscription'; // subscription | one_time
    public ?int $planId = null;
    public ?int $productId = null;
    public ?CheckoutSession $session = null;

    // Form state
    public string $email = '';
    public string $name = '';
    public ?array $taxAddress = null;
    public ?string $couponCode = null;
    public bool $agreedToTerms = false;

    // UI state
    public bool $processing = false;
    public ?string $error = null;
    public bool $showCouponForm = false;

    protected CheckoutService $checkoutService;

    public function boot(CheckoutService $checkoutService): void
    {
        $this->checkoutService = $checkoutService;
    }

    public function mount(): void
    {
        $user = auth()->user();

        if ($user) {
            $this->email = $user->email;
            $this->name = $user->name;
        }

        $this->createSession();
    }

    public function createSession(): void
    {
        $user = auth()->user();

        if ($this->type === 'subscription' && $this->planId) {
            $plan = Plan::findOrFail($this->planId);
            $this->session = $this->checkoutService->createSubscriptionSession(
                user: $user,
                plan: $plan,
                couponCode: $this->couponCode,
                taxAddress: $this->taxAddress,
            );
        } elseif ($this->type === 'one_time' && $this->productId) {
            $product = Product::findOrFail($this->productId);
            $this->session = $this->checkoutService->createOneTimeSession(
                user: $user,
                product: $product,
                couponCode: $this->couponCode,
                taxAddress: $this->taxAddress,
            );
        }
    }

    public function applyCoupon(): void
    {
        $this->validate([
            'couponCode' => 'required|string',
        ]);

        $this->createSession(); // Recalculate with coupon
    }

    public function updateTaxAddress(): void
    {
        $this->createSession(); // Recalculate with tax
    }

    public function checkout(): RedirectResponse
    {
        $this->validate([
            'agreedToTerms' => 'accepted',
        ]);

        if (!$this->session) {
            $this->error = 'Checkout session not found. Please try again.';
            return redirect()->back();
        }

        $this->processing = true;

        return redirect($this->session->stripe_url);
    }

    public function render(): View
    {
        return view('livewire.checkout.checkout-page', [
            'session' => $this->session,
        ]);
    }
}
```

```php
// app/Services/CheckoutError Handler.php
class CheckoutErrorHandler
{
    public function handleError(\Stripe\Exception\ApiErrorException $e, CheckoutSession $session): CheckoutError
    {
        $errorType = match (true) {
            $e instanceof \Stripe\Exception\CardException => 'card_declined',
            $e instanceof \Stripe\Exception\InvalidRequestException => 'invalid_request',
            $e instanceof \Stripe\Exception\AuthenticationException => 'authentication_error',
            $e instanceof \Stripe\Exception\ApiConnectionException => 'connection_error',
            $e instanceof \Stripe\Exception\RateLimitException => 'rate_limit',
            default => 'api_error',
        };

        $userMessage = $this->getUserMessage($errorType, $e);
        $retryable = $this->isRetryable($errorType);

        $session->events()->create([
            'event_type' => 'error',
            'payload' => [
                'type' => $errorType,
                'message' => $e->getMessage(),
                'stripe_code' => $e->getStripeCode(),
            ],
        ]);

        event(new CheckoutFailed($session, $errorType));

        return new CheckoutError(
            type: $errorType,
            message: $userMessage,
            retryable: $retryable,
            stripeCode: $e->getStripeCode(),
        );
    }

    private function getUserMessage(string $type, \Stripe\Exception\ApiErrorException $e): string
    {
        return match ($type) {
            'card_declined' => match ($e->getStripeCode()) {
                'card_declined' => 'Your card was declined. Please try a different payment method.',
                'expired_card' => 'Your card has expired. Please update your payment method.',
                'incorrect_cvc' => 'The CVC number is incorrect. Please check and try again.',
                'insufficient_funds' => 'Insufficient funds. Please try a different payment method.',
                'processing_error' => 'An error occurred processing your card. Please try again.',
                default => 'Payment failed. Please try a different payment method.',
            },
            'invalid_request' => 'There was an error with your request. Please try again.',
            'authentication_error' => 'Authentication failed. Please refresh and try again.',
            'connection_error' => 'Network error. Please check your connection and try again.',
            'rate_limit' => 'Too many requests. Please wait a moment and try again.',
            default => 'An unexpected error occurred. Please try again.',
        };
    }

    private function isRetryable(string $type): bool
    {
        return in_array($type, ['connection_error', 'rate_limit', 'processing_error']);
    }
}

class CheckoutError
{
    public function __construct(
        public readonly string $type,
        public readonly string $message,
        public readonly bool $retryable,
        public readonly ?string $stripeCode = null,
    ) {}
}
```

```php
class DetectAbandonedCheckouts implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public function handle(): void
    {
        // Find sessions abandoned for 30+ minutes
        $abandoned = CheckoutSession::where('status', 'active')
            ->where('created_at', '<=', now()->subMinutes(30))
            ->where('created_at', '>=', now()->subHours(24))
            ->whereNull('abandoned_at')
            ->with('user')
            ->get();

        foreach ($abandoned as $session) {
            $session->update(['abandoned_at' => now()]);

            // Send recovery email
            if ($session->user) {
                CheckoutAbandonmentMail::to($session->user)
                    ->delay(now()->addMinutes(5))
                    ->queue($session);
            }

            event(new CheckoutAbandoned($session));
        }

        // Send reminder after 24 hours
        $reminderDue = CheckoutSession::where('status', 'active')
            ->where('abandoned_at', '<=', now()->subHours(24))
            ->where('abandoned_at', '>=', now()->subHours(48))
            ->whereNull('reminder_sent_at')
            ->with('user')
            ->get();

        foreach ($reminderDue as $session) {
            $session->update(['reminder_sent_at' => now()]);

            if ($session->user) {
                CheckoutReminderMail::to($session->user)
                    ->queue($session);
            }
        }
    }
}
```

```php
// app/Services/CheckoutFraudDetection.php
class CheckoutFraudDetection
{
    public function assessRisk(CheckoutSession $session, User $user): RiskAssessment
    {
        $score = 0;
        $flags = [];

        // Velocity check: too many checkouts in short time
        $recentCheckouts = CheckoutSession::where('user_id', $user->id)
            ->where('created_at', '>=', now()->subHours(1))
            ->count();

        if ($recentCheckouts > 3) {
            $score += 30;
            $flags[] = 'high_velocity';
        }

        // Email age check
        if ($user->created_at->diffInDays(now()) < 1) {
            $score += 20;
            $flags[] = 'new_account';
        }

        // IP geolocation mismatch
        if ($this->detectGeolocationMismatch($user, $session)) {
            $score += 25;
            $flags[] = 'geo_mismatch';
        }

        // Known fraud patterns
        if ($this->matchesKnownPattern($session)) {
            $score += 40;
            $flags[] = 'known_pattern';
        }

        // Assess risk level
        $level = match (true) {
            $score >= 70 => 'high',
            $score >= 40 => 'medium',
            default => 'low',
        };

        return new RiskAssessment(
            score: $score,
            level: $level,
            flags: $flags,
            action: $this->determineAction($level),
        );
    }

    private function determineAction(string $level): string
    {
        return match ($level) {
            'high' => 'block',
            'medium' => 'review',
            default => 'allow',
        };
    }
}

class RiskAssessment
{
    public function __construct(
        public readonly int $score,
        public readonly string $level,
        public readonly array $flags,
        public readonly string $action,
    ) {}
}
```

```php
class CheckoutAnalytics
{
    public function getFunnel(string $type, Carbon $start, Carbon $end): array
    {
        $pageViews = CheckoutEvent::where('event_type', 'page_view')
            ->where('created_at', '>=', $start)
            ->where('created_at', '<=', $end)
            ->count();

        $started = CheckoutSession::where('type', $type)
            ->where('created_at', '>=', $start)
            ->where('created_at', '<=', $end)
            ->count();

        $completed = CheckoutSession::where('type', $type)
            ->where('status', 'completed')
            ->where('completed_at', '>=', $start)
            ->where('completed_at', '<=', $end)
            ->count();

        $abandoned = CheckoutSession::where('type', $type)
            ->where('status', 'active')
            ->where('abandoned_at', '>=', $start)
            ->where('abandoned_at', '<=', $end)
            ->count();

        return [
            'page_views' => $pageViews,
            'started' => $started,
            'completed' => $completed,
            'abandoned' => $abandoned,
            'start_rate' => $pageViews > 0 ? round(($started / $pageViews) * 100, 2) : 0,
            'completion_rate' => $started > 0 ? round(($completed / $started) * 100, 2) : 0,
            'abandonment_rate' => $started > 0 ? round(($abandoned / $started) * 100, 2) : 0,
        ];
    }

    public function getDropoffPoints(string $type): array
    {
        return CheckoutEvent::where('created_at', '>=', now()->subDays(30))
            ->select('event_type', DB::raw('COUNT(*) as count'))
            ->groupBy('event_type')
            ->get()
            ->pluck('count', 'event_type')
            ->toArray();
    }

    public function getRevenueBySource(): array
    {
        return Order::where('status', 'completed')

            ->select('metadata->source as source', DB::raw('SUM(total) as revenue'), DB::raw('COUNT(*) as orders'))
            ->groupBy('source')
            ->get()
```

## A/B Test Results

```php
// app/Services/CheckoutABTest.php
class CheckoutABTest
{
    public function getResults(string $experiment): array
    {
        $variants = CheckoutABAssignment::where('experiment', $experiment)
            ->with('conversions')
            ->get()
            ->groupBy('variant');

        $results = [];

        foreach ($variants as $variant => $assignments) {
            $total = $assignments->count();
            $conversions = $assignments->filter(fn($a) => $a->converted)->count();

            $results[$variant] = [
                'total' => $total,
                'conversions' => $conversions,
                'conversion_rate' => $total > 0 ? round(($conversions / $total) * 100, 2) : 0,
                'revenue' => $assignments->sum('revenue'),
                'average_order_value' => $conversions > 0
                    ? round($assignments->where('converted')->sum('revenue') / $conversions, 2)
                    : 0,
            ];
        }

        // Statistical significance
        $control = $results['control'] ?? reset($results);
        $variant = collect($results)->except('control')->first();

        if ($control && $variant) {
            $results['significance'] = $this->calculateSignificance(
                $control['conversions'],
                $control['total'],
                $variant['conversions'],
                $variant['total'],
            );
        }

        return $results;
    }

    private function calculateSignificance(int $c1, int $n1, int $c2, int $n2): float
    {
        $p1 = $c1 / $n1;
        $p2 = $c2 / $n2;
        $pooled = ($c1 + $c2) / ($n1 + $n2);

        $se = sqrt($pooled * (1 - $pooled) * (1/$n1 + 1/$n2));

        $z = $se > 0 ? ($p2 - $p1) / $se : 0;

        // Approximate p-value from z-score
        return round((1 - $this->normalCDF(abs($z))) * 2, 4);
    }
}
```

```php
// app/Services/CheckoutPerformanceMonitor.php
class CheckoutPerformanceMonitor
{
    public function recordMetric(string $metric, float $value, array $tags = []): void
    {
        // Store in time-series database
        PerformanceMetric::create([
            'metric' => $metric,
            'value' => $value,
            'tags' => $tags,
            'recorded_at' => now(),
        ]);

        // Alert if threshold exceeded
        $threshold = config("checkout.thresholds.{$metric}");
        if ($threshold && $value > $threshold) {
            CheckoutPerformanceAlert::dispatch($metric, $value, $threshold);
        }
    }

    public function getDashboard(): array
    {
        return [
            'page_load_time' => $this->getAverageMetric('page_load_time', 24),
            'payment_element_render' => $this->getAverageMetric('payment_element_render', 24),
            'session_creation_time' => $this->getAverageMetric('session_creation_time', 24),
            'checkout_completion_time' => $this->getAverageMetric('checkout_completion_time', 24),
            'error_rate' => $this->getErrorRate(24),
            'success_rate' => $this->getSuccessRate(24),
        ];
    }
}
```

# Checkout Data Export

## Checkout Analytics Export

```php
// app/Services/CheckoutExportService.php
class CheckoutExportService
{
    public function exportFunnelData(string $type, Carbon $start, Carbon $end): StreamedResponse
    {
        return response()->streamDownload(function () use ($type, $start, $end) {
            $handle = fopen('php://output', 'w');

            fputcsv($handle, [
                'Date', 'Page Views', 'Started', 'Completed',
                'Abandoned', 'Start Rate', 'Completion Rate',
            ]);

            $days = $start->diffInDays($end);

            for ($i = 0; $i <= $days; $i++) {
                $date = $start->copy()->addDays($i);
                $nextDate = $date->copy()->addDay();

                $data = app(CheckoutAnalytics::class)
                    ->getFunnel($type, $date, $nextDate);

                fputcsv($handle, [
                    $date->format('Y-m-d'),
                    $data['page_views'],
                    $data['started'],
                    $data['completed'],
                    $data['abandoned'],
                    $data['start_rate'] . '%',
                    $data['completion_rate'] . '%',
                ]);
            }

            fclose($handle);
        }, "checkout-funnel-{$type}-{$start->format('Y-m-d')}.csv");
    }
}
```
