# Purpose

Subscription management handles the lifecycle of user subscriptions including creation, modification, renewal, and cancellation.

---

# Subscription Lifecycle

## Creation Flow

1. User selects plan
2. Checkout page loads
3. Payment method collected
4. Subscription created in payment gateway
5. Webhook confirms creation
6. Local subscription record updated
7. User granted access

## Modification Flow

1. User requests upgrade/downgrade
2. Proration calculated
3. New price confirmed
4. Subscription updated in gateway
5. Webhook confirms change
6. Local record updated

## Renewal Flow

1. Subscription due for renewal
2. Payment processed
3. Webhook confirms success
4. User access confirmed
5. Invoice generated

## Cancellation Flow

1. User requests cancellation
2. Option: cancel now or at period end
3. Subscription updated in gateway
4. Webhook confirms
5. Access revoked (if immediate)

## State Descriptions

| State | Access | Billing | Webhooks | Description |
|-------|--------|---------|----------|-------------|
| `incomplete` | No | Not billed | `customer.subscription.created` | Checkout in progress |
| `trialing` | Yes | Not billed | `customer.subscription.trial_will_end` | Free trial period |
| `active` | Yes | Billed | `invoice.payment_succeeded` | Fully active subscription |
| `past_due` | Grace period | Retrying | `invoice.payment_failed` | Payment failed, retrying |
| `canceled` | No | Stopped | `customer.subscription.deleted` | User or system canceled |
| `unpaid` | No | Stopped | `customer.subscription.updated` | Payment permanently failed |
| `paused` | No | Suspended | `customer.subscription.paused` | User-initiated pause |

## Grace Period Logic

```php
// app/Services/SubscriptionGracePeriod.php
class SubscriptionGracePeriod
{
    private const GRACE_PERIOD_DAYS = 3;
    private const MAX_RETRY_ATTEMPTS = 3;
    private const RETRY_INTERVALS = [1, 3, 7]; // days

    public function handlePaymentFailure(Subscription $subscription): void
    {
        $subscription->update([
            'status' => 'past_due',
            'payment_failure_count' => $subscription->payment_failure_count + 1,
            'last_payment_attempt_at' => now(),
        ]);

        // Schedule retry
        $retryDay = self::RETRY_INTERVALS[
            min($subscription->payment_failure_count - 1, count(self::RETRY_INTERVALS) - 1)
        ];

        RetryPaymentJob::dispatch($subscription)
            ->delay(now()->addDays($retryDay));

        // Send notification
        PaymentFailedNotification::to($subscription->user)
            ->queue();

        // Check if grace period expired
        if ($subscription->payment_failure_count >= self::MAX_RETRY_ATTEMPTS) {
            $this->expireGracePeriod($subscription);
        }
    }

    public function handleRetrySuccess(Subscription $subscription): void
    {
        $subscription->update([
            'status' => 'active',
            'payment_failure_count' => 0,
            'last_payment_attempt_at' => now(),
        ]);

        GracePeriodRecoveredNotification::to($subscription->user)
            ->queue();
    }

    private function expireGracePeriod(Subscription $subscription): void
    {
        $subscription->update([
            'status' => 'canceled',
            'ended_at' => now(),
            'cancellation_reason' => 'Payment failure after grace period',
        ]);

        event(new SubscriptionExpired($subscription));
    }

    public function isInGracePeriod(Subscription $subscription): bool
    {
        return $subscription->status === 'past_due'
            && $subscription->payment_failure_count < self::MAX_RETRY_ATTEMPTS;
    }

    public function getDaysRemainingInGrace(Subscription $subscription): int
    {
        if (!$this->isInGracePeriod($subscription)) return 0;

        $lastAttempt = Carbon::parse($subscription->last_payment_attempt_at);
        $nextRetry = $lastAttempt->addDays(self::RETRY_INTERVALS[
            min($subscription->payment_failure_count, count(self::RETRY_INTERVALS) - 1)
        ]);

        return max(0, $nextRetry->diffInDays(now()));
    }
}
```

## Stripe Webhook Processor

```php
// app/Services/StripeWebhookProcessor.php
class StripeWebhookProcessor
{
    private const HANDLED_EVENTS = [
        'customer.subscription.created',
        'customer.subscription.updated',
        'customer.subscription.deleted',
        'customer.subscription.trial_will_end',
        'customer.subscription.paused',
        'customer.subscription.resumed',
        'invoice.created',
        'invoice.payment_succeeded',
        'invoice.payment_failed',
        'invoice.finalized',
        'payment_method.attached',
        'payment_method.detached',
    ];

    public function process(\Stripe\Event $event): void
    {
        $webhookEvent = $this->recordEvent($event);

        if ($webhookEvent->status === 'processed') {
            Log::info('Webhook already processed', ['event_id' => $event->id]);
            return;
        }

        try {
            DB::transaction(function () use ($event) {
                match ($event->type) {
                    'customer.subscription.created' => $this->subscriptionCreated($event),
                    'customer.subscription.updated' => $this->subscriptionUpdated($event),
                    'customer.subscription.deleted' => $this->subscriptionDeleted($event),
                    'customer.subscription.trial_will_end' => $this->trialWillEnd($event),
                    'customer.subscription.paused' => $this->subscriptionPaused($event),
                    'customer.subscription.resumed' => $this->subscriptionResumed($event),
                    'invoice.payment_succeeded' => $this->invoicePaid($event),
                    'invoice.payment_failed' => $this->invoiceFailed($event),
                    default => null,
                };
            });

            $webhookEvent->markProcessed();
        } catch (\Exception $e) {
            $webhookEvent->markFailed($e->getMessage());
            throw $e;
        }
    }

    private function subscriptionCreated(\Stripe\Event $event): void
    {
        $stripeSub = $event->data->object;

        $user = $this->findUserFromCustomer($stripeSub->customer);
        if (!$user) {
            Log::warning('Subscription created for unknown user', [
                'stripe_customer' => $stripeSub->customer,
            ]);
            return;
        }

        $plan = $this->resolvePlan($stripeSub);

        $subscription = Subscription::updateOrCreate(
            ['stripe_id' => $stripeSub->id],
            [
                'user_id' => $user->id,
                'plan_id' => $plan?->id,
                'status' => $stripeSub->status,
                'stripe_status' => $stripeSub->status,
                'stripe_price_id' => $stripeSub->items->data[0]->price->id,
                'trial_starts_at' => $this->timestampToCarbon($stripeSub->trial_start),
                'trial_ends_at' => $this->timestampToCarbon($stripeSub->trial_end),
                'current_period_start' => $this->timestampToCarbon($stripeSub->current_period_start),
                'current_period_end' => $this->timestampToCarbon($stripeSub->current_period_end),
            ]
        );

        // Record event
        $subscription->events()->create([
            'event_type' => 'created',
            'payload' => $stripeSub->toArray(),
        ]);

        event(new SubscriptionCreated($user, $plan, $subscription));
    }

    private function subscriptionUpdated(\Stripe\Event $event): void
    {
        $stripeSub = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeSub->id)->first();
        if (!$subscription) {
            Log::warning('Subscription updated but not found', [
                'stripe_id' => $stripeSub->id,
            ]);
            return;
        }

        $oldPlan = $subscription->plan;
        $newPlan = $this->resolvePlan($stripeSub);

        $subscription->update([
            'status' => $stripeSub->status,
            'stripe_status' => $stripeSub->status,
            'plan_id' => $newPlan?->id,
            'current_period_start' => $this->timestampToCarbon($stripeSub->current_period_start),
            'current_period_end' => $this->timestampToCarbon($stripeSub->current_period_end),
            'cancel_at_period_end' => $stripeSub->cancel_at_period_end,
            'ended_at' => $this->timestampToCarbon($stripeSub->ended_at),
        ]);

        $subscription->events()->create([
            'event_type' => 'updated',
            'payload' => array_merge($stripeSub->toArray(), [
                'old_plan_id' => $oldPlan?->id,
                'new_plan_id' => $newPlan?->id,
            ]),
        ]);

        // Detect plan change
        if ($oldPlan?->id !== $newPlan?->id) {
            event(new SubscriptionPlanChanged($subscription, $oldPlan, $newPlan));
        }

        event(new SubscriptionUpdated($subscription));
    }

    private function subscriptionDeleted(\Stripe\Event $event): void
    {
        $stripeSub = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeSub->id)->first();
        if (!$subscription) return;

        $subscription->update([
            'status' => 'canceled',
            'ended_at' => now(),
        ]);

        $subscription->events()->create([
            'event_type' => 'canceled',
            'payload' => $stripeSub->toArray(),
        ]);

        event(new SubscriptionCanceled($subscription));
    }

    private function trialWillEnd(\Stripe\Event $event): void
    {
        $stripeSub = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeSub->id)->first();
        if (!$subscription) return;

        $subscription->events()->create([
            'event_type' => 'trial_will_end',
            'payload' => $stripeSub->toArray(),
        ]);

        event(new SubscriptionTrialEnding($subscription));
    }

    private function invoicePaid(\Stripe\Event $event): void
    {
        $stripeInvoice = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeInvoice->subscription)->first();

        if ($subscription) {
            $subscription->update([
                'status' => 'active',
                'payment_failure_count' => 0,
            ]);

            $subscription->events()->create([
                'event_type' => 'payment_succeeded',
                'payload' => $stripeInvoice->toArray(),
            ]);

            event(new SubscriptionPaymentSucceeded($subscription, $stripeInvoice->amount_paid));
        }

        // Generate invoice
        $this->generateLocalInvoice($stripeInvoice, $subscription);
    }

    private function invoiceFailed(\Stripe\Event $event): void
    {
        $stripeInvoice = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeInvoice->subscription)->first();

        if ($subscription) {
            $subscription->update([
                'status' => 'past_due',
                'payment_failure_count' => $subscription->payment_failure_count + 1,
                'last_payment_attempt_at' => now(),
            ]);

            $subscription->events()->create([
                'event_type' => 'payment_failed',
                'payload' => $stripeInvoice->toArray(),
            ]);

            event(new SubscriptionPaymentFailed($subscription));
        }
    }

    private function resolvePlan(\Stripe\Subscription $stripeSub): ?Plan
    {
        $priceId = $stripeSub->items->data[0]->price->id;
        return Plan::where('stripe_price_id', $priceId)->first();
    }

    private function findUserFromCustomer(string $customerId): ?User
    {
        return User::where('stripe_id', $customerId)->first()
            ?? User::whereHas('subscriptions', fn($q) => $q->where('stripe_id', 'LIKE', "%{$customerId}%"))->first();
    }

    private function timestampToCarbon(?int $timestamp): ?Carbon
    {
        return $timestamp ? Carbon::createFromTimestamp($timestamp) : null;
    }
}
```

## Subscription Dashboard

```php
// app/Livewire/Billing/SubscriptionDashboard.php
class SubscriptionDashboard extends Component
{
    public ?Subscription $subscription = null;
    public bool $showCancelModal = false;
    public bool $showUpgradeModal = false;
    public string $cancellationReason = '';
    public bool $cancelAtPeriodEnd = true;

    public function mount(): void
    {
        $this->subscription = auth()->user()->subscription()
            ->with('plan')
            ->first();
    }

    public function getUpcomingInvoiceProperty(): ?array
    {
        if (!$this->subscription || !$this->subscription->isActive()) {
            return null;
        }

        return [
            'amount' => $this->subscription->plan->price,
            'date' => $this->subscription->current_period_end,
            'days_until' => now()->diffInDays($this->subscription->current_period_end),
        ];
    }

    public function cancelSubscription(): void
    {
        $this->validate([
            'cancellationReason' => 'required|string|max:500',
        ]);

        app(SubscriptionService::class)->cancel(
            user: auth()->user(),
            immediate: !$this->cancelAtPeriodEnd,
            reason: $this->cancellationReason,
        );

        $this->showCancelModal = false;
        $this->subscription->refresh();

        $this->dispatch('subscription.canceled');
    }

    public function resumeSubscription(): void
    {
        app(SubscriptionService::class)->resume(auth()->user());

        $this->subscription->refresh();
        $this->dispatch('subscription.resumed');
    }

    public function render(): View
    {
        return view('livewire.billing.subscription-dashboard', [
            'subscription' => $this->subscription,
            'upcoming_invoice' => $this->upcomingInvoice,
        ]);
    }
}
```

```php
class RetentionService
{
    public function getRetentionOffers(User $user): array
    {
        $subscription = $user->subscription;
        if (!$subscription) return [];

        $offers = [];

        // Discount offer
        if ($subscription->plan->price > 0) {
            $offers[] = new RetentionOffer(
                type: 'discount',
                title: 'Get 20% off for 3 months',
                description: 'We\'ll reduce your plan price by 20% for the next 3 billing cycles.',
                discount_percentage: 20,
                discount_months: 3,
                action_url: route('billing.apply-offer', ['type' => 'discount']),
            );
        }

        // Plan downgrade offer
        $lowerPlan = Plan::active()
            ->where('price', '<', $subscription->plan->price)
            ->where('price', '>', 0)
            ->orderByDesc('price')
            ->first();

        if ($lowerPlan) {
            $offers[] = new RetentionOffer(
                type: 'downgrade',
                title: 'Switch to ' . $lowerPlan->name,
                description: "Keep access to key features at a lower price: \${$lowerPlan->price}/mo",
                target_plan: $lowerPlan,
                action_url: route('billing.switch-plan', ['plan' => $lowerPlan->slug]),
            );
        }

        // Pause offer
        $offers[] = new RetentionOffer(
            type: 'pause',
            title: 'Pause your subscription',
            description: 'Take a break for up to 3 months without losing your data.',
            action_url: route('billing.pause'),
        );

        return $offers;
    }

    public function applyRetentionOffer(User $user, string $offerType): void
    {
        match ($offerType) {
            'discount' => $this->applyDiscount($user),
            'pause' => $this->pauseSubscription($user),
            default => null,
        };
    }

    private function applyDiscount(User $user): void
    {
        $subscription = $user->subscription;
        $discountedPrice = (int) round($subscription->plan->price * 0.8);

        // Create a discount coupon for 3 months
        $coupon = Coupon::create([
            'code' => 'RETENTION-' . strtoupper(Str::random(8)),
            'type' => 'fixed_amount',
            'value' => $subscription->plan->price - $discountedPrice,
            'max_uses' => 1,
            'max_uses_per_user' => 1,
            'expires_at' => now()->addMonths(3),
            'metadata' => ['type' => 'retention', 'user_id' => $user->id],
        ]);

        // Apply to subscription
        \Stripe\Subscription::update($subscription->stripe_id, [
            'coupon' => $coupon->code,
        ]);
    }
}
```

```php
class SubscriptionInvoiceService
{
    public function generateForPayment(Subscription $subscription, \Stripe\Invoice $stripeInvoice): Invoice
    {
        $user = $subscription->user;
        $plan = $subscription->plan;

        $invoiceNumber = $this->generateNumber();

        $invoice = Invoice::create([
            'user_id' => $user->id,
            'subscription_id' => $subscription->id,
            'number' => $invoiceNumber,
            'status' => 'paid',
            'subtotal' => $stripeInvoice->subtotal,
            'tax_amount' => $stripeInvoice->tax,
            'total' => $stripeInvoice->amount_paid,
            'currency' => $stripeInvoice->currency,
            'issued_at' => Carbon::createFromTimestamp($stripeInvoice->created),
            'paid_at' => now(),
            'billing_address' => $this->getBillingAddress($user),
            'line_items' => $this->buildLineItems($subscription, $stripeInvoice),
        ]);

        // Generate PDF
        $pdfUrl = app(InvoicePdfService::class)->generate($invoice);
        $invoice->update(['pdf_path' => $pdfUrl]);

        // Send email
        InvoiceMail::to($user)->queue($invoice);

        return $invoice;
    }

    private function buildLineItems(Subscription $subscription, \Stripe\Invoice $stripeInvoice): array
    {
        $items = [];

        foreach ($stripeInvoice->lines->data as $line) {
            $items[] = [
                'description' => $line->description ?? "{$subscription->plan->name} subscription",
                'amount' => $line->amount,
                'quantity' => $line->quantity,
                'period_start' => Carbon::createFromTimestamp($line->period->start)->toDateString(),
                'period_end' => Carbon::createFromTimestamp($line->period->end)->toDateString(),
            ];
        }

        return $items;
    }

    private function generateNumber(): string
    {
        $year = now()->year;
        $sequence = Invoice::whereYear('created_at', $year)->count() + 1;
        return "INV-{$year}-" . str_pad($sequence, 6, '0', STR_PAD_LEFT);
    }
}
```

```php
// app/Services/SubscriptionMetrics.php
class SubscriptionMetrics
{
    public function getDashboard(): array
    {
        return [
            'mrr' => $this->getMrr(),
            'arr' => $this->getMrr() * 12,
            'active_subscribers' => $this->getActiveSubscriberCount(),
            'churn_rate' => $this->getMonthlyChurnRate(),
            'trial_conversion' => $this->getTrialConversionRate(),
            'average_lifetime' => $this->getAverageLifetime(),
            'ltv' => $this->getLifetimeValue(),
            'revenue_by_plan' => $this->getRevenueByPlan(),
            'subscriber_growth' => $this->getSubscriberGrowth(),
            'mrr_growth' => $this->getMrrGrowth(),
        ];
    }

    public function getMrr(): float
    {
        return Subscription::where('status', 'active')
            ->join('plans', 'plans.id', '=', 'subscriptions.plan_id')
            ->sum('plans.price') / 100;
    }

    public function getMonthlyChurnRate(): float
    {
        $startOfMonth = now()->startOfMonth();
        $endOfMonth = now()->endOfMonth();

        $starting = Subscription::where('created_at', '<', $startOfMonth)
            ->where(function ($q) use ($endOfMonth) {
                $q->whereNull('canceled_at')
                  ->orWhere('canceled_at', '>', $endOfMonth);
            })
            ->count();

        $canceled = Subscription::where('canceled_at', '>=', $startOfMonth)
            ->where('canceled_at', '<=', $endOfMonth)
            ->count();

        return $starting > 0 ? round(($canceled / $starting) * 100, 2) : 0;
    }

    public function getTrialConversionRate(): float
    {
        $trials = Subscription::where('status', 'trialing')
            ->where('trial_ends_at', '<', now())
            ->count();

        $converted = Subscription::whereIn('status', ['active', 'trialing'])
            ->where('trial_ends_at', '<', now())
            ->where('created_at', '<', now()->subDays(14))
            ->count();

        return $trials > 0 ? round(($converted / $trials) * 100, 2) : 0;
    }

    public function getLifetimeValue(): float
    {
        $mrr = $this->getMrr();
        $subscribers = $this->getActiveSubscriberCount();
        $avgMonthlyRevenue = $subscribers > 0 ? $mrr / $subscribers : 0;
        $avgLifetimeMonths = 12 / max($this->getMonthlyChurnRate() / 100, 0.01);

        return round($avgMonthlyRevenue * $avgLifetimeMonths, 2);
    }

    public function getSubscriberGrowth(int $months = 12): Collection
    {
        return Subscription::select(
            DB::raw("DATE_FORMAT(created_at, '%Y-%m') as month"),
            DB::raw('COUNT(*) as new_subscribers'),
            DB::raw('SUM(CASE WHEN status = "canceled" THEN 1 ELSE 0 END) as churned')
        )
        ->where('created_at', '>=', now()->subMonths($months))
        ->groupBy('month')
        ->orderBy('month')
        ->get();
    }
}
```

```php
// app/Services/CohortAnalysis.php
class CohortAnalysis
{
    public function getRetentionCohorts(int $months = 12): Collection
    {
        $cohorts = collect();

        for ($i = 0; $i < $months; $i++) {
            $cohortDate = now()->subMonths($i)->startOfMonth();

            $subscribers = Subscription::where('created_at', '>=', $cohortDate)
                ->where('created_at', '<', $cohortDate->copy()->addMonth())
                ->count();

            $retained = [];
            for ($m = 0; $m <= $i; $m++) {
                $checkDate = $cohortDate->copy()->addMonths($m);
                $active = Subscription::where('created_at', '>=', $cohortDate)
                    ->where('created_at', '<', $cohortDate->copy()->addMonth())
                    ->where(function ($q) use ($checkDate) {
                        $q->whereNull('canceled_at')
                          ->orWhere('canceled_at', '>', $checkDate);
                    })
                    ->count();

                $retained[] = $subscribers > 0 ? round(($active / $subscribers) * 100, 1) : 0;
            }

            $cohorts->push([
                'cohort' => $cohortDate->format('Y-m'),
                'subscribers' => $subscribers,
                'retention' => $retained,
            ]);
        }

        return $cohorts->reverse();
    }
}
```

```php
// app/Console/Commands/SubscriptionCommands.php
class SubscriptionCommands extends Command
{
    protected $signature = 'subscriptions:{action}';

    public function handle(): int
    {
        return match ($this->argument('action')) {
            'sync' => $this->sync(),
            'reconcile' => $this->reconcile(),
            'expire-trials' => $this->expireTrials(),
            'renewals' => $this->processRenewals(),
            'report' => $this->report(),
            default => $this->error("Unknown action"),
        };
    }

    private function sync(): int
    {
        $this->info('Syncing subscriptions with Stripe...');

        $subscriptions = Subscription::whereNotNull('stripe_id')->get();

        foreach ($subscriptions as $local) {
            try {
                $stripe = \Stripe\Subscription::retrieve($local->stripe_id);

                if ($stripe->status !== $local->status) {
                    $local->update(['status' => $stripe->status]);
                    $this->line("Updated subscription {$local->id}: {$local->status} -> {$stripe->status}");
                }
            } catch (\Exception $e) {
                $this->error("Failed to sync subscription {$local->id}: {$e->getMessage()}");
            }
        }

        $this->info('Sync complete.');
        return 0;
    }

    private function expireTrials(): int
    {
        $this->info('Expiring trials...');

        $expired = Subscription::where('status', 'trialing')
            ->where('trial_ends_at', '<', now())
            ->update(['status' => 'canceled']);

        $this->info("Expired {$expired} trials.");
        return 0;
    }

    private function processRenewals(): int
    {
        $this->info('Processing renewals...');

        $due = Subscription::where('status', 'active')
            ->where('current_period_end', '<=', now()->addDays(3))
            ->count();

        $this->info("{$due} subscriptions due for renewal in next 3 days.");
        return 0;
    }

    private function report(): int
    {
        $metrics = app(SubscriptionMetrics::class)->getDashboard();

        $this->table(
            ['Metric', 'Value'],
            [
                ['MRR', '$' . number_format($metrics['mrr'], 2)],
                ['ARR', '$' . number_format($metrics['arr'], 2)],
                ['Active Subscribers', number_format($metrics['active_subscribers'])],
                ['Monthly Churn', $metrics['churn_rate'] . '%'],
                ['Trial Conversion', $metrics['trial_conversion'] . '%'],
                ['LTV', '$' . number_format($metrics['ltv'], 2)],
            ]
        );

        return 0;
    }
}
```

```php
// app/Services/SubscriptionUpgradePreview.php
class SubscriptionUpgradePreview
{
    public function preview(Subscription $subscription, Plan $newPlan): UpgradePreviewData
    {
        $currentPlan = $subscription->plan;

        $periodStart = Carbon::parse($subscription->current_period_start);
        $periodEnd = Carbon::parse($subscription->current_period_end);
        $now = now();

        $totalDays = $periodStart->diffInDays($periodEnd);
        $daysUsed = $periodStart->diffInDays($now);
        $daysRemaining = $now->diffInDays($periodEnd);

        // Credit for unused time on current plan
        $dailyCurrentRate = $currentPlan->price / $totalDays;
        $credit = (int) round($dailyCurrentRate * $daysRemaining);

        // Charge for remaining time on new plan
        $dailyNewRate = $newPlan->price / $totalDays;
        $charge = (int) round($dailyNewRate * $daysRemaining);

        $proration = $charge - $credit;

        return new UpgradePreviewData(
            current_plan: $currentPlan,
            new_plan: $newPlan,
            credit: $credit,
            charge: $charge,
            proration: $proration,
            effective_date: $periodEnd,
            immediate_access: true,
            days_used: $daysUsed,
            days_remaining: $daysRemaining,
            total_days: $totalDays,
        );
    }
}

    public function __construct(
        public Plan $current_plan,
```

```php
// app/Jobs/WebhookRetryJob.php
class WebhookRetryJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $tries = 5;
    public int $timeout = 30;

    public function backoff(): array
    {
        return [30, 120, 300, 600, 1800]; // 30s, 2m, 5m, 10m, 30m
    }

    public function handle(): void
    {
        $webhookEvent = $this->webhookEvent;

        try {
            app(StripeWebhookProcessor::class)->process($webhookEvent->toStripeEvent());

            $webhookEvent->update([
                'status' => 'processed',
                'processed_at' => now(),
            ]);
        } catch (\Exception $e) {
            if ($this->attempts() >= $this->tries) {
                $webhookEvent->update(['status' => 'failed']);

                Log::critical('Webhook permanently failed', [
                    'event_id' => $webhookEvent->stripe_event_id,
                    'type' => $webhookEvent->type,
                    'error' => $e->getMessage(),
                ]);

                return;
            }

            throw $e;
        }
    }

    public function failed(\Throwable $exception): void
    {
        Log::error('Webhook retry job failed', [
            'event_id' => $this->webhookEvent->stripe_event_id,
            'error' => $exception->getMessage(),
        ]);
    }
}
```

## Cancellation Service

```php
// app/Services/SubscriptionCancellationService.php
class SubscriptionCancellationService
{
    public function __construct(
        private StripeClient $stripe,
        private RetentionService $retentionService,
    ) {}

    public function initiateCancellation(User $user, string $reason, bool $immediate = false): CancellationResult
    {
        $subscription = $user->subscription;

        if (!$subscription || !$subscription->isActive()) {
            throw new \Exception('No active subscription to cancel');
        }

        // Check for retention offers
        $offers = $this->retentionService->getRetentionOffers($user);

        if (!empty($offers) && !$immediate) {
            return new CancellationResult(
                status: 'offers_available',
                offers: $offers,
                subscription: $subscription,
            );
        }

        // Process cancellation
        $this->processCancellation($subscription, $reason, $immediate);

        return new CancellationResult(
            status: 'canceled',
            subscription: $subscription->fresh(),
        );
    }

    private function processCancellation(Subscription $subscription, string $reason, bool $immediate): void
    {
        if ($immediate) {
            $this->stripe->subscriptions->cancel($subscription->stripe_id);
            $subscription->update([
                'status' => 'canceled',
                'canceled_at' => now(),
                'cancellation_reason' => $reason,
            ]);
        } else {
            $this->stripe->subscriptions->update($subscription->stripe_id, [
                'cancel_at_period_end' => true,
            ]);
            $subscription->update([
                'cancel_at_period_end' => true,
                'cancellation_reason' => $reason,
            ]);
        }

        event(new SubscriptionCancellationInitiated($subscription, $reason, $immediate));
    }
}

class CancellationResult
{
    public function __construct(
        public readonly string $status,
        public readonly ?array $offers = null,
        public readonly ?Subscription $subscription = null,
    ) {}
}
```

```php
// app/Services/SubscriptionPauseService.php
class SubscriptionPauseService
{
    public function pause(User $user, int $months = 1): void
    {
        $subscription = $user->subscription;

        if (!$subscription || !$subscription->isActive()) {
            throw new \Exception('No active subscription to pause');
        }

        // Pause in Stripe
        $this->stripe->subscriptions->update($subscription->stripe_id, [
            'pause_collection' => [
                'behavior' => 'void',
            ],
        ]);

        // Update local record
        $subscription->update([
            'status' => 'paused',
            'metadata' => array_merge($subscription->metadata ?? [], [
                'paused_at' => now()->toIso8601String(),
                'resume_at' => now()->addMonths($months)->toIso8601String(),
            ]),
        ]);

        // Revoke access temporarily
        $user->revokePremiumAccess();

        event(new SubscriptionPaused($subscription, $months));
    }

    public function resume(User $user): void
    {
        $subscription = $user->subscription;

        if (!$subscription || $subscription->status !== 'paused') {
            throw new \Exception('No paused subscription to resume');
        }

        // Resume in Stripe
        $this->stripe->subscriptions->update($subscription->stripe_id, [
            'pause_collection' => '',
        ]);

        // Update local record
        $subscription->update([
            'status' => 'active',
        ]);

        // Grant access back
        $user->grantPremiumAccess();

        event(new SubscriptionResumed($subscription));
    }
}
```

```php
// app/Services/DunningService.php
class DunningService
{
    private const RETRY_SCHEDULE = [
        1 => 1,  // Day 1: First retry
        2 => 3,  // Day 3: Second retry
        3 => 7,  // Day 7: Third retry
        4 => 14, // Day 14: Final retry
    ];

    public function handleFailedPayment(Subscription $subscription): void
    {
        $attempt = $subscription->payment_failure_count + 1;

        $subscription->update([
            'status' => 'past_due',
            'payment_failure_count' => $attempt,
            'last_payment_attempt_at' => now(),
        ]);

        // Send dunning email
        DunningEmail::to($subscription->user)->queue($subscription, $attempt);

        // Schedule next retry
        if (isset(self::RETRY_SCHEDULE[$attempt])) {
            RetryPaymentJob::dispatch($subscription)
                ->delay(now()->addDays(self::RETRY_SCHEDULE[$attempt]));
        } else {
            // Max retries reached - cancel subscription
            $this->cancelForNonPayment($subscription);
        }
    }

    private function cancelForNonPayment(Subscription $subscription): void
    {
        $subscription->update([
            'status' => 'canceled',
            'ended_at' => now(),
            'cancellation_reason' => 'Non-payment after maximum retry attempts',
        ]);

        event(new SubscriptionCanceledForNonPayment($subscription));
    }
}
```
