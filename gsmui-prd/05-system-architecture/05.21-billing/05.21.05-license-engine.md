# Purpose

The License Engine manages digital license generation, validation, and revocation for one-time purchases. It provides a secure, scalable system for controlling access to purchased digital products including templates, themes, and component packs.

---

# License Generation

## Algorithm

- UUID-based license keys
- Prefix for product type
- Checksum for validation
- Example: `GSMUI-TEMPLATE-ABCD-1234`

## Generation Steps

1. Product purchase completed
2. License key generated
3. Key stored encrypted in database
4. Key sent to user (email + download page)
5. Key logged for audit

# Detailed License Architecture

## License Key Format

```text
GSMUI - [PREFIX] - [RANDOM] - [CHECKSUM]
 │         │          │          │
 │         │          │          └── 4-char checksum for validation
 │         │          └── 8 random alphanumeric characters
 │         └── 2-4 char product type prefix
 └── Product family identifier

Examples:
GSMUI-TPLT-A1B2C3D4-X7K9    (Template)
GSMUI-THME-M3N4O5P6-Q2W8    (Theme)
GSMUI-CMPN-R7S8T9U0-V4Y6    (Component Pack)
GSMUI-PROJ-H1J2K3L4-Z5A1    (Project License)
```

## Key Generation Service

```php
// app/Services/LicenseKeyGenerator.php
class LicenseKeyGenerator
{
    private const PREFIXES = [
        'template' => 'TPLT',
        'theme' => 'THME',
        'component' => 'CMPN',
        'project' => 'PROJ',
        'bundle' => 'BNDL',
    ];

    private const CHARS = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // Removed ambiguous: I, O, 0, 1

    public function generate(string $productType): string
    {
        $prefix = self::PREFIXES[$productType] ?? 'GENR';

        // Generate random segment
        $random = '';
        for ($i = 0; $i < 8; $i++) {
            $random .= self::CHARS[random_int(0, strlen(self::CHARS) - 1)];
        }

        // Calculate checksum
        $checksum = $this->calculateChecksum($random);

        return "GSMUI-{$prefix}-{$random}-{$checksum}";
    }

    private function calculateChecksum(string $data): string
    {
        $hash = hash('xxh3', $data);
        // Take first 4 chars, uppercase
        return strtoupper(substr($hash, 0, 4));
    }

    public function validate(string $key): bool
    {
        if (!preg_match('/^GSMUI-[A-Z]{2,4}-[A-Z0-9]{8}-[A-Z0-9]{4}$/', $key)) {
            return false;
        }

        $parts = explode('-', $key);
        $random = $parts[2];
        $checksum = $parts[3];

        return $this->calculateChecksum($random) === $checksum;
    }
}
```

```php
class License extends Model
{
    use HasFactory;

    protected $fillable = [
        'key', 'order_id', 'product_id', 'user_id',
        'type', 'status', 'domain', 'ip_address',
        'activated_at', 'last_validated_at',
        'expires_at', 'metadata',
    ];

    protected $casts = [
        'activated_at' => 'datetime',
        'last_validated_at' => 'datetime',
        'expires_at' => 'datetime',
        'metadata' => 'array',
    ];

    protected $hidden = ['key'];

    public function order(): BelongsTo
    {
        return $this->belongsTo(Order::class);
    }

    public function product(): BelongsTo
    {
        return $this->belongsTo(Product::class);
    }

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class);
    }

    public function activations(): HasMany
    {
        return $this->hasMany(LicenseActivation::class);
    }

    public function scopeActive(Builder $query): Builder
    {
        return $query->where('status', 'active');
    }

    public function scopeValid(Builder $query): Builder
    {
        return $query->where('status', 'active')
            ->where(function ($q) {
                $q->whereNull('expires_at')
                  ->orWhere('expires_at', '>', now());
            });
    }

    public function isValid(): bool
    {
        if ($this->status !== 'active') return false;
        if ($this->expires_at && $this->expires_at->isPast()) return false;
        return true;
    }

    public function isExpired(): bool
    {
        return $this->expires_at && $this->expires_at->isPast();
    }

    public function getDaysUntilExpiry(): ?int
    {
        return $this->expires_at?->diffInDays(now());
    }
}
```

## License Service

```php
// app/Services/LicenseService.php
class LicenseService
{
    public function __construct(
        private LicenseKeyGenerator $keyGenerator,
        private AuditLogger $auditLogger,
    ) {}

    public function generate(Order $order, Product $product, User $user): License
    {
        $licenseType = $this->determineLicenseType($product, $order);
        $key = $this->keyGenerator->generate($product->type);

        $license = License::create([
            'key' => $key,
            'order_id' => $order->id,
            'product_id' => $product->id,
            'user_id' => $user->id,
            'type' => $licenseType,
            'status' => 'active',
            'expires_at' => $this->calculateExpiry($licenseType),
            'metadata' => [
                'product_name' => $product->name,
                'product_version' => $product->version,
                'order_total' => $order->total,
            ],
        ]);

        $this->auditLogger->log('license.generated', $license, [
            'product_id' => $product->id,
            'order_id' => $order->id,
            'type' => $licenseType,
        ]);

        return $license;
    }

    public function validate(string $key, ?string $domain = null, ?string $ip = null): LicenseValidationResult
    {
        // Validate key format
        if (!$this->keyGenerator->validate($key)) {
            return LicenseValidationResult::invalid('Invalid license key format');
        }

        $license = License::where('key', $key)->first();

        if (!$license) {
            return LicenseValidationResult::invalid('License not found');
        }

        if (!$license->isValid()) {
            return LicenseValidationResult::invalid('License is ' . $license->status);
        }

        // Domain binding check
        if ($domain && $license->domain && $license->domain !== $domain) {
            return LicenseValidationResult::invalid('License is bound to a different domain');
        }

        // Activation limit check
        if (!$this->canActivate($license, $domain)) {
            return LicenseValidationResult::atLimit('Activation limit reached');
        }

        // Record validation
        $license->update(['last_validated_at' => now()]);

        $this->auditLogger->log('license.validated', $license, [
            'domain' => $domain,
            'ip' => $ip,
        ]);

        return LicenseValidationResult::valid(
            license: $license,
            product: $license->product,
            expires: $license->expires_at,
        );
    }

    public function activate(License $license, ?string $domain = null, ?string $ip = null): LicenseActivation
    {
        if (!$license->isValid()) {
            throw new InvalidLicenseException('License is not valid');
        }

        if (!$this->canActivate($license, $domain)) {
            throw new LicenseLimitReachedException('Activation limit reached');
        }

        $activation = LicenseActivation::create([
            'license_id' => $license->id,
            'domain' => $domain,
            'ip_address' => $ip,
            'user_agent' => request()->userAgent(),
            'activated_at' => now(),
        ]);

        if (!$license->activated_at) {
            $license->update(['activated_at' => now()]);
        }

        if ($domain && !$license->domain) {
            $license->update(['domain' => $domain]);
        }

        $this->auditLogger->log('license.activated', $license, [
            'domain' => $domain,
            'ip' => $ip,
        ]);

        return $activation;
    }

    public function deactivate(License $license, string $domain): void
    {
        $license->activations()
            ->where('domain', $domain)
            ->update(['deactivated_at' => now()]);

        $this->auditLogger->log('license.deactivated', $license, [
            'domain' => $domain,
        ]);
    }

    public function revoke(License $license, ?string $reason = null): void
    {
        $license->update([
            'status' => 'revoked',
            'metadata' => array_merge($license->metadata ?? [], [
                'revoked_at' => now()->toIso8601String(),
                'revoked_reason' => $reason,
            ]),
        ]);

        // Deactivate all activations
        $license->activations()
            ->whereNull('deactivated_at')
            ->update(['deactivated_at' => now()]);

        $this->auditLogger->log('license.revoked', $license, [
            'reason' => $reason,
        ]);

        event(new LicenseRevoked($license, $reason));
    }

    public function revokeAllForOrder(Order $order): void
    {
        $licenses = License::where('order_id', $order->id)
            ->where('status', 'active')
            ->get();

        foreach ($licenses as $license) {
            $this->revoke($license, 'Order refunded');
        }
    }

    public function transfer(License $license, User $newUser): void
    {
        $oldUser = $license->user;

        $license->update([
            'user_id' => $newUser->id,
            'domain' => null,
            'metadata' => array_merge($license->metadata ?? [], [
                'transferred_from' => $oldUser->id,
                'transferred_at' => now()->toIso8601String(),
            ]),
        ]);

        // Reset activations
        $license->activations()->update(['deactivated_at' => now()]);

        $this->auditLogger->log('license.transferred', $license, [
            'from_user' => $oldUser->id,
            'to_user' => $newUser->id,
        ]);

        event(new LicenseTransferred($license, $oldUser, $newUser));
    }

    private function canActivate(License $license, ?string $domain): bool
    {
        $activeActivations = $license->activations()
            ->whereNull('deactivated_at')
            ->count();

        $maxActivations = match ($license->type) {
            'single' => 1,
            'multiple' => 5,
            'team' => 10,
            default => 1,
        };

        // If domain already activated, allow re-validation
        if ($domain && $license->activations()->where('domain', $domain)->whereNull('deactivated_at')->exists()) {
            return true;
        }

        return $activeActivations < $maxActivations;
    }

    private function determineLicenseType(Product $product, Order $order): string
    {
        // Check order metadata for license type override
        if (isset($order->metadata['license_type'])) {
            return $order->metadata['license_type'];
        }

        // Default based on product type
        return match ($product->type) {
            'template' => 'single',
            'theme' => 'single',
            'component' => 'single',
            'bundle' => 'team',
            default => 'single',
        };
    }

    private function calculateExpiry(string $type): ?Carbon
    {
        return match ($type) {
            'subscription' => now()->addYear(),
            default => null, // Perpetual license
        };
    }
}
```

## External License Verification

For third-party integrations (e.g., WordPress plugins, Laravel packages):

```php
// app/Http/Controllers/LicenseWebhookController.php
class LicenseWebhookController extends Controller
{
    public function verify(Request $request): JsonResponse
    {
        // Validate API key
        $apiKey = $request->header('X-License-API-Key');
        $partner = Partner::where('api_key', $apiKey)->first();

        if (!$partner) {
            return response()->json(['error' => 'Invalid API key'], 401);
        }

        $request->validate([
            'key' => 'required|string',
            'domain' => 'nullable|string',
        ]);

        $result = app(LicenseService::class)->validate(
            key: $request->key,
            domain: $request->domain,
        );

        return response()->json([
            'valid' => $result->valid,
            'product' => $result->product?->name,
            'expires' => $result->expires?->toIso8601String(),
            'error' => $result->error,
        ], $result->valid ? 200 : 422);
    }

    public function activate(Request $request): JsonResponse
    {
        $apiKey = $request->header('X-License-API-Key');
        $partner = Partner::where('api_key', $apiKey)->first();

        if (!$partner) {
            return response()->json(['error' => 'Invalid API key'], 401);
        }

        $request->validate([
            'key' => 'required|string',
            'domain' => 'required|string',
        ]);

        $license = License::where('key', $request->key)->first();

        if (!$license || !$license->isValid()) {
            return response()->json(['error' => 'Invalid license'], 422);
        }

        try {
            $activation = app(LicenseService::class)->activate(
                license: $license,
                domain: $request->domain,
                ip: $request->ip(),
            );

            return response()->json([
                'success' => true,
                'activation_id' => $activation->id,
            ]);
        } catch (LicenseLimitReachedException $e) {
            return response()->json(['error' => $e->getMessage()], 422);
        }
    }
}
```

## Anti-Piracy Measures

```php
// app/Services/AntiPiracyService.php
class AntiPiracyService
{
    public function detectSuspiciousActivity(License $license): array
    {
        $activations = $license->activations()
            ->where('activated_at', '>=', now()->subDays(7))
            ->get();

        $flags = [];

        // Multiple IPs in short time
        $uniqueIps = $activations->pluck('ip_address')->unique();
        if ($uniqueIps->count() > 3) {
            $flags[] = 'multiple_ips';
        }

        // Multiple domains in short time
        $uniqueDomains = $activations->pluck('domain')->unique();
        if ($uniqueDomains->count() > 5) {
            $flags[] = 'multiple_domains';
        }

        // Rapid activation/deactivation cycles
        $deactivations = $license->activations()
            ->where('deactivated_at', '>=', now()->subDays(7))
            ->count();

        if ($deactivations > 10) {
            $flags[] = 'rapid_cycling';
        }

        // Geographic impossibility
        if ($this->detectImpossibleTravel($activations)) {
            $flags[] = 'impossible_travel';
        }

        return $flags;
    }

    public function handleSuspiciousActivity(License $license, array $flags): void
    {
        // Log security event
        SecurityEvent::create([
            'type' => 'suspicious_license_activity',
            'license_id' => $license->id,
            'flags' => $flags,
            'severity' => count($flags) > 2 ? 'high' : 'medium',
        ]);

        // Auto-suspend if high severity
        if (count($flags) > 2) {
            $license->update(['status' => 'suspended']);

            LicenseSuspendedNotification::to($license->user)
                ->queue($license, $flags);
        }

        // Notify admin
        SuspiciousLicenseActivityNotification::toAdmins()
            ->queue($license, $flags);
    }

    private function detectImpossibleTravel(Collection $activations): bool
    {
        $sorted = $activations->sortBy('activated_at');

        for ($i = 1; $i < $sorted->count(); $i++) {
            $prev = $sorted[$i - 1];
            $curr = $sorted[$i];

            $timeDiff = $prev->activated_at->diffInMinutes($curr->activated_at);
            $distance = $this->estimateDistance(
                $prev->ip_address,
                $curr->ip_address
            );

            // Assume max 1000 km/h travel speed
            if ($timeDiff > 0 && $distance / ($timeDiff / 60) > 1000) {
                return true;
            }
        }

        return false;
    }
}
```

```php
// app/Services/LicenseExportService.php
class LicenseExportService
{
    public function export(array $filters, string $format = 'csv'): StreamedResponse
    {
        $licenses = License::query()
            ->applyFilters($filters)
            ->with(['user', 'product', 'order'])
            ->cursor();

        return response()->streamDownload(function () use ($licenses, $format) {
            $handle = fopen('php://output', 'w');

            // Header
            fputcsv($handle, [
                'ID', 'Key', 'User Email', 'Product', 'Type',
                'Status', 'Domain', 'Activated At', 'Expires At', 'Created At',
            ]);

            foreach ($licenses as $license) {
                fputcsv($handle, [
                    $license->id,
                    $license->key,
                    $license->user->email,
                    $license->product->name,
                    $license->type,
                    $license->status,
                    $license->domain,
                    $license->activated_at?->toIso8601String(),
                    $license->expires_at?->toIso8601String(),
                    $license->created_at->toIso8601String(),
                ]);
            }

            fclose($handle);
        }, "licenses-" . now()->format('Y-m-d') . ".{$format}");
    }

    public function import(string $filePath, int $userId): ImportResult
    {
        $rows = array_map('str_getcsv', file($filePath));
        $header = array_shift($rows);

        $imported = 0;
        $errors = [];

        foreach ($rows as $index => $row) {
            try {
                $data = array_combine($header, $row);

                License::create([
                    'key' => $data['Key'],
                    'user_id' => $userId,
                    'product_id' => $this->resolveProduct($data['Product']),
                    'type' => $data['Type'] ?? 'single',
                    'status' => 'active',
                ]);

                $imported++;
            } catch (\Exception $e) {
                $errors[] = ['row' => $index + 2, 'error' => $e->getMessage()];
            }
        }

        return new ImportResult(
            imported: $imported,
            errors: $errors,
        );
    }
}
```

```php
        $activations = $license->activations()->get();
        $maxActivations = $license->product->max_activations ?? 1;

        $status = [
            'license_id' => $license->id,
            'key' => $license->key,
            'status' => $license->status,
            'activations_count' => $activations->count(),
            'max_activations' => $maxActivations,
            'is_overused' => $activations->count() > $maxActivations,
            'last_validated' => $license->last_validated_at,
            'expires_at' => $license->expires_at,
            'is_expired' => $license->expires_at && $license->expires_at->isPast(),
        ];

        // Check for suspicious activity
        $uniqueIPs = $activations->pluck('ip_address')->unique()->count();
        $uniqueHardware = $activations->pluck('hardware_id')->unique()->count();

        $status['unique_ips'] = $uniqueIPs;
        $status['unique_hardware_profiles'] = $uniqueHardware;
        $status['potential_piracy'] = $uniqueHardware > $maxActivations * 2;

        // Cache health status for quick lookups
        Cache::put("license_health:{$license->id}", $status, now()->addMinutes(5));

        if ($status['potential_piracy']) {
            Log::warning("Potential piracy detected for license {$license->key}", $status);
            $this->flagForReview($license);
        }

        return $status;
    }

    public function getSystemHealth(): array
    {
        $totalLicenses = License::count();
        $activeLicenses = License::where('status', 'active')->count();
        $expiredLicenses = License::where('status', 'expired')->count();
        $revokedLicenses = License::where('status', 'revoked')->count();

        $totalActivations = Activation::count();
        $recentActivations = Activation::where('created_at', '>=', now()->subHours(24))->count();
        $failedValidations = Cache::get('license_validation_failures_24h', 0);

        return [
            'licenses' => [
                'total' => $totalLicenses,
                'active' => $activeLicenses,
                'expired' => $expiredLicenses,
                'revoked' => $revokedLicenses,
            ],
            'activations' => [
                'total' => $totalActivations,
                'last_24h' => $recentActivations,
            ],
            'validations' => [
                'failures_24h' => $failedValidations,
            ],
            'health_score' => $this->calculateHealthScore($totalLicenses, $activeLicenses, $failedValidations),
        ];
    }

    private function calculateHealthScore(int $total, int $active, int $failures): float
    {
        if ($total === 0) return 100.0;

        $activeRatio = $active / $total;
        $failurePenalty = min(1.0, $failures / 100);

        return round(($activeRatio * 100) - ($failurePenalty * 20), 1);
    }

    private function flagForReview(License $license): void
    {
        $license->update([
            'metadata' => array_merge($license->metadata ?? [], [
                'flagged_for_review' => true,
                'flagged_at' => now()->toISOString(),
                'reason' => 'potential_piracy',
            ]),
        ]);
    }
}
```

## License Export/Import for Migration

```php
<?php
// app/Services/LicenseMigrationService.php
namespace App\Services;

use App\Models\License;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Facades\Crypt;

class LicenseMigrationService
{
    public function exportLicenses(string $format = 'json'): string
    {
        $licenses = License::with(['product', 'activations', 'user'])->get();

        $data = $licenses->map(function ($license) {
            return [
                'key' => $license->key,
                'product_slug' => $license->product->slug,
                'user_email' => $license->user->email,
                'status' => $license->status,
                'expires_at' => $license->expires_at?->toISOString(),
                'activations' => $license->activations->map(function ($activation) {
                    return [
                        'hardware_id' => $activation->hardware_id,
                        'ip_address' => $activation->ip_address,
                        'activated_at' => $activation->created_at->toISOString(),
                    ];
                }),
                'metadata' => $license->metadata,
            ];
        });

        $encrypted = Crypt::encryptString($data->toJson());
        $filename = 'licenses_export_' . now()->format('Y-m-d_His') . '.enc';

        Storage::disk('exports')->put($filename, $encrypted);

        return $filename;
    }

    public function importLicenses(string $encryptedFilename): array
    {
        $encrypted = Storage::disk('exports')->get($encryptedFilename);
        $decrypted = Crypt::decryptString($encrypted);
        $data = json_decode($decrypted, true);

        $imported = 0;
        $skipped = 0;
        $errors = [];

        foreach ($data as $record) {
            try {
                $existing = License::where('key', $record['key'])->first();
                if ($existing) {
                    $skipped++;
                    continue;
                }

                $user = \App\Models\User::where('email', $record['user_email'])->first();
                $product = \App\Models\Product::where('slug', $record['product_slug'])->first();

                if (! $user || ! $product) {
                    $errors[] = "Missing user or product for key {$record['key']}";
                    continue;
                }

                License::create([
                    'key' => $record['key'],
                    'user_id' => $user->id,
                    'product_id' => $product->id,
                    'status' => $record['status'],
                    'expires_at' => $record['expires_at'] ? \Carbon\Carbon::parse($record['expires_at']) : null,
                    'metadata' => $record['metadata'],
                ]);

                $imported++;
            } catch (\Exception $e) {
                $errors[] = "Error importing key {$record['key']}: {$e->getMessage()}";
            }
        }

        return [
            'imported' => $imported,
            'skipped' => $skipped,
            'errors' => $errors,
        ];
    }
}
```
