```php
class Plan extends Model
{
    use HasFactory;

    protected $fillable = [
        'name', 'slug', 'description', 'price', 'currency',
        'interval', 'trial_days', 'is_active', 'sort_order',
        'features', 'limits', 'stripe_price_id', 'stripe_product_id',
    ];

    protected $casts = [
        'price' => 'integer',
        'trial_days' => 'integer',
        'is_active' => 'boolean',
        'features' => 'array',
        'limits' => 'array',
    ];

    public function subscriptions(): HasMany
    {
        return $this->hasMany(Subscription::class);
    }

    public function scopeActive(Builder $query): Builder
    {
        return $query->where('is_active', true);
    }

    public function scopeOrdered(Builder $query): Builder
    {
        return $query->orderBy('sort_order')->orderBy('price');
    }

    public function getPriceFormattedAttribute(): string
    {
        return '$' . number_format($this->price / 100, 2);
    }

    public function getMonthlyPriceAttribute(): int
    {
        return $this->interval === 'yearly'
            ? (int) round($this->price / 12)
            : $this->price;
    }

    public function includesFeature(string $feature): bool
    {
        return in_array($feature, $this->features ?? []);
    }

    public function getLimit(string $key): ?int
    {
        return $this->limits[$key] ?? null;
    }
}

    ];

    {
        return $this->belongsTo(User::class);
    }

    public function plan(): BelongsTo
    {
        return $this->belongsTo(Plan::class);
    }

    public function invoices(): HasMany
    {
        return $this->hasMany(Invoice::class);
    }

    public function events(): HasMany
    {
        return $this->hasMany(SubscriptionEvent::class);
    }

    public function scopeActive(Builder $query): Builder
    {
        return $query->whereIn('status', ['active', 'trialing']);
    }

    public function scopeDueForRenewal(Builder $query, int $daysAhead = 3): Builder
    {
        return $query->where('status', 'active')
            ->where('current_period_end', '<=', now()->addDays($daysAhead));
    }

    public function isTrialing(): bool
    {
        return $this->status === 'trialing';
    }

    public function isActive(): bool
    {
        return in_array($this->status, ['active', 'trialing']);
    }

    public function isCanceled(): bool
    {
        return $this->status === 'canceled';
    }

    public function daysUntilRenewal(): int
    {
        return now()->diffInDays($this->current_period_end);
    }

    public function canUpgrade(): bool
    {
        return $this->isActive() && !$this->cancel_at_period_end;
    }
}
```

# Webhook Processing Architecture

## Webhook Handler

```php
// app/Http/Controllers/WebhookController.php
class WebhookController extends Controller
{
    public function handleStripe(Request $request): Response
    {
        $payload = $request->getContent();
        $sigHeader = $request->header('Stripe-Signature');

        try {
            $event = \Stripe\Webhook::constructEvent(
                $payload,
                $sigHeader,
                config('services.stripe.webhook_secret')
            );
        } catch (\UnexpectedValueException $e) {
            Log::warning('Invalid webhook payload');
            return response('Invalid payload', 400);
        } catch (\Stripe\Exception\SignatureVerificationException $e) {
            Log::warning('Invalid webhook signature');
            return response('Invalid signature', 400);
        }

        // Record event for idempotency
        $webhookEvent = WebhookEvent::firstOrCreate(
            ['stripe_event_id' => $event->id],
            [
                'type' => $event->type,
                'payload' => $event->data->toArray(),
                'status' => 'pending',
            ]
        );

        if ($webhookEvent->status === 'processed') {
            return response('Already processed', 200);
        }

        // Process event
        try {
            $this->processEvent($event);
            $webhookEvent->update([
                'status' => 'processed',
                'processed_at' => now(),
            ]);
        } catch (\Exception $e) {
            $webhookEvent->update([
                'status' => 'failed',
                'error' => $e->getMessage(),
                'attempts' => $webhookEvent->attempts + 1,
            ]);

            Log::error('Webhook processing failed', [
                'event_id' => $event->id,
                'type' => $event->type,
                'error' => $e->getMessage(),
            ]);

            throw $e; // Will be retried by Stripe
        }

        return response('OK', 200);
    }

    private function processEvent(\Stripe\Event $event): void
    {
        match ($event->type) {
            'customer.subscription.created' => $this->handleSubscriptionCreated($event),
            'customer.subscription.updated' => $this->handleSubscriptionUpdated($event),
            'customer.subscription.deleted' => $this->handleSubscriptionDeleted($event),
            'invoice.payment_succeeded' => $this->handleInvoicePaid($event),
            'invoice.payment_failed' => $this->handleInvoiceFailed($event),
            'checkout.session.completed' => $this->handleCheckoutCompleted($event),
            'charge.refunded' => $this->handleRefund($event),
            default => Log::info("Unhandled webhook type: {$event->type}"),
        };
    }

    private function handleSubscriptionCreated(\Stripe\Event $event): void
    {
        $stripeSubscription = $event->data->object;

        $user = User::where('email', $stripeSubscription->metadata->email ?? '')
            ->orWhereHas('subscriptions', fn($q) => $q->where('stripe_id', $stripeSubscription->id))
            ->first();

        if (!$user) {
            Log::warning('Subscription created for unknown user', [
                'stripe_id' => $stripeSubscription->id,
            ]);
            return;
        }

        $plan = Plan::where('stripe_price_id', $stripeSubscription->items->data[0]->price->id)->first();

        Subscription::updateOrCreate(
            ['stripe_id' => $stripeSubscription->id],
            [
                'user_id' => $user->id,
                'plan_id' => $plan?->id,
                'status' => $stripeSubscription->status,
                'stripe_status' => $stripeSubscription->status,
                'stripe_price_id' => $stripeSubscription->items->data[0]->price->id,
                'trial_starts_at' => $stripeSubscription->trial_start
                    ? Carbon::createFromTimestamp($stripeSubscription->trial_start)
                    : null,
                'trial_ends_at' => $stripeSubscription->trial_end
                    ? Carbon::createFromTimestamp($stripeSubscription->trial_end)
                    : null,
                'current_period_start' => Carbon::createFromTimestamp($stripeSubscription->current_period_start),
                'current_period_end' => Carbon::createFromTimestamp($stripeSubscription->current_period_end),
            ]
        );

        event(new SubscriptionCreated($user, $plan));
    }

    private function handleSubscriptionUpdated(\Stripe\Event $event): void
    {
        $stripeSubscription = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeSubscription->id)->first();

        if (!$subscription) {
            Log::warning('Subscription updated but not found locally', [
                'stripe_id' => $stripeSubscription->id,
            ]);
            return;
        }

        $plan = Plan::where('stripe_price_id', $stripeSubscription->items->data[0]->price->id)->first();

        $subscription->update([
            'status' => $stripeSubscription->status,
            'stripe_status' => $stripeSubscription->status,
            'plan_id' => $plan?->id,
            'current_period_start' => Carbon::createFromTimestamp($stripeSubscription->current_period_start),
            'current_period_end' => Carbon::createFromTimestamp($stripeSubscription->current_period_end),
            'cancel_at_period_end' => $stripeSubscription->cancel_at_period_end,
            'ended_at' => $stripeSubscription->ended_at
                ? Carbon::createFromTimestamp($stripeSubscription->ended_at)
                : null,
        ]);

        event(new SubscriptionUpdated($subscription));
    }

    private function handleInvoicePaid(\Stripe\Event $event): void
    {
        $stripeInvoice = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeInvoice->subscription)->first();

        if ($subscription) {
            $subscription->update([
                'status' => 'active',
                'payment_failure_count' => 0,
            ]);
        }

        // Generate local invoice
        $this->generateInvoice($stripeInvoice, $subscription);

        event(new PaymentSucceeded($subscription, $stripeInvoice->amount_paid));
    }

    private function handleInvoiceFailed(\Stripe\Event $event): void
    {
        $stripeInvoice = $event->data->object;

        $subscription = Subscription::where('stripe_id', $stripeInvoice->subscription)->first();

        if ($subscription) {
            $subscription->update([
                'status' => 'past_due',
                'payment_failure_count' => $subscription->payment_failure_count + 1,
                'last_payment_attempt_at' => now(),
            ]);

            event(new PaymentFailed($subscription));
        }
    }

    private function generateInvoice(\StripeInvoice $stripeInvoice, ?Subscription $subscription): void
    {
        $user = $subscription?->user ?? User::firstWhere('stripe_id', $stripeInvoice->customer);

        if (!$user) return;

        $invoiceNumber = 'INV-' . now()->format('Y') . '-' . str_pad(
            Invoice::whereYear('created_at', now()->year)->count() + 1,
            6, '0', STR_PAD_LEFT
        );

        Invoice::create([
            'user_id' => $user->id,
            'subscription_id' => $subscription?->id,
            'number' => $invoiceNumber,
            'status' => 'paid',
            'subtotal' => $stripeInvoice->subtotal,
            'tax_amount' => $stripeInvoice->tax,
            'total' => $stripeInvoice->amount_paid,
            'currency' => $stripeInvoice->currency,
            'issued_at' => Carbon::createFromTimestamp($stripeInvoice->created),
            'paid_at' => now(),
            'line_items' => collect($stripeInvoice->lines->data)->map(fn($item) => [
                'description' => $item->description,
                'amount' => $item->amount,
                'quantity' => $item->quantity,
            ])->toArray(),
        ]);

        // Send invoice email
        InvoiceMail::to($user)->queue($invoice);
    }
}
```

```php
{
    public function convert(int $amount, string $from, string $to): int
    {
        if ($from === $to) return $amount;

        $rate = $this->getExchangeRate($from, $to);
        return (int) round($amount * $rate);
    }

    public function getExchangeRate(string $from, string $to): float
    {
        $cacheKey = "exchange_rate:{$from}:{$to}";

        return Cache::remember($cacheKey, 3600, function () use ($from, $to) {
            $response = Http::get('https://api.exchangerate-api.com/v4/latest/' . strtoupper($from));

            if ($response->failed()) {
                throw new ExchangeRateException("Failed to fetch exchange rate");
            }

            return $response->json('rates.' . strtoupper($to));
        });
    }

    {
        $config = config("currencies.supported.{$currency}");
        $symbol = $config['symbol'] ?? '$';
        $decimals = $config['decimals'] ?? 2;

        return $symbol . number_format($amount / 100, $decimals, '.', ',');
    }

    {
        return $user->currency
            ?? $this->getCurrencyFromLocale($user->locale)
```

```php
// app/Services/TaxService.php
class TaxService
{
    public function __construct(private TaxJarClient $taxjar) {}

    public function calculateTax(int $amount, string $currency, array $address): TaxResult
    {
        if (!$this->isTaxApplicable($address)) {
            return new TaxResult(amount: 0, rate: 0, jurisdiction: null);
        }

        $taxResponse = $this->taxjar->taxForOrder([
            'from_country' => 'US',
            'from_zip' => config('app.tax_from_zip'),
            'to_country' => $address['country'],
            'to_state' => $address['state'] ?? null,
            'to_zip' => $address['zip'] ?? null,
            'amount' => $amount / 100,
            'shipping' => 0,
        ]);

        return new TaxResult(
            amount: (int) round($taxResponse->tax->amount_to_collect * 100),
            rate: $taxResponse->tax->rate,
            jurisdiction: $taxResponse->tax->jurisdiction,
        );
    }

    private function isTaxApplicable(array $address): bool
    {
        // EU VAT for digital services
        if ($this->isEuCountry($address['country'])) {
            return true;
        }

        // US state tax
        if ($address['country'] === 'US' && in_array($address['state'], $this->getTaxableStates())) {
            return true;
        }

        // Other tax jurisdictions
        return in_array($address['country'], config('tax.taxable_countries', []));
    }
}
```

```php
// app/Services/RevenueAnalytics.php
class RevenueAnalytics
{
    public function getDashboard(): array
    {
        return [
            'mrr' => $this->getMrr(),
            'arr' => $this->getMrr() * 12,
            'churn_rate' => $this->getChurnRate(),
            'ltv' => $this->getLifetimeValue(),
            'revenue_by_plan' => $this->getRevenueByPlan(),
            'revenue_trend' => $this->getRevenueTrend(),
            'top_countries' => $this->getTopCountries(),
            'payment_methods' => $this->getPaymentMethodBreakdown(),
        ];
    }

    public function getMrr(): float
    {
        return DB::table('subscriptions')
            ->join('plans', 'plans.id', '=', 'subscriptions.plan_id')
            ->where('subscriptions.status', 'active')
            ->sum(DB::raw('plans.price / 100'));
    }

    public function getChurnRate(int $months = 1): float
    {
        $start = now()->subMonths($months);
        $starting = Subscription::where('created_at', '<', $start)->count();
        $canceled = Subscription::where('canceled_at', '>=', $start)
            ->where('canceled_at', '<', now())
            ->count();

        return $starting > 0 ? round(($canceled / $starting) * 100, 2) : 0;
    }

    public function getLifetimeValue(): float
    {
        $avgMonthlyRevenue = $this->getMrr() / max(Subscription::where('status', 'active')->count(), 1);
        $avgLifetimeMonths = 12 / max($this->getChurnRate() / 100, 0.01);

        return round($avgMonthlyRevenue * $avgLifetimeMonths, 2);
    }

    public function getRevenueTrend(int $months = 12): Collection
    {
        return DB::table('orders')
            ->select(
                DB::raw("DATE_FORMAT(paid_at, '%Y-%m') as month"),
                DB::raw('SUM(total) / 100 as revenue'),
                DB::raw('COUNT(*) as orders')
            )
            ->where('status', 'completed')
            ->where('paid_at', '>=', now()->subMonths($months))
            ->groupBy('month')
            ->orderBy('month')
            ->get();
    }
}
```

```php
// app/Jobs/SubscriptionReconciliationJob.php
class SubscriptionReconciliationJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public function handle(): void
    {
        $subscriptions = Subscription::where('status', 'active')
            ->with('user')
            ->get();

        foreach ($subscriptions as $local) {
            try {
                $stripeSub = \Stripe\Subscription::retrieve($local->stripe_id);

                if ($stripeSub->status !== $local->status) {
                    Log::warning('Subscription status mismatch', [
                        'local_id' => $local->id,
                        'local_status' => $local->status,
                        'stripe_status' => $stripeSub->status,
                    ]);

                    $local->update([
                        'status' => $stripeSub->status,
                        'stripe_status' => $stripeSub->status,
                    ]);

                    event(new SubscriptionSynced($local, $stripeSub->status));
                }

                // Check for period drift
                $localEnd = Carbon::parse($local->current_period_end);
                $stripeEnd = Carbon::createFromTimestamp($stripeSub->current_period_end);

                if ($localEnd->diffInMinutes($stripeEnd) > 5) {
                    Log::warning('Subscription period drift detected', [
                        'local_id' => $local->id,
                        'local_end' => $localEnd->toIso8601String(),
                        'stripe_end' => $stripeEnd->toIso8601String(),
                    ]);

                    $local->update([
                        'current_period_start' => $stripeEnd->copy()->subMonth(),
                        'current_period_end' => $stripeEnd,
                    ]);
                }
            } catch (\Exception $e) {
                Log::error('Reconciliation failed for subscription', [
                    'local_id' => $local->id,
                    'error' => $e->getMessage(),
                ]);
            }
        }
    }
}
```

```php
        $events->listen(PaymentFailed::class, [self::class, 'handlePaymentFailed']);
        $events->listen(OrderCompleted::class, [self::class, 'handleOrderCompleted']);
        $events->listen(RefundProcessed::class, [self::class, 'handleRefundProcessed']);

    {
        $event->user->grantPremiumAccess();
        WelcomeEmail::to($event->user)->queue($event->plan);

        analytics()->track('subscription_created', [
            'user_id' => $event->user->id,
            'plan' => $event->plan->name,
            'price' => $event->plan->price,
        ]);

        UserMetrics::update($event->user->id, [
            'subscription_started_at' => now(),
            'subscription_plan' => $event->plan->slug,
        ]);
    }

    public function handleSubscriptionCanceled(SubscriptionCanceled $event): void
    {
        $event->user->revokePremiumAccess();
        CancellationSurvey::to($event->user)->queue($event->subscription);

        analytics()->track('subscription_canceled', [
            'user_id' => $event->user->id,
            'plan' => $event->subscription->plan->name,
            'reason' => $event->reason,
            'months_subscribed' => $event->subscription->created_at->diffInMonths(now()),
        ]);

        UserMetrics::update($event->user->id, [
            'subscription_ended_at' => now(),
            'subscription_plan' => null,
        ]);
    }

    public function handlePaymentSucceeded(PaymentSucceeded $event): void
    {
        InvoiceService::generateForPayment($event->subscription, $event->amount);
        PaymentReceipt::to($event->subscription->user)->queue($event->amount);
        RevenueMetrics::recordPayment($event->amount, $event->subscription->plan);
        $event->subscription->update(['payment_failure_count' => 0]);
    }

    public function handlePaymentFailed(PaymentFailed $event): void
    {
        PaymentFailedNotification::to($event->subscription->user)->queue();

        if ($event->subscription->payment_failure_count >= 2) {
            CriticalPaymentFailureNotification::toAdmins()->queue($event->subscription);
        }

        analytics()->track('payment_failed', [
            'user_id' => $event->subscription->user_id,
            'plan' => $event->subscription->plan->name,
            'attempt' => $event->subscription->payment_failure_count,
        ]);
    }

    public function handleOrderCompleted(OrderCompleted $event): void
    {
        LicenseService::generateForOrder($event->order);
        DownloadLinksEmail::to($event->order->user)->queue($event->order);

        analytics()->track('order_completed', [
            'user_id' => $event->order->user_id,
            'total' => $event->order->total,
            'product_count' => $event->order->items->count(),
        ]);
    }

    public function handleRefundProcessed(RefundProcessed $event): void
    {
        LicenseService::revokeForOrder($event->order);
        RefundConfirmation::to($event->order->user)->queue($event->refund);

        analytics()->track('refund_processed', [
            'user_id' => $event->order->user_id,
            'amount' => $event->refund->amount,
            'reason' => $event->refund->reason,
        ]);

        RevenueMetrics::recordRefund($event->refund->amount);
    }
}
```

```php
// app/Http/Middleware/CheckSubscription.php
class CheckSubscription
{
    public function handle(Request $request, Closure $next, string $plan = null): Response
    {
        $user = $request->user();

        if (!$user) {
            return redirect()->route('login');
        }

        if (!$user->hasActiveSubscription()) {
            if ($request->expectsJson()) {
                return response()->json(['error' => 'Active subscription required'], 403);
            }
            return redirect()->route('pricing');
        }

        if ($plan && !$user->subscription->plan->matches($plan)) {
            if ($request->expectsJson()) {
                return response()->json(['error' => "Plan {$plan} required"], 403);
            }
            return redirect()->route('pricing');
        }

        if ($user->subscription->status === 'past_due') {
            if ($request->expectsJson()) {
                return response()->json(['error' => 'Payment required'], 402);
            }
            return redirect()->route('billing.payment-method');
        }

        return $next($request);
    }
}
```

```php
// app/Jobs/BillingDataRetentionJob.php
class BillingDataRetentionJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public function handle(): void
    {
        $this->archiveOldOrders();
        $this->cleanupCheckoutSessions();
        $this->cleanupWebhookEvents();
        $this->archiveSubscriptionEvents();
    }

    private function archiveOldOrders(): void
    {
        $cutoff = now()->subYear();

        Order::where('created_at', '<', $cutoff)
            ->where('status', 'completed')
            ->chunkById(1000, function ($orders) {
                foreach ($orders as $order) {
                    ArchiveService::archive($order);
                    $order->update([
                        'metadata' => array_merge($order->metadata ?? [], [
                            'archived_at' => now()->toIso8601String(),
                        ]),
                    ]);
                }
            });
    }

    private function cleanupCheckoutSessions(): void
    {
        CheckoutSession::where('created_at', '<', now()->subDays(90))
            ->where('status', '!=', 'completed')
            ->delete();
    }

    private function cleanupWebhookEvents(): void
    {
        WebhookEvent::where('created_at', '<', now()->subDays(30))
            ->where('status', 'processed')
            ->delete();
    }

    private function archiveSubscriptionEvents(): void
    {
        SubscriptionEvent::where('created_at', '<', now()->subMonths(6))
            ->chunkById(1000, function ($events) {
                foreach ($events as $event) {
                    ArchiveService::archive($event);
                    $event->delete();
                }
            });
    }
}
```

```php
        ];

        $this->postJson('/webhooks/stripe', $payload, [

    protected function generateStripeSignature(array $payload): string
    {
        $secret = config('services.stripe.webhook_secret');
```

```php
// app/Exceptions/BillingExceptionHandler.php
class BillingExceptionHandler
{
    public function handle(Throwable $e): Response
    {
        return match (true) {
            $e instanceof PaymentFailedException => $this->paymentFailed($e),
            $e instanceof SubscriptionNotFoundException => $this->subscriptionNotFound($e),
            $e instanceof PlanNotFoundException => $this->planNotFound($e),
            $e instanceof CouponExpiredException => $this->couponExpired($e),
            $e instanceof RefundWindowClosedException => $this->refundWindowClosed($e),
            default => $this->genericError($e),
        };
    }

    private function paymentFailed(PaymentFailedException $e): Response
    {
        Log::error('Payment failed', [
            'user_id' => $e->getUserId(),
            'error' => $e->getMessage(),
        ]);

        return redirect()->route('billing.payment-method')
            ->with('error', 'Your payment was declined. Please try a different payment method.');
    }
}
```
