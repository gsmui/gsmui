**Project:** GsmUi

# Purpose

The Plugin System allows third-party developers to extend GsmUi functionality with custom plugins. Plugins can add new components, integrations, features, and customizations to the platform while maintaining security and compatibility.

---

# Architecture Overview

The plugin system follows a service provider pattern where each plugin is a Laravel package that registers itself with the platform. Plugins declare their capabilities through a manifest file and are loaded dynamically at runtime.

The architecture enforces isolation between plugins through namespace segregation, permission scoping, and sandboxed execution. Plugins can hook into platform events, register routes and views, and interact with the database through migration support.

The plugin lifecycle includes discovery, installation, activation, updates, and deactivation. Each stage has validation and safety checks to prevent malicious or incompatible plugins from affecting platform stability.

- Plugins run in isolated namespaces
- File system access restricted to plugin directory
- Database access through platform repositories only
- No direct access to platform internals

## Manifest Validation Rules

```php
<?php

namespace App\Plugins\Validation;

class ManifestValidator
{
    private array $rules = [
        'name' => 'required|string|min:3|max:100',
        'slug' => 'required|string|regex:/^[a-z0-9\-]+$/|max:50',
        'description' => 'required|string|min:10|max:500',
        'version' => 'required|regex:/^\d+\.\d+\.\d+$/',
        'minimum_platform_version' => 'required|regex:/^\d+\.\d+\.\d+$/',
        'providers' => 'required|array|min:1',
        'providers.*' => 'required|string|ends_with:ServiceProvider',
        'permissions' => 'sometimes|array',
        'permissions.*' => 'required|string|regex:/^[a-z0-9\.]+$/',
        'events.listens.*' => 'required|string|regex:/^[a-z0-9\.]+$/',
        'events.dispatches.*' => 'required|string|regex:/^[a-z0-9\.]+$/',
        'routes' => 'sometimes|array',
        'routes.web' => 'sometimes|string|exists:file',
        'routes.api' => 'sometimes|string|exists:file',
        'migrations' => 'sometimes|array',
        'migrations.*' => 'required|string|exists:file',
        'config' => 'sometimes|string|exists:file',
        'config_key' => 'sometimes|string',
        'settings' => 'sometimes|array',
        'settings.*.key' => 'required|string',
        'settings.*.label' => 'required|string',
        'settings.*.type' => 'required|in:text,textarea,number,boolean,password,select',
        'scripts.post_install' => 'sometimes|string|exists:callable',
        'scripts.post_update' => 'sometimes|string|exists:callable',
        'scripts.pre_uninstall' => 'sometimes|string|exists:callable',
    ];

    public function validate(array $manifest): array
    {
        $errors = [];

        foreach ($this->rules as $field => $rules) {
            $value = data_get($manifest, $field);
            // Simplified validation logic
            if (str_contains($rules, 'required') && empty($value)) {
                $errors[$field] = "{$field} is required";
            }
        }

        // Version format validation
        if (isset($manifest['version']) && !preg_match('/^\d+\.\d+\.\d+$/', $manifest['version'])) {
            $errors['version'] = 'Version must follow semantic versioning (x.y.z)';
        }

        // Slug format validation
        if (isset($manifest['slug']) && !preg_match('/^[a-z0-9\-]+$/', $manifest['slug'])) {
            $errors['slug'] = 'Slug must contain only lowercase letters, numbers, and hyphens';
        }

        // Permission format validation
        if (isset($manifest['permissions'])) {
            foreach ($manifest['permissions'] as $permission) {
                if (!preg_match('/^[a-z0-9\.]+$/', $permission)) {
                    $errors['permissions'][] = "Invalid permission format: {$permission}";
                }
            }
        }

        // Event name format validation
        if (isset($manifest['events']['listens'])) {
            foreach ($manifest['events']['listens'] as $event) {
                if (!preg_match('/^[a-z0-9\.]+$/', $event)) {
                    $errors['events.listens'][] = "Invalid event name format: {$event}";
                }
            }
        }

        return $errors;
    }

    public function validateCompatibility(array $manifest, string $platformVersion): array
    {
        $errors = [];

        if (isset($manifest['minimum_platform_version'])) {
            if (version_compare($platformVersion, $manifest['minimum_platform_version'], '<')) {
                $errors[] = "Requires GsmUi {$manifest['minimum_platform_version']} or later";
            }
        }

        if (isset($manifest['maximum_platform_version'])) {
            if (version_compare($platformVersion, $manifest['maximum_platform_version'], '>')) {
                $errors[] = "Incompatible with GsmUi {$manifest['maximum_platform_version']} or later";
            }
        }

        if (isset($manifest['requirements']['php'])) {
            $requiredPhp = ltrim($manifest['requirements']['php'], '>=');
            if (version_compare(PHP_VERSION, $requiredPhp, '<')) {
                $errors[] = "Requires PHP {$requiredPhp} or later";
            }
        }

        if (isset($manifest['requirements']['extensions'])) {
            foreach ($manifest['requirements']['extensions'] as $ext) {
                if (!extension_loaded($ext)) {
                    $errors[] = "Requires PHP extension: {$ext}";
                }
            }
        }

        return $errors;
    }
}
```

# Plugin Loader

## Discovery and Loading

The plugin loader is responsible for discovering installed plugins, validating their manifests, and registering their services with the Laravel application container.

```php
<?php

namespace App\Plugins;

use App\Plugins\Models\Plugin;
use App\Plugins\Validation\ManifestValidator;
use Illuminate\Support\Facades\File;
use Illuminate\Support\ServiceProvider;

class PluginLoader
{
    private array $loadedPlugins = [];
    private ManifestValidator $validator;

    public function __construct(ManifestValidator $validator)
    {
        $this->validator = $validator;
    }

    public function discoverPlugins(string $pluginsPath): array
    {
        $plugins = [];
        $directories = File::directories($pluginsPath);

        foreach ($directories as $directory) {
            $manifestPath = $directory . '/gsmui-plugin.json';

            if (!File::exists($manifestPath)) {
                continue;
            }

            $manifest = json_decode(File::get($manifestPath), true);

            if (json_last_error() !== JSON_ERROR_NONE) {
                \Illuminate\Support\Facades\Log::warning("Invalid manifest in {$directory}: " . json_last_error_msg());
                continue;
            }

            $errors = $this->validator->validate($manifest);

            if (!empty($errors)) {
                \Illuminate\Support\Facades\Log::warning("Manifest validation failed for {$directory}", $errors);
                continue;
            }

            $plugins[] = [
                'path' => $directory,
                'manifest' => $manifest,
            ];
        }

        return $plugins;
    }

    public function loadPlugin(string $path, array $manifest): bool
    {
        $slug = $manifest['slug'];

        if (isset($this->loadedPlugins[$slug])) {
            return true;
        }

        // Register service providers
        if (isset($manifest['providers'])) {
            foreach ($manifest['providers'] as $providerClass) {
                if (!class_exists($providerClass)) {
                    // Auto-load from plugin's src directory
                    $autoLoadPath = $path . '/src/' . str_replace('\\', '/', $providerClass) . '.php';
                    if (File::exists($autoLoadPath)) {
                        require_once $autoLoadPath;
                    }
                }

                if (class_exists($providerClass)) {
                    $provider = app()->register($providerClass);
                    $this->loadedPlugins[$slug]['providers'][] = $provider;
                }
            }
        }

        // Register routes
        $this->registerRoutes($path, $manifest);

        // Register views
        $this->registerViews($path, $manifest);

        // Publish assets
        $this->publishAssets($path, $manifest);

        // Register config
        $this->registerConfig($path, $manifest);

        $this->loadedPlugins[$slug] = [
            'path' => $path,
            'manifest' => $manifest,
            'loaded_at' => now(),
        ];

        return true;
    }

    private function registerRoutes(string $path, array $manifest): void
    {
        if (!isset($manifest['routes'])) {
            return;
        }

        $routeFiles = is_array($manifest['routes'])
            ? $manifest['routes']
            : [$manifest['routes']];

        foreach ($routeFiles as $routeFile) {
            $routePath = $path . '/routes/' . $routeFile;
            if (File::exists($routePath)) {
                require $routePath;
            }
        }
    }

    private function registerViews(string $path, array $manifest): void
    {
        if (!isset($manifest['views'])) {
            return;
        }

        $namespace = $manifest['views']['namespace'] ?? $manifest['slug'];
        $viewPath = $manifest['views']['path'] ?? 'resources/views';

        \Illuminate\Support\Facades\View::addNamespace(
            $namespace,
            $path . '/' . $viewPath
        );
    }

    private function publishAssets(string $path, array $manifest): void
    {
        if (!isset($manifest['assets']['publishes'])) {
            return;
        }

        foreach ($manifest['assets']['publishes'] as $publish) {
            $from = $path . '/' . $publish['from'];
            $to = $publish['to'];

            if (File::exists($from)) {
                $this->publishFile($from, $to);
            }
        }
    }

    private function publishFile(string $from, string $to): void
    {
        $directory = dirname($to);

        if (!File::isDirectory($directory)) {
            File::makeDirectory($directory, 0755, true);
        }

        File::copy($from, $to);
    }

    private function registerConfig(string $path, array $manifest): void
    {
        if (!isset($manifest['config'])) {
            return;
        }

        $configPath = $path . '/' . $manifest['config'];
        $configKey = $manifest['config_key'] ?? $manifest['slug'];

        if (File::exists($configPath)) {
            $config = require $configPath;
            \Illuminate\Support\Facades\Config::set($configKey, $config);
        }
    }

    public function getLoadedPlugins(): array
    {
        return $this->loadedPlugins;
    }

    public function getPluginBySlug(string $slug): ?array
    {
        return $this->loadedPlugins[$slug] ?? null;
    }
```

# Plugin Sandbox

## Security Sandbox Implementation

The sandbox isolates plugins from the core platform and from each other, preventing unauthorized access to files, databases, and system resources.

```php
<?php

namespace App\Plugins\Sandbox;

use App\Plugins\Models\Plugin;
use Illuminate\Support\Facades\File;

class PluginSandbox
{
    private string $pluginsRoot;
    private array $allowedOperations = [
        'file.read',
        'file.write',
        'database.read',
        'database.write',
        'cache.read',
        'cache.write',
        'queue.dispatch',
        'mail.send',
        'http.request',
    ];

    public function __construct(string $pluginsRoot)
    {
        $this->pluginsRoot = $pluginsRoot;
    }

    public function canAccessFile(Plugin $plugin, string $path): bool
    {
        $pluginDir = $this->pluginsRoot . '/' . $plugin->slug;

        $realPath = realpath($path);

        if ($realPath === false) {
            return false;
        }

        $allowedPaths = [
            $pluginDir . '/src',
            $pluginDir . '/resources',
            $pluginDir . '/config',
            $pluginDir . '/database',
            storage_path('app/plugins/' . $plugin->slug),
        ];

        foreach ($allowedPaths as $allowedPath) {
            if (str_starts_with($realPath, $allowedPath)) {
                return true;
            }
        }

        return false;
    }

    public function canAccessTable(Plugin $plugin, string $table): bool
    {
        $allowedPrefixes = [
            $plugin->slug . '_',
            'gsmui_',
            'cache_',
            'sessions',
            'jobs',
            'failed_jobs',
        ];

        foreach ($allowedPrefixes as $prefix) {
            if (str_starts_with($table, $prefix)) {
                return true;
            }
        }

        return false;
    }

    public function canPerformOperation(Plugin $plugin, string $operation): bool
    {
        $pluginPermissions = $plugin->getPermissions();

        $requiredPermission = "plugin.{$plugin->slug}.{$operation}";

        return in_array($requiredPermission, $pluginPermissions)
            || in_array($operation, $this->allowedOperations);
    }

    public function validateMigration(Plugin $plugin, string $migrationPath): bool
    {
        $tablePattern = '/create_(\w+)_table/';

        if (preg_match($tablePattern, $migrationPath, $matches)) {
            $tableName = $matches[1];

            if (!$this->canAccessTable($plugin, $tableName)) {
                return false;
            }
        }

        $contents = File::get($migrationPath);

        $dangerousPatterns = [
            '/DROP\s+TABLE/i',
            '/TRUNCATE\s+TABLE/i',
            '/DELETE\s+FROM.*WHERE\s*1\s*=\s*1/i',
            '/ALTER\s+TABLE.*ADD\s+COLUMN.*password/i',
            '/INTO\s+OUTFILE/i',
            '/LOAD\s+DATA/i',
        ];

        foreach ($dangerousPatterns as $pattern) {
            if (preg_match($pattern, $contents)) {
                return false;
            }
        }

        return true;
    }

    public function isolateNamespace(Plugin $plugin, callable $callback): mixed
    {
        $originalNamespace = $this->getCurrentNamespace();

        try {
            $pluginNamespace = "Plugin\\{$this->toPascalCase($plugin->slug)}";

            $this->setNamespace($pluginNamespace);

            $result = $callback();

            return $result;
        } finally {
            $this->setNamespace($originalNamespace);
        }
    }

    private function getCurrentNamespace(): string
    {
        return app()->getNamespace();
    }

    private function setNamespace(string $namespace): void
    {
        app()->bind('app.namespace', $namespace);
    }

    private function toPascalCase(string $slug): string
    {
        return str_replace('-', '', ucwords($slug, '-'));
    }
}
```

# Plugin Event System

## Event Registry and Dispatch

Plugins register event listeners and can dispatch custom events. The event system ensures proper ordering, priority handling, and error isolation.

```php
<?php

namespace App\Plugins\Events;

use App\Plugins\Models\Plugin;
use Illuminate\Support\Facades\Event;

class PluginEventRegistry
{
    private array $listeners = [];
    private array $pluginListeners = [];

    public function registerPluginListeners(Plugin $plugin, array $events): void
    {
        foreach ($events as $eventName => $handler) {
            $priority = is_array($handler) && isset($handler['priority'])
                ? $handler['priority']
                : 0;

            $handlerClass = is_array($handler) ? $handler['handler'] : $handler;

            $this->listeners[$eventName][] = [
                'plugin' => $plugin->slug,
                'handler' => $handlerClass,
                'priority' => $priority,
            ];

            $this->pluginListeners[$plugin->slug][] = [
                'event' => $eventName,
                'handler' => $handlerClass,
            ];
        }
    }

    public function dispatch(string $eventName, mixed $payload = null): void
    {
        $listeners = $this->getSortedListeners($eventName);

        foreach ($listeners as $listener) {
            try {
                $handler = app($listener['handler']);
                $handler->handle($payload);
            } catch (\Throwable $e) {
                \Illuminate\Support\Facades\Log::error(
                    "Plugin event handler failed",
                    [
                        'event' => $eventName,
                        'plugin' => $listener['plugin'],
                        'handler' => $listener['handler'],
                        'error' => $e->getMessage(),
                    ]
                );
            }
        }
    }

    private function getSortedListeners(string $eventName): array
    {
        $listeners = $this->listeners[$eventName] ?? [];

        usort($listeners, fn($a, $b) => $b['priority'] <=> $a['priority']);

        return $listeners;
    }

    public function removePluginListeners(string $pluginSlug): void
    {
        $eventsToClean = $this->pluginListeners[$pluginSlug] ?? [];

        foreach ($eventsToClean as $eventInfo) {
            $eventName = $eventInfo['event'];
            $this->listeners[$eventName] = array_filter(
                $this->listeners[$eventName],
                fn($listener) => $listener['plugin'] !== $pluginSlug
            );
        }

        unset($this->pluginListeners[$pluginSlug]);
    }
}
```

# Plugin Database Layer

## Migration Management

Plugins register migrations that run in isolation with their own table prefixes.

```php
<?php

namespace App\Plugins\Database;

use App\Plugins\Models\Plugin;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Schema;

class PluginMigrationManager
{
    private string $migrationsPath;

    public function __construct(string $migrationsPath)
    {
        $this->migrationsPath = $migrationsPath;
    }

    public function getPluginMigrations(Plugin $plugin): array
    {
        $migrationDir = $plugin->getPath() . '/database/migrations';

        if (!File::isDirectory($migrationDir)) {
            return [];
        }

        $migrations = File::files($migrationDir);

        return collect($migrations)
            ->filter(fn($file) => $file->getExtension() === 'php')
            ->map(fn($file) => $file->getRealPath())
            ->toArray();
    }

    public function runMigrations(Plugin $plugin): array
    {
        $migrations = $this->getPluginMigrations($plugin);
        $ran = [];

        foreach ($migrations as $migrationPath) {
            $migration = require $migrationPath;

            if ($migration instanceof \Illuminate\Database\Migrations\Migration) {
                Schema::defaultConnection(function () use ($migration, $plugin) {
                    $migration->up();
                });
                $ran[] = basename($migrationPath);
            }
        }

        $this->recordMigrations($plugin, $ran);

        return $ran;
    }

    public function rollbackMigrations(Plugin $plugin): array
    {
        $ran = $this->getRanMigrations($plugin);
        $rolledBack = [];

        foreach (array_reverse($ran) as $migrationName) {
            $migrationPath = $this->findMigrationFile($plugin, $migrationName);

            if ($migrationPath) {
                $migration = require $migrationPath;

                if ($migration instanceof \Illuminate\Database\Migrations\Migration) {
                    Schema::defaultConnection(function () use ($migration) {
                        $migration->down();
                    });
                    $rolledBack[] = $migrationName;
                }
            }
        }

        $this->removeMigrationRecords($plugin, $rolledBack);

        return $rolledBack;
    }

    private function recordMigrations(Plugin $plugin, array $migrations): void
    {
        DB::table('plugin_migrations')->insert(
            array_map(fn($name) => [
                'plugin_slug' => $plugin->slug,
                'migration' => $name,
                'ran_at' => now(),
            ], $migrations)
        );
    }

    private function getRanMigrations(Plugin $plugin): array
    {
        return DB::table('plugin_migrations')
            ->where('plugin_slug', $plugin->slug)
            ->pluck('migration')
            ->toArray();
    }

    private function removeMigrationRecords(Plugin $plugin, array $migrations): void
    {
        DB::table('plugin_migrations')
            ->where('plugin_slug', $plugin->slug)
            ->whereIn('migration', $migrations)
            ->delete();
    }

    private function findMigrationFile(Plugin $plugin, string $migrationName): ?string
    {
        $migrationDir = $plugin->getPath() . '/database/migrations';
        $files = File::glob($migrationDir . "/*{$migrationName}*");

        return $files[0] ?? null;
    }
}
```

# Plugin Settings Management

## Settings Repository

Plugins can define configurable settings that admins can manage through the admin panel.

```php
<?php

namespace App\Plugins\Settings;

use App\Plugins\Models\Plugin;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class PluginSettingsRepository
{
    public function get(Plugin $plugin, string $key, mixed $default = null): mixed
    {
        $cacheKey = "plugin_settings:{$plugin->slug}:{$key}";

        return Cache::tags(["plugin:{$plugin->slug}"])->remember(
            $cacheKey,
            3600,
            function () use ($plugin, $key, $default) {
                $setting = DB::table('plugin_settings')
                    ->where('plugin_slug', $plugin->slug)
                    ->where('key', $key)
                    ->first();

                return $setting ? $this->castValue($setting->value, $setting->type) : $default;
            }
        );
    }

    public function set(Plugin $plugin, string $key, mixed $value): void
    {
        $type = $this->detectType($value);

        DB::table('plugin_settings')->updateOrInsert(
            [
                'plugin_slug' => $plugin->slug,
                'key' => $key,
            ],
            [
                'value' => $this->serializeValue($value, $type),
                'type' => $type,
                'updated_at' => now(),
            ]
        );

        Cache::tags(["plugin:{$plugin->slug}"])->forget("plugin_settings:{$plugin->slug}:{$key}");
    }

    public function getMany(Plugin $plugin, array $keys): array
    {
        $results = [];

        foreach ($keys as $key) {
            $results[$key] = $this->get($plugin, $key);
        }

        return $results;
    }

    public function setMany(Plugin $plugin, array $settings): void
    {
        foreach ($settings as $key => $value) {
            $this->set($plugin, $key, $value);
        }
    }

    public function getAll(Plugin $plugin): array
    {
        $settings = DB::table('plugin_settings')
            ->where('plugin_slug', $plugin->slug)
            ->get();

        $results = [];

        foreach ($settings as $setting) {
            $results[$setting->key] = $this->castValue($setting->value, $setting->type);
        }

        return $results;
    }

    public function delete(Plugin $plugin, string $key): void
    {
        DB::table('plugin_settings')
            ->where('plugin_slug', $plugin->slug)
            ->where('key', $key)
            ->delete();

        Cache::tags(["plugin:{$plugin->slug}"])->forget("plugin_settings:{$plugin->slug}:{$key}");
    }

    public function resetAll(Plugin $plugin): void
    {
        DB::table('plugin_settings')
            ->where('plugin_slug', $plugin->slug)
            ->delete();

        Cache::tags(["plugin:{$plugin->slug}"])->flush();
    }

    private function detectType(mixed $value): string
    {
        return match (true) {
            is_bool($value) => 'boolean',
            is_int($value) || is_float($value) => 'number',
            is_array($value) => 'json',
            default => 'string',
        };
    }

    private function serializeValue(mixed $value, string $type): string
    {
        return match ($type) {
            'json' => json_encode($value),
            'boolean' => $value ? '1' : '0',
            default => (string) $value,
        };
    }

    private function castValue(string $value, string $type): mixed
    {
        return match ($type) {
            'boolean' => (bool) $value,
            'number' => is_numeric($value) ? (float) $value : $value,
            'json' => json_decode($value, true),
            default => $value,
        };
    }
}
```

# Plugin Manager Service

## Core Plugin Manager

The PluginManager coordinates all plugin lifecycle operations including installation, activation, updates, and removal.

```php
<?php

namespace App\Plugins\Services;

use App\Plugins\Models\Plugin;
use App\Plugins\PluginLoader;
use App\Plugins\PluginSandbox;
use App\Plugins\Database\PluginMigrationManager;
use App\Plugins\Settings\PluginSettingsRepository;
use App\Plugins\Validation\ManifestValidator;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\File;

class PluginManager
{
    public function __construct(
        private PluginLoader $loader,
        private PluginSandbox $sandbox,
        private PluginMigrationManager $migrations,
        private PluginSettingsRepository $settings,
        private ManifestValidator $validator
    ) {}

    public function install(array $data): Plugin
    {
        $manifest = $this->resolveManifest($data);

        $errors = $this->validator->validate($manifest);
        if (!empty($errors)) {
            throw new \InvalidArgumentException('Manifest validation failed: ' . implode(', ', array_flatten($errors)));
        }

        $compatErrors = $this->validator->validateCompatibility(
            $manifest,
            config('gsmui.version', '2.15.0')
        );
        if (!empty($compatErrors)) {
            throw new \InvalidArgumentException('Compatibility check failed: ' . implode(', ', $compatErrors));
        }

        $plugin = Plugin::create([
            'name' => $manifest['name'],
            'slug' => $manifest['slug'],
            'description' => $manifest['description'],
            'version' => $manifest['version'],
            'author' => $manifest['author']['name'] ?? null,
            'status' => 'inactive',
            'manifest' => $manifest,
            'installed_at' => now(),
        ]);

        $pluginPath = $this->extractPlugin($data, $manifest['slug']);

        $this->migrations->runMigrations($plugin);

        $this->publishPluginConfig($plugin, $manifest);

        $this->loader->loadPlugin($pluginPath, $manifest);

        $plugin->update(['status' => 'inactive']);

        return $plugin;
    }

    public function enable(Plugin $plugin): void
    {
        if ($plugin->status === 'active') {
            return;
        }

        $this->loader->loadPlugin($plugin->getPath(), $plugin->manifest);

        $plugin->update([
            'status' => 'active',
            'activated_at' => now(),
        ]);

        event('plugin.activated', ['plugin' => $plugin]);
    }

    public function disable(Plugin $plugin): void
    {
        if ($plugin->status === 'inactive') {
            return;
        }

        $plugin->update([
            'status' => 'inactive',
            'deactivated_at' => now(),
        ]);

        event('plugin.deactivated', ['plugin' => $plugin]);
    }

    public function uninstall(Plugin $plugin): void
    {
        $this->disable($plugin);

        $this->migrations->rollbackMigrations($plugin);

        $this->removePluginAssets($plugin);

        $this->settings->resetAll($plugin);

        $plugin->delete();

        event('plugin.uninstalled', ['plugin' => $plugin]);
    }

    public function update(Plugin $plugin, array $newManifest): void
    {
        $errors = $this->validator->validate($newManifest);
        if (!empty($errors)) {
            throw new \InvalidArgumentException('New manifest validation failed');
        }

        $plugin->update([
            'version' => $newManifest['version'],
            'manifest' => $newManifest,
        ]);

        if (isset($newManifest['migrations'])) {
            $this->migrations->runMigrations($plugin);
        }

        $this->loader->loadPlugin($plugin->getPath(), $newManifest);

        event('plugin.updated', ['plugin' => $plugin, 'old_version' => $plugin->version]);
    }

    public function getSettings(Plugin $plugin): array
    {
        $manifestSettings = $plugin->manifest['settings'] ?? [];
        $currentValues = $this->settings->getAll($plugin);

        $result = [];

        foreach ($manifestSettings as $setting) {
            $result[$setting['key']] = [
                'label' => $setting['label'],
                'type' => $setting['type'],
                'value' => $currentValues[$setting['key']] ?? $setting['default'] ?? null,
                'description' => $setting['description'] ?? '',
                'required' => $setting['required'] ?? false,
            ];
        }

        return $result;
    }

    public function updateSettings(Plugin $plugin, array $settings): void
    {
        foreach ($settings as $key => $value) {
            $this->settings->set($plugin, $key, $value);
        }
    }

    private function resolveManifest(array $data): array
    {
        if (isset($data['manifest_file'])) {
            $content = File::get($data['manifest_file']->getRealPath());
            return json_decode($content, true);
        }

        if (isset($data['manifest_url'])) {
            $response = \Illuminate\Support\Facades\Http::get($data['manifest_url']);
            return $response->json();
        }

        return $data;
    }

    private function extractPlugin(array $data, string $slug): string
    {
        $pluginPath = base_path("plugins/{$slug}");

        if (!File::isDirectory($pluginPath)) {
            File::makeDirectory($pluginPath, 0755, true);
        }

        if (isset($data['manifest_file'])) {
            $data['manifest_file']->move($pluginPath, 'gsmui-plugin.json');
        }

        return $pluginPath;
    }

    private function publishPluginConfig(Plugin $plugin, array $manifest): void
    {
        if (!isset($manifest['config'])) {
            return;
        }

        $sourcePath = $plugin->getPath() . '/' . $manifest['config'];
        $configKey = $manifest['config_key'] ?? $manifest['slug'];

        if (File::exists($sourcePath)) {
            $this->publishFile(
                $sourcePath,
                config_path("{$configKey}.php")
            );
        }
    }

    private function removePluginAssets(Plugin $plugin): void
    {
        $assetPaths = [
            public_path("vendor/{$plugin->slug}"),
            config_path("{$plugin->slug}.php"),
        ];

        foreach ($assetPaths as $path) {
            if (File::isDirectory($path)) {
                File::deleteDirectory($path);
            } elseif (File::isFile($path)) {
                File::delete($path);
            }
        }
    }

    private function publishFile(string $from, string $to): void
    {
        $directory = dirname($to);

        if (!File::isDirectory($directory)) {
            File::makeDirectory($directory, 0755, true);
        }

        File::copy($from, $to);
    }
}
```

```php
        '/TRUNCATE\s+TABLE/i',
        '/DROP\s+DATABASE/i',
        '/INTO\s+(OUTFILE|DUMPFILE)/i',
        '/LOAD\s+DATA\s+(LOCAL\s+)?INFILE/i',
        '/GRANT\s+ALL/i',
    ];

    private array $xssPatterns = [
        '/<script[\s>]/i',
        '/javascript:/i',
        '/on\w+\s*=/i',
        '/data:text\/html/i',
    ];

    public function scan(string $pluginPath): array
    {
        $results = [
            'passed' => true,
            'findings' => [],
            'files_scanned' => 0,
            'scan_time' => 0,
        ];

        $startTime = microtime(true);

        $phpFiles = $this->getPhpFiles($pluginPath);

        foreach ($phpFiles as $file) {
            $results['files_scanned']++;
            $fileFindings = $this->scanFile($file, $pluginPath);
            $results['findings'] = array_merge($results['findings'], $fileFindings);
        }

        $bladeFiles = $this->getBladeFiles($pluginPath);

        foreach ($bladeFiles as $file) {
            $results['files_scanned']++;
            $fileFindings = $this->scanBladeFile($file, $pluginPath);
            $results['findings'] = array_merge($results['findings'], $fileFindings);
        }

        $results['passed'] = empty($results['findings']);
        $results['scan_time'] = round(microtime(true) - $startTime, 2);

        return $results;
    }

    private function scanFile(string $filePath, string $pluginPath): array
    {
        $findings = [];
        $contents = File::get($filePath);
        $relativePath = str_replace($pluginPath . '/', '', $filePath);

        // Check for dangerous functions
        foreach ($this->dangerousFunctions as $function) {
            if (is_array($function)) {
                continue;
            }

            if (preg_match_all('/\b' . $function . '\s*\(/', $contents, $matches)) {
                $findings[] = [
                    'severity' => 'critical',
                    'type' => 'dangerous_function',
                    'file' => $relativePath,
                    'message' => "Use of dangerous function: {$function}",
                    'matches' => count($matches[0]),
                ];
            }
        }

        // Check for SQL injection patterns
        foreach ($this->sqlPatterns as $pattern) {
            if (preg_match($pattern, $contents, $matches)) {
                $findings[] = [
                    'severity' => 'critical',
                    'type' => 'sql_injection',
                    'file' => $relativePath,
                    'message' => 'Potential SQL injection pattern detected',
                    'match' => $matches[0],
                ];
            }
        }

        // Check for hardcoded secrets
        $secretPatterns = [
            '/password\s*=\s*["\'][^"\']+["\']/i',
            '/api_key\s*=\s*["\'][^"\']+["\']/i',
            '/secret\s*=\s*["\'][^"\']+["\']/i',
            '/token\s*=\s*["\'][^"\']+["\']/i',
        ];

        foreach ($secretPatterns as $pattern) {
            if (preg_match($pattern, $contents, $matches)) {
                $findings[] = [
                    'severity' => 'high',
                    'type' => 'hardcoded_secret',
                    'file' => $relativePath,
                    'message' => 'Potential hardcoded secret detected',
                    'match' => preg_replace('/(.{10}).*(.{10})/', '$1...$2', $matches[0]),
                ];
            }
        }

        // Check for direct file system access outside plugin directory
        if (preg_match('/(?:file_get_contents|file_put_contents|fopen|fwrite)\s*\(\s*["\'](?:\/|\.\.\/)/', $contents)) {
            $findings[] = [
                'severity' => 'high',
                'type' => 'path_traversal',
                'file' => $relativePath,
                'message' => 'Direct file system access outside plugin directory',
            ];
        }

        return $findings;
    }

    private function scanBladeFile(string $filePath, string $pluginPath): array
    {
        $findings = [];
        $contents = File::get($filePath);
        $relativePath = str_replace($pluginPath . '/', '', $filePath);

        foreach ($this->xssPatterns as $pattern) {
            if (preg_match($pattern, $contents, $matches)) {
                $findings[] = [
                    'severity' => 'high',
                    'type' => 'xss',
                    'file' => $relativePath,
                    'message' => 'Potential XSS pattern detected in Blade template',
                    'match' => $matches[0],
                ];
            }
        }

        // Check for {!! !!} (unescaped output)
        if (preg_match_all('/\{!!\s*(.+?)\s*!!\}/', $contents, $matches)) {
            foreach ($matches[1] as $expression) {
                if (preg_match('/\$', $expression)) {
                    $findings[] = [
                        'severity' => 'medium',
                        'type' => 'unescaped_output',
                        'file' => $relativePath,
                        'message' => 'Unescaped Blade output may contain user data',
                        'expression' => $expression,
                    ];
                }
            }
        }

        return $findings;
    }

    private function getPhpFiles(string $path): array
    {
        return collect(File::allFiles($path))
            ->filter(fn($file) => $file->getExtension() === 'php')
            ->map(fn($file) => $file->getRealPath())
            ->toArray();
    }

    private function getBladeFiles(string $path): array
    {
        return collect(File::allFiles($path))
            ->filter(fn($file) => $file->getExtension() === 'blade.php')
            ->map(fn($file) => $file->getRealPath())
            ->toArray();
    }
}
```

```php
<?php

namespace Tests\Plugin;

use App\Plugins\Models\Plugin;
use Orchestra\Testbench\TestCase;

abstract class PluginTestCase extends TestCase
{
    protected Plugin $plugin;

        }

    {
        $seederPath = $this->plugin->getPath() . '/database/seeders';

        if (is_dir($seederPath)) {
            $this->seedDatabase([$seederPath]);
        }
    }

    protected function getPluginSetting(string $key, mixed $default = null): mixed
    {
        return setting("{$this->pluginSlug}.{$key}", $default);
    }

    protected function assertPluginSettingEquals(string $key, mixed $expected): void
    {
        $actual = $this->getPluginSetting($key);
        $this->assertEquals($expected, $actual);
    }

    protected function assertPluginTableExists(string $tableName): void
    {
        $fullTable = "{$this->pluginSlug}_{$tableName}";
        $this->assertDatabaseHas('information_schema.tables', [
            'table_schema' => $this->app['config']['database.connections.mysql.database'],
            'table_name' => $fullTable,
        ]);
    }
}
```

# Plugin Migration Guide

## Version Upgrade Procedures

When a plugin is updated, the migration process ensures backward compatibility and smooth transitions.

```php
<?php

namespace App\Plugins\Migration;

use App\Plugins\Models\Plugin;

class PluginMigrationGuide
{
    public function getUpgradeInstructions(Plugin $plugin, string $fromVersion, string $toVersion): array
    {
        $instructions = [];

        $migrationFiles = $this->getMigrationFiles($plugin, $fromVersion, $toVersion);

        foreach ($migrationFiles as $file) {
            $migration = require $file;

            $instructions[] = [
                'version' => $migration->getVersion(),
                'description' => $migration->getDescription(),
                'up' => $migration->getUpInstructions(),
                'down' => $migration->getDownInstructions(),
                'breaking' => $migration->isBreaking(),
            ];
        }

        return $instructions;
    }

    private function getMigrationFiles(Plugin $plugin, string $from, string $to): array
    {
        $migrationsPath = $plugin->getPath() . '/database/migrations/versions';

        if (!is_dir($migrationsPath)) {
            return [];
        }

        $files = glob($migrationsPath . '/*.php');

        return array_filter($files, function ($file) use ($from, $to) {
            $filename = basename($file, '.php');
            preg_match('/(\d+\.\d+\.\d+)/', $filename, $matches);
            $version = $matches[1] ?? '0.0.0';

            return version_compare($version, $from, '>')
                && version_compare($version, $to, '<=');
        });
    }
}
```

# Plugin Performance

## Resource Limits

```php
<?php

namespace App\Plugins\Performance;

class PluginResourceLimiter
{
    private array $limits = [
        'memory_mb' => 128,
        'execution_time_seconds' => 30,
        'query_count' => 100,
        'file_operations' => 50,
        'cache_operations' => 200,
    ];

    public function enforce(string $pluginSlug): void
    {
        $originalMemoryLimit = ini_get('memory_limit');
        $originalExecutionTime = ini_get('max_execution_time');

        ini_set('memory_limit', $this->limits['memory_mb'] . 'M');
        ini_set('max_execution_time', $this->limits['execution_time_seconds']);

        register_shutdown_function(function () use ($originalMemoryLimit, $originalExecutionTime) {
            ini_set('memory_limit', $originalMemoryLimit);
            ini_set('max_execution_time', $originalExecutionTime);
        });
    }

    public function trackQueryCount(string $pluginSlug): void
    {
        $key = "plugin_queries:{$pluginSlug}";
        $count = cache()->increment($key);

        if ($count > $this->limits['query_count']) {
            throw new \RuntimeException(
                "Plugin {$pluginSlug} exceeded query limit ({$this->limits['query_count']})"
            );
        }
    }

    public function getUsageStats(string $pluginSlug): array
    {
        return [
            'memory_used_mb' => round(memory_get_usage(true) / 1024 / 1024, 2),
            'memory_peak_mb' => round(memory_get_peak_usage(true) / 1024 / 1024, 2),
            'queries_executed' => cache()->get("plugin_queries:{$pluginSlug}", 0),
            'file_operations' => cache()->get("plugin_files:{$pluginSlug}", 0),
        ];
    }
}
```
