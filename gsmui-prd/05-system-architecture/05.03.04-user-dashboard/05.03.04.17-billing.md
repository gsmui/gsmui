The billing page must handle complex scenarios (proration, refunds, multiple payment methods) while remaining simple and unintuitive.

## Billing Dashboard Component

```php
// app/Livewire/Billing/BillingDashboard.php
class BillingDashboard extends Component
{
    use AuthorizesRequests;

    public ?array $summary = null;
    public array $paymentMethods = [];
    public array $recentInvoices = [];
    public bool $showUpgradeModal = false;
    public bool $showPaymentMethodModal = false;
    public bool $showCancelModal = false;
    public ?string $selectedPlan = null;
    public string $search = '';
    public int $invoicePage = 1;
    public int $invoicesPerPage = 25;
    public bool $loading = true;
    public ?string $exportStatus = null;

    public function mount(): void
    {
        $this->loadBillingData();
    }

    public function loadBillingData(): void
    {
        $this->loading = true;

        $this->summary = $this->getBillingSummary();
        $this->paymentMethods = $this->getPaymentMethods();
        $this->recentInvoices = $this->getRecentInvoices();

        $this->loading = false;
    }

    private function getBillingSummary(): array
    {
        $user = auth()->user();
        $subscription = $user->currentSubscription();

        return [
            'plan_name' => $subscription?->name ?? 'Free',
            'price' => $subscription?->monthlyPrice() ?? 0,
            'billing_cycle' => $subscription?->billingCycle ?? 'monthly',
            'next_billing_date' => $subscription?->nextBillingAt?->format('M j, Y') ?? 'N/A',
            'status' => $subscription?->status ?? 'active',
            'total_spend' => $user->invoices()->sum('amount'),
            'usage' => $this->getUsageStats(),
            'seats' => $this->getSeatUsage(),
        ];
    }

    private function getUsageStats(): array
    {
        $user = auth()->user();
        $subscription = $user->currentSubscription();

        return [
            'projects_used' => $user->projects()->count(),
            'projects_limit' => $subscription?->project_limit ?? 3,
            'storage_used' => $user->storageUsedInBytes(),
            'storage_limit' => $subscription?->storage_limit ?? 1073741824, // 1GB
            'api_calls_used' => $this->getApiCallsThisMonth(),
            'api_calls_limit' => $subscription?->api_calls_limit ?? 10000,
        ];
    }

    private function getSeatUsage(): array
    {
        $user = auth()->user();
        $team = $user->ownedTeam;

        if (!$team) {
            return ['used' => 1, 'limit' => 1];
        }

        return [
            'used' => $team->members()->count(),
            'limit' => $team->seat_limit,
        ];
    }

    private function getPaymentMethods(): array
    {
        return auth()->user()
            ->paymentMethods()
            ->orderByDesc('is_default')
            ->get()
            ->map(fn($pm) => [
                'id' => $pm->id,
                'brand' => $pm->brand,
                'last4' => $pm->last4,
                'exp_month' => $pm->exp_month,
                'exp_year' => $pm->exp_year,
                'is_default' => $pm->is_default,
            ])
            ->toArray();
    }

    private function getRecentInvoices(): array
    {
        return auth()->user()
            ->invoices()
            ->orderByDesc('created_at')
            ->limit(10)
            ->get()
            ->map(fn($invoice) => [
                'id' => $invoice->id,
                'number' => $invoice->number,
                'amount' => $invoice->amount,
                'status' => $invoice->status,
                'created_at' => $invoice->created_at->format('M j, Y'),
            ])
            ->toArray();
    }

    public function upgradePlan(string $planSlug): void
    {
        $this->authorize('manageBilling');

        $user = auth()->user();
        $currentPlan = $user->currentSubscription();
        $newPlan = Plan::where('slug', $planSlug)->firstOrFail();

        $proration = $this->calculateProration($currentPlan, $newPlan);

        $this->selectedPlan = $planSlug;
        $this->showUpgradeModal = true;

        $this->dispatch('plan-selected', [
            'plan' => $newPlan->toArray(),
            'proration' => $proration,
        ]);
    }

    public function confirmUpgrade(): void
    {
        $user = auth()->user();
        $currentPlan = $user->currentSubscription();
        $newPlan = Plan::where('slug', $this->selectedPlan)->firstOrFail();

        try {
            $user->subscription($currentPlan->stripe_price_id)
                ->swap($newPlan->stripe_price_id);

            $this->showUpgradeModal = false;
            $this->loadBillingData();

            $this->dispatch('plan-upgraded', ['plan' => $newPlan->name]);
        } catch (\Exception $e) {
            $this->dispatch('upgrade-failed', ['message' => $e->getMessage()]);
        }
    }

    public function cancelSubscription(string $reason, ?string $feedback): void
    {
        $user = auth()->user();
        $subscription = $user->currentSubscription();

        if (!$subscription) {
            return;
        }

        $subscription->cancel();

        activity()
            ->performedOn($subscription)
            ->withProperties([
                'reason' => $reason,
                'feedback' => $feedback,
            ])
            ->log('subscription.cancelled');

        $this->showCancelModal = false;
        $this->loadBillingData();

        $this->dispatch('subscription-cancelled');
    }

    public function setDefaultPaymentMethod(int $paymentMethodId): void
    {
        $user = auth()->user();

        $user->paymentMethods()->update(['is_default' => false]);
        $user->paymentMethods()->where('id', $paymentMethodId)->update(['is_default' => true]);

        $this->paymentMethods = $this->getPaymentMethods();

        $this->dispatch('default-updated');
    }

    public function removePaymentMethod(int $paymentMethodId): void
    {
        $user = auth()->user();
        $pm = $user->paymentMethods()->findOrFail($paymentMethodId);

        if ($pm->is_default && $user->paymentMethods()->count() > 1) {
            $this->dispatch('cannot-remove-default');
            return;
        }

        $pm->delete();
        $this->paymentMethods = $this->getPaymentMethods();

        $this->dispatch('payment-method-removed');
    }

    public function exportBillingData(string $format = 'csv'): void
    {
        $user = auth()->user();

        $export = BillingExport::create([
            'user_id' => $user->id,
            'format' => $format,
            'status' => 'pending',
        ]);

        GenerateBillingExport::dispatch($user, $export, $format);

        $this->exportStatus = 'processing';

        $this->dispatch('export-started', ['export_id' => $export->id]);
    }

    public function render(): View
    {
        return view('gsmui::billing.dashboard', [
            'summary' => $this->summary,
            'paymentMethods' => $this->paymentMethods,
            'recentInvoices' => $this->recentInvoices,
            'loading' => $this->loading,
        ]);
    }
}
```

## Subscription Upgrade Modal Component

```php
// app/Livewire/Billing/UpgradeModal.php
class UpgradeModal extends Component
{
    public bool $show = false;
    public ?array $plan = null;
    public ?array $proration = null;
    public bool $processing = false;
    public ?string $error = null;
    public bool $confirmed = false;

    protected $listeners = ['plan-selected' => 'open'];

    public function open(array $data): void
    {
        $this->plan = $data['plan'];
        $this->proration = $data['proration'];
        $this->show = true;
        $this->error = null;
        $this->confirmed = false;
    }

    public function confirm(): void
    {
        $this->processing = true;
        $this->error = null;

        try {
            $user = auth()->user();
            $currentPlan = $user->currentSubscription();

            if ($currentPlan && $this->proration['amount'] > 0) {
                $this->confirmed = true;
            }

            $this->dispatch('confirm-upgrade');

            $this->show = false;
        } catch (\Exception $e) {
            $this->error = $e->getMessage();
        } finally {
            $this->processing = false;
        }
    }

    public function close(): void
    {
        $this->show = false;
        $this->plan = null;
        $this->proration = null;
        $this->error = null;
    }

    public function render(): View
    {
        return view('gsmui::billing.upgrade-modal', [
            'plan' => $this->plan,
            'proration' => $this->proration,
            'processing' => $this->processing,
            'error' => $this->error,
        ]);
    }
}
```

```php
class InvoiceDetail extends Component
{
    public int $invoiceId;
    public ?array $invoice = null;
    public bool $loading = true;
    public bool $showDownloadProgress = false;

    public function mount(int $invoiceId): void
    {
        $this->invoiceId = $invoiceId;
        $this->loadInvoice();
    }

    public function loadInvoice(): void
    {
        $invoice = auth()->user()
            ->invoices()
            ->with('items')
            ->findOrFail($this->invoiceId);

        $this->invoice = [
            'id' => $invoice->id,
            'number' => $invoice->number,
            'amount' => $invoice->amount,
            'currency' => $invoice->currency,
            'status' => $invoice->status,
            'created_at' => $invoice->created_at->format('M j, Y'),
            'items' => $invoice->items->map(fn($item) => [
                'description' => $item->description,
                'amount' => $item->amount,
                'quantity' => $item->quantity,
            ])->toArray(),
            'pdf_url' => $invoice->pdf_path
                ? route('billing.invoice.pdf', $invoice->id)
                : null,
        ];

        $this->loading = false;
    }

    public function downloadPdf(): StreamedResponse
    {
        $invoice = auth()->user()
            ->invoices()
            ->findOrFail($this->invoiceId);

        if (!$invoice->pdf_path) {
            abort(404, 'Invoice PDF not available');
        }

        return Storage::disk('s3')
            ->download($invoice->pdf_path, "invoice-{$invoice->number}.pdf");
    }

    public function render(): View
    {
        return view('gsmui::billing.invoice-detail', [
            'invoice' => $this->invoice,
            'loading' => $this->loading,
        ]);
    }
}
```

```php
// app/Livewire/Billing/PaymentMethodForm.php
class PaymentMethodForm extends Component
{
    use WithFileUploads;

    public bool $show = false;
    public bool $processing = false;
    public ?string $error = null;
    public string $cardholderName = '';
    public string $billingEmail = '';
    public string $addressLine1 = '';
    public string $addressLine2 = '';
    public string $city = '';
    public string $state = '';
    public string $postalCode = '';
    public string $country = 'US';
    public bool $setAsDefault = true;

    protected array $rules = [
        'cardholderName' => 'required|string|max:255',
        'billingEmail' => 'required|email',
        'addressLine1' => 'required|string|max:255',
        'city' => 'required|string|max:255',
        'state' => 'required|string|max:255',
        'postalCode' => 'required|string|max:20',
        'country' => 'required|string|size:2',
    ];

    protected array $messages = [
        'cardholderName.required' => 'Cardholder name is required',
        'billingEmail.required' => 'Billing email is required',
        'billingEmail.email' => 'Please enter a valid email address',
        'addressLine1.required' => 'Address line 1 is required',
        'city.required' => 'City is required',
        'state.required' => 'State is required',
        'postalCode.required' => 'Postal code is required',
        'country.required' => 'Country is required',
    ];

    public function open(): void
    {
        $user = auth()->user();
        $this->cardholderName = $user->name;
        $this->billingEmail = $user->email;
        $this->show = true;
        $this->error = null;
    }

    public function submitPaymentMethod(string $stripeToken): void
    {
        $this->validate();

        $this->processing = true;
        $this->error = null;

        try {
            $user = auth()->user();

            $paymentMethod = $user->addPaymentMethod($stripeToken);

            if ($this->setAsDefault) {
                $user->updateDefaultPaymentMethod($paymentMethod->id);
            }

            $this->show = false;
            $this->dispatch('payment-method-added', [
                'payment_method_id' => $paymentMethod->id,
            ]);
        } catch (\Exception $e) {
            $this->error = $this->getStripeErrorMessage($e);
        } finally {
            $this->processing = false;
        }
    }

    private function getStripeErrorMessage(\Exception $e): string
    {
        return match($e->getStripeCode()) {
            'card_declined' => 'Your card was declined. Please try a different card.',
            'expired_card' => 'Your card has expired. Please update your card details.',
            'incorrect_cvc' => 'The CVC number is incorrect. Please check and try again.',
            'insufficient_funds' => 'Your card has insufficient funds.',
            'invalid_card_number' => 'The card number is invalid. Please check and try again.',
            'invalid_expiry_date' => 'The expiration date is invalid.',
            default => 'An error occurred processing your card. Please try again.',
        };
    }

    public function render(): View
    {
        return view('gsmui::billing.payment-method-form');
    }
}
```

# User Flows

## Upgrade Flow

1. User clicks "Upgrade to Pro" on current plan card
2. Modal opens showing new plan details and proration calculation
3. User reviews proration breakdown (credit, new plan cost, amount due)
4. User clicks "Confirm Upgrade"
5. System processes upgrade via Stripe
6. Modal closes, billing data refreshes
7. Success notification appears
8. User sees updated plan details

## Payment Method Flow

1. User clicks "Add Payment Method"
2. Modal opens with cardholder name and billing email pre-filled from profile
3. User enters card details (Stripe Elements)
4. User enters billing address
5. User checks/unchecks "Set as default"
6. User clicks "Add Payment Method"
7. Stripe tokenizes card, saves to Stripe
8. Modal closes, payment methods list refreshes
9. If default, existing default badge moves to new card

## Cancel Subscription Flow

1. User clicks "Cancel Subscription"
2. Warning modal shows what they'll lose
3. User selects cancellation reason from dropdown
4. User enters optional feedback
5. User clicks "Confirm Cancellation"
6. Subscription cancelled at end of current period
7. User sees confirmation
8. Dashboard updates to show "Free" plan

## Download Invoice PDF Flow

1. User clicks "PDF" on invoice row
2. Browser downloads PDF file
3. PDF contains invoice details, line items, payment info
4. PDF is cached after first generation

## Export Billing Data Flow

1. User clicks "Export CSV"
2. Background job starts generating export
3. Status indicator shows "Processing..."
4. Job completes, notification sent
5. User clicks download link in notification
6. CSV file downloads with all billing data

```php
        }
```

## Proration Calculation

```php
// app/Services/Billing/ProrationCalculator.php
class ProrationCalculator
{
    public function calculate(User $user, string $fromPlanSlug, string $toPlanSlug): array
    {
        $fromPlan = Plan::where('slug', $fromPlanSlug)->first();
        $toPlan = Plan::where('slug', $toPlanSlug)->first();

        if (!$fromPlan || !$toPlan) {
            throw new \InvalidArgumentException('Invalid plan specified');
        }

        $subscription = $user->currentSubscription();

        if (!$subscription) {
            return [
                'credit' => 0,
                'amount' => $toPlan->price,
                'total' => $toPlan->price,
            ];
        }

        // Calculate remaining days in current period
        $periodStart = $subscription->current_period_start;
        $periodEnd = $subscription->current_period_end;
        $now = now();

        $totalDays = $periodEnd->diffInDays($periodStart);
        $remainingDays = $periodEnd->diffInDays($now);

        // Calculate daily rates
        $fromDailyRate = $fromPlan->price / $totalDays;
        $toDailyRate = $toPlan->price / $totalDays;

        // Calculate credit for remaining time
        $credit = $fromDailyRate * $remainingDays;

        // Calculate new plan cost for remaining time
        $newPlanCost = $toDailyRate * $remainingDays;

        // Calculate amount due
        $amountDue = max(0, $newPlanCost - $credit);

        return [
            'credit' => round($credit, 2),
            'amount' => round($newPlanCost, 2),
            'total' => round($amountDue, 2),
            'remaining_days' => $remainingDays,
            'period_end' => $periodEnd->format('M j, Y'),
        ];
    }
}
```

## Dunning Management

```php
// app/Services/Billing/DunningManager.php
class DunningManager
{
    private array $retrySchedule = [
        1 => 3,    // Retry after 3 days
        2 => 7,    // Retry after 7 days
        3 => 14,   // Retry after 14 days
        4 => 21,   // Retry after 21 days
    ];

    public function startDunningCycle(Invoice $invoice, array $stripePayload): void
    {
        $dunningCycle = DunningCycle::create([
            'invoice_id' => $invoice->id,
            'user_id' => $invoice->user_id,
            'attempt' => 1,
            'status' => 'pending',
            'next_retry_at' => now()->addDays($this->retrySchedule[1]),
            'last_error' => $payload['last_finalization_error']['message'] ?? null,
        ]);

        // Send first dunning email
        $this->sendDunningEmail($invoice, $dunningCycle);
    }

    public function processRetry(DunningCycle $cycle): void
    {
        $invoice = $cycle->invoice;
        $user = $invoice->user;

        try {
            // Attempt to charge again
            $user->invoiceFor($invoice->description, $invoice->amount);

            // Success - update status
            $cycle->update(['status' => 'succeeded']);
            $invoice->update(['status' => 'paid']);

            // Send success email
            PaymentRecoveredMail::to($user)->send();

        } catch (\Exception $e) {
            // Failed again
            $nextAttempt = $cycle->attempt + 1;

            if ($nextAttempt > count($this->retrySchedule)) {
                // Max retries reached
                $cycle->update([
                    'status' => 'failed',
                    'last_error' => $e->getMessage(),
                ]);

                // Cancel subscription
                $user->currentSubscription()?->cancel();

                // Send final notice
                PaymentFailedFinalMail::to($user)->send();
            } else {
                $cycle->update([
                    'attempt' => $nextAttempt,
                    'status' => 'pending',
                    'next_retry_at' => now()->addDays($this->retrySchedule[$nextAttempt]),
                    'last_error' => $e->getMessage(),
                ]);

                // Send next dunning email
                $this->sendDunningEmail($invoice, $cycle);
            }
        }
    }

    private function sendDunningEmail(Invoice $invoice, DunningCycle $cycle): void
    {
        $attemptNumber = $cycle->attempt;

        match($attemptNumber) {
            1 => DunningEmailAttempt1::to($invoice->user)->send(),
            2 => DunningEmailAttempt2::to($invoice->user)->send(),
            3 => DunningEmailAttempt3::to($invoice->user)->send(),
            default => DunningEmailFinal::to($invoice->user)->send(),
        };
    }
}
```

```php
    private int $summaryTtl = 300; // 5 minutes
    private int $invoicesTtl = 120; // 2 minutes
    private int $paymentMethodsTtl = 600; // 10 minutes

    public function getSummary(User $user): array
    {
        $cacheKey = "billing_summary:{$user->id}";

        return Cache::remember($cacheKey, $this->summaryTtl, function () use ($user) {
            return $this->computeSummary($user);
        });
    }

    public function getInvoices(User $user, int $page = 1, int $perPage = 25): array
    {
        $cacheKey = "recent_invoices:{$user->id}:{$page}:{$perPage}";

        return Cache::remember($cacheKey, $this->invoicesTtl, function () use ($user, $page, $perPage) {
            return $user->invoices()
                ->orderByDesc('created_at')
                ->offset(($page - 1) * $perPage)
                ->limit($perPage)
                ->get()
                ->toArray();
        });
    }

    public function getPaymentMethods(User $user): array
    {
        $cacheKey = "payment_methods:{$user->id}";

        return Cache::remember($cacheKey, $this->paymentMethodsTtl, function () use ($user) {
            return $user->paymentMethods()
                ->orderByDesc('is_default')
                ->get()
                ->toArray();
        });
    }

    public function invalidate(User $user): void
    {
        Cache::forget("billing_summary:{$user->id}");
        Cache::forget("recent_invoices:{$user->id}");
        Cache::forget("payment_methods:{$user->id}");

        // Invalidate paginated caches
        for ($i = 1; $i <= 10; $i++) {
            Cache::forget("recent_invoices:{$user->id}:{$i}:25");
        }
    }

    public function invalidateAll(): void
    {
        $user = auth()->user();
        if ($user) {
            $this->invalidate($user);
        }
    }
}
```

```php
class GenerateBillingExport implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public function __construct(
        public User $user,
        public BillingExport $export,
        public string $format,
    ) {}

    public function handle(): void
    {
        try {
            $this->export->update(['status' => 'processing']);

            $data = $this->gatherBillingData();

            $filePath = $this->generateFile($data);

            $this->export->update([
                'status' => 'completed',
                'file_path' => $filePath,
                'expires_at' => now()->addDays(7),
            ]);

            BillingExportReadyMail::to($this->user)->send();

        } catch (\Exception $e) {
            $this->export->update([
                'status' => 'failed',
                'error' => $e->getMessage(),
            ]);
        }
    }

    private function gatherBillingData(): array
    {
        return [
            'invoices' => $this->user->invoices()->get()->toArray(),
            'payments' => $this->user->payments()->get()->toArray(),
            'payment_methods' => $this->user->paymentMethods()->get()->toArray(),
            'subscriptions' => $this->user->subscriptions()->get()->toArray(),
        ];
    }

    private function generateFile(array $data): string
    {
        $fileName = "billing-export-{$this->user->id}-" . now()->format('Y-m-d') . ".{$this->format}";
        $filePath = "exports/{$this->user->id}/{$fileName}";

        if ($this->format === 'csv') {
            $csv = $this->convertToCsv($data);
            Storage::disk('s3')->put($filePath, $csv);
        } else {
            $json = json_encode($data, JSON_PRETTY_PRINT);
            Storage::disk('s3')->put($filePath, $json);
        }

        return $filePath;
    }

    private function convertToCsv(array $data): string
    {
        $output = fopen('php://temp', 'r+');

        // Invoice headers
        fputcsv($output, ['Invoice Number', 'Date', 'Amount', 'Status']);

        foreach ($data['invoices'] as $invoice) {
            fputcsv($output, [
                $invoice['number'],
                $invoice['created_at'],
                $invoice['amount'],
                $invoice['status'],
            ]);
        }

        rewind($output);
        $csv = stream_get_contents($output);
        fclose($output);

        return $csv;
    }
}
```

```php
class ReconcileStripeBilling implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public function handle(): void
    {
        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        // Fetch recent Stripe invoices
        $stripeInvoices = $stripe->invoices->all([
            'limit' => 100,
            'created' => ['gte' => now()->subDay()->timestamp],
        ]);

        foreach ($stripeInvoices->data as $stripeInvoice) {
            $localInvoice = Invoice::where('stripe_id', $stripeInvoice->id)->first();

            if ($localInvoice && $localInvoice->status !== $stripeInvoice->status) {
                $localInvoice->update(['status' => $stripeInvoice->status]);

                Log::warning('Invoice status mismatch reconciled', [
                    'invoice_id' => $localInvoice->id,
                    'local_status' => $localInvoice->getOriginal('status'),
                    'stripe_status' => $stripeInvoice->status,
                ]);
            }
        }
    }
}
```

```php
// app/Http/Middleware/BillingSecurity.php
class BillingSecurity
{
    public function handle(Request $request, Closure $next): Response
    {
        // Rate limiting
        if (RateLimiter::tooManyAttempts('billing:' . auth()->id(), 30)) {
            abort(429, 'Too many billing requests. Please try again later.');
        }

        RateLimiter::hit('billing:' . auth()->id(), 60);

        // CSRF protection (handled by Laravel)

        // Verify Stripe webhook signature
        if ($request->is('api/stripe/webhook')) {
            $payload = @file_get_contents('php://input');
            $sigHeader = $_SERVER['HTTP_STRIPE_SIGNATURE'];

            try {
                \Stripe\Webhook::constructEvent(
                    $payload,
                    $sigHeader,
                    config('services.stripe.webhook_secret')
                );
            } catch (\UnexpectedValueException $e) {
                abort(400, 'Invalid payload');
            } catch (\Stripe\Exception\SignatureVerificationException $e) {
                abort(400, 'Invalid signature');
            }
        }

        return $next($request);
    }
}
```

```php
// app/Http/Controllers/BillingController.php
class BillingController extends Controller
{
    public function summary(): JsonResponse
    {
        $user = auth()->user();
        $summary = app(BillingCache::class)->getSummary($user);

        return response()->json($summary);
    }

    public function invoices(): JsonResponse
    {
        $user = auth()->user();
        $page = request()->input('page', 1);
        $perPage = request()->input('per_page', 25);

        $invoices = app(BillingCache::class)->getInvoices($user, $page, $perPage);

        return response()->json([
            'invoices' => $invoices,
            'meta' => [
                'current_page' => $page,
                'per_page' => $perPage,
                'total' => $user->invoices()->count(),
            ],
        ]);
    }

    public function showInvoice(int $id): JsonResponse
    {
        $invoice = auth()->user()->invoices()->findOrFail($id);

        return response()->json([
            'id' => $invoice->id,
            'number' => $invoice->number,
            'amount' => $invoice->amount,
            'currency' => $invoice->currency,
            'status' => $invoice->status,
            'created_at' => $invoice->created_at->toISOString(),
            'items' => $invoice->items,
        ]);
    }

    public function addPaymentMethod(Request $request): JsonResponse
    {
        $request->validate([
            'token' => 'required|string',
            'set_as_default' => 'boolean',
        ]);

        $user = auth()->user();
        $paymentMethod = $user->addPaymentMethod($request->token);

        if ($request->boolean('set_as_default')) {
            $user->updateDefaultPaymentMethod($paymentMethod->id);
        }

        return response()->json([
            'id' => $paymentMethod->id,
            'brand' => $paymentMethod->card->brand,
            'last4' => $paymentMethod->card->last4,
        ]);
    }

    public function upgrade(Request $request): JsonResponse
    {
        $request->validate([
            'plan' => 'required|string|exists:plans,slug',
        ]);

        $user = auth()->user();
        $plan = Plan::where('slug', $request->plan)->firstOrFail();

        $user->subscription($user->currentSubscription()->stripe_price_id)
            ->swap($plan->stripe_price_id);

        return response()->json(['message' => 'Plan upgraded successfully']);
    }

    public function export(Request $request): JsonResponse
    {
        $request->validate([
            'format' => 'required|in:csv,json',
        ]);

        $export = BillingExport::create([
            'user_id' => auth()->id(),
            'format' => $request->format,
            'status' => 'pending',
        ]);

        GenerateBillingExport::dispatch(auth()->user(), $export, $request->format);

        return response()->json([
            'export_id' => $export->id,
            'status' => 'processing',
        ]);
    }
}
```
