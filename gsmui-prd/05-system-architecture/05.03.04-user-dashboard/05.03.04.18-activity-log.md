# Conclusion

Activity Log provides users with a comprehensive, searchable history of every action taken on their GsmUi account. It serves as both a personal audit trail and a transparency tool — users should never wonder "what did I do?" or "when did that happen?" The log must be fast to query, easy to filter, and never lose data. For team accounts, it doubles as an accountability mechanism.

# Performance

| Metric | Target | Strategy |
|--------|--------|----------|
| Activity log page load | < 300ms | Paginated (50/page), indexed query |
| Search results | < 200ms | Full-text index on action + metadata |
| Activity write (log event) | < 10ms | Async queue, fire-and-forget |
| Export 5000 records | < 10s | Background job, streamed CSV |
| Real-time broadcast | < 2s | Laravel Echo, minimal payload |
| Archive old records | < 1hr/100K records | Weekly compression job |

**Storage Strategy:**
- Recent (0-30 days): Hot storage, full detail, indexed
- Warm (30-60 days): Compressed, reduced index
- Cold (60-90 days): Archive table, export-only
- Purge (90+ days): Auto-deleted, configurable retention

**Caching:**
- Activity count: Redis counter per user
- Recent activities: 30-second cache per page
- Search results: 1-minute cache per query
- Team activity: 2-minute cache per team

## Activity Log Component

```php
// app/Livewire/Activity/ActivityLog.php
class ActivityLog extends Component
{
    use AuthorizesRequests;

    public array $activities = [];
    public int $totalCount = 0;
    public int $currentPage = 1;
    public int $perPage = 50;
    public string $search = '';
    public ?string $filterType = null;
    public ?string $filterDateFrom = null;
    public ?string $filterDateTo = null;
    public bool $loading = true;
    public bool $exporting = false;
    public ?string $exportStatus = null;
    public array $activityTypes = [
        'account' => 'Account',
        'content' => 'Content',
        'billing' => 'Billing',
        'security' => 'Security',
    ];

    protected $listeners = [
        'activity-logged' => 'loadActivities',
        'refresh-activities' => 'loadActivities',
    ];

    public function mount(): void
    {
        $this->loadActivities();
    }

    public function loadActivities(): void
    {
        $this->loading = true;

        $query = auth()->user()->activities();

        if ($this->search) {
            $query->where(function ($q) {
                $q->where('action', 'LIKE', "%{$this->search}%")
                  ->orWhere('description', 'LIKE', "%{$this->search}%");
            });
        }

        if ($this->filterType) {
            $query->where('type', $this->filterType);
        }

        if ($this->filterDateFrom) {
            $query->where('created_at', '>=', $this->filterDateFrom);
        }

        if ($this->filterDateTo) {
            $query->where('created_at', '<=', $this->filterDateTo);
        }

        $this->totalCount = $query->count();

        $this->activities = $query
            ->orderByDesc('created_at')
            ->offset(($this->currentPage - 1) * $this->perPage)
            ->limit($this->perPage)
            ->get()
            ->map(fn($activity) => [
                'id' => $activity->id,
                'type' => $activity->type,
                'action' => $activity->action,
                'description' => $activity->description,
                'metadata' => $activity->metadata,
                'ip_address' => $activity->ip_address,
                'user_agent' => $activity->user_agent,
                'created_at' => $activity->created_at->format('M j, Y g:i A'),
                'relative_time' => $activity->created_at->diffForHumans(),
            ])
            ->toArray();

        $this->loading = false;
    }

    public function updatedSearch(): void
    {
        $this->currentPage = 1;
        $this->loadActivities();
    }

    public function updatedFilterType(): void
    {
        $this->currentPage = 1;
        $this->loadActivities();
    }

    public function clearFilters(): void
    {
        $this->search = '';
        $this->filterType = null;
        $this->filterDateFrom = null;
        $this->filterDateTo = null;
        $this->currentPage = 1;
        $this->loadActivities();
    }

    public function setPage(int $page): void
    {
        $this->currentPage = $page;
        $this->loadActivities();
    }

    public function getTotalPages(): int
    {
        return (int) ceil($this->totalCount / $this->perPage);
    }

    public function exportActivities(string $format = 'csv'): void
    {
        $this->exporting = true;

        $user = auth()->user();

        $export = ActivityExport::create([
            'user_id' => $user->id,
            'format' => $format,
            'filters' => [
                'type' => $this->filterType,
                'date_from' => $this->filterDateFrom,
                'date_to' => $this->filterDateTo,
                'search' => $this->search,
            ],
            'status' => 'pending',
        ]);

        GenerateActivityExport::dispatch($user, $export, $format);

        $this->exportStatus = 'processing';
        $this->exporting = false;

        $this->dispatch('export-started', ['export_id' => $export->id]);
    }

    public function getActivityIcon(string $type): string
    {
        return match($type) {
            'account' => 'user',
            'content' => 'download',
            'billing' => 'credit-card',
            'security' => 'shield',
            default => 'activity',
        };
    }

    public function getActivityColor(string $type): string
    {
        return match($type) {
            'account' => 'text-blue-600 dark:text-blue-400',
            'content' => 'text-green-600 dark:text-green-400',
            'billing' => 'text-amber-600 dark:text-amber-400',
            'security' => 'text-red-600 dark:text-red-400',
            default => 'text-zinc-600 dark:text-zinc-400',
        };
    }

    public function getSummaryStats(): array
    {
        $user = auth()->user();

        return [
            'total_activities' => $user->activities()->count(),
            'today_count' => $user->activities()->whereDate('created_at', today())->count(),
            'week_count' => $user->activities()->where('created_at', '>=', now()->subWeek())->count(),
            'month_count' => $user->activities()->where('created_at', '>=', now()->subMonth())->count(),
            'by_type' => $user->activities()
                ->selectRaw('type, COUNT(*) as count')
                ->groupBy('type')
                ->pluck('count', 'type')
                ->toArray(),
        ];
    }

    public function render(): View
    {
        return view('gsmui::activity.log', [
            'activities' => $this->activities,
            'totalCount' => $this->totalCount,
            'totalPages' => $this->getTotalPages(),
            'activityTypes' => $this->activityTypes,
            'summaryStats' => $this->getSummaryStats(),
        ]);
    }
}
```

```php
// app/Livewire/Activity/ActivityDetail.php
class ActivityDetail extends Component
{
    public int $activityId;
    public ?array $activity = null;
    public bool $loading = true;

    public function mount(int $activityId): void
    {
        $this->activityId = $activityId;
        $this->loadActivity();
    }

    public function loadActivity(): void
    {
        $activity = auth()->user()
            ->activities()
            ->findOrFail($this->activityId);

        $this->activity = [
            'id' => $activity->id,
            'type' => $activity->type,
            'action' => $activity->action,
            'description' => $activity->description,
            'metadata' => $activity->metadata,
            'ip_address' => $activity->ip_address,
            'user_agent' => $activity->user_agent,
            'created_at' => $activity->created_at->format('M j, Y g:i A'),
            'created_at_iso' => $activity->created_at->toIso8601String(),
            'browser_info' => $this->parseUserAgent($activity->user_agent),
        ];

        $this->loading = false;
    }

    private function parseUserAgent(?string $userAgent): array
    {
        if (!$userAgent) {
            return ['browser' => 'Unknown', 'os' => 'Unknown', 'device' => 'Unknown'];
        }

        // Simple parser - in production use a proper UA parser
        $browser = 'Unknown';
        $os = 'Unknown';
        $device = 'Desktop';

        if (str_contains($userAgent, 'Chrome')) $browser = 'Chrome';
        elseif (str_contains($userAgent, 'Firefox')) $browser = 'Firefox';
        elseif (str_contains($userAgent, 'Safari')) $browser = 'Safari';
        elseif (str_contains($userAgent, 'Edge')) $browser = 'Edge';

        if (str_contains($userAgent, 'Windows')) $os = 'Windows';
        elseif (str_contains($userAgent, 'Mac OS')) $os = 'macOS';
        elseif (str_contains($userAgent, 'Linux')) $os = 'Linux';
        elseif (str_contains($userAgent, 'Android')) $os = 'Android';
        elseif (str_contains($userAgent, 'iPhone') || str_contains($userAgent, 'iPad')) $os = 'iOS';

        if (str_contains($userAgent, 'Mobile') || str_contains($userAgent, 'iPhone')) $device = 'Mobile';
        elseif (str_contains($userAgent, 'iPad')) $device = 'Tablet';

        return [
            'browser' => $browser,
            'os' => $os,
            'device' => $device,
            'full_string' => $userAgent,
        ];
    }

    public function render(): View
    {
        return view('gsmui::activity.detail', [
            'activity' => $this->activity,
            'loading' => $this->loading,
        ]);
    }
}
```

```php
// app/Livewire/Activity/TeamActivity.php
class TeamActivity extends Component
{
    use AuthorizesRequests;

    public int $teamId;
    public array $activities = [];
    public int $totalCount = 0;
    public int $currentPage = 1;
    public int $perPage = 50;
    public string $search = '';
    public ?string $filterMemberId = null;
    public bool $loading = true;

    public function mount(int $teamId): void
    {
        $this->teamId = $teamId;
        $this->authorize('viewActivity', Team::find($teamId));
        $this->loadActivities();
    }

    public function loadActivities(): void
    {
        $this->loading = true;

        $query = TeamActivity::where('team_id', $this->teamId)->with('actor');

        if ($this->search) {
            $query->where(function ($q) {
                $q->where('action', 'LIKE', "%{$this->search}%")
                  ->orWhere('description', 'LIKE', "%{$this->search}%");
            });
        }

        if ($this->filterMemberId) {
            $query->where('actor_id', $this->filterMemberId);
        }

        $this->totalCount = $query->count();

        $this->activities = $query
            ->orderByDesc('created_at')
            ->offset(($this->currentPage - 1) * $this->perPage)
            ->limit($this->perPage)
            ->get()
            ->map(fn($activity) => [
                'id' => $activity->id,
                'actor' => $activity->actor ? [
                    'id' => $activity->actor->id,
                    'name' => $activity->actor->name,
                    'avatar' => $activity->actor->avatar_url,
                ] : null,
                'action' => $activity->action,
                'description' => $activity->description,
                'metadata' => $activity->metadata,
                'ip_address' => $activity->ip_address,
                'created_at' => $activity->created_at->format('M j, Y g:i A'),
                'relative_time' => $activity->created_at->diffForHumans(),
            ])
            ->toArray();

        $this->loading = false;
    }

    public function getTeamMembers(): array
    {
        return TeamMember::where('team_id', $this->teamId)
            ->with('user')
            ->get()
            ->map(fn($member) => [
                'id' => $member->user_id,
                'name' => $member->user->name,
            ])
            ->toArray();
    }

    public function render(): View
    {
        return view('gsmui::activity.team-activity', [
            'activities' => $this->activities,
            'totalCount' => $this->totalCount,
            'teamMembers' => $this->getTeamMembers(),
        ]);
    }
}
```

```php
// app/Listeners/LogUserActivity.php
class LogUserActivity
{
    public function __construct(
        private ActivityLogger $logger,
    ) {}

    public function handle(object $event): void
    {
        $user = $this->resolveUser($event);

        if (!$user) {
            return;
        }

        $this->logger->log(
            user: $user,
            type: $this->resolveType($event),
            action: $this->resolveAction($event),
            description: $this->resolveDescription($event),
            metadata: $this->resolveMetadata($event),
        );
    }

    private function resolveUser(object $event): ?User
    {
        return match(true) {
            $event instanceof Login => $event->user,
            $event instanceof PasswordChanged => $event->user,
            $event instanceof ProfileUpdated => $event->user,
            $event instanceof ComponentDownloaded => $event->user,
            $event instanceof InvoiceGenerated => $event->user,
            default => null,
        };
    }

    private function resolveType(object $event): string
    {
        return match(true) {
            $event instanceof Login,
            $event instanceof Logout,
            $event instanceof PasswordChanged,
            $event instanceof TwoFactorChanged => 'account',

            $event instanceof ComponentDownloaded,
            $event instanceof TemplatePurchased,
            $event instanceof PlaygroundSaved,
            $event instanceof FavoriteAdded => 'content',

            $event instanceof InvoiceGenerated,
            $event instanceof PaymentSucceeded,
            $event instanceof PaymentFailed,
            $event instanceof SubscriptionChanged => 'billing',

            $event instanceof SuspiciousActivity,
            $event instanceof FailedLoginAttempt => 'security',

            default => 'account',
        };
    }

    private function resolveAction(object $event): string
    {
        return class_basename($event);
    }

    private function resolveDescription(object $event): string
    {
        return match(true) {
            $event instanceof Login => 'Logged in from ' . $event->ipAddress,
            $event instanceof Logout => 'Logged out',
            $event instanceof PasswordChanged => 'Changed password',
            $event instanceof ComponentDownloaded => 'Downloaded ' . $event->component->name,
            $event instanceof TemplatePurchased => 'Purchased template ' . $event->template->name,
            $event instanceof InvoiceGenerated => 'Invoice ' . $event->invoice->number . ' generated',
            $event instanceof PaymentSucceeded => 'Payment of $' . $event->payment->amount . ' succeeded',
            $event instanceof PaymentFailed => 'Payment failed',
            default => class_basename($event),
        };
    }

    private function resolveMetadata(object $event): array
    {
        return match(true) {
            $event instanceof Login => [
                'ip_address' => $event->ipAddress,
                'user_agent' => $event->userAgent,
            ],
            $event instanceof ComponentDownloaded => [
                'component_id' => $event->component->id,
                'component_name' => $event->component->name,
                'version' => $event->component->version,
            ],
            default => [],
        };
    }
}
```

```php
// app/Services/ActivityLogger.php
class ActivityLogger
{
    public function log(
        User $user,
        string $type,
        string $action,
        string $description,
        array $metadata = [],
    ): ActivityLogEntry {
        $activity = ActivityLogEntry::create([
            'user_id' => $user->id,
            'type' => $type,
            'action' => $action,
            'description' => $description,
            'metadata' => $metadata,
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
            'created_at' => now(),
        ]);

        // Update counter cache
        $this->updateCounterCache($user);

        // Broadcast real-time update
        broadcast(new ActivityLoggedEvent($user, $activity))->toOthers();

        // Check for anomalies
        $this->checkForAnomalies($user, $activity);

        return $activity;
    }

    private function updateCounterCache(User $user): void
    {
        $key = "activity_count:{$user->id}";
        Cache::increment($key);
    }

    private function checkForAnomalies(User $user, ActivityLogEntry $activity): void
    {
        // Check for multiple failed logins
        if ($activity->action === 'FailedLoginAttempt') {
            $recentFailed = ActivityLogEntry::where('user_id', $user->id)
                ->where('action', 'FailedLoginAttempt')
                ->where('created_at', '>=', now()->subMinutes(15))
                ->count();

            if ($recentFailed >= 5) {
                // Alert user about suspicious activity
                SuspiciousActivityAlert::dispatch($user, $recentFailed);
            }
        }

        // Check for login from new location
        if ($activity->action === 'Login') {
            $knownIps = ActivityLogEntry::where('user_id', $user->id)
                ->where('action', 'Login')
                ->where('created_at', '>=', now()->subDays(30))
                ->pluck('ip_address')
                ->unique()
                ->toArray();

            if (!in_array($activity->ip_address, $knownIps)) {
                // Send new location notification
                NewLocationLoginMail::to($user)->send();
            }
        }
    }
}
```

```php
        $this->searchIndex = app('meilisearch.index.activities');
    }

    public function indexActivity(ActivityLogEntry $activity): void
    {
        $this->searchIndex->addDocuments([
            [
                'id' => $activity->id,
                'user_id' => $activity->user_id,
                'type' => $activity->type,
                'action' => $activity->action,
                'description' => $activity->description,
                'metadata' => json_encode($activity->metadata),
                'created_at' => $activity->created_at->timestamp,
                'searchable_content' => $this->buildSearchableContent($activity),
            ],
        ]);
    }

    public function search(User $user, string $query, array $filters = []): array
    {
        $searchFilters = ["user_id = {$user->id}"];

        if (isset($filters['type'])) {
            $searchFilters[] = "type = '{$filters['type']}'";
        }

        if (isset($filters['from'])) {
            $searchFilters[] = "created_at >= " . strtotime($filters['from']);
        }

        if (isset($filters['to'])) {
            $searchFilters[] = "created_at <= " . strtotime($filters['to']);
        }

        $results = $this->searchIndex->search($query, [
            'filter' => implode(' AND ', $searchFilters),
            'limit' => $filters['limit'] ?? 50,
            'offset' => $filters['offset'] ?? 0,
        ]);

        return [
            'hits' => $results['hits'],
            'nbHits' => $results['nbHits'],
            'processingTimeMs' => $results['processingTimeMs'],
        ];
    }

    public function deleteActivity(ActivityLogEntry $activity): void
    {
        $this->searchIndex->deleteDocument($activity->id);
    }

    private function buildSearchableContent(ActivityLogEntry $activity): string
    {
        return implode(' ', [
            $activity->action,
            $activity->description,
            $activity->ip_address ?? '',
            $activity->user_agent ?? '',
        ]);
    }
}
```

```php
// app/Services/ActivityExportService.php
class ActivityExportService
{
    public function generateCsv(User $user, array $filters = []): string
    {
        $query = $user->activities();

        $this->applyFilters($query, $filters);

        $activities = $query->orderByDesc('created_at')->get();

        $output = fopen('php://temp', 'r+');

        // Headers
        fputcsv($output, [
            'Date',
            'Time',
            'Type',
            'Action',
            'Description',
            'IP Address',
            'Browser',
        ]);

        // Data
        foreach ($activities as $activity) {
            fputcsv($output, [
                $activity->created_at->format('Y-m-d'),
                $activity->created_at->format('H:i:s'),
                $activity->type,
                $activity->action,
                $activity->description,
                $activity->ip_address ?? '',
                $activity->user_agent ?? '',
            ]);
        }

        rewind($output);
        $csv = stream_get_contents($output);
        fclose($output);

        return $csv;
    }

    public function generateJson(User $user, array $filters = []): string
    {
        $query = $user->activities();

        $this->applyFilters($query, $filters);

        $activities = $query->orderByDesc('created_at')->get();

        $data = $activities->map(fn($activity) => [
            'id' => $activity->id,
            'type' => $activity->type,
            'action' => $activity->action,
            'description' => $activity->description,
            'metadata' => $activity->metadata,
            'ip_address' => $activity->ip_address,
            'user_agent' => $activity->user_agent,
            'created_at' => $activity->created_at->toIso8601String(),
        ])->toArray();

        return json_encode($data, JSON_PRETTY_PRINT);
    }

    private function applyFilters($query, array $filters): void
    {
        if (isset($filters['type'])) {
            $query->where('type', $filters['type']);
        }

        if (isset($filters['date_from'])) {
            $query->where('created_at', '>=', $filters['date_from']);
        }

        if (isset($filters['date_to'])) {
            $query->where('created_at', '<=', $filters['date_to']);
        }

        if (isset($filters['search'])) {
            $query->where(function ($q) use ($filters) {
                $q->where('action', 'LIKE', "%{$filters['search']}%")
                  ->orWhere('description', 'LIKE', "%{$filters['search']}%");
            });
        }
    }
}
```

```php
                'status' => 'failed',

            ]);

    }
```

```php
// app/Http/Middleware/ActivitySecurity.php
class ActivitySecurity
{
    public function handle(Request $request, Closure $next): Response
    {
        // Rate limiting for activity queries
        if (RateLimiter::tooManyAttempts('activity:' . auth()->id(), 100)) {
            abort(429, 'Too many requests. Please try again later.');
        }

        RateLimiter::hit('activity:' . auth()->id(), 60);

        // Verify user can only view their own activities
        if ($request->route('activityId')) {
            $activityId = $request->route('activityId');
            $activity = ActivityLogEntry::find($activityId);

            if (!$activity || $activity->user_id !== auth()->id()) {
                abort(404);
            }
        }

        return $next($request);
    }
}
```

```php
// app/Http/Controllers/ActivityController.php
class ActivityController extends Controller
{
    public function index(): JsonResponse
    {
        $user = auth()->user();
        $page = request()->input('page', 1);
        $perPage = request()->input('per_page', 50);

        $query = $user->activities()->orderByDesc('created_at');

        if (request()->has('type')) {
            $query->where('type', request()->input('type'));
        }

        if (request()->has('from')) {
            $query->where('created_at', '>=', request()->input('from'));
        }

        if (request()->has('to')) {
            $query->where('created_at', '<=', request()->input('to'));
        }

        $total = $query->count();

        $activities = $query
            ->offset(($page - 1) * $perPage)
            ->limit($perPage)
            ->get()
            ->map(fn($activity) => [
                'id' => $activity->id,
                'type' => $activity->type,
                'action' => $activity->action,
                'description' => $activity->description,
                'metadata' => $activity->metadata,
                'created_at' => $activity->created_at->toISOString(),
            ]);

        return response()->json([
            'activities' => $activities,
            'meta' => [
                'current_page' => $page,
                'per_page' => $perPage,
                'total' => $total,
            ],
        ]);
    }

    public function search(): JsonResponse
    {
        $query = request()->input('q', '');

        if (strlen($query) < 2) {
            return response()->json(['error' => 'Query too short'], 400);
        }

        $results = app(ActivitySearchService::class)->search(
            auth()->user(),
            $query,
            [
                'type' => request()->input('type'),
                'from' => request()->input('from'),
                'to' => request()->input('to'),
                'limit' => request()->input('limit', 50),
            ]
        );

        return response()->json($results);
    }

    public function show(int $id): JsonResponse
    {
        $activity = auth()->user()->activities()->findOrFail($id);

        return response()->json([
            'id' => $activity->id,
            'type' => $activity->type,
            'action' => $activity->action,
            'description' => $activity->description,
            'metadata' => $activity->metadata,
            'ip_address' => $activity->ip_address,
            'user_agent' => $activity->user_agent,
            'created_at' => $activity->created_at->toISOString(),
        ]);
    }

    public function export(): JsonResponse
    {
        $request->validate([
            'format' => 'required|in:csv,json',
        ]);

        $export = ActivityExport::create([
            'user_id' => auth()->id(),
            'format' => $request->format,
            'filters' => request()->only(['type', 'from', 'to', 'search']),
            'status' => 'pending',
        ]);

        GenerateActivityExport::dispatch(auth()->user(), $export, $request->format);

        return response()->json([
            'export_id' => $export->id,
            'status' => 'processing',
        ]);
    }

    public function downloadExport(int $id): StreamedResponse
    {
        $export = ActivityExport::where('user_id', auth()->id())->findOrFail($id);

        if ($export->status !== 'completed') {
            abort(404, 'Export not ready');
        }

        if ($export->expires_at < now()) {
            abort(410, 'Export expired');
        }

        return Storage::disk('s3')
            ->download($export->file_path, "activity-export.{$export->format}");
    }
}
```

```php
// app/Services/ActivityStorageOptimizer.php
class ActivityStorageOptimizer
{
    public function compressActivity(ActivityLogEntry $activity): array
    {
        return [
            'id' => $activity->id,
            'user_id' => $activity->user_id,
            'type' => $activity->type,
            'action' => $activity->action,
            'description' => $activity->description,
            'metadata' => gzcompress(json_encode($activity->metadata), 6),
            'created_at' => $activity->created_at->timestamp,
        ];
    }

    public function getStorageStats(): array
    {
        $totalRecords = ActivityLogEntry::count();
        $totalArchives = ActivityArchive::count();

        $recentSize = $this->estimateTableSize('activity_log');
        $archiveSize = $this->estimateTableSize('activity_archives');

        return [
            'total_records' => $totalRecords,
            'total_archives' => $totalArchives,
            'recent_size_mb' => round($recentSize / 1024 / 1024, 2),
            'archive_size_mb' => round($archiveSize / 1024 / 1024, 2),
            'estimated_monthly_growth' => $this->estimateMonthlyGrowth(),
        ];
    }

    private function estimateTableSize(string $table): int
    {
        $result = DB::select("SELECT pg_total_relation_size(?) as size", [$table]);
        return $result[0]->size ?? 0;
    }

    private function estimateMonthlyGrowth(): int
    {
        return ActivityLogEntry::where('created_at', '>=', now()->subMonth())->count();
    }
}
```
