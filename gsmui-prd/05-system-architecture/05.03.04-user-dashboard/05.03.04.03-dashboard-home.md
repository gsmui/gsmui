## DashboardHome Component

```php
// app/Livewire/Dashboard/Home.php
class DashboardHome extends Component
{
    public ?string $greeting = null;
    public array $subscription = [];
    public array $quickActions = [];
    public array $activities = [];
    public array $recommendations = [];
    public array $widgets = [];
    public int $notificationCount = 0;
    public bool $loading = false;

    protected $listeners = [
        'widgetReordered' => 'refreshWidgets',
        'notificationReceived' => 'updateNotificationCount',
        'subscriptionUpdated' => 'refreshSubscription',
    ];

    public function mount(): void
    {
        $this->loading = true;
        $this->loadDashboardData();
        $this->loading = false;
    }

    public function loadDashboardData(): void
    {
        $user = auth()->user();

        $this->greeting = $this->generateGreeting();
        $this->subscription = $this->getSubscriptionStatus($user);
        $this->quickActions = $this->getQuickActions($user);
        $this->activities = $this->getRecentActivities($user);
        $this->recommendations = $this->getRecommendations($user);
        $this->widgets = $this->getWidgetConfig($user);
        $this->notificationCount = $this->getUnreadNotificationCount($user);
    }

    private function generateGreeting(): string
    {
        $hour = (int) now()->format('H');
        $name = auth()->user()->first_name;

        return match(true) {
            $hour < 12 => "Good morning, {$name}",
            $hour < 17 => "Good afternoon, {$name}",
            default => "Good evening, {$name}",
        };
    }

    private function getSubscriptionStatus(User $user): array
    {
        return Cache::remember(
            "dashboard:subscription:{$user->id}",
            300,
            fn() => $user->currentSubscription()?->toArray() ?? [
                'plan' => 'free',
                'status' => 'active',
                'expires_at' => null,
                'usage' => ['downloads' => 0, 'projects' => 0],
            ]
        );
    }

    private function getQuickActions(User $user): array
    {
        return [
            ['key' => 'playground', 'label' => 'New Playground', 'icon' => 'beaker', 'url' => '/playground/create'],
            ['key' => 'downloads', 'label' => 'Downloads', 'icon' => 'arrow-down-tray', 'url' => '/downloads'],
            ['key' => 'showcase', 'label' => 'Upload to Showcase', 'icon' => 'cloud-arrow-up', 'url' => '/showcase/upload'],
            ['key' => 'team', 'label' => 'Manage Team', 'icon' => 'users', 'url' => '/team'],
            ['key' => 'templates', 'label' => 'Template Library', 'icon' => 'document-duplicate', 'url' => '/templates'],
            ['key' => 'billing', 'label' => 'Billing', 'icon' => 'credit-card', 'url' => '/billing'],
        ];
    }

    private function getRecentActivities(User $user): array
    {
        return Cache::remember(
            "dashboard:activities:{$user->id}",
            300,
            fn() => ActivityLog::where('user_id', $user->id)
                ->orderBy('created_at', 'desc')
                ->limit(10)
                ->get()
                ->map(fn($activity) => [
                    'id' => $activity->id,
                    'type' => $activity->type,
                    'action' => $activity->action,
                    'description' => $activity->description,
                    'icon' => $this->getActivityIcon($activity->type),
                    'timestamp' => $activity->created_at->diffForHumans(),
                ])
                ->toArray()
        );
    }

    public function refreshWidgets(): void
    {
        $this->widgets = $this->getWidgetConfig(auth()->user());
    }

    public function updateNotificationCount(int $count): void
    {
        $this->notificationCount = $count;
    }

    public function refreshSubscription(): void
    {
        Cache::forget("dashboard:subscription:" . auth()->id());
        $this->subscription = $this->getSubscriptionStatus(auth()->user());
    }

    public function render()
    {
        return view('livewire.dashboard.home')->layout('layouts.dashboard');
    }
}
```

```php
// app/Livewire/Dashboard/SubscriptionStatusCard.php
class SubscriptionStatusCard extends Component
{
    public array $subscription = [];
    public float $usagePercent = 0;
    public string $usageColor = 'green';
    public bool $showUpgradeModal = false;

    public function mount(array $subscription): void
    {
        $this->subscription = $subscription;
        $this->calculateUsage();
    }

    private function calculateUsage(): void
    {
        $used = $this->subscription['usage']['downloads'] ?? 0;
        $limit = $this->getDownloadLimit();
        $this->usagePercent = $limit > 0 ? ($used / $limit) * 100 : 0;

        $this->usageColor = match(true) {
            $this->usagePercent < 50 => 'green',
            $this->usagePercent < 80 => 'amber',
            default => 'red',
        };
    }

    private function getDownloadLimit(): int
    {
        return match($this->subscription['plan'] ?? 'free') {
            'free' => 10,
            'starter' => 100,
            'pro' => 500,
            'business' => 2000,
            'enterprise' => PHP_INT_MAX,
            default => 10,
        };
    }

    public function toggleUpgradeModal(): void
    {
        $this->showUpgradeModal = !$this->showUpgradeModal;
    }

    public function render()
    {
        return view('livewire.dashboard.subscription-status-card');
    }
}
```

```php
class QuickActionsGrid extends Component
{
    public array $actions = [];
    public bool $showAll = false;

    public function mount(array $actions): void
    {
        $this->actions = $actions;
    }

    public function getVisibleActions(): array
    {
        return $this->showAll
            ? $this->actions
            : array_slice($this->actions, 0, 4);
    }

    public function toggleShowAll(): void
    {
        $this->showAll = !$this->showAll;
    }

    public function performAction(string $key): void
    {
        $action = collect($this->actions)->firstWhere('key', $key);

        if ($action) {
            event(new QuickActionPerformed(auth()->id(), $key));

            if (str_starts_with($action['url'], '/')) {
                return redirect()->to($action['url']);
            }
        }
    }

    public function render()
    {
        return view('livewire.dashboard.quick-actions-grid');
    }
}
```

## ActivityFeed Component

```php
// app/Livewire/Dashboard/ActivityFeed.php
class ActivityFeed extends Component
{
    public int $perPage = 10;
    public int $page = 1;
    public array $activities = [];
    public bool $hasMore = true;
    public bool $loading = false;

    protected $listeners = [
        'loadMore' => 'loadMoreActivities',
    ];

    public function mount(array $initialActivities): void
    {
        $this->activities = $initialActivities;
        $this->hasMore = count($initialActivities) >= $this->perPage;
    }

    public function loadMoreActivities(): void
    {
        $this->loading = true;
        $this->page++;

        $newActivities = ActivityLog::where('user_id', auth()->id())
            ->orderBy('created_at', 'desc')
            ->skip($this->page * $this->perPage)
            ->take($this->perPage)
            ->get()
            ->map(fn($activity) => [
                'id' => $activity->id,
                'type' => $activity->type,
                'action' => $activity->action,
                'description' => $activity->description,
                'icon' => $this->getActivityIcon($activity->type),
                'timestamp' => $activity->created_at->diffForHumans(),
            ])
            ->toArray();

        $this->activities = array_merge($this->activities, $newActivities);
        $this->hasMore = count($newActivities) >= $this->perPage;
        $this->loading = false;
    }

    public function render()
    {
        return view('livewire.dashboard.activity-feed');
    }
}
```

```php
class RecommendationsPanel extends Component
{
    public array $recommendations = [];
    public array $dismissedIds = [];

    public function mount(array $recommendations): void
    {
        $this->recommendations = $recommendations;
    }

    public function dismiss(int $id): void
    {
        $this->dismissedIds[] = $id;
        $this->recommendations = array_filter(
            $this->recommendations,
            fn($r) => $r['id'] !== $id
        );

        event(new RecommendationDismissed(auth()->id(), $id));
    }

    public function getVisibleRecommendations(): array
    {
        return array_values(array_filter(
            $this->recommendations,
            fn($r) => !in_array($r['id'], $this->dismissedIds)
        ));
    }

    public function render()
    {
        return view('livewire.dashboard.recommendations-panel');
    }
}
```

```php
// app/Livewire/Dashboard/WidgetContainer.php
class WidgetContainer extends Component
{
    public array $widgets = [];
    public bool $editMode = false;

    protected $listeners = [
        'widgetToggled' => 'toggleWidget',
        'editModeToggled' => 'toggleEditMode',
    ];

    public function mount(array $widgets): void
    {
        $this->widgets = $widgets;
    }

    public function toggleWidget(string $widgetKey): void
    {
        $this->widgets = array_map(function ($widget) use ($widgetKey) {
            if ($widget['key'] === $widgetKey) {
                $widget['visible'] = !$widget['visible'];
            }
            return $widget;
        }, $this->widgets);

        $this->saveWidgetConfig();
    }

    public function reorderWidgets(array $newOrder): void
    {
        $this->widgets = collect($this->widgets)
            ->sortBy(fn($w) => array_search($w['key'], $newOrder) ?? 999)
            ->values()
            ->toArray();

        $this->saveWidgetConfig();
    }

    private function saveWidgetConfig(): void
    {
        UserWidgetPreference::updateOrCreate(
            ['user_id' => auth()->id()],
            ['widgets' => $this->widgets]
        );

        Cache::forget("dashboard:widgets:" . auth()->id());

        event(new WidgetReordered(auth()->id(), $this->widgets));
    }

    public function toggleEditMode(): void
    {
        $this->editMode = !$this->editMode;
    }

    public function render()
    {
        return view('livewire.dashboard.widget-container');
    }
}
```

# User Flows

## Flow 1: First-Time Dashboard Visit

```
User logs in for first time
  ↓
Dashboard loads with empty state
  ↓
Welcome card displayed with onboarding steps
  ↓
Step 1: "Complete your profile" → Profile page
  Step 2: "Browse the marketplace" → Marketplace
  Step 3: "Try the playground" → Playground
  ↓
User completes steps
  ↓
Progress bar updates (33%, 66%, 100%)
  ↓
Onboarding complete → Normal dashboard view
```

**Implementation Notes:**
- Track onboarding progress in `users.onboarding_completed_at`
- Use `Cache::remember` for onboarding state (5-minute TTL)
- Hide onboarding card once completed
- Allow users to re-trigger onboarding from profile settings

## Flow 2: Dashboard Widget Customization

```
User clicks "Customize" button
  ↓
Widget container enters edit mode
  ↓
Widgets become draggable (Alpine.js Sortable.js)
  ↓
User drags widgets to reorder
  ↓
Debounced save (300ms delay)
  ↓
POST /api/dashboard/widgets with new order
  ↓
Server updates user_widget_preferences
  ↓
Broadcast WidgetReordered event to other tabs
  ↓
Other tabs refresh widget order
```

**Implementation Notes:**
- Use `@sortable` Alpine.js directive
- Optimistic UI: update order immediately, revert on error
- Store widget order as JSON array in `user_widget_preferences`
- Cache widget config per user (5-minute TTL)

## Flow 3: Subscription Upgrade from Dashboard

```
User sees subscription card with "Upgrade Plan" CTA
  ↓
Clicks "Upgrade Plan"
  ↓
Modal opens with plan comparison
  ↓
User selects new plan
  ↓
Proration calculated via Stripe API
  ↓
Confirmation dialog shows proration amount
  ↓
User confirms upgrade
  ↓
Stripe Checkout session created
  ↓
Redirect to Stripe Checkout
  ↓
Payment successful
  ↓
Webhook received → Subscription updated
  ↓
Dashboard refreshes → New plan active
```

**Implementation Notes:**
- Use Stripe Checkout for PCI compliance
- Handle webhook signature verification
- Optimistic UI: show "processing" state during checkout
- Invalidate dashboard cache on subscription update

## Flow 4: Notification Interaction

```
User sees notification badge on bell icon
  ↓
Clicks bell icon
  ↓
Notification dropdown opens (Alpine.js)
  ↓
Notifications loaded via Livewire
  ↓
User clicks a notification
  ↓
Notification marked as read
  ↓
Redirect to relevant page
  ↓
Badge count decremented
  ↓
Other tabs sync via broadcast
```

**Implementation Notes:**
- Use `wire:model` for notification dropdown state
- Mark as read on click (server-side)
- Real-time badge updates via Laravel Echo
- Notification count cached in Redis (2-minute TTL)

## Flow 5: Dashboard Data Refresh

```
User lands on dashboard
  ↓
Initial data loaded via Livewire mount()
  ↓
Real-time: Notifications (WebSocket)
  ↓
5-minute: Activity feed (polling)
  ↓
On-demand: Statistics (user click)
  ↓
Cache invalidated on any mutation
  ↓
Background: Recommendation refresh (15-min interval)
```

**Implementation Notes:**
- Use `wire:poll` for activity feed refresh
- WebSocket for notification badge updates
- Cache keys: `dashboard:{type}:{user_id}`
- Invalidate on relevant events (download, purchase, etc.)

```javascript
// Alpine.js dashboard store
Alpine.store('dashboard', {
    loading: false,
    greeting: '',
    subscription: {
        plan: 'free',

        expires_at: null,

    widgets: [],

        this.loadDashboard();
    },

    async loadDashboard() {
        this.loading = true;
        const response = await fetch('/api/dashboard/summary');
        const data = await response.json();
        Object.assign(this, data);
        this.loading = false;
    },

    refreshNotifications(count) {
        this.notificationCount = count;
    },

    toggleEditMode() {
        this.editMode = !this.editMode;
    }
});
```

## Widget State Persistence

```javascript
// Widget drag-and-drop state
Alpine.data('widgetDragger', () => ({
    widgets: [],
    sortable: null,

    init() {
        this.widgets = this.$wire.get('widgets');
        this.initSortable();
    },

    initSortable() {
        this.sortable = new Sortable(this.$refs.widgetContainer, {
            animation: 200,
            ghostClass: 'opacity-50',
            dragClass: 'shadow-xl',
            onEnd: (evt) => {
                this.reorderWidgets();
            }
        });
    },

    async reorderWidgets() {
        const newOrder = Array.from(this.$refs.widgetContainer.children)
            .map(el => el.dataset.widgetKey);

        await this.$wire.reorderWidgets(newOrder);
    }
}));
```

## Widget Rendering Pipeline

```
1. WidgetContainer mounts
  ↓
2. Fetch user widget preferences from cache/DB
  ↓
3. For each visible widget:
  a. Check widget-specific cache
  b. If cached → render from cache
  c. If not cached → fetch data → cache → render
  ↓
4. Render widgets in order
  ↓
5. Attach event listeners (Alpine.js)
  ↓
6. Schedule refresh intervals
```

## Widget Data Loading Strategy

```php
// Widget data loading with cache-first approach
class WidgetDataLoader
{
    public function load(string $widgetKey, User $user): array
    {
        $cacheKey = "widget:{$widgetKey}:{$user->id}";
        $ttl = $this->getCacheTtl($widgetKey);

        return Cache::remember($cacheKey, $ttl, function () use ($widgetKey, $user) {
            return $this->fetchWidgetData($widgetKey, $user);
        });
    }

    private function fetchWidgetData(string $widgetKey, User $user): array
    {
        return match($widgetKey) {
            'subscription-status' => $this->getSubscriptionData($user),
            'recent-downloads' => $this->getRecentDownloads($user),
            'activity-timeline' => $this->getActivityTimeline($user),
            'recommendations' => $this->getRecommendations($user),
            'team-overview' => $this->getTeamOverview($user),
            'project-count' => $this->getProjectCount($user),
            'billing-summary' => $this->getBillingSummary($user),
            'usage-meters' => $this->getUsageMeters($user),
            'notifications-preview' => $this->getNotificationPreviews($user),
            default => [],
        };
    }
}
```

## Recommendation Algorithm

```php
class RecommendationEngine
{
    public function generate(User $user): array
    {
        $scores = [];

        // Factor 1: Download history similarity (weight: 0.3)
        $downloadHistory = $user->downloads()->pluck('component_id');
        $similarComponents = $this->findSimilarComponents($downloadHistory);
        foreach ($similarComponents as $component => $score) {
            $scores[$component] = ($scores[$component] ?? 0) + ($score * 0.3);
        }

        // Factor 2: Subscription tier compatibility (weight: 0.25)
        $tierComponents = $this->getTierCompatibleComponents($user->subscription_tier);
        foreach ($tierComponents as $component => $score) {
            $scores[$component] = ($scores[$component] ?? 0) + ($score * 0.25);
        }

        // Factor 3: Usage pattern matching (weight: 0.25)
        $usagePatterns = $this->analyzeUsagePatterns($user);
        $patternMatches = $this->matchByUsagePatterns($usagePatterns);
        foreach ($patternMatches as $component => $score) {
            $scores[$component] = ($scores[$component] ?? 0) + ($score * 0.25);
        }

        // Factor 4: Team preferences (weight: 0.2)
        if ($user->team_id) {
            $teamPreferences = $this->getTeamPreferences($user->team_id);
            foreach ($teamPreferences as $component => $score) {
                $scores[$component] = ($scores[$component] ?? 0) + ($score * 0.2);
            }
        }

        // Sort by score and return top 5
        arsort($scores);
        return array_slice(array_keys($scores), 0, 5);
    }

    private function analyzeUsagePatterns(User $user): array
    {
        return [
            'most_used_components' => $user->downloads()
                ->selectRaw('component_id, COUNT(*) as count')
                ->groupBy('component_id')
                ->orderByDesc('count')
                ->limit(5)
                ->pluck('count', 'component_id')
                ->toArray(),
            'download_frequency' => $user->downloads()
                ->where('created_at', '>=', now()->subDays(30))
                ->count(),
            'preferred_categories' => $user->downloads()
                ->join('components', 'downloads.component_id', '=', 'components.id')
                ->pluck('components.category')
                ->countBy()
                ->toArray(),
            'last_download_category' => $user->downloads()
                ->join('components', 'downloads.component_id', '=', 'components.id')
                ->latest()
                ->value('components.category'),
        ];
    }
}
```

## Cold-Start Handling

```php
class ColdStartRecommendations
{
    public function generate(): array
    {
        // For new users with no history, use popularity-based recommendations
        return Cache::remember('recommendations:popular', 900, function () {
            return Component::query()
                ->select('id', 'name', 'slug')
                ->withCount('downloads')
                ->orderByDesc('downloads_count')
                ->limit(5)
                ->get()
                ->map(fn($component) => [
                    'id' => $component->id,
                    'type' => 'component',
                    'name' => $component->name,
                    'reason' => 'Popular component',
                    'score' => $component->downloads_count,
                ])
                ->toArray();
        });
    }
}
```

```php
// app/Http/Middleware/EnsureDashboardAccess.php
class EnsureDashboardAccess
{
    public function handle(Request $request, Closure $next)
    {
        // Check if user has dashboard access
        if (!auth()->check()) {
            return redirect()->route('login');
        }

        // Check if user's account is active
        if (auth()->user()->isSuspended()) {
            abort(403, 'Your account has been suspended.');
        }

        // Rate limiting
        if (RateLimiter::tooManyAttempts('dashboard:' . auth()->id(), 60)) {
            abort(429, 'Too many requests. Please try again later.');
        }

        RateLimiter::hit('dashboard:' . auth()->id(), 60);

        return $next($request);
    }
}
```

```php
class DashboardViewed implements ShouldBroadcast
```
