// app/Livewire/Templates/TemplateCatalog.php
class TemplateCatalog extends Component
{
    public string $search = '';
    public string $category = '';
    public string $priceFilter = '';
    public string $ratingFilter = '';
    public string $sortBy = 'popular';
    public int $currentPage = 1;
    public int $perPage = 24;
    public array $templates = [];
    public int $totalTemplates = 0;
    public array $categories = [];
    public array $featuredTemplates = [];

    public function mount(): void
    {
        $this->loadCategories();
        $this->loadTemplates();
        $this->loadFeatured();
    }

    private function loadCategories(): void
    {
        $this->categories = TemplateCategory::orderBy('position')
            ->get()
            ->map(fn($cat) => [
                'id' => $cat->id,
                'name' => $cat->name,
                'slug' => $cat->slug,
                'count' => $cat->templates()->count(),
            ])
            ->toArray();
    }

    private function loadTemplates(): void
    {
        $query = Template::query()
            ->with(['creator', 'category'])
            ->where('status', 'published');

        // Search
        if ($this->search) {
            $query->where(function ($q) {
                $q->where('name', 'like', "%{$this->search}%")
                  ->orWhere('description', 'like', "%{$this->search}%")
                  ->orWhereHas('tags', fn($q) => $q->where('name', 'like', "%{$this->search}%"));
            });
        }

        // Category filter
        if ($this->category) {
            $query->where('category_id', $this->category);
        }

        // Price filter
        if ($this->priceFilter) {
            match($this->priceFilter) {
                'free' => $query->where('price', 0),
                'paid' => $query->where('price', '>', 0),
                'under25' => $query->where('price', '<=', 25),
                'under50' => $query->where('price', '<=', 50),
                'under100' => $query->where('price', '<=', 100),
            };
        }

        // Rating filter
        if ($this->ratingFilter) {
            $query->where('rating', '>=', (float) $this->ratingFilter);
        }

        // Sort
        match($this->sortBy) {
            'popular' => $query->orderByDesc('download_count'),
            'newest' => $query->orderByDesc('created_at'),
            'rating' => $query->orderByDesc('rating'),
            'price_low' => $query->orderBy('price'),
            'price_high' => $query->orderByDesc('price'),
        };

        $this->totalTemplates = $query->count();
        $this->templates = $query
            ->skip(($this->currentPage - 1) * $this->perPage)
            ->take($this->perPage)
            ->get()
            ->map(fn($template) => [
                'id' => $template->id,
                'name' => $template->name,
                'slug' => $template->slug,
                'description' => $template->description,
                'price' => $template->price,
                'rating' => $template->rating,
                'review_count' => $template->review_count,
                'download_count' => $template->download_count,
                'preview_url' => $template->preview_screenshots[0] ?? null,
                'creator' => [
                    'name' => $template->creator->name,
                    'avatar' => $template->creator->avatar_url,
                ],
                'category' => $template->category->name,
                'laravel_version' => $template->laravel_version,
                'components_included' => $template->components_included,
                'last_updated' => $template->updated_at->diffForHumans(),
                'is_featured' => $template->is_featured,
            ])
            ->toArray();
    }

    private function loadFeatured(): void
    {
        $this->featuredTemplates = Cache::remember('templates:featured', 3600, function () {
            return Template::where('is_featured', true)
                ->where('status', 'published')
                ->with('creator')
                ->limit(6)
                ->get()
                ->toArray();
        });
    }

    public function updatedSearch(): void
    {
        $this->currentPage = 1;
        $this->loadTemplates();
    }

    public function updatedCategory(): void
    {
        $this->currentPage = 1;
        $this->loadTemplates();
    }

    public function updatedPriceFilter(): void
    {
        $this->currentPage = 1;
        $this->loadTemplates();
    }

    public function updatedRatingFilter(): void
    {
        $this->currentPage = 1;
        $this->loadTemplates();
    }

    public function updatedSortBy(): void
    {
        $this->currentPage = 1;
        $this->loadTemplates();
    }

    public function goToPage(int $page): void
    {
        $this->currentPage = $page;
        $this->loadTemplates();
    }

    public function render()
    {
        return view('livewire.templates.catalog')->layout('layouts.dashboard');
    }
}

// app/Livewire/Templates/TemplateDetail.php
class TemplateDetail extends Component
{
    public string $templateId;
    public array $template = [];
    public int $currentScreenshot = 0;
    public string $previewDevice = 'desktop';
    public bool $showPurchaseModal = false;
    public bool $showInstallModal = false;
    public bool $hasPurchased = false;
    public array $reviews = [];
    public int $reviewRating = 5;
    public string $reviewText = '';
    public bool $submittingReview = false;

    public function mount(string $templateId): void
    {
        $this->templateId = $templateId;
        $this->loadTemplate();
        $this->checkPurchase();
        $this->loadReviews();
    }

    private function loadTemplate(): void
    {
        $template = Template::with(['creator', 'category', 'tags'])
            ->where('slug', $this->templateId)
            ->orWhere('id', $this->templateId)
            ->firstOrFail();

        $this->template = [
            'id' => $template->id,
            'name' => $template->name,
            'slug' => $template->slug,
            'description' => $template->description,
            'price' => $template->price,
            'rating' => $template->rating,
            'review_count' => $template->review_count,
            'download_count' => $template->download_count,
            'preview_screenshots' => $template->preview_screenshots,
            'live_preview_url' => $template->live_preview_url,
            'demo_url' => $template->demo_url,
            'creator' => [
                'id' => $template->creator->id,
                'name' => $template->creator->name,
                'avatar' => $template->creator->avatar_url,
                'templates_count' => $template->creator->templates()->count(),
            ],
            'category' => $template->category->name,
            'tags' => $template->tags->pluck('name')->toArray(),
            'laravel_version' => $template->laravel_version,
            'components_included' => $template->components_included,
            'features' => $template->features,
            'compatibility' => $template->compatibility,
            'changelog' => $template->changelog,
            'documentation_url' => $template->documentation_url,
            'last_updated' => $template->updated_at->diffForHumans(),
            'created_at' => $template->created_at->format('M j, Y'),
        ];

        // Track view
        event(new TemplateViewed($template, auth()->id()));
    }

    private function checkPurchase(): void
    {
        $this->hasPurchased = TemplatePurchase::where('user_id', auth()->id())
            ->where('template_id', $this->template['id'])
            ->exists();
    }

    private function loadReviews(): void
    {
        $this->reviews = TemplateReview::where('template_id', $this->template['id'])
            ->with('user')
            ->orderByDesc('created_at')
            ->limit(10)
            ->get()
            ->map(fn($review) => [
                'id' => $review->id,
                'rating' => $review->rating,
                'review' => $review->review,
                'user' => [
                    'name' => $review->user->name,
                    'avatar' => $review->user->avatar_url,
                ],
                'created_at' => $review->created_at->diffForHumans(),
            ])
            ->toArray();
    }

    public function nextScreenshot(): void
    {
        $count = count($this->template['preview_screenshots']);
        $this->currentScreenshot = ($this->currentScreenshot + 1) % $count;
    }

    public function prevScreenshot(): void
    {
        $count = count($this->template['preview_screenshots']);
        $this->currentScreenshot = ($this->currentScreenshot - 1 + $count) % $count;
    }

    public function setPreviewDevice(string $device): void
    {
        $this->previewDevice = $device;
    }

    public function submitReview(): void
    {
        $this->validate([
            'reviewRating' => 'required|integer|between:1,5',
            'reviewText' => 'required|string|min:10|max:1000',
        ]);

        $this->submittingReview = true;

        TemplateReview::create([
            'template_id' => $this->template['id'],
            'user_id' => auth()->id(),
            'rating' => $this->reviewRating,
            'review' => $this->reviewText,
        ]);

        // Update template average rating
        $avgRating = TemplateReview::where('template_id', $this->template['id'])->avg('rating');
        $reviewCount = TemplateReview::where('template_id', $this->template['id'])->count();

        Template::where('id', $this->template['id'])->update([
            'rating' => round($avgRating, 1),
            'review_count' => $reviewCount,
        ]);

        $this->reviewRating = 5;
        $this->reviewText = '';
        $this->submittingReview = false;

        $this->loadTemplate();
        $this->loadReviews();

        $this->dispatch('profileSaved', message: 'Review submitted successfully');
    }

    public function render()
    {
        return view('livewire.templates.detail');
    }
}

// app/Livewire/Templates/TemplatePurchaseModal.php
class TemplatePurchaseModal extends Component
{
    public string $templateId;
    public array $template = [];
    public string $selectedPaymentMethod = '';
    public array $paymentMethods = [];
    public bool $processing = false;
    public bool $completed = false;
    public ?string $licenseKey = null;

    public function mount(string $templateId): void
    {
        $this->templateId = $templateId;
        $this->loadTemplate();
        $this->loadPaymentMethods();
    }

    private function loadTemplate(): void
    {
        $this->template = Template::findOrFail($this->templateId)->toArray();
    }

    private function loadPaymentMethods(): void
    {
        $user = auth()->user();

        if (!$user->stripe_id) {
            return;
        }

        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        $methods = $stripe->paymentMethods->all([
            'customer' => $user->stripe_id,
            'type' => 'card',
        ]);

        $this->paymentMethods = collect($methods->data)->map(fn($method) => [
            'id' => $method->id,
            'brand' => $method->card->brand,
            'last4' => $method->card->last4,
            'exp_month' => $method->card->exp_month,
            'exp_year' => $method->card->exp_year,
        ])->toArray();

        if (!empty($this->paymentMethods)) {
            $this->selectedPaymentMethod = $this->paymentMethods[0]['id'];
        }
    }

    public function purchase(): void
    {
        if ($this->template['price'] > 0 && empty($this->selectedPaymentMethod)) {
            $this->addError('payment', 'Please select a payment method');
            return;
        }

        $this->processing = true;

        $user = auth()->user();
        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        try {
            // Create payment intent
            $paymentIntent = $stripe->paymentIntents->create([
                'amount' => $this->template['price'] * 100,
                'currency' => 'usd',
                'customer' => $user->stripe_id,
                'payment_method' => $this->selectedPaymentMethod,
                'confirm' => true,
                'automatic_payment_methods' => [
                    'enabled' => true,
                    'allow_redirects' => 'never',
                ],
            ]);

            // Create purchase record
            $purchase = TemplatePurchase::create([
                'user_id' => $user->id,
                'template_id' => $this->template['id'],
                'stripe_id' => $paymentIntent->id,
                'amount' => $this->template['price'],
                'license_key' => Str::random(32),
                'purchased_at' => now(),
            ]);

            $this->licenseKey = $purchase->license_key;
            $this->completed = true;

            // Increment download count
            Template::where('id', $this->template['id'])->increment('download_count');

            event(new TemplatePurchased($user, $purchase));

            $this->dispatch('profileSaved', message: 'Template purchased successfully');
        } catch (\Exception $e) {
            $this->addError('purchase', 'Purchase failed: ' . $e->getMessage());
        }

        $this->processing = false;
    }

    public function render()
    {
        return view('livewire.templates.purchase-modal');
    }
}

// app/Livewire/Templates/TemplateInstallModal.php
class TemplateInstallModal extends Component
{
    public string $templateId;
    public array $template = [];
    public array $requirements = [];
    public bool $requirementsMet = false;
    public int $installProgress = 0;
    public string $installStatus = '';
    public bool $installing = false;
    public bool $installed = false;
    public ?string $installPath = null;

    public function mount(string $templateId): void
    {
        $this->templateId = $templateId;
        $this->loadTemplate();
        $this->checkRequirements();
    }

    private function loadTemplate(): void
    {
        $this->template = Template::findOrFail($this->templateId)->toArray();
        $this->requirements = $this->template['compatibility'] ?? [];
    }

    private function checkRequirements(): void
    {
        // Check Laravel version
        $laravelVersion = app()->version();
        $requiredVersion = $this->template['laravel_version'] ?? '*';

        $this->requirementsMet = true;

        foreach ($this->requirements as $requirement) {
            if (!$this->checkRequirement($requirement)) {
                $this->requirementsMet = false;
                break;
            }
        }

    {
        return match($requirement['type']) {
            'php_version' => version_compare(PHP_VERSION, $requirement['constraint'], '>='),
            'laravel_version' => version_compare(app()->version(), $requirement['constraint'], '>='),
            'extension' => extension_loaded($requirement['name']),
            default => true,
        };
    }

    {
        $this->installing = true;
        $this->installProgress = 0;

            'install_path' => $extractPath,

        $this->installing = false;
    }

    private function downloadTemplate(): string

class TemplateScreenshotGenerator
{
    private string $screenshotPath = 'templates/screenshots';
    private array $devices = [
        'desktop' => ['width' => 1920, 'height' => 1080],
        'tablet' => ['width' => 768, 'height' => 1024],
        'mobile' => ['width' => 375, 'height' => 812],
    ];

    public function generateScreenshots(Template $template): array
    {
        $screenshots = [];

        foreach ($this->devices as $device => $dimensions) {
            $screenshot = $this->captureScreenshot(
                $template->demo_url,
                $dimensions['width'],
                $dimensions['height']
            );

            $path = "{$this->screenshotPath}/{$template->id}/{$device}.webp";
            Storage::disk('s3')->put($path, $screenshot);

            $screenshots[$device] = Storage::disk('s3')->url($path);
        }

        $template->update(['preview_screenshots' => $screenshots]);

        return $screenshots;
    }

    private function captureScreenshot(string $url, int $width, int $height): string
    {
        $browser = app(Browser::class);

        $page = $browser->newPage();
        $page->setViewportSize($width, $height);
        $page->navigate($url);
        $page->waitForLoadState('networkidle');

        // Wait for animations
        usleep(500000);

        $screenshot = $page->screenshot([
            'type' => 'webp',
            'quality' => 80,
        ]);

        $page->close();

        return $screenshot;
    }

    public function generateThumbnail(string $screenshotUrl): string
    {
        $image = Image::make($screenshotUrl);
        $image->resize(400, 225, function ($constraint) {
            $constraint->aspectRatio();
            $constraint->upsize();
        });

        $thumbnailPath = 'templates/thumbnails/' . Str::uuid() . '.webp';
        Storage::disk('s3')->put($thumbnailPath, (string) $image->encode('webp', 85));

        return Storage::disk('s3')->url($thumbnailPath);
    }
}

// app/Services/TemplateSearchIndex.php
class TemplateSearchIndex
{
    public function index(Template $template): void
    {
        $searchable = [
            'id' => $template->id,
            'name' => $template->name,
            'description' => $template->description,
            'category' => $template->category->name,
            'tags' => $template->tags->pluck('name')->toArray(),
            'price' => $template->price,
            'rating' => $template->rating,
            'download_count' => $template->download_count,
            'laravel_version' => $template->laravel_version,
            'components_included' => $template->components_included,
            'creator_name' => $template->creator->name,
            'is_featured' => $template->is_featured,
            'created_at' => $template->created_at->timestamp,
        ];

        $index = Meilisearch::index('templates');
        $index->addDocuments([$searchable]);
    }

    public function search(string $query, array $filters = []): array
    {
        $index = Meilisearch::index('templates');

        $searchParams = [
            'q' => $query,
            'limit' => $filters['limit'] ?? 24,
            'offset' => ($filters['page'] ?? 1) - 1,
        ];

        // Faceted filtering
        $facets = [];
        if (isset($filters['category'])) {
            $facets[] = "category = '{$filters['category']}'";
        }
        if (isset($filters['price_filter'])) {
            $facets[] = $this->buildPriceFilter($filters['price_filter']);
        }
        if (isset($filters['min_rating'])) {
            $facets[] = "rating >= {$filters['min_rating']}";
        }
        if (!empty($facets)) {
            $searchParams['filter'] = implode(' AND ', $facets);
        }

        // Sorting
        $searchParams['sort'] = match($filters['sort'] ?? 'relevance') {
            'popular' => ['download_count:desc'],
            'newest' => ['created_at:desc'],
            'rating' => ['rating:desc'],
            'price_low' => ['price:asc'],
            'price_high' => ['price:desc'],
            default => ['_ranking:typo', '_ranking:words', '_ranking:attributes', '_ranking:proximity', '_ranking:exactness'],
        };

        $results = $index->search($query, $searchParams);

        return [
            'hits' => $results['hits'],
            'total' => $results['nbHits'],
            'page' => $filters['page'] ?? 1,
            'per_page' => $searchParams['limit'],
            'facets' => $results['facetsDistribution'] ?? [],
        ];
    }

    private function buildPriceFilter(string $priceFilter): string
    {
        return match($priceFilter) {
            'free' => 'price = 0',
            'paid' => 'price > 0',
            'under25' => 'price <= 25',
            'under50' => 'price <= 50',
            'under100' => 'price <= 100',
            default => '1=1',
        };
    }

    public function deleteIndex(Template $template): void
    {
        $index = Meilisearch::index('templates');
        $index->deleteDocument($template->id);
    }
}

    public function getRevenueByTemplate(Template $template): array
    {
        $purchases = TemplatePurchase::where('template_id', $template->id);

        return [
            'total_revenue' => (clone $purchases)->sum('amount'),
            'total_purchases' => (clone $purchases)->count(),
            'average_order_value' => (clone $purchases)->avg('amount'),
            'revenue_by_month' => (clone $purchases)
                ->selectRaw('DATE_FORMAT(purchased_at, "%Y-%m") as month, SUM(amount) as revenue, COUNT(*) as purchases')
                ->groupBy('month')
                ->orderByDesc('month')
                ->limit(12)
                ->get()
                ->toArray(),
            'conversion_rate' => $this->calculateConversionRate($template),
        ];
    }

    private function calculateConversionRate(Template $template): float
    {
        $views = TemplateView::where('template_id', $template->id)->count();
        $purchases = TemplatePurchase::where('template_id', $template->id)->count();

        if ($views === 0) {
            return 0;
        }

        return round(($purchases / $views) * 100, 2);
    }

    {
        return [
            'total_revenue' => Template::where('creator_id', $creator->id)
                ->sum('total_revenue'),
            'total_sales' => TemplatePurchase::whereHas('template', fn($q) => $q->where('creator_id', $creator->id))

## Payout System

```php
// app/Services/TemplatePayoutService.php
class TemplatePayoutService
{
    private float $creatorShare = 0.70;
    private float $platformShare = 0.30;
    private int $minimumPayout = 50;
    private string $payoutSchedule = 'monthly';

    public function calculatePayout(User $creator): array
    {
        $totalRevenue = $creator->templates->sum('total_revenue');
        $previousPayouts = $creator->payouts->sum('amount');
        $pendingAmount = $totalRevenue - $previousPayouts;

        return [
            'total_revenue' => $totalRevenue,
            'previous_payouts' => $previousPayouts,
            'pending_amount' => $pendingAmount,
            'creator_share' => $pendingAmount * $this->creatorShare,
            'platform_share' => $pendingAmount * $this->platformShare,
            'eligible_for_payout' => $pendingAmount >= $this->minimumPayout,
            'next_payout_date' => $this->getNextPayoutDate(),
        ];
    }

    public function processPayout(User $creator): ?Payout
    {
        $payoutInfo = $this->calculatePayout($creator);

        if (!$payoutInfo['eligible_for_payout']) {
            return null;
        }

        $payout = Payout::create([
            'user_id' => $creator->id,
            'amount' => $payoutInfo['creator_share'],
            'status' => 'pending',
            'scheduled_at' => $this->getNextPayoutDate(),
        ]);

        // Process via Stripe Connect
        $this->processStripePayout($creator, $payout);

        event(new PayoutCreated($creator, $payout));

        return $payout;
    }

    private function processStripePayout(User $creator, Payout $payout): void
    {
        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        $stripe->transfers->create([
            'amount' => $payout->amount * 100,
            'currency' => 'usd',
            'destination' => $creator->stripe_account_id,
            'description' => 'GsmUi Template Payout',
        ]);

        $payout->update(['status' => 'completed', 'processed_at' => now()]);
    }

    private function getNextPayoutDate(): Carbon
    {
        $now = now();

        return match($this->payoutSchedule) {
            'weekly' => $now->next(Carbon::MONDAY),
            'monthly' => $now->addMonth()->startOfMonth(),
            default => $now->addMonth()->startOfMonth(),
        };
    }
}
