// app/Livewire/Subscription/CurrentPlanCard.php
class CurrentPlanCard extends Component
{
    public array $subscription = [];
    public array $usage = [];
    public string $nextBillingDate = '';
    public float $monthlySpend = 0;
    public bool $showUpgradeModal = false;
    public bool $showCancellationModal = false;

    public function mount(): void
    {
        $this->loadSubscriptionData();
    }

    private function loadSubscriptionData(): void
    {
        $user = auth()->user();
        $subscription = $user->currentSubscription();

        if ($subscription) {
            $this->subscription = [
                'id' => $subscription->id,
                'plan' => $subscription->stripe_plan,
                'plan_name' => $this->getPlanName($subscription->stripe_plan),
                'status' => $subscription->stripe_status,
                'price' => $subscription->price ?? 0,
                'currency' => $subscription->currency ?? 'USD',
                'billing_cycle' => $subscription->billing_cycle ?? 'monthly',
                'next_billing_date' => $subscription->next_billing_date,
                'ends_at' => $subscription->ends_at,
                'created_at' => $subscription->created_at,
            ];

            $this->nextBillingDate = Carbon::parse($subscription->next_billing_date)->format('F j, Y');
            $this->monthlySpend = $subscription->billing_cycle === 'yearly'
                ? $subscription->price / 12
                : $subscription->price;
        }

        $this->usage = $this->getUsageData($user);
    }

    private function getUsageData(User $user): array
    {
        return [
            'downloads' => [
                'used' => $user->downloads()->count(),
                'limit' => $this->getDownloadLimit($user),
                'percentage' => $this->calculateUsagePercentage(
                    $user->downloads()->count(),
                    $this->getDownloadLimit($user)
                ),
            ],
            'projects' => [
                'used' => $user->playgroundProjects()->count(),
                'limit' => -1, // Unlimited
                'percentage' => 0,
            ],
            'team_seats' => [
                'used' => $user->team ? $user->team->members()->count() : 0,
                'limit' => $this->getTeamSeatLimit($user),
                'percentage' => $this->calculateUsagePercentage(
                    $user->team ? $user->team->members()->count() : 0,
                    $this->getTeamSeatLimit($user)
                ),
            ],
        ];
    }

    private function getDownloadLimit(User $user): int
    {
        return match($user->subscription_tier ?? 'free') {
            'free' => 10,
            'starter' => 100,
            'pro' => 500,
            'business' => 2000,
            'enterprise' => PHP_INT_MAX,
            default => 10,
        };
    }

    private function getTeamSeatLimit(User $user): int
    {
        return match($user->subscription_tier ?? 'free') {
            'free' => 1,
            'starter' => 3,
            'pro' => 10,
            'business' => 50,
            'enterprise' => PHP_INT_MAX,
            default => 1,
        };
    }

    private function calculateUsagePercentage(int $used, int $limit): int
    {
        if ($limit <= 0 || $limit === PHP_INT_MAX) {
            return 0;
        }

        return min(100, round(($used / $limit) * 100));
    }

    public function toggleUpgradeModal(): void
    {
        $this->showUpgradeModal = !$this->showUpgradeModal;
    }

    public function toggleCancellationModal(): void
    {
        $this->showCancellationModal = !$this->showCancellationModal;
    }

    public function render()
    {
        return view('livewire.subscription.current-plan-card');
    }
}

// app/Livewire/Subscription/PlanComparison.php
class PlanComparison extends Component
{
    public array $plans = [];
    public string $selectedCycle = 'monthly';
    public ?string $selectedPlan = null;
    public bool $loading = false;

    public function mount(): void
    {
        $this->loadPlans();
    }

    private function loadPlans(): void
    {
        $this->plans = Plan::orderBy('price_monthly')
            ->get()
            ->map(fn($plan) => [
                'id' => $plan->id,
                'slug' => $plan->slug,
                'name' => $plan->name,
                'description' => $plan->description,
                'price_monthly' => $plan->price_monthly,
                'price_yearly' => $plan->price_yearly,
                'features' => $plan->features,
                'limits' => $plan->limits,
                'is_popular' => $plan->is_popular,
                'is_current' => $this->isCurrentPlan($plan),
            ])
            ->toArray();
    }

    private function isCurrentPlan(Plan $plan): bool
    {
        $currentPlan = auth()->user()->currentSubscription()?->stripe_plan;
        return $currentPlan === $plan->slug;
    }

    public function selectPlan(string $planSlug): void
    {
        $this->selectedPlan = $planSlug;
    }

    public function getSelectedPlanPrice(): float
    {
        $plan = collect($this->plans)->firstWhere('slug', $this->selectedPlan);

        if (!$plan) {
            return 0;
        }

        return $this->selectedCycle === 'yearly'
            ? $plan['price_yearly']
            : $plan['price_monthly'];
    }

    public function getSavingsPercentage(): int
    {
        $plan = collect($this->plans)->firstWhere('slug', $this->selectedPlan);

        if (!$plan || $plan['price_yearly'] <= 0) {
            return 0;
        }

        $monthlyCost = $plan['price_monthly'] * 12;
        $yearlyCost = $plan['price_yearly'];

        return round((1 - ($yearlyCost / $monthlyCost)) * 100);
    }

    public function render()
    {
        return view('livewire.subscription.plan-comparison');
    }
}

// app/Livewire/Subscription/UpgradeFlow.php
class UpgradeFlow extends Component
{
    public string $targetPlan = '';
    public string $billingCycle = 'monthly';
    public float $prorationAmount = 0;
    public string $prorationCredit = '';
    public bool $processing = false;
    public bool $confirmed = false;

    protected $listeners = [
        'planSelected' => 'handlePlanSelected',
    ];

    public function handlePlanSelected(string $plan, string $cycle): void
    {
        $this->targetPlan = $plan;
        $this->billingCycle = $cycle;
        $this->calculateProration();
    }

    private function calculateProration(): void
    {
        $user = auth()->user();
        $currentSubscription = $user->currentSubscription();

        if (!$currentSubscription) {
            $this->prorationAmount = $this->getPlanPrice($this->targetPlan, $this->billingCycle);
            return;
        }

        // Use Stripe API for proration calculation
        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        $subscription = $stripe->subscriptions->retrieve($currentSubscription->stripe_id);

        $invoice = $stripe->invoices->create([
            'customer' => $subscription->customer,
            'subscription' => $subscription->id,
            'subscription_proration_behavior' => 'create_prorations',
            'subscription_items' => [
                [
                    'id' => $subscription->items->data[0]->id,
                    'price' => $this->getStripePriceId($this->targetPlan, $this->billingCycle),
                ],
            ],
        ]);

        $this->prorationAmount = $invoice->amount_due / 100;
        $this->prorationCredit = $invoice->lines->data[0]->amount / 100;
    }

    public function confirmUpgrade(): void
    {
        $this->processing = true;

        $user = auth()->user();
        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        try {
            $subscription = $user->currentSubscription();

            if ($subscription) {
                // Upgrade existing subscription
                $stripe->subscriptions->update($subscription->stripe_id, [
                    'items' => [
                        [
                            'id' => $subscription->asStripeSubscription()->items->data[0]->id,
                            'price' => $this->getStripePriceId($this->targetPlan, $this->billingCycle),
                        ],
                    ],
                    'proration_behavior' => 'create_prorations',
                ]);
            } else {
                // Create new subscription
                $stripe->subscriptions->create([
                    'customer' => $user->stripe_id,
                    'items' => [
                        ['price' => $this->getStripePriceId($this->targetPlan, $this->billingCycle)],
                    ],
                    'payment_behavior' => 'error_if_incomplete',
                ]);
            }

            event(new SubscriptionUpgraded($user, $this->targetPlan));

            $this->confirmed = true;
            $this->processing = false;

            $this->dispatch('subscriptionUpdated');
            $this->dispatch('profileSaved', message: 'Subscription upgraded successfully');
        } catch (\Exception $e) {
            $this->processing = false;
            $this->addError('upgrade', 'Upgrade failed: ' . $e->getMessage());
        }
    }

    private function getStripePriceId(string $plan, string $cycle): string
    {
        return config("services.stripe.plans.{$plan}.{$cycle}");
    }

    public function render()
    {
        return view('livewire.subscription.upgrade-flow');
    }
}

// app/Livewire/Subscription/CancellationFlow.php
class CancellationFlow extends Component
{
    public int $step = 1;
    public string $reason = '';
    public string $feedback = '';
    public bool $cancelImmediately = false;
    public bool $processing = false;
    public array $retentionOffer = [];

    public array $cancellationReasons = [
        'too_expensive' => 'Too expensive',
        'not_using' => 'Not using enough',
        'missing_features' => 'Missing features',
        'poor_quality' => 'Poor quality',
        'found_alternative' => 'Found an alternative',
        'other' => 'Other',
    ];

    public function mount(): void
    {
        $this->generateRetentionOffer();
    }

    private function generateRetentionOffer(): void
    {
        $user = auth()->user();
        $subscription = $user->currentSubscription();

        if (!$subscription) {
            return;
        }

        // Generate personalized retention offer based on churn risk
        $churnRisk = $this->calculateChurnRisk($user);

        $this->retentionOffer = match(true) {
            $churnRisk > 0.8 => [
                'type' => 'discount',
                'value' => 50,
                'duration' => 3,
                'message' => 'Get 50% off for the next 3 months',
            ],
            $churnRisk > 0.5 => [
                'type' => 'discount',
                'value' => 25,
                'duration' => 2,
                'message' => 'Get 25% off for the next 2 months',
            ],
            default => [
                'type' => 'feature_access',
                'feature' => 'priority_support',
                'duration' => 1,
                'message' => 'Get free priority support for 1 month',
            ],
        };
    }

    private function calculateChurnRisk(User $user): float
    {
        $score = 0;

        // Low engagement
        $recentDownloads = $user->downloads()
            ->where('created_at', '>=', now()->subDays(30))
            ->count();

        if ($recentDownloads < 2) {
            $score += 0.3;
        }

        // Support tickets
        $recentTickets = $user->supportTickets()
            ->where('created_at', '>=', now()->subDays(30))
            ->count();

        if ($recentTickets > 3) {
            $score += 0.2;
        }

        // Failed payments
        $failedPayments = $user->invoices()
            ->where('status', 'failed')
            ->count();

        if ($failedPayments > 0) {
            $score += 0.3;
        }

        // Account age
        if ($user->created_at->diffInMonths(now()) < 3) {
            $score += 0.2;
        }

        return min(1.0, $score);
    }

    public function acceptRetentionOffer(): void
    {
        $user = auth()->user();
        $subscription = $user->currentSubscription();

        if ($this->retentionOffer['type'] === 'discount') {
            // Apply discount coupon
            $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

            $stripe->subscriptions->update($subscription->stripe_id, [
                'coupon' => $this->getCouponId($this->retentionOffer),
            ]);

            event(new RetentionOfferAccepted($user, $this->retentionOffer));
        }

        $this->dispatch('profileSaved', message: 'Discount applied to your subscription');
        $this->dispatch('cancelFlowComplete');
    }

    public function nextStep(): void
    {
        $this->step++;
    }

    public function previousStep(): void
    {
        $this->step--;
    }

    public function cancelSubscription(): void
    {
        $this->processing = true;

        $user = auth()->user();
        $subscription = $user->currentSubscription();

        if (!$subscription) {
            $this->processing = false;
            return;
        }

        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        try {
            if ($this->cancelImmediately) {
                // Cancel immediately
                $stripe->subscriptions->cancel($subscription->stripe_id);
            } else {
                // Cancel at period end
                $stripe->subscriptions->update($subscription->stripe_id, [
                    'cancel_at_period_end' => true,
                ]);
            }

            // Log cancellation
            CancellationLog::create([
                'user_id' => $user->id,
                'subscription_id' => $subscription->id,
                'reason' => $this->reason,
                'feedback' => $this->feedback,
                'cancel_immediately' => $this->cancelImmediately,
                'cancelled_at' => now(),
            ]);

            event(new SubscriptionCancelled($user, $subscription, $this->reason));

            $this->processing = false;
            $this->step = 4; // Confirmation step

            $this->dispatch('subscriptionUpdated');
        } catch (\Exception $e) {

## Failed Payment Recovery

```php
// app/Services/DunningManager.php
class DunningManager
{
    private array $dunningSchedule = [
        1 => ['delay' => 1, 'email' => 'dunning.first_attempt'],
        2 => ['delay' => 3, 'email' => 'dunning.second_attempt'],
        3 => ['delay' => 7, 'email' => 'dunning.final_notice'],
    ];

    public function handleFailedPayment(Invoice $invoice): void
    {
        $subscription = $invoice->subscription;
        $user = $subscription->user;

        // Track attempt
        $attempt = DunningCycle::create([
            'subscription_id' => $subscription->id,
            'invoice_id' => $invoice->id,
            'attempt' => $this->getAttemptNumber($subscription),
            'status' => 'pending',
            'next_retry_at' => $this->getNextRetryTime($subscription),
        ]);

        // Send dunning email
        $emailClass = $this->dunningSchedule[$attempt->attempt]['email'];
        Mail::to($user->email)->send(new $emailClass($user, $invoice, $attempt));

        // Schedule retry
        $this->scheduleRetry($subscription, $attempt);

        event(new PaymentFailed($user, $invoice, $attempt));
    }

    private function getAttemptNumber(Subscription $subscription): int
    {
        return DunningCycle::where('subscription_id', $subscription->id)
            ->where('status', 'pending')
            ->count() + 1;
    }

    private function getNextRetryTime(Subscription $subscription): Carbon
    {
        $attempt = $this->getAttemptNumber($subscription);
        $delay = $this->dunningSchedule[$attempt]['delay'] ?? 7;

        return now()->addDays($delay);
    }

    private function scheduleRetry(Subscription $subscription, DunningCycle $attempt): void
    {
        RetryPayment::dispatch($subscription, $attempt)
            ->delay($attempt->next_retry_at);
    }

    public function handleSuccessfulRecovery(Subscription $subscription, Invoice $invoice): void
    {
        // Mark all pending dunning cycles as recovered
        DunningCycle::where('subscription_id', $subscription->id)
            ->where('status', 'pending')
            ->update(['status' => 'recovered', 'recovered_at' => now()]);

        // Reactivate subscription if suspended
        if ($subscription->stripe_status === 'past_due') {
            $subscription->update(['stripe_status' => 'active']);
        }

        event(new PaymentRecovered($subscription->user, $invoice));
    }

    public function handleMaxAttemptsReached(Subscription $subscription): void
    {
        // Suspend subscription
        $subscription->update(['stripe_status' => 'past_due']);

        // Send final notice
        Mail::to($subscription->user->email)
            ->send(new FinalNoticeEmail($subscription->user, $subscription));

        // Schedule cancellation after grace period
        CancelSubscription::dispatch($subscription)
            ->delay(now()->addDays(7));

        event(new SubscriptionSuspended($subscription));
    }
}

## Metrics Dashboard

```php
// app/Services/SubscriptionAnalytics.php
class SubscriptionAnalytics
{
    public function getMetrics(): array
    {
        return [
            'mrr' => $this->calculateMRR(),
            'arr' => $this->calculateARR(),
            'churn_rate' => $this->calculateChurnRate(),
            'upgrade_rate' => $this->calculateUpgradeRate(),
            'average_revenue_per_user' => $this->calculateARPU(),
            'customer_lifetime_value' => $this->calculateCLV(),
            'ltv_to_cac_ratio' => $this->calculateLtvCacRatio(),
        ];
    }

    private function calculateMRR(): float
    {
        return Subscription::where('stripe_status', 'active')
            ->where('ends_at', '>', now())
            ->sum('price') / 12; // Convert yearly to monthly
    }

    private function calculateARR(): float
    {
        return $this->calculateMRR() * 12;
    }

    private function calculateChurnRate(): float
    {
        $startOfMonth = now()->startOfMonth();
        $endOfMonth = now()->endOfMonth();

        $activeAtStart = Subscription::where('created_at', '<', $startOfMonth)
            ->where('stripe_status', 'active')
            ->count();

        $cancelledThisMonth = CancellationLog::where('cancelled_at', '>=', $startOfMonth)
            ->where('cancelled_at', '<=', $endOfMonth)
            ->count();

        if ($activeAtStart === 0) {
            return 0;
        }

        return round(($cancelledThisMonth / $activeAtStart) * 100, 2);
    }

    private function calculateUpgradeRate(): float
    {
        $totalSubscribers = Subscription::where('stripe_status', 'active')->count();
        $upgradedThisMonth = SubscriptionUpgrade::where('upgraded_at', '>=', now()->startOfMonth())->count();

        if ($totalSubscribers === 0) {
            return 0;
        }

        return round(($upgradedThisMonth / $totalSubscribers) * 100, 2);
    }

    private function calculateARPU(): float
    {
        $activeSubscriptions = Subscription::where('stripe_status', 'active')->count();

        if ($activeSubscriptions === 0) {
            return 0;
        }

        return $this->calculateMRR() / $activeSubscriptions;
    }

    private function calculateCLV(): float
    {
        $arpu = $this->calculateARPU();
        $churnRate = $this->calculateChurnRate() / 100;

        if ($churnRate === 0) {
            return PHP_FLOAT_MAX;
        }

        return $arpu / $churnRate;
    }

    public function getRevenueByPlan(): array
    {
        return Subscription::where('stripe_status', 'active')
            ->selectRaw('stripe_plan, COUNT(*) as count, SUM(price) as revenue')
            ->groupBy('stripe_plan')
            ->get()
            ->toArray();
    }

    public function getChurnByReason(): array
    {
        return CancellationLog::selectRaw('reason, COUNT(*) as count')
            ->groupBy('reason')
            ->orderByDesc('count')
            ->get()
            ->toArray();
    }

// app/Http/Controllers/Webhooks/StripeWebhookController.php
class StripeWebhookController extends Controller
{
    public function handleWebhook(Request $request)
    {
        $payload = $request->all();
        $sigHeader = $request->header('Stripe-Signature');

        try {
            $event = \Stripe\Webhook::constructEvent(
                $request->getContent(),
                $sigHeader,
                config('services.stripe.webhook_secret')
            );
        } catch (\Exception $e) {
            return response('Invalid signature', 400);
        }

        match($event->type) {
            'invoice.payment_succeeded' => $this->handlePaymentSucceeded($event->data->object),
            'invoice.payment_failed' => $this->handlePaymentFailed($event->data->object),
            'customer.subscription.created' => $this->handleSubscriptionCreated($event->data->object),
            'customer.subscription.updated' => $this->handleSubscriptionUpdated($event->data->object),
            'customer.subscription.deleted' => $this->handleSubscriptionDeleted($event->data->object),
            'customer.subscription.trial_will_end' => $this->handleTrialWillEnd($event->data->object),
            'payment_method.attached' => $this->handlePaymentMethodAttached($event->data->object),
            'payment_method.detached' => $this->handlePaymentMethodDetached($event->data->object),
            default => null,
        };

        return response('Webhook processed', 200);
    }

    private function handlePaymentSucceeded(\Stripe\Invoice $invoice): void
    {
        $subscription = Subscription::where('stripe_id', $invoice->subscription)->first();

        if (!$subscription) {
            return;
        }

        // Update subscription status
        $subscription->update(['stripe_status' => 'active']);

        // Create invoice record
        Invoice::create([
            'user_id' => $subscription->user_id,
            'subscription_id' => $subscription->id,
            'stripe_id' => $invoice->id,
            'number' => $invoice->number,
            'amount' => $invoice->amount_paid / 100,
            'currency' => strtoupper($invoice->currency),
            'status' => 'paid',
            'pdf_path' => null,
        ]);

        // Generate PDF
        $this->generateInvoicePdf($subscription->user, $invoice);

        // Handle dunning recovery
        app(DunningManager::class)->handleSuccessfulRecovery($subscription, $invoice);

        event(new PaymentSucceeded($subscription->user, $invoice));
    }

    private function handlePaymentFailed(\Stripe\Invoice $invoice): void
    {
        $subscription = Subscription::where('stripe_id', $invoice->subscription)->first();

        if (!$subscription) {
            return;
        }

        // Update subscription status
        $subscription->update(['stripe_status' => 'past_due']);

        // Create failed invoice record
        Invoice::create([
            'user_id' => $subscription->user_id,
            'subscription_id' => $subscription->id,
            'stripe_id' => $invoice->id,
            'number' => $invoice->number,
            'amount' => $invoice->amount_due / 100,
            'currency' => strtoupper($invoice->currency),
            'status' => 'failed',
        ]);

        // Initiate dunning process
        app(DunningManager::class)->handleFailedPayment(
            Invoice::where('stripe_id', $invoice->id)->first()
        );

        event(new PaymentFailed($subscription->user, $invoice));
    }

    private function handleSubscriptionCreated(\Stripe\Subscription $stripeSubscription): void
    {
        $user = User::where('stripe_id', $stripeSubscription->customer)->first();

        if (!$user) {
            return;
        }

        Subscription::create([
            'user_id' => $user->id,
            'stripe_id' => $stripeSubscription->id,
            'stripe_status' => $stripeSubscription->status,
            'stripe_plan' => $stripeSubscription->items->data[0]->price->id,
            'quantity' => $stripeSubscription->items->data[0]->quantity,
            'trial_ends_at' => $stripeSubscription->trial_end
                ? Carbon::createFromTimestamp($stripeSubscription->trial_end)
                : null,
            'ends_at' => null,
        ]);

        event(new SubscriptionCreated($user, $stripeSubscription));
    }

    private function handleSubscriptionUpdated(\Stripe\Subscription $stripeSubscription): void
    {
        $subscription = Subscription::where('stripe_id', $stripeSubscription->id)->first();

        if (!$subscription) {
            return;
        }

        $subscription->update([
            'stripe_status' => $stripeSubscription->status,
            'stripe_plan' => $stripeSubscription->items->data[0]->price->id,
            'quantity' => $stripeSubscription->items->data[0]->quantity,
            'ends_at' => $stripeSubscription->canceled_at
                ? Carbon::createFromTimestamp($stripeSubscription->canceled_at)
                : null,
        ]);

        // Sync entitlements
        $this->syncEntitlements($subscription);

        event(new SubscriptionUpdated($subscription->user, $subscription));
    }

    private function handleSubscriptionDeleted(\Stripe\Subscription $stripeSubscription): void
    {
        $subscription = Subscription::where('stripe_id', $stripeSubscription->id)->first();

        if (!$subscription) {
            return;
        }

        $subscription->update([
            'stripe_status' => 'cancelled',
            'ends_at' => now(),
        ]);

        // Remove entitlements
        $this->removeEntitlements($subscription);

        event(new SubscriptionCancelled($subscription->user, $subscription));
    }

    private function handleTrialWillEnd(\Stripe\Subscription $stripeSubscription): void
    {
        $subscription = Subscription::where('stripe_id', $stripeSubscription->id)->first();

        if (!$subscription) {
            return;
        }

        // Send trial ending notification
        Mail::to($subscription->user->email)
            ->send(new TrialEndingEmail($subscription->user, $subscription));

        event(new TrialWillEnd($subscription->user, $subscription));
    }

    private function syncEntitlements(Subscription $subscription): void
    {
        $plan = Plan::where('slug', $subscription->stripe_plan)->first();

        if (!$plan) {
            return;
        }

        // Update user entitlements
        $subscription->user->update([
            'subscription_tier' => $plan->slug,
            'download_limit' => $plan->limits['downloads'] ?? -1,
            'team_seat_limit' => $plan->limits['team_seats'] ?? 1,
        ]);
    }

    private function removeEntitlements(Subscription $subscription): void
    {
        $subscription->user->update([
            'subscription_tier' => 'free',
            'download_limit' => 10,
            'team_seat_limit' => 1,
        ]);
    }

    private function generateInvoicePdf(User $user, \Stripe\Invoice $stripeInvoice): void
    {
        $invoice = Invoice::where('stripe_id', $stripeInvoice->id)->first();

        if (!$invoice) {
            return;
        }

        $pdfPath = app(InvoicePdfGenerator::class)->generate($invoice->order ?? $invoice);

        $invoice->update(['pdf_path' => $pdfPath]);
    }
}

```php
// app/Services/EntitlementManager.php
class EntitlementManager
{
    public function check(User $user, string $feature): bool
    {
        $subscription = $user->currentSubscription();

        if (!$subscription || $subscription->stripe_status !== 'active') {
            return $this->checkFreeTierAccess($feature);
        }

        $plan = Plan::where('slug', $subscription->stripe_plan)->first();

        if (!$plan) {
            return false;
        }

        return in_array($feature, $plan->features ?? []);
    }

    public function getLimit(User $user, string $resource): int
    {
        $subscription = $user->currentSubscription();

        if (!$subscription || $subscription->stripe_status !== 'active') {
            return $this->getFreeTierLimit($resource);
        }

        $plan = Plan::where('slug', $subscription->stripe_plan)->first();

        if (!$plan) {
            return 0;
        }

        return $plan->limits[$resource] ?? 0;
    }

    public function getUsage(User $user, string $resource): int
    {
        return match($resource) {
            'downloads' => $user->downloads()->count(),
            'projects' => $user->playgroundProjects()->count(),
            'team_seats' => $user->team ? $user->team->members()->count() : 0,
            default => 0,
        };
    }

    public function hasAccess(User $user, string $feature): bool
    {
        return $this->check($user, $feature);
    }

    public function getRemaining(User $user, string $resource): int
    {
        $limit = $this->getLimit($user, $resource);
        $used = $this->getUsage($user, $resource);

        if ($limit === PHP_INT_MAX) {
            return PHP_INT_MAX;
        }

        return max(0, $limit - $used);
    }

    private function checkFreeTierAccess(string $feature): bool
    {
        $freeFeatures = [
            'basic_components',
            'community_support',
            'limited_downloads',
        ];

        return in_array($feature, $freeFeatures);
    }

    private function getFreeTierLimit(string $resource): int
    {
        return match($resource) {
            'downloads' => 10,
            'projects' => 3,
            'team_seats' => 1,
            default => 0,
        };
    }
}
