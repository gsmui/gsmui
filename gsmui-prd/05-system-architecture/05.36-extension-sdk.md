# Purpose

The Extension SDK provides tools and documentation for building custom extensions and integrations with the GsmUi platform. It offers PHP, JavaScript, and CLI tools that enable developers to create plugins, themes, and integrations that work seamlessly with the platform.

---

# Architecture Overview

The Extension SDK is a layered toolkit that provides different levels of access to the GsmUi platform. The PHP SDK offers deep integration with the Laravel backend, the JavaScript SDK provides frontend component APIs, and the CLI tools automate common development tasks.

The SDK architecture follows the principle of progressive disclosure: basic extensions use simple APIs while advanced extensions can access deeper platform internals. This approach lowers the barrier to entry while enabling complex integrations.

The SDK is versioned independently of the platform to allow for more frequent updates and breaking changes when necessary. Semantic versioning ensures compatibility across minor versions, with clear migration guides for major version upgrades.

# PHP SDK Deep Dive

## Service Provider Base Class

The SDK provides an enhanced service provider base class that adds plugin-specific capabilities on top of Laravel's standardServiceProvider.

```php
<?php

namespace Gsmui\Extension;

use Illuminate\Support\ServiceProvider;

abstract class ExtensionServiceProvider extends ServiceProvider
{
    protected string $extensionSlug;
    protected string $extensionVersion;
    protected array $extensionManifest = [];

    public function register(): void
    {
        $this->mergeConfigFrom(
            $this->getConfigPath(),
            $this->extensionSlug
        );

        $this->registerServices();
        $this->registerEvents();
        $this->registerMiddleware();
        $this->registerViews();
        $this->registerPublishes();
    }

    public function boot(): void
    {
        $this->bootRoutes();
        $this->bootMigrations();
        $this->bootCommands();
        $this->bootPublishes();
    }

    protected function registerServices(): void
    {
        // Override in your extension
    }

    protected function registerEvents(): void
    {
        $listeners = $this->extensionManifest['events']['listeners'] ?? [];

        foreach ($listeners as $event => $handler) {
            if (is_string($handler)) {
                $this->app['events']->listen($event, $handler);
            }
        }
    }

    protected function registerMiddleware(): void
    {
        $middleware = $this->extensionManifest['middleware'] ?? [];

        foreach ($middleware as $middlewareClass) {
            $this->app['router']->pushMiddlewareToGroup('web', $middlewareClass);
        }
    }

    protected function registerViews(): void
    {
        $viewPath = $this->extensionManifest['views']['path'] ?? resource_path('views');
        $namespace = $this->extensionManifest['views']['namespace'] ?? $this->extensionSlug;

        $this->loadViewsFrom($viewPath, $namespace);
    }

    protected function registerPublishes(): void
    {
        // Publish config
        $this->publishes([
            $this->getConfigPath() => config_path("{$this->extensionSlug}.php"),
        ], "{$this->extensionSlug}-config");

        // Publish views
        $viewPath = $this->extensionManifest['views']['path'] ?? resource_path('views');
        $this->publishes([
            $viewPath => resource_path("views/vendor/{$this->extensionSlug}"),
        ], "{$this->extensionSlug}-views");

        // Publish assets
        if (isset($this->extensionManifest['assets']['publishes'])) {
            foreach ($this->extensionManifest['assets']['publishes'] as $publish) {
                $this->publishes([
                    $publish['from'] => $publish['to'],
                ], "{$this->extensionSlug}-assets");
            }
        }
    }

    protected function bootRoutes(): void
    {
        $routeFiles = $this->extensionManifest['routes'] ?? [];

        if (!is_array($routeFiles)) {
            $routeFiles = [$routeFiles];
        }

        foreach ($routeFiles as $routeFile) {
            $routePath = $this->getRoutePath($routeFile);

            if (file_exists($routePath)) {
                $this->loadRoutesFrom($routePath);
            }
        }
    }

    protected function bootMigrations(): void
    {
        $this->loadMigrationsFrom($this->getMigrationPath());
    }

    protected function bootCommands(): void
    {
        if ($this->app->runningInConsole()) {
            $commands = $this->extensionManifest['commands'] ?? [];

            foreach ($commands as $commandClass) {
                $this->commands([$commandClass]);
            }
        }
    }

    protected function getConfigPath(): string
    {
        return $this->extensionManifest['config']
            ?? $this->getBasePath() . "/config/{$this->extensionSlug}.php";
    }

    protected function getRoutePath(string $file): string
    {
        return $this->getBasePath() . "/routes/{$file}";
    }

    protected function getMigrationPath(): string
    {
        return $this->getBasePath() . '/database/migrations';
    }

    public function getBasePath(): string
    {
        return dirname((new \ReflectionClass($this))->getFileName());
    }

    public function getExtensionManifest(): array
    {
        return $this->extensionManifest;
    }
}
```

## Trait: HasSettings

```php
<?php

namespace Gsmui\Extension\Traits;

use App\Plugins\Settings\PluginSettingsRepository;
use App\Plugins\Models\Plugin;

trait HasSettings
{
    protected string $settingsKey;
    protected PluginSettingsRepository $settingsRepository;
    protected ?Plugin $plugin = null;

    public function getSetting(string $key, mixed $default = null): mixed
    {
        return $this->getSettingsRepository()->get(
            $this->getPlugin(),
            $key,
            $default
        );
    }

    public function setSetting(string $key, mixed $value): void
    {
        $this->getSettingsRepository()->set(
            $this->getPlugin(),
            $key,
            $value
        );
    }

    public function getSettings(): array
    {
        return $this->getSettingsRepository()->getAll($this->getPlugin());
    }

    public function updateSettings(array $settings): void
    {
        $this->getSettingsRepository()->setMany(
            $this->getPlugin(),
            $settings
        );
    }

    protected function getSettingsRepository(): PluginSettingsRepository
    {
        if (!isset($this->settingsRepository)) {
            $this->settingsRepository = app(PluginSettingsRepository::class);
        }

        return $this->settingsRepository;
    }

    protected function getPlugin(): Plugin
    {
        if (!isset($this->plugin)) {
            $this->plugin = Plugin::where('slug', $this->settingsKey)->firstOrFail();
        }

        return $this->plugin;
    }
}
```

## Trait: IsExtendable

```php
<?php

namespace Gsmui\Extension\Traits;

trait IsExtendable
{
    protected array $extensionPoints = [];
    protected array $registeredExtensions = [];

    public function registerExtension(string $point, string $handler, int $priority = 0): void
    {
        $this->registeredExtensions[$point][] = [
            'handler' => $handler,
            'priority' => $priority,
        ];

        usort(
            $this->registeredExtensions[$point],
            fn($a, $b) => $b['priority'] <=> $a['priority']
        );
    }

    public function executeExtensionPoint(string $point, mixed $payload = null): mixed
    {
        $handlers = $this->registeredExtensions[$point] ?? [];

        foreach ($handlers as $handler) {
            $handlerInstance = app($handler['handler']);
            $payload = $handlerInstance->handle($payload);
        }

        return $payload;
    }

    public function hasExtensionPoint(string $point): bool
    {
        return in_array($point, $this->extensionPoints);
    }

    public function getExtensionPoints(): array
    {
        return $this->extensionPoints;
    }

    public function getRegisteredExtensions(string $point): array
    {
        return $this->registeredExtensions[$point] ?? [];
    }

    public function removeExtension(string $point, string $handler): void
    {
        $this->registeredExtensions[$point] = array_filter(
            $this->registeredExtensions[$point],
            fn($ext) => $ext['handler'] !== $handler
        );
    }
}
```

## Trait: ValidatesInput

```php
<?php

namespace Gsmui\Extension\Traits;

use Illuminate\Support\Facades\Validator;
use Illuminate\Validation\ValidationException;

trait ValidatesInput
{
    protected bool $throwOnValidationFailure = true;

    public function validate(array $data, array $rules, array $messages = []): array
    {
        $validator = Validator::make($data, $rules, $messages);

        if ($validator->fails()) {
            if ($this->throwOnValidationFailure) {
                throw ValidationException::withMessages($validator->errors()->toArray());
            }

            return [];
        }

        return $validator->validated();
    }

    public function validateField(mixed $value, string $rule, string $field = 'field'): ?string
    {
        $validator = Validator::make(
            [$field => $value],
            [$field => $rule]
        );

        if ($validator->fails()) {
            return $validator->errors()->first($field);
        }

        return null;
    }

    public function validateArray(array $data, array $schema): array
    {
        $errors = [];

        foreach ($schema as $field => $rules) {
            $fieldValue = data_get($data, $field);
            $fieldRules = is_array($rules) ? $rules : explode('|', $rules);

            $fieldValidator = Validator::make(
                [$field => $fieldValue],
                [$field => $fieldRules]
            );

            if ($fieldValidator->fails()) {
                $errors[$field] = $fieldValidator->errors()->first($field);
            }
        }

        return $errors;
    }
}
```

## Trait: Cacheable

```php
<?php

namespace Gsmui\Extension\Traits;

use Illuminate\Support\Facades\Cache;

trait Cacheable
{
    protected int $cacheTtl = 3600;
    protected string $cachePrefix = '';

    public function cacheGet(string $key, ?callable $callback = null): mixed
    {
        $fullKey = $this->getCacheKey($key);

        if (Cache::has($fullKey)) {
            return Cache::get($fullKey);
        }

        if ($callback) {
            $value = $callback();
            Cache::put($fullKey, $value, $this->cacheTtl);
            return $value;
        }

        return null;
    }

    public function cachePut(string $key, mixed $value, ?int $ttl = null): void
    {
        Cache::put(
            $this->getCacheKey($key),
            $value,
            $ttl ?? $this->cacheTtl
        );
    }

    public function cacheForget(string $key): void
    {
        Cache::forget($this->getCacheKey($key));
    }

    public function cacheFlush(): void
    {
        $prefix = $this->getCachePrefix();
        $keys = Cache::tags([$prefix])->keys();

        foreach ($keys as $key) {
            Cache::forget($key);
        }
    }

    protected function getCacheKey(string $key): string
    {
        return "{$this->getCachePrefix()}:{$key}";
    }

    protected function getCachePrefix(): string
    {
        return $this->cachePrefix ?: class_basename(static::class);
    }
}
```

```php
<?php

namespace Gsmui\Extension\Api;

use Illuminate\Http\Client\PendingRequest;
use Illuminate\Http\Client\Response;
use Illuminate\Support\Facades\Http;

class GsmuiApiClient
{
    private string $baseUrl;
    private string $token;
    private int $timeout;
    private int $retryAttempts;

    public function __construct(array $config = [])
    {
        $this->baseUrl = rtrim($config['base_url'] ?? config('gsmui.api_url', 'http://localhost:8000'), '/');
        $this->token = $config['token'] ?? '';
        $this->timeout = $config['timeout'] ?? 30;
        $this->retryAttempts = $config['retry_attempts'] ?? 3;
    }

    public function users(): UserApi
    {
        return new UserApi($this);
    }

    public function components(): ComponentApi
    {
        return new ComponentApi($this);
    }

    public function plugins(): PluginApi
    {
        return new PluginApi($this);
    }

    public function analytics(): AnalyticsApi
    {
        return new AnalyticsApi($this);
    }

    public function themes(): ThemeApi
    {
        return new ThemeApi($this);
    }

    public function request(string $method, string $endpoint, array $data = []): Response
    {
        $request = $this->buildRequest();

        $response = $request->withHeaders([
            'Authorization' => "Bearer {$this->token}",
            'Accept' => 'application/json',
            'Content-Type' => 'application/json',
            'X-Gsmui-SDK-Version' => '1.0.0',
            'X-Gsmui-Client' => 'php-sdk',
        ])->timeout($this->timeout)
          ->retry($this->retryAttempts, 1000)

                $response->json('error')
            );
        }

        return $response;
    }
```

```javascript
export class ComponentRegistry {
    constructor() {
        this.components = new Map();
        this.hooks = new Map();
    }

    register(name, definition) {
        if (this.components.has(name)) {
            console.warn(`Component "${name}" is already registered. Overwriting.`);
        }

        const validated = this.validateDefinition(definition);
        this.components.set(name, validated);

        this.executeHook('component:registered', { name, definition: validated });

        if (typeof Alpine !== 'undefined') {
            this.registerAlpineComponent(name, validated);
        }

        return this;
    }

    unregister(name) {
        this.components.delete(name);
        this.executeHook('component:unregistered', { name });
        return this;
    }

    get(name) {
        return this.components.get(name);
    }

    has(name) {
        return this.components.has(name);
    }

    list() {
        return Array.from(this.components.keys());
    }

    validateDefinition(definition) {
        const required = ['template'];
        const missing = required.filter(key => !(key in definition));

        if (missing.length > 0) {
            throw new Error(`Component definition missing required keys: ${missing.join(', ')}`);
        }

        return {
            template: definition.template,
            data: definition.data || (() => ({})),
            methods: definition.methods || {},
            computed: definition.computed || {},
            watch: definition.watch || {},
            mounted: definition.mounted || (() => {}),
            destroyed: definition.destroyed || (() => {}),
            props: definition.props || {},
        };
    }

    registerAlpineComponent(name, definition) {
        Alpine.data(name, () => ({
            ...definition.data(),
            init() {
                definition.mounted.call(this);
                this.$watch('$data', (value) => {
                    this.executeHook('component:data-changed', { name, data: value });
                });
            },
            destroy() {
                definition.destroyed.call(this);
            },
            ...definition.methods,
            ...Object.fromEntries(
                Object.entries(definition.computed || {}).map(([key, getter]) => [
                    key,
                    {
                        get() {
                            return getter.call(this);
                        },
                    },
                ])
            ),
        }));
    }

    on(hookName, callback) {
        if (!this.hooks.has(hookName)) {
            this.hooks.set(hookName, []);
        }
        this.hooks.get(hookName).push(callback);
        return () => {
            const hooks = this.hooks.get(hookName);
            const index = hooks.indexOf(callback);
            if (index > -1) hooks.splice(index, 1);
        };
    }

    executeHook(hookName, data) {
        const hooks = this.hooks.get(hookName) || [];
        hooks.forEach(callback => {
            try {
                callback(data);
            } catch (error) {
                console.error(`Hook "${hookName}" error:`, error);
            }
        });
    }
}
```

```javascript
// @gsmui/extension-sdk - Event System

export class EventBus {
    constructor() {
        this.listeners = new Map();
        this.wildcardListeners = [];
    }

    on(event, callback, options = {}) {
        const { priority = 0, once = false } = options;

        if (!this.listeners.has(event)) {
            this.listeners.set(event, []);
        }

        this.listeners.get(event).push({
            callback,
            priority,
            once,
        });

        this.listeners.get(event).sort((a, b) => b.priority - a.priority);

        return () => this.off(event, callback);
    }

    once(event, callback, options = {}) {
        return this.on(event, callback, { ...options, once: true });
    }

    off(event, callback) {
        if (!this.listeners.has(event)) return;

        if (callback) {
            const listeners = this.listeners.get(event);
            const index = listeners.findIndex(l => l.callback === callback);
            if (index > -1) listeners.splice(index, 1);
        } else {
            this.listeners.delete(event);
        }
    }

    emit(event, data = {}) {
        const listeners = this.listeners.get(event) || [];
        const toRemove = [];

        listeners.forEach((listener, index) => {
            try {
                listener.callback(data);
                if (listener.once) {
                    toRemove.push(index);
                }
            } catch (error) {
                console.error(`Event "${event}" handler error:`, error);
            }
        });

        toRemove.reverse().forEach(index => listeners.splice(index, 1));

        this.wildcardListeners.forEach(listener => {
            try {
                listener.callback(event, data);
            } catch (error) {
                console.error(`Wildcard listener error for event "${event}":`, error);
            }
        });
    }

    onAny(callback) {
        this.wildcardListeners.push({ callback });
        return () => {
            const index = this.wildcardListeners.findIndex(l => l.callback === callback);
            if (index > -1) this.wildcardListeners.splice(index, 1);
        };
    }

    removeAllListeners(event) {
        if (event) {
            this.listeners.delete(event);
        } else {
            this.listeners.clear();
            this.wildcardListeners = [];
        }
    }
}
```

## State Management

```javascript
// @gsmui/extension-sdk - State Management

export class ExtensionState {
    constructor(initialState = {}) {
        this.state = new Proxy(initialState, {
            set: (target, property, value) => {
                const oldValue = target[property];
                target[property] = value;
                this.notifyWatchers(property, value, oldValue);
                return true;
            },
        });

        this.watchers = new Map();
        this.persistenceKeys = new Set();
    }

    get(key) {
        return this.state[key];
    }

    set(key, value) {
        this.state[key] = value;
        if (this.persistenceKeys.has(key)) {
            this.persist(key, value);
        }
        return this;
    }

    watch(key, callback) {
        if (!this.watchers.has(key)) {
            this.watchers.set(key, []);
        }
        this.watchers.get(key).push(callback);

        return () => {
            const watchers = this.watchers.get(key);
            const index = watchers.indexOf(callback);
            if (index > -1) watchers.splice(index, 1);
        };
    }

    notifyWatchers(key, newValue, oldValue) {
        const watchers = this.watchers.get(key) || [];
        watchers.forEach(callback => {
            try {
                callback(newValue, oldValue, key);
            } catch (error) {
                console.error(`State watcher error for key "${key}":`, error);
            }
        });
    }

    persist(key, ttl = 3600) {
        this.persistenceKeys.add(key);
        const storageKey = `gsmui_ext_${key}`;
        const value = this.state[key];
        const expiry = Date.now() + ttl * 1000;

        try {
            localStorage.setItem(storageKey, JSON.stringify({ value, expiry }));
        } catch (error) {
            console.warn(`Failed to persist state key "${key}":`, error);
        }
    }

    restore(key) {
        const storageKey = `gsmui_ext_${key}`;
        try {
            const stored = JSON.parse(localStorage.getItem(storageKey));
            if (stored && stored.expiry > Date.now()) {
                this.state[key] = stored.value;
                return stored.value;
            }
        } catch (error) {
            console.warn(`Failed to restore state key "${key}":`, error);
        }
        return null;
    }

    clear() {
        Object.keys(this.state).forEach(key => {
            delete this.state[key];
        });
        this.watchers.clear();
        this.persistenceKeys.clear();
    }

    snapshot() {
        return JSON.parse(JSON.stringify(this.state));
    }

    restoreSnapshot(snapshot) {
        Object.entries(snapshot).forEach(([key, value]) => {
            this.state[key] = value;
        });
    }
}
```

```javascript
// @gsmui/extension-sdk - HTTP Client

export class HttpClient {
    constructor(options = {}) {
        this.baseUrl = options.baseUrl || '';
        this.headers = options.headers || {};
        this.timeout = options.timeout || 30000;
        this.retries = options.retries || 3;
    }

    async request(method, endpoint, data = null, options = {}) {
        const url = `${this.baseUrl}${endpoint}`;
        const config = {
            method,
            headers: {
                'Content-Type': 'application/json',
                'Accept': 'application/json',
                'X-Gsmui-SDK-Version': '1.0.0',
                ...this.headers,
                ...options.headers,
            },
            signal: AbortSignal.timeout(options.timeout || this.timeout),
        };

        if (data && ['POST', 'PUT', 'PATCH'].includes(method)) {
            config.body = JSON.stringify(data);
        }

        let lastError;
        for (let attempt = 0; attempt < (options.retries || this.retries); attempt++) {
            try {
                const response = await fetch(url, config);

                if (!response.ok) {
                    const errorBody = await response.json().catch(() => null);
                    throw new HttpError(
                        `HTTP ${response.status}: ${response.statusText}`,
                        response.status,
                        errorBody
                    );
                }

                return await response.json();
            } catch (error) {
                lastError = error;
                if (error instanceof HttpError) {
                    throw error;
                }
                if (attempt < (options.retries || this.retries) - 1) {
                    await this.delay(Math.pow(2, attempt) * 1000);
                }
            }
        }

        throw lastError;
    }

    get(endpoint, params = {}) {
        const queryString = new URLSearchParams(params).toString();
        const url = queryString ? `${endpoint}?${queryString}` : endpoint;
        return this.request('GET', url);
    }

    post(endpoint, data) {
        return this.request('POST', endpoint, data);
    }

    put(endpoint, data) {
        return this.request('PUT', endpoint, data);
    }

    patch(endpoint, data) {
        return this.request('PATCH', endpoint, data);
    }

    delete(endpoint) {
        return this.request('DELETE', endpoint);
    }

    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

export class HttpError extends Error {
    constructor(message, status, body) {
        super(message);
        this.name = 'HttpError';
        this.status = status;
        this.body = body;
    }
}
```

```php
<?php

namespace Gsmui\Extension\Exceptions;

class ExtensionException extends \RuntimeException
{
    protected string $extensionSlug;
    protected array $context = [];

    public function __construct(string $message, string $extensionSlug, array $context = [], ?\Throwable $previous = null)
    {
        parent::__construct($message, 0, $previous);
        $this->extensionSlug = $extensionSlug;
        $this->context = $context;
    }

    public function getExtensionSlug(): string
    {
        return $this->extensionSlug;
    }

    public function getContext(): array
    {
        return $this->context;
    }

    public function toArray(): array
    {
        return [
            'error' => static::class,
            'message' => $this->getMessage(),
            'extension' => $this->extensionSlug,
            'context' => $this->context,
        ];
    }
}
```

# SDK Testing

## Extension Test Helpers

```php
<?php

namespace Gsmui\Extension\Testing;

use App\Plugins\Models\Plugin;
use Orchestra\Testbench\TestCase;

abstract class ExtensionTestCase extends TestCase
{
    protected string $extensionSlug;
    protected string $extensionPath;

    protected function setUp(): void
    {
        parent::setUp();

        $this->extensionPath = base_path("extensions/{$this->extensionSlug}");
        $this->loadExtensionConfig();
        $this->loadExtensionMigrations();
    }

    protected function loadExtensionConfig(): void
    {
        $configPath = "{$this->extensionPath}/config/{$this->extensionSlug}.php";

        if (file_exists($configPath)) {
            $config = require $configPath;
            $this->app['config']->set($this->extensionSlug, $config);
        }
    }

    protected function loadExtensionMigrations(): void
    {
        $migrationPath = "{$this->extensionPath}/database/migrations";

        if (is_dir($migrationPath)) {
            $this->loadMigrationsFrom($migrationPath);
        }
    }

    protected function getExtensionPath(): string
    {
        return $this->extensionPath;
    }

    protected function assertExtensionConfigExists(string $key): void
    {
        $this->assertNotNull(
            config("{$this->extensionSlug}.{$key}"),
            "Extension config key '{$key}' not found"
        );
    }

    protected function assertExtensionRouteRegistered(string $routeName): void
    {
        $this->assertTrue(
            $this->app['router']->hasRoutes($routeName),
            "Extension route '{$routeName}' not registered"
        );
    }

    protected function assertExtensionViewExists(string $viewName): void
    {
        $this->assertTrue(
            view()->exists("{$this->extensionSlug}::{$viewName}"),
            "Extension view '{$viewName}' not found"
        );
    }

    protected function assertExtensionEventListened(string $eventName): void
    {
        $listeners = $this->app['events']->getListeners($eventName);
        $this->assertNotEmpty($listeners, "No listeners registered for event '{$eventName}'");
    }
}
```
