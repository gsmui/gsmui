# 13.24 Packaging — GsmUi Packaging Specification

## Overview

This document defines the packaging, distribution, and installation specifications for GsmUi components across multiple channels: Composer (PHP), NPM (JavaScript), CDN, and design tools.

## Package Structure

### Monorepo Layout

```

├── packages/
│   ├── gsmui-core/           # Base components (separate package)
```

### Service Provider

```php
<?php

namespace GsmUi;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\Blade;
use GsmUi\Support\GsmuiClasses;
use GsmUi\Support\IconRegistry;
use GsmUi\Services\ThemeManager;

class GsmUiProServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        // Register component namespace
        $this->bootComponentPath();
        
        // Publish assets
        $this->bootPublishing();
        
        // Register commands
        $this->bootCommands();
        
        // Register view composers
        $this->bootViewComposers();
    }
    
    protected function bootComponentPath(): void
    {
        Blade::anonymousComponentPath(
            __DIR__ . '/../resources/views/components',
            'gsmui'
        );
        
        // Also register pro-specific components
        Blade::anonymousComponentPath(
            __DIR__ . '/../resources/views/pro-components',
            'gsmui'
        );
    }
    
    protected function bootPublishing(): void
    {
        // Config
        $this->publishes([
            __DIR__ . '/../config/gsmui-pro.php' => config_path('gsmui-pro.php'),
        ], 'gsmui-pro-config');
        
        // CSP Config
        $this->publishes([
            __DIR__ . '/../stubs/config/gsmui-pro-csp.php' => config_path('gsmui-pro-csp.php'),
        ], 'gsmui-pro-csp-config');
        
        // Views (for customization)
        $this->publishes([
            __DIR__ . '/../stubs/resources/views' => resource_path('views/vendor/gsmui'),
        ], 'gsmui-pro-views');
        
        // Assets (for CDN/self-hosting)
        $this->publishes([
            __DIR__ . '/../dist' => public_path('vendor/gsmui'),
        ], 'gsmui-pro-assets');
    }
    
    protected function bootCommands(): void
    {
        if ($this->app->runningInConsole()) {
            $this->commands([
                \GsmUi\Commands\CompileTheme::class,
                \GsmUi\Commands\GenerateDocs::class,
                \GsmUi\Commands\InstallCommand::class,
            ]);
        }
    }
    
    protected function bootViewComposers(): void
    {
        View::composer('gsmui::components.*', function ($view) {
            $view->with([
                'gsmuiVersion' => '1.5.0',
                'gsmuiConfig' => config('gsmui-pro'),
                'gsmuiTheme' => app('gsmui.theme')->current(),
            ]);
        });
    }
    
    public function register(): void
    {
        $this->mergeConfigFrom(__DIR__ . '/../config/gsmui-pro.php', 'gsmui-pro');
        
        $this->app->singleton('gsmui.theme', fn() => new ThemeManager());
        $this->app->singleton('gsmui.icons', fn() => new IconRegistry());
    }
}
```

### Install Command

```php
<?php

namespace GsmUi\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\File;

class InstallCommand extends Command
{
    protected $signature = 'gsmui:install 
                            {--force : Overwrite existing files}
                            {--theme= : Theme to compile (default: default)}
                            {--no-assets : Skip asset publishing}';
    
    protected $description = 'Install GsmUi Pro in your Laravel application';
    
    public function handle(): int
    {
        $this->info('Installing GsmUi Pro...');
        
        // Publish config
        $this->call('vendor:publish', [
            '--tag' => 'gsmui-pro-config',
            '--force' => $this->option('force'),
        ]);
        
        // Publish CSP config
        $this->call('vendor:publish', [
            '--tag' => 'gsmui-pro-csp-config',
            '--force' => $this->option('force'),
        ]);
        
        // Publish assets
        if (!$this->option('no-assets')) {
            $this->call('vendor:publish', [
                '--tag' => 'gsmui-pro-assets',
                '--force' => $this->option('force'),
            ]);
        }
        
        // Compile theme
        $theme = $this->option('theme') ?? 'default';
        $this->call('gsmui:compile-theme', ['--theme' => $theme]);
        
        // Update .gitignore
        $this->updateGitIgnore();
        
        // Install JS dependencies
        $this->installJsDependencies();
        
        $this->info('GsmUi Pro installed successfully!');
        $this->line('Next steps:');
        $this->line('  1. Add @vite([\'vendor/gsmui/gsmui.js\', \'vendor/gsmui/gsmui.css\']) to your layout');
        $this->line('  2. Run npm run dev to compile assets');
        $this->line('  3. Configure CSP middleware if needed: config/gsmui-pro-csp.php');
        
        return self::SUCCESS;
    }
    
    protected function installJsDependencies(): void
    {
        $packageJson = base_path('package.json');
        
        if (File::exists($packageJson)) {
            $content = File::get($packageJson);
            $json = json_decode($content, true);
            
            $dependencies = [
                '@gsmui/pro' => '^1.5.0',
                '@gsmui/core' => '^2.15.0',
                'alpinejs' => '^3.13.0',
            ];
            
            foreach ($dependencies as $pkg => $version) {
                if (!isset($json['dependencies'][$pkg]) && !isset($json['devDependencies'][$pkg])) {
                    $this->line("Installing {$pkg}...");
                    exec("npm install {$pkg}@{$version}");
                }
            }
        }
    }
}
```

```yaml
            - run: cd packages/gsmui-pro && npm publish
```

## Tree-Shaking

### ES Module Tree-Shaking Configuration

GsmUi is designed for optimal tree-shaking. Only used components are included in the final bundle.

### Package Exports for Tree-Shaking

```json

    "type": "module",
    "sideEffects": [

    ],
    "exports": {
        ".": {
            "import": "./dist/gsmui.module.js",
            "require": "./dist/gsmui.umd.js",
```

### CDN Fallback Strategy

```html

    src="https://cdn.jsdelivr.net/npm/@gsmui/pro@1.5.0/dist/gsmui.min.js"
    integrity="sha384-{hash}"
    crossorigin="anonymous"

<script>
function loadFallback(failedScript) {
    // Fallback to unpkg
    const fallback = document.createElement('script');
    fallback.src = failedScript.src
        .replace('cdn.jsdelivr.net', 'unpkg.com')
        .replace('@gsmui/pro@', '@gsmui/pro@');
    fallback.onerror = function() {
        // Fallback to local
        const local = document.createElement('script');
        local.src = '/vendor/gsmui/gsmui.min.js';
        document.head.appendChild(local);
    };
    document.head.appendChild(fallback);
}
</script>
```

## Breaking Changes Policy

### Definition of Breaking Change

A breaking change is any modification that requires consumers to change their code to continue working:

- **Removed component** or **removed prop**
- **Changed prop type** (e.g., string to object)
- **Changed default behavior** (e.g., validation rules)
- **Changed event payload** structure
- **Changed CSS class names** (consumers targeting internal classes)
- **Changed ARIA attributes** (consumers testing accessibility)
- **Minimum version bump** of Laravel/Livewire/PHP

### Deprecation Process

```php
// Step 1: Mark as deprecated (current version)
/**
 * @deprecated Use `newProp` instead. Will be removed in v2.0.
 */
public string $oldProp = '';

// Step 2: Add migration path in docs
// Step 3: Add runtime deprecation notice (dev mode only)
public function mount(): void
{
    if (config('app.debug') && isset($this->oldProp) && $this->oldProp !== '') {
        trigger_error(
            "gsmui: Property `oldProp` is deprecated. Use `newProp` instead.",
            E_USER_DEPRECATED
        );
    }
}

// Step 4: Remove in next major version
```

### Automated Changelog Script

```javascript
// scripts/generate-changelog.js
import { readFileSync, writeFileSync } from 'fs';
import { execSync } from 'child_process';

function generateChangelog(fromTag, toTag) {
    const commits = execSync(
        `git log ${fromTag}..${toTag} --pretty=format:"%s|%h|%an" --no-merges`,
        { encoding: 'utf-8' }
    ).split('\n').filter(Boolean);

    const groups = {
        'Added': [],
        'Changed': [],
        'Fixed': [],
        'Deprecated': [],
        'Removed': [],
        'Security': [],
    };

    const prefixMap = {
        'feat': 'Added',
        'fix': 'Fixed',
        'docs': null,
        'style': null,
        'refactor': 'Changed',
        'perf': 'Changed',
        'test': null,
        'chore': null,
        'breaking-change': 'Removed',
    };

    for (const line of commits) {
        const [message, hash, author] = line.split('|');
        const match = message.match(/^(\w+)(?:\(.+\))?:\s*(.+)/);
        if (!match) continue;

        const [, prefix, description] = match;
        const group = prefixMap[prefix];
        if (!group) continue;

        groups[group].push(`- ${description} (${hash})`);
    }

    let changelog = `## [${toTag.replace('v', '')}] - ${new Date().toISOString().split('T')[0]}\n\n`;
    for (const [group, items] of Object.entries(groups)) {
        if (items.length > 0) {
            changelog += `### ${group}\n${items.join('\n')}\n\n`;
        }
    }

    return changelog;
}

const fromTag = process.argv[2] || execSync('git describe --tags --abbrev=0 HEAD~1', { encoding: 'utf-8' }).trim();
const toTag = process.argv[3] || 'HEAD';
const changelog = generateChangelog(fromTag, toTag);

const existing = readFileSync('CHANGELOG.md', 'utf-8');
writeFileSync('CHANGELOG.md', `# Changelog\n\n${changelog}\n${existing.replace(/^# Changelog\n\n/, '')}`);

console.log('Changelog generated!');
```

```yaml
# .github/workflows/post-release-monitor.yml
name: Post-Release Monitor

on:
  release:
    types: [published]

jobs:
  monitor:
    runs-on: ubuntu-latest
    steps:
      - name: Wait for Packagist
        run: sleep 60

      - name: Check Packagist availability
        run: |
          RESPONSE=$(curl -s -o /dev/null -w "%{http_code}" \
            "https://packagist.org/packages/gsmui/gsmui-pro.json")
          if [ "$RESPONSE" != "200" ]; then
            echo "⚠️ Package not yet available on Packagist"
          fi

      - name: Check NPM availability
        run: |
          TAG=${GITHUB_REF#refs/tags/}
          RESPONSE=$(curl -s -o /dev/null -w "%{http_code}" \
            "https://registry.npmjs.org/@gsmui/pro/${TAG}")
          if [ "$RESPONSE" != "200" ]; then
            echo "⚠️ Package not yet available on NPM"
          fi

      - name: Notify Discord
        if: success()
        run: |
          TAG=${GITHUB_REF#refs/tags/}
          curl -X POST "${{ secrets.DISCORD_WEBHOOK }}" \
            -H "Content-Type: application/json" \
```

### Changeset Workflow

```bash
# 1. Add changeset for your changes
npx changeset

# 2. Changesets are automatically tracked
git add .changeset/
git commit -m "fix: correct select positioning"

# 3. When ready to release, version packages

# 4. This updates CHANGELOG.md and bumps versions in package.json

# 5. Build and publish
npm run build:prod
npx changeset publish

# 6. Push tags
git push --follow-tags
```

## Design Tool Packages

### Figma Token Sync

```php
<?php

namespace GsmUi\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\File;

class ExportFigmaTokens extends Command
{
    protected $signature = 'gsmui:export-figma-tokens {--output=figma-tokens.json} {--format=tokens-studio}';

    public function handle(): int
    {
        $theme = config('gsmui.theme.themes.default');
        $tokens = $this->convertToFigmaFormat($theme);
        
        File::put(
            base_path($this->option('output')),
            json_encode($tokens, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES)
        );

        $this->info("Figma tokens exported to {$this->option('output')}");
        return self::SUCCESS;
    }

    protected function convertToFigmaFormat(array $theme): array
    {
        $tokens = ['color' => [], 'spacing' => [], 'borderRadius' => [], 'typography' => []];
        
        // Colors
        if (isset($theme['colors'])) {
            foreach ($theme['colors'] as $name => $shades) {
                if (is_array($shades) && isset($shades['light'])) {
                    $tokens['color'][$name] = [
                        'value' => $shades['light'],
                        'type' => 'color',
                        'description' => "Semantic color: {$name}",
                    ];
                }
            }
        }
        
        // Spacing
        $spacing = config('gsmui.theme.spacing', [1 => '0.25rem', 2 => '0.5rem', 3 => '0.75rem', 4 => '1rem']);
        foreach ($spacing as $step => $value) {
            $tokens['spacing'][$step] = ['value' => $value, 'type' => 'dimension'];
        }
        
        // Border radius
        $radius = config('gsmui.theme.radius', ['sm' => '0.125rem', 'md' => '0.25rem', 'lg' => '0.375rem']);
        foreach ($radius as $name => $value) {
            $tokens['borderRadius'][$name] = ['value' => $value, 'type' => 'dimension'];
        }
        
        // Typography
        $fontFamily = $theme['typography']['fontFamily'] ?? 'Inter, system-ui, sans-serif';
        $tokens['typography']['fontFamily'] = ['value' => $fontFamily, 'type' => 'fontFamily'];
        
        return [
            'name' => 'GsmUi Design Tokens',
            'version' => config('gsmui.version', '1.0.0'),
            'tokenSets' => $tokens,
        ];
    }
}
```

### Sketch/Library Sync

```javascript
// scripts/export-sketch-tokens.js
const fs = require('fs');
const tokens = require('../resources/css/tokens.json');

function exportSketchPalette() {
    const palette = Object.entries(tokens.color)
        .filter(([_, config]) => config.type === 'color')
        .map(([name, config]) => ({
            name,
            red: parseInt(config.value.slice(1, 3), 16),
            green: parseInt(config.value.slice(3, 5), 16),
            blue: parseInt(config.value.slice(5, 7), 16),
            alpha: 1,
        }));

    fs.writeFileSync(
        'dist/gsmui-sketch-palette.sketchpalette',
        JSON.stringify({ compatibleVersion: '2.0', pluginVersion: '2.22', colors: palette })
    );
}

module.exports = { exportSketchPalette };
```

### Integrity Checking Script

```bash
#!/bin/bash
# scripts/verify-integrity.sh

echo "Verifying package integrity..."

# Check Composer
echo "Composer packages:"
composer show --direct --format=json | jq -r '.installed[] | "\(.name)@\(.version)"' | while read pkg; do
    echo "  ✓ $pkg"
done

# Check NPM
echo "NPM packages:"
npm ls --depth=0 --json | jq -r '.dependencies // {} | to_entries[] | "\(.key)@\(.value.version)"' | while read pkg; do
    echo "  ✓ $pkg"
done

# Verify no unexpected modifications
echo "Checking for uncommitted changes..."
if [ -n "$(git status --porcelain)" ]; then
    echo "⚠️ Uncommitted changes detected"
    git status --short
else
    echo "✓ Working tree clean"
fi
```
