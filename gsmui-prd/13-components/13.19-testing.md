# 13.19 Testing — GsmUi Testing Specification

## Overview

GsmUi follows a comprehensive testing strategy covering unit tests, render tests, accessibility tests, visual regression tests, and browser/E2E tests. Every component must pass all test layers before release.

## Test Pyramid

```
                    ┌─────────────┐
                    │  E2E Tests  │  ← Few, critical paths
                    └──────┬──────┘
                           │
              ┌────────────┴────────────┐
              │   Visual Regression     │  ← Per component, key states
              └────────────┬────────────┘
                           │
       ┌───────────────────┴───────────────────┐
       │           Accessibility               │  ← Every component
       └───────────────────┬───────────────────┘
                           │
       ┌───────────────────┴───────────────────┐
       │            Render Tests               │  ← Every component + variants
       └───────────────────┬───────────────────┘
                           │
       ┌───────────────────┴───────────────────┐
       │            Unit Tests                 │  ← Logic, traits, utilities
       └───────────────────────────────────────┘
```

## Unit Tests (Pest/PHPUnit)

### Base Test Case

```php
<?php

namespace Tests\Unit;

use GsmUi\Components\BaseComponent;
use Livewire\Component;
use Pest\Laravel\Livewire;

abstract class ComponentTestCase extends TestCase
{
    use InteractsWithLivewire;
    
    protected function component(string $class, array $props = []): Livewire
    {
        return Livewire::test($class, $props);
    }
    
    protected function assertPropsSet(Livewire $component, array $expected): void
    {
        foreach ($expected as $prop => $value) {
            $this->assertEquals($value, $component->get($prop), "Prop {$prop} mismatch");
        }
    }
    
    protected function assertEmittedEventsDispatched(Livewire $component, string $event, array $expectedPayload = []): void
    {
        $events = $component->dispatchedEvents($event);
        $this->assertNotEmpty($events);
        
        if ($expectedPayload) {
            $this->assertEquals($expectedPayload, $events[0]['params'] ?? []);
        }
    }
}
```

## Render Tests

### Blade Render Test Case

```php
<?php

namespace Tests\Render;

use Illuminate\Support\Facades\Blade;
use Orchestra\Testbench\TestCase as BaseTestCase;

abstract class BladeTestCase extends BaseTestCase
{
    protected function render(string $blade, array $data = []): string
    {
        return Blade::render($blade, $data);
    }
    
    protected function renderComponent(string $component, array $props = [], string $slot = ''): string
    {
        $attrs = collect($props)->map(fn($v, $k) => ":{$k}=\"{$v}\"")->implode(' ');
        $slotContent = $slot ? ">{$slot}</x-gsmui:{$component}>" : '/>';
        
        return $this->render("<x-gsmui:{$component} {$attrs}{$slotContent}");
    }
}
```

## Accessibility Tests

### Axe-Core Integration

```php
<?php

namespace Tests\Browser\A11y;

use Tests\Browser\BrowserTestCase;

class AccessibilityTest extends BrowserTestCase
{
    /** @var \Axe\Runner */
    protected $axe;
    
    protected function setUp(): void
    {
        parent::setUp();
        $this->axe = new \Axe\Runner();
    }
    
    public function test_all_components_pass_axe_audit()
    {
        $components = [
            'button' => '<x-gsmui:button>Test</x-gsmui:button>',
            'input' => '<x-gsmui:input label="Email" type="email" />',
            'select' => '<x-gsmui:select :options="[[\'value\' => 1, \'label\' => \'One\']]" />',
            'checkbox' => '<x-gsmui:checkbox label="Agree" />',
            'modal' => '<x-gsmui:modal :open="true"><x-slot name="title">Title</x-slot></x-gsmui:modal>',
        ];
        
        foreach ($components as $name => $blade) {
            $html = Blade::render($blade);
            $results = $this->axe->analyze($html);
            
            $this->assertEmpty($results->violations(), "{$name} has accessibility violations: " . json_encode($results->violations()));
        }
    }
    
    public function test_focus_order_is_logical()
    {
        $this->visit('/docs/forms');
        
        $focusable = $this->browser->elements('button, input, select, textarea, a[href], [tabindex]:not([tabindex="-1"])');
        
        $previousTabIndex = -1;
        foreach ($focusable as $element) {
            $tabIndex = $element->getAttribute('tabindex') ?? 0;
            $this->assertGreaterThanOrEqual($previousTabIndex, (int)$tabIndex);
            $previousTabIndex = (int)$tabIndex;
        }
    }
    
    public function test_keyboard_navigation_works()
    {
        $this->visit('/docs/select');
        $select = $this->browser->element('x-gsmui-select');
        
        // Tab to select
        $select->keys('Tab');
        $this->assertTrue($select->hasFocus());
        
        // Open with Enter
        $select->keys('Enter');
        $this->assertTrue($select->getAttribute('aria-expanded') === 'true');
        
        // Navigate
        $select->keys('ArrowDown');
        $this->assertTrue($select->getAttribute('aria-activedescendant') !== '');
        
        // Select
        $select->keys('Enter');
        $this->assertEquals('option-value', $select->getValue());
        
        // Close with Escape
        $select->keys('Escape');
        $this->assertEquals('false', $select->getAttribute('aria-expanded'));
    }
}
```

```typescript
import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';

test.describe('Accessibility', () => {
    const components = [
        { name: 'button', url: '/docs/button' },
        { name: 'input', url: '/docs/input' },
        { name: 'select', url: '/docs/select' },
        { name: 'modal', url: '/docs/modal' },

            await page.waitForLoadState('networkidle');
            
            const results = await new AxeBuilder({ page }).analyze();
            
            expect(results.violations).toHaveLength(0);

        const interactive = await page.locator('button, input, select, textarea, a[href], [tabindex]:not([tabindex="-1"])').all();
        
        for (const element of interactive) {
            await element.focus();
            await expect(element).toBeFocused();
            
            // Check focus ring is visible
            const styles = await element.evaluate(el => getComputedStyle(el));
            expect(styles.outlineWidth).not.toBe('0px');
        }
    });
    
    test('select keyboard navigation', async ({ page }) => {
        await page.goto('/docs/select');

        await trigger.focus();

        await page.keyboard.press('ArrowDown');
```

### Visual Test Runner

```typescript
// tests/Browser/Visual/visual-regression.spec.ts
import { test, expect } from '@playwright/test';
import pixelmatch from 'pixelmatch';
import { PNG } from 'pngjs';
import fs from 'fs';
import path from 'path';

const VIEWPORTS = [
    { name: 'mobile', width: 375, height: 667 },
    { name: 'tablet', width: 768, height: 1024 },
    { name: 'desktop', width: 1280, height: 720 },
];

const THEMES = ['light', 'dark'];

test.describe.configure({ retries: 0 });

for (const viewport of VIEWPORTS) {
    for (const theme of THEMES) {
        test.describe(`${viewport.name} - ${theme}`, () => {
            test.use({ viewport, colorScheme: theme });
            
            for (const state of STATES) {
                test(`button-${state}`, async ({ page }) => {
                    await page.goto('/visual/button');
                    await page.waitForLoadState('networkidle');
                    
                    // Apply state
                    await applyState(page, 'button', state);
                    
                    // Screenshot
                    const screenshot = await page.locator('[data-testid="button-primary"]').screenshot();
                    
                    // Compare
                    await compareScreenshot('button', 'primary', state, viewport.name, theme, screenshot);
                });
            }
        });
    }
}

async function applyState(page, component, state) {
    const actions = {
        hover: () => page.hover('[data-testid="button-primary"]'),
        focus: () => page.focus('[data-testid="button-primary"]'),
        disabled: () => page.evaluate(() => document.querySelector('[data-testid="button-primary"]').setAttribute('disabled', '')),
        loading: () => page.evaluate(() => document.querySelector('[data-testid="button-primary"]').setAttribute('data-loading', 'true')),
        invalid: () => page.evaluate(() => document.querySelector('[data-testid="button-primary"]').setAttribute('data-invalid', 'true')),
    };
    
    if (actions[state]) {
        await actions[state]();
        await page.waitForTimeout(100); // Wait for transition
    }
}

async function compareScreenshot(component, variant, state, viewport, theme, currentBuffer) {
    const baselinePath = path.join(
        __dirname, 
        '../../Visual/__snapshots__', 
        `${component}-${variant}-${state}-${viewport}-${theme}.png`
    );
    
    if (!fs.existsSync(baselinePath)) {
        fs.writeFileSync(baselinePath, currentBuffer);
        return; // First run - create baseline
    }
    
    const baseline = PNG.sync.read(fs.readFileSync(baselinePath));
    const current = PNG.sync.read(currentBuffer);
    
    const { width, height } = baseline;
    const diff = new PNG({ width, height });
    
    const numDiffPixels = pixelmatch(
        baseline.data, 
        current.data, 
        diff.data, 
        width, 
        height, 
        { threshold: 0.01 }
    );
    
    if (numDiffPixels > 0) {
        // Save diff for debugging
        fs.writeFileSync(
            baselinePath.replace('.png', '-diff.png'), 
            PNG.sync.write(diff)
        );
        
        throw new Error(
            `Visual regression: ${numDiffPixels} pixels differ in ${component}-${variant}-${state}-${viewport}-${theme}`
        );
    }
}
```

```typescript
import { test, expect, Page } from '@playwright/test';

test.describe('Select - Advanced Interactions', () => {

    test.beforeEach(async ({ browser }) => {

        await trigger.click();
        await expect(page.locator('[role="listbox"]')).toBeVisible();

        const firstOption = page.locator('[role="option"]').first();
        const optionText = await firstOption.textContent();
        await firstOption.click();

        await expect(trigger).toContainText(optionText!);
        await expect(page.locator('[role="listbox"]')).toBeHidden();
    });

    test('searchable select filters options', async () => {
        await page.goto('/docs/select-searchable');
        await page.waitForLoadState('networkidle');

        const trigger = page.locator('[role="combobox"]');
        await trigger.click();

        const search = page.locator('[role="searchbox"]');
        await search.fill('opt');

        const options = page.locator('[role="option"]');
        const count = await options.count();
        expect(count).toBeLessThan(10);

        for (let i = 0; i < count; i++) {
            await expect(options.nth(i)).toContainText(/opt/i);
        }
    });

        await page.goto('/docs/select-multiple');
        await page.waitForLoadState('networkidle');

        await page.locator('[role="option"]').nth(0).click();
        await page.locator('[role="option"]').nth(1).click();
        await page.locator('[role="option"]').nth(2).click();
```

```php
namespace Tests\Helpers;

use Livewire\Livewire;
use Livewire\Features\SupportTesting\Testable;

class LivewireMockHelper
{
    public static function mockComponent(string $componentClass, array $props = []): Testable
    {
        return Livewire::test($componentClass, $props);
    }

    public static function assertEventDispatched(Testable $component, string $event, $payload = null): void
    {
        $component->assertDispatched($event, $payload);
    }

    public static function assertNotEventDispatched(Testable $component, string $event): void
    {
        $component->assertNotDispatched($event);
    }

    {
        return $component->set($property, $value);
    }

    public static function simulateFormSubmit(Testable $component, array $data = []): Testable
    {
        if (!empty($data)) {
            foreach ($data as $key => $value) {
                $component->set($key, $value);
            }
        }
        return $component->call('submit');
    }

    public static function assertValidationErrors(Testable $component, array $fields): void
    {
        $component->assertHasErrors($fields);
    }

    public static function assertNoValidationErrors(Testable $component): void
    {
        $component->assertHasNoErrors();
    }

    public static function assertViewHas(Testable $component, string $key, $value = null): void
    {
        if ($value === null) {
            $component->assertViewHas($key);
        } else {
            $component->assertViewHas($key, $value);
        }
    }
```

```php
class GsmuiTestHelper

    public static function renderComponent(string $component, array $props = [], string $slot = ''): string
    {
        $attrs = collect($props)
            ->map(fn($v, $k) => is_bool($v) ? ":{$k}=\"{$v ? 'true' : 'false'}\"" : ":{$k}=\"{$v}\"")
            ->implode(' ');

        $tag = "x-gsmui:{$component}";
        if ($slot) {
            return Blade::render("<{$tag} {$attrs}>{$slot}</{$tag}>");
        }
        return Blade::render("<{$tag} {$attrs} />");

    public static function assertHtmlContainsClasses(string $html, array $expectedClasses): void
    {
        foreach ($expectedClasses as $class) {
            if (!str_contains($html, $class)) {
                throw new \PHPUnit\Framework\AssertionFailedError(
                    "Expected HTML to contain class '{$class}' but it did not.\nHTML: " . substr($html, 0, 500)
                );
            }
        }
    }

    public static function assertHtmlHasAriaAttribute(string $html, string $attribute, string $expectedValue = null): void
    {
        if ($expectedValue !== null) {
            $pattern = preg_quote("{$attribute}=\"{$expectedValue}\"", '/');
        } else {
            $pattern = preg_quote($attribute, '/') . '\s*=';
        }
        if (!preg_match("/{$pattern}/", $html)) {
            throw new \PHPUnit\Framework\AssertionFailedError(
                "Expected HTML to have aria attribute '{$attribute}'" .
                ($expectedValue ? " with value '{$expectedValue}'" : '') .
                ".\nHTML: " . substr($html, 0, 500)
            );
        }
    }

    public static function assertHtmlHasRole(string $html, string $role): void
    {
        self::assertHtmlHasAriaAttribute($html, 'role', $role);
    }

    public static function assertHtmlHasId(string $html, string $id): void
    {
        if (!str_contains($html, "id=\"{$id}\"")) {
            throw new \PHPUnit\Framework\AssertionFailedError(
                "Expected HTML to have id='{$id}'.\nHTML: " . substr($html, 0, 500)
            );
        }
    }
```

```typescript
// tests/Browser/Helpers/test-utils.ts
import { Page, Locator, expect } from '@playwright/test';

export async function openComponent(page: Page, name: string, props: Record<string, string> = {}): Promise<void> {
    const params = new URLSearchParams(props);
    await page.goto(`/docs/${name}?${params.toString()}`);

}

export async function waitForLivewire(page: Page): Promise<void> {
    await page.waitForFunction(() => {
        return window.Livewire && !window.Livewire.hooks?.message?.pending;
    }, { timeout: 10000 });
}

export async function assertNoConsoleErrors(page: Page): Promise<void> {
    const errors: string[] = [];
    page.on('console', (msg) => {
        if (msg.type() === 'error') errors.push(msg.text());
    });
    await page.waitForTimeout(1000);
    expect(errors).toHaveLength(0);
}

export async function assertAriaLabel(page: Page, selector: string, expected: string): Promise<void> {
    const element = page.locator(selector);
    await expect(element).toHaveAttribute('aria-label', expected);
}

export async function assertAriaDescribedBy(page: Page, selector: string, expectedId: string): Promise<void> {
    const element = page.locator(selector);
    const describedBy = await element.getAttribute('aria-describedby');
    expect(describedBy).toContain(expectedId);
}

export async function focusAndAssert(page: Page, selector: string): Promise<void> {
    const element = page.locator(selector);
    await element.focus();
    await expect(element).toBeFocused();
}

export async function tabThroughElements(page: Page, selectors: string[]): Promise<void> {
    for (const selector of selectors) {
        await page.keyboard.press('Tab');
        const element = page.locator(selector);
        await expect(element).toBeFocused();
    }
}

export async function assertNoFocusTrap(page: Page, dialogSelector: string): Promise<void> {
    await page.keyboard.press('Tab');
    const isInDialog = await page.evaluate((sel) => {
        const dialog = document.querySelector(sel);
        return dialog?.contains(document.activeElement);
    }, dialogSelector);
    expect(isInDialog).toBe(false);
}

export async function measureRenderTime(page: Page, action: () => Promise<void>): Promise<number> {
    const start = Date.now();
    await action();
    return Date.now() - start;
}
```
