# 13.15 Blade Integration — GsmUi Blade Specification

## Overview

GsmUi components are implemented as **Laravel Anonymous Blade Components** with the `gsmui:` prefix. This document specifies the Blade integration patterns, conventions, and APIs.

## Component Registration

### Service Provider

```php
<?php

namespace GsmUi;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\Blade;

class GsmuiProServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        // Register component namespace
        $this->bootComponentPath();
        
        // Register custom directives
        $this->bootDirectives();
        
        // Publish assets
        $this->bootPublishing();
    }
    
    protected function bootComponentPath(): void
    {
        Blade::anonymousComponentPath(
            __DIR__ . '/../stubs/resources/views',
            'gsmui'
        );
        
        // Also register pro components
        Blade::anonymousComponentPath(
            __DIR__ . '/../resources/views',
            'gsmui'
        );
    }
}
```

## Component Authoring

### Basic Component

```blade
{{-- resources/views/components/gsmui/button/index.blade.php --}}
@props([
    'variant' => 'primary',
    'size' => 'md',
    'color' => 'primary',
    'disabled' => false,
    'loading' => false,
    'as' => 'button',           // button, a, div
    'href' => null,             // for <a> tag
    'type' => 'button',         // for <button> tag
    'icon' => null,
    'iconPosition' => 'left',   // left, right
    'fullWidth' => false,
    'class' => '',
    'attributes' => [],
])

<?php
$classes = \Gsmui\Support\GsmuiClasses::join([
    'inline-flex items-center justify-center font-medium transition-colors',
    'focus-visible:outline-2 focus-visible:outline-[-webkit-focus-ring-color]',
    $this->variantClasses($variant),
    $this->sizeClasses($size),
    $this->colorClasses($variant, $color),
    $disabled ? 'opacity-50 cursor-not-allowed pointer-events-none' : '',
    $loading ? 'relative pr-8' : '',
    $fullWidth ? 'w-full' : '',
    $class,
    $attributes->get('class', ''),
]);

$tag = $as;
$isButton = $tag === 'button';
$isLink = $tag === 'a';
$isDiv = $tag === 'div';
?>

<{{ $tag }}
    {{ $attributes->merge(['class' => $classes])->except(['class', 'href', 'type']) }}
    @if($isButton)
        type="{{ $type }}"
        @if($disabled) disabled @endif
    @endif
    @if($isLink)
        href="{{ $href }}"
        @if($disabled) tabindex="-1" aria-disabled="true" @endif
    @endif
    @if($isDiv)
        role="button"
        tabindex="0"
        @if($disabled) aria-disabled="true" @endif
    @endif
>
    {{-- Loading State --}}
    @if($loading)
        <svg class="animate-spin -ml-1 mr-2 h-4 w-4" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" aria-hidden="true">
            <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle>
            <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
        </svg>
    @endif

    {{-- Icon (left) --}}
    @if($icon && $iconPosition === 'left')
        <x-gsmui:icon :name="$icon" class="w-4 h-4 @if(!$loading) mr-2 @endif" aria-hidden="true" />
    @endif

    {{-- Content --}}
    <span class="flex items-center justify-center {{ $fullWidth ? 'w-full' : '' }}">
        @if($label)
            <span class="font-medium">{{ $label }}</span>
        @else
            {{ $slot }}
        @endif
    </span>

    {{-- Icon (right) --}}
    @if($icon && $iconPosition === 'right')
        <x-gsmui:icon :name="$icon" class="w-4 h-4 @if(!$loading) ml-2 @endif" aria-hidden="true" />
    @endif
</{{ $tag }}>

 {{-- Description / Help Text --}}
@if($description)
    <p id="{{ $attributes->id ?? '' }}-description" class="mt-1 text-sm text-zinc-500 dark:text-zinc-400">
        {{ $description }}
    </p>
@endif
```

### Class Merging Helper

```php
// Gsmui\Support\GsmuiClasses
class GsmuiClasses
{
    public static function join(...$classes): string
    {
        return collect($classes)
            ->flatten()
            ->filter(fn($c) => filled($c))
            ->implode(' ');
    }
    
    public static function when(bool $condition, string $trueClass, string $falseClass = ''): string
    {
        return $condition ? $trueClass : $falseClass;
    }
}
```

## Directive Registration

### Custom Directives

```php
// In ServiceProvider
protected function bootDirectives(): void
{
    Blade::directive('gsmuiClass', function ($expression) {
        return "<?php echo \\Gsmui\\Support\\GsmuiClasses::join({$expression}); ?>";
    });
```

## Anonymous Component Registration

### Deep Component Path Registration

GsmUi registers Blade anonymous components from multiple paths with priority resolution:

### Multi-Path Registration

```php
<?php

namespace GsmUi;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\Blade;

class GsmuiProServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        $this->registerComponentPaths();
        $this->registerComponentOverrides();
        $this->registerComponentFallbacks();
    }

    protected function registerComponentPaths(): void
    {
        // Core components (lowest priority)
        Blade::anonymousComponentPath(
            __DIR__ . '/../stubs/resources/views/components',
            'gsmui'
        );

        // Pro components (medium priority)
        Blade::anonymousComponentPath(
            __DIR__ . '/../resources/views/pro-components',
            'gsmui'
        );

        // User overrides (highest priority)
        if (is_dir(resource_path('views/vendor/gsmui'))) {
            Blade::anonymousComponentPath(
                resource_path('views/vendor/gsmui'),
                'gsmui'
            );
        }
    }

    protected function registerComponentOverrides(): void
    {
        Blade::anonymousComponentPath(
            resource_path('views/vendor/gsmui-overrides'),
            'gsmui'
        );
    }

    protected function registerComponentFallbacks(): void
    {
        Blade::anonymousComponentPath(
            __DIR__ . '/../resources/views/fallbacks',
            'gsmui'
        );
    }
}
```

### Runtime Component Registration

```php
<?php

namespace GsmUi\Support;

class ComponentRegistry
{
    protected array $registered = [];
    protected array $aliases = [];
    protected array $disabled = [];

    public function register(string $name, string $view, array $aliases = []): void
    {
        $this->registered[$name] = $view;
        foreach ($aliases as $alias) {
            $this->aliases[$alias] = $name;
        }
        Blade::anonymousComponentPath($view, 'gsmui');
    }

    public function alias(string $alias, string $component): void
    {
        $this->aliases[$alias] = $component;
    }

    public function disable(string $component): void
    {
        $this->disabled[] = $component;
    }

    public function isDisabled(string $component): bool
    {
        return in_array($component, $this->disabled);
    }

    public function resolve(string $name): ?string
    {
        if ($this->isDisabled($name)) return null;
        if (isset($this->aliases[$name])) {
            return $this->resolve($this->aliases[$name]);
        }
        return $this->registered[$name] ?? null;
    }

    public function all(): array
    {
        return $this->registered;
    }
}
```

## Component Attributes

### Advanced Attribute Handling

GsmUi components support complex attribute patterns including nested attributes, conditional merging, and attribute bags.

### Attribute Bag Extension

```php
<?php

namespace GsmUi\Support;

use Illuminate\View\ComponentAttributeBag;

class GsmuiAttributeBag extends ComponentAttributeBag
{
    public function mergeClasses(string ...$classes): self
    {
        $existing = $this->get('class', '');
        $merged = GsmuiClasses::join($existing, ...$classes);
        return $this->merge(['class' => $merged]);
    }

    public function mergeStyles(string ...$styles): self
    {
        $existing = $this->get('style', '');
        $merged = trim($existing . '; ' . implode('; ', $styles), '; ');
        return $this->merge(['style' => $merged]);
    }

    public function exceptProps(array $props): self
    {
        $bag = clone $this;
        foreach ($props as $prop) {
            unset($bag->$prop);
        }
        return $bag;
    }

    public function onlyProps(array $props): self
    {
        $bag = new self();
        foreach ($props as $prop) {
            if ($this->has($prop)) {
                $bag->$prop = $this->$prop;
            }
        }
        return $bag;
    }

    public function dataAttributes(): self
    {
        $bag = new self();
        foreach ($this->getAttributes() as $key => $value) {
            if (str_starts_with($key, 'data-')) {
                $bag->$key = $value;
            }
        }
        return $bag;
    }

    public function ariaAttributes(): self
    {
        $bag = new self();
        foreach ($this->getAttributes() as $key => $value) {
            if (str_starts_with($key, 'aria-')) {
                $bag->$key = $value;
            }
        }
        return $bag;
    }

    public function mergeAria(array $aria): self
    {
        $merged = [];
        foreach ($this->getAttributes() as $key => $value) {
            if (str_starts_with($key, 'aria-')) {
                $merged[$key] = $value;
            }
        }
        $merged = array_merge($merged, $aria);
        $bag = clone $this;
        foreach ($merged as $key => $value) {
            $bag->$key = $value;
        }
        return $bag;
    }

    public function mergeData(array $data): self
    {
        $merged = [];
        foreach ($this->getAttributes() as $key => $value) {
            if (str_starts_with($key, 'data-')) {
                $merged[$key] = $value;
            }
        }
        $merged = array_merge($merged, $data);
        $bag = clone $this;
        foreach ($merged as $key => $value) {
            $bag->$key = $value;
        }
        return $bag;
    }

    public function withId(string $suffix): self
    {
        $id = $this->get('id', '');
        return $this->merge(['id' => "{$id}-{$suffix}"]);
    }

    public function withDefault(string $key, mixed $default): self
    {
        if (!$this->has($key)) {
            return $this->merge([$key => $default]);
        }
        return $this;
    }
}
```

### Component Memoization

```php
<?php

namespace GsmUi\Concerns;

trait MemoizesRendering
{
    protected ?string $renderCache = null;
    protected int $lastRenderTimestamp = 0;

    public function render(): \Illuminate\Contracts\View\View
    {
        $cacheKey = $this->getRenderCacheKey();
        $now = microtime(true);

        // Cache for 1 second
        if ($this->renderCache && ($now - $this->lastRenderTimestamp) < 1.0) {
            return view('gsmui::cached', ['html' => $this->renderCache]);
        }

        $view = $this->buildView();
        $this->renderCache = $view->render();
        $this->lastRenderTimestamp = $now;

        return $view;
    }

    protected function getRenderCacheKey(): string
    {
        return md5(json_encode([
            $this->getName(),
            get_object_vars($this),
        ]));
    }
}
```

### Custom Directive Implementation

```php
// In GsmuiProServiceProvider
protected function bootDirectives(): void
{
    Blade::directive('gsmuiClass', function ($expression) {
        return "<?php echo \\Gsmui\\Support\\GsmuiClasses::join({$expression}); ?>";
    });

    Blade::directive('gsmuiVariant', function ($expression) {
        return "<?php echo \$this->variantClasses({$expression}); ?>";
    });

    Blade::if('gsmuiDark', function () {
        return request()->hasSession() && session('gsmui_dark_mode', false);
    });

    Blade::if('gsmuiRtl', function () {
        return app('request')->input('dir') === 'rtl' || 
               (request()->hasSession() && session('gsmui_rtl', false));
    });

    Blade::if('gsmuiFeature', function ($expression) {
        return config("features.{$expression}", false);
    });

    Blade::if('gsmuiPermission', function ($expression) {
        return auth()->check() && auth()->user()->can($expression);
    });

    Blade::if('gsmuiRole', function ($expression) {
        return auth()->check() && auth()->user()->hasRole($expression);
    });

    Blade::if('gsmuiEnv', function ($expression) {
        return app()->environment($expression);
    });
}
```

## Blade Component Testing Utilities

### Component Test Helpers

```php
<?php

namespace Tests\Helpers;

use Illuminate\Support\Facades\Blade;
use Illuminate\Support\Str;

class BladeComponentTestHelper
{
    public static function renderWithProps(string $component, array $props = [], string $slot = ''): string
    {
        $attrs = collect($props)->map(function ($value, $key) {
            if (is_bool($value)) {
                return $value ? ":{$key}=\"true\"" : ":{$key}=\"false\"";
            }
            if (is_null($value)) {
                return ":{$key}=\"null\"";
            }
            if (is_array($value)) {
                $arrayStr = var_export($value, true);
                return ":{$key}=\"{$arrayStr}\"";
            }
            return "{$key}=\"{$value}\"";
        })->implode(' ');

        $tag = "x-gsmui:{$component}";
        $content = $slot ? ">{$slot}</{$tag}>" : ' />';
        return Blade::render("<{$tag} {$attrs}{$content}");
    }

    public static function assertContainsClass(string $html, string $class): void
    {
        $pattern = '/class="[^"]*\b' . preg_quote($class, '/') . '\b[^"]*"|class=\'[^\']*\b' . preg_quote($class, '/') . '\b[^\']*\'/';
        if (!preg_match($pattern, $html)) {
            throw new \PHPUnit\Framework\AssertionFailedError(
                "Expected HTML to contain class '{$class}'"
            );
        }
    }

    public static function assertNotContainsClass(string $html, string $class): void
    {
        $pattern = '/class="[^"]*\b' . preg_quote($class, '/') . '\b[^"]*"|class=\'[^\']*\b' . preg_quote($class, '/') . '\b[^\']*\'/';
        if (preg_match($pattern, $html)) {
            throw new \PHPUnit\Framework\AssertionFailedError(
                "Expected HTML to NOT contain class '{$class}'"
            );
        }
    }

    public static function assertHasAriaRole(string $html, string $role): void
    {
        if (!str_contains($html, "role=\"{$role}\"")) {
            throw new \PHPUnit\Framework\AssertionFailedError(
                "Expected HTML to have role='{$role}'"
            );
        }
    }

    public static function assertHasAriaAttribute(string $html, string $attr, string $value = null): void
    {
        if ($value !== null) {
            $expected = "{$attr}=\"{$value}\"";
        } else {
            $expected = "{$attr}=\"";
        }
        if (!str_contains($html, $expected)) {
            throw new \PHPUnit\Framework\AssertionFailedError(
                "Expected HTML to have {$attr}" . ($value ? "=\"{$value}\"" : '')
            );
        }
    }

    public static function assertHasDataAttribute(string $html, string $attr, string $value = null): void
    {
        $fullAttr = "data-{$attr}";
        if ($value !== null) {
            $expected = "{$fullAttr}=\"{$value}\"";
        } else {
            $expected = "{$fullAttr}=\"";
        }
        if (!str_contains($html, $expected)) {
            throw new \PHPUnit\Framework\AssertionFailedError(
                "Expected HTML to have {$fullAttr}" . ($value ? "=\"{$value}\"" : '')
            );
        }
    }

    public static function extractTag(string $html, string $tag): ?string
    {
        $pattern = "/<{$tag}[^>]*>(.*?)<\/{$tag}>/s";
        if (preg_match($pattern, $html, $matches)) {
            return $matches[0];
        }
        return null;
    }

    public static function extractAllTags(string $html, string $tag): array
    {
        $pattern = "/<{$tag}[^>]*>(.*?)<\/{$tag}>/s";
        preg_match_all($pattern, $html, $matches);
        return $matches[0] ?? [];
    }
}
```
