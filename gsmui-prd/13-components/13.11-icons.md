# 13.11 Icons System — GsmUi Icons Specification

## Overview

The GsmUi icon system provides a consistent, performant way to use icons across all components. It uses inline SVG with a unified API, supporting both built-in icon sets and custom icons.

## Architecture

### Icon Component

```blade
{{-- resources/views/components/icon/index.blade.php --}}
@props([
    'name' => '',
    'class' => 'w-5 h-5',
    'variant' => 'outline',  // outline, solid, mini
    'ariaHidden' => true,
    'attributes' => [],
])

<?php
$iconPath = __DIR__ . "/../../icons/{$variant}/{$name}.svg";
$fallbackPath = __DIR__ . "/../../icons/outline/{$name}.svg";

if (!file_exists($iconPath) && file_exists($fallbackPath)) {
    $iconPath = $fallbackPath;
}

if (!file_exists($iconPath)) {
    // Render nothing or fallback
    return;
}

$svg = file_get_contents($iconPath);
$svg = preg_replace('/<\?xml.*?\?>/', '', $svg);
$svg = preg_replace('/width=".*?"/', '', $svg);
$svg = preg_replace('/height=".*?"/', '', $svg);
$svg = str_replace('<svg', '<svg ' . $attributes->merge(['class' => $class])->class($class), $svg);
if ($ariaHidden) {
    $svg = str_replace('<svg', '<svg aria-hidden="true"', $svg);
}
?>

{!! $svg !!}
```

### Icon Registry

```php
<?php

namespace GsmUi\Support;

class IconRegistry
{
    protected array $icons = [];
    protected array $aliases = [];
    protected string $defaultVariant = 'outline';
    
    public function register(string $name, string $path, string $variant = 'outline'): void
    {
        $this->icons["{$variant}:{$name}"] = $path;
    }
    
    public function alias(string $alias, string $target): void
    {
        $this->aliases[$alias] = $target;
    }
    
    public function resolve(string $name, string $variant = null): ?string
    {
        $variant = $variant ?? $this->defaultVariant;
        $key = "{$variant}:{$name}";
        
        // Check direct
        if (isset($this->icons[$key])) {
            return $this->icons[$key];
        }
        
        // Check alias
        if (isset($this->aliases[$name])) {
            return $this->resolve($this->aliases[$name], $variant);
        }
        
        // Fallback to outline
        if ($variant !== 'outline' && isset($this->icons["outline:{$name}"])) {
            return $this->icons["outline:{$name}"];
        }
        
        return null;
    }
    
    public function exists(string $name, string $variant = null): bool
    {
        return $this->resolve($name, $variant) !== null;
    }
    
    public function getAll(string $variant = null): array
    {
        $variant = $variant ?? $this->defaultVariant;
        return array_keys(array_filter($this->icons, fn($k) => str_starts_with($k, "{$variant}:"), ARRAY_FILTER_USE_KEY));
    }
}
```

## Icon Sizing

### Standard Sizes

| Size | Class | Dimensions | Use Case |
|------|-------|------------|----------|
| xs | `w-3 h-3` | 12×12px | Inline text, badges |
| sm | `w-4 h-4` | 16×16px | Buttons, inputs (sm) |
| md | `w-5 h-5` | 20×20px | **Default**, buttons (md) |
| lg | `w-6 h-6` | 24×24px | Buttons (lg), cards |
| xl | `w-8 h-8` | 32×32px | Hero, feature cards |
| 2xl | `w-10 h-10` | 40×40px | Large illustrations |
| 3xl | `w-12 h-12` | 48×48px | Empty states |

## SVG Optimization

### Build-Time Optimization

```javascript
// vite.config.js / mix
import { optimizeSvg } from 'svgo';

export default {
    plugins: [
        // SVG optimization
        {
            name: 'optimize-svgs',
            generateBundle(options, bundle) {
                for (const fileName of Object.keys(bundle)) {
                    if (fileName.endsWith('.svg')) {
                        const svg = bundle[fileName].source;
                        bundle[fileName].source = optimizeSvg(svg, {
                            plugins: [
                                'removeDimensions',
                                'removeXMLNS',
                                'cleanupAttrs',
                                'removeComments',
                                'minifyStyles',
                                'convertColors',
                                'convertShapeToPath',
                                'mergePaths',
                            ],
                        }).data;
                    }
                }
            },
        },
    ],
};
```

### Runtime Optimization

```php
class OptimizedIconComponent extends Component
{
    protected static array $cache = [];
    
    public function render(): View
    {
        $cacheKey = "{$this->variant}:{$this->name}";
        
        if (!isset(self::$cache[$cacheKey])) {
            $path = $this->iconRegistry->resolve($this->name, $this->variant);
            $svg = file_get_contents($path);
            
            // Optimize once
            $svg = $this->optimizeSvg($svg);
            self::$cache[$cacheKey] = $svg;
        }
        
        return view('gsmui::components.icon', [
            'svg' => self::$cache[$cacheKey],
        ]);
    }
    
    protected function optimizeSvg(string $svg): string
    {
        $svg = preg_replace('/<\?xml.*?\?>/', '', $svg);
        $svg = preg_replace('/width=".*?"/', '', $svg);
        $svg = preg_replace('/height=".*?"/', '', $svg);
        $svg = preg_replace('/\s+/', ' ', $svg);
        return trim($svg);
    }
}
```

## Icon Registry Advanced

### Dynamic Icon Loading

```php
<?php

namespace GsmUi\Support;

class DynamicIconRegistry extends IconRegistry
{
    protected array $lazyIcons = [];
    protected array $loadedIcons = [];
    
    public function registerLazy(string $name, callable $loader, string $variant = 'outline'): void
    {
        $this->lazyIcons["{$variant}:{$name}"] = $loader;
    }
    
    public function resolve(string $name, string $variant = null): ?string
    {
        $variant = $variant ?? $this->defaultVariant;
        $key = "{$variant}:{$name}";
        
        // Check loaded icons cache
        if (isset($this->loadedIcons[$key])) {
            return $this->loadedIcons[$key];
        }
        
        // Check parent registry
        $resolved = parent::resolve($name, $variant);
        if ($resolved) {
            return $resolved;
        }
        
        // Check lazy icons
        if (isset($this->lazyIcons[$key])) {
            $path = ($this->lazyIcons[$key])();
            if ($path && file_exists($path)) {
                $this->loadedIcons[$key] = $path;
                return $path;
            }
        }
        
        return null;
    }
    
    public function preload(array $icons): void
    {
        foreach ($icons as $name => $variant) {
            $variant = $variant ?? $this->defaultVariant;
            $this->resolve($name, $variant);
        }
    }
}
```

### Icon Collections Manager

```php
<?php

namespace GsmUi\Support;

class IconCollectionManager
{
    protected array $collections = [];
    protected string $activeCollection = 'heroicons';
    
    public function registerCollection(string $name, string $path): void
    {
        $this->collections[$name] = $path;
    }
    
    public function setActiveCollection(string $name): void
    {
        if (!isset($this->collections[$name])) {
            throw new \InvalidArgumentException("Collection '{$name}' not registered");
        }
        $this->activeCollection = $name;
    }
    
    public function getActiveCollection(): string
    {
        return $this->activeCollection;
    }
    
    public function getCollectionPath(string $name): ?string
    {
        return $this->collections[$name] ?? null;
    }
    
    public function listCollections(): array
    {
        return array_keys($this->collections);
    }
    
    public function listIcons(string $collection = null): array
    {
        $collection = $collection ?? $this->activeCollection;
        $path = $this->getCollectionPath($collection);
        
        if (!$path || !is_dir($path)) {
            return [];
        }
        
        $icons = [];
        $files = glob($path . '/*.svg');
        
        foreach ($files as $file) {
            $icons[] = basename($file, '.svg');
        }
        
        sort($icons);
        return $icons;
    }
    
    public function collectionStats(string $collection): array
    {
        $path = $this->getCollectionPath($collection);
        
        if (!$path || !is_dir($path)) {
            return ['total' => 0, 'variants' => []];
        }
        
        $variants = ['outline' => 0, 'solid' => 0, 'mini' => 0];
        
        foreach ($variants as $variant => $count) {
            $variantPath = "{$path}/{$variant}";
            if (is_dir($variantPath)) {
                $variants[$variant] = count(glob("{$variantPath}/*.svg"));
            }
        }
        
        return [
            'total' => array_sum($variants),
            'variants' => $variants,
        ];
    }
}
```

## SVG Icon Processing

### SVG Sprite Builder

```php
<?php

namespace GsmUi\Icons;

class SvgSpriteBuilder
{
    protected array $icons = [];
    protected string $spritePath;
    
    public function __construct(string $spritePath = null)
    {
        $this->spritePath = $spritePath ?? storage_path('app/gsmui/sprite.svg');
    }
    
    public function addIcon(string $name, string $svgContent): void
    {
        $this->icons[$name] = $this->cleanSvg($svgContent);
    }
    
    public function build(): string
    {
        $sprite = '<?xml version="1.0" encoding="UTF-8"?>';
        $sprite .= '<svg xmlns="http://www.w3.org/2000/svg" style="display: none;">';
        
        foreach ($this->icons as $name => $svg) {
            $sprite .= $this->wrapSymbol($name, $svg);
        }
        
        $sprite .= '</svg>';
        
        file_put_contents($this->spritePath, $sprite);
        
        return $this->spritePath;
    }
    
    protected function cleanSvg(string $svg): string
    {
        // Remove XML declaration
        $svg = preg_replace('/<\?xml.*?\?>/', '', $svg);
        
        // Remove dimensions
        $svg = preg_replace('/width=".*?"/', '', $svg);
        $svg = preg_replace('/height=".*?"/', '', $svg);
        
        // Remove xmlns (will be added to symbol)
        $svg = preg_replace('/xmlns=".*?"/', '', $svg);
        
        // Clean whitespace
        $svg = preg_replace('/\s+/', ' ', $svg);
        
        return trim($svg);
    }
    
    protected function wrapSymbol(string $name, string $svg): string
    {
        // Extract viewBox from original SVG
        preg_match('/viewBox="([^"]+)"/', $svg, $matches);
        $viewBox = $matches[1] ?? '0 0 24 24';
        
        // Extract inner content (everything between <svg> tags)
        preg_match('/<svg[^>]*>(.*?)<\/svg>/s', $svg, $matches);
        $content = $matches[1] ?? $svg;
        
        return "<symbol id=\"icon-{$name}\" viewBox=\"{$viewBox}\">{$content}</symbol>";
    }
    
    public function getSpriteUrl(): string
    {
        return asset('storage/gsmui/sprite.svg');
    }
}
```

### SVG Optimization Pipeline

```php
<?php

namespace GsmUi\Icons;

class SvgOptimizer
{
    protected array $options;
    
    public function __construct(array $options = [])
    {
        $this->options = array_merge([
            'removeDimensions' => true,
            'removeXmlDeclaration' => true,
            'removeComments' => true,
            'removeMetadata' => true,
            'removeEditorsNSData' => true,
            'cleanupAttrs' => true,
            'mergeStyles' => true,
            'minifyStyles' => true,
            'removeUselessDefs' => true,
            'cleanupNumericValues' => true,
            'convertColors' => true,
            'removeUnknownsAndDefaults' => true,
            'removeNonInheritableGroupAttrs' => true,
            'removeUselessStrokeAndFill' => true,
            'cleanupEnableBackground' => true,
            'removeHiddenElems' => true,
            'removeEmptyText' => true,
            'convertShapeToPath' => true,
            'moveGroupAttrsToElems' => true,
            'collapseGroups' => true,
            'convertEllipseToCircle' => true,
            'moveElemsAttrsToGroup' => true,
            'moveGroupAttrsToElems' => true,
            'pathUtils' => true,
        ], $options);
    }
    
    public function optimize(string $svg): string
    {
        $svg = $this->removeXmlDeclaration($svg);
        $svg = $this->removeComments($svg);
        $svg = $this->removeMetadata($svg);
        $svg = $this->removeDimensions($svg);
        $svg = $this->cleanupAttrs($svg);
        $svg = $this->minifyStyles($svg);
        $svg = $this->convertColors($svg);
        $svg = $this->cleanupNumericValues($svg);
        $svg = $this->removeEmptyAttrs($svg);
        $svg = $this->removeEmptyText($svg);
        $svg = $this->collapseWhitespace($svg);
        
        return trim($svg);
    }
    
    protected function removeXmlDeclaration(string $svg): string
    {
        return $this->options['removeXmlDeclaration']
            ? preg_replace('/<\?xml.*?\?>/', '', $svg)
            : $svg;
    }
    
    protected function removeComments(string $svg): string
    {
        return $this->options['removeComments']
            ? preg_replace('/<!--.*?-->/s', '', $svg)
            : $svg;
    }
    
    protected function removeMetadata(string $svg): string
    {
        return $this->options['removeMetadata']
            ? preg_replace('/<metadata.*?<\/metadata>/s', '', $svg)
            : $svg;
    }
    
    protected function removeDimensions(string $svg): string
    {
        if (!$this->options['removeDimensions']) {
            return $svg;
        }
        
        $svg = preg_replace('/width="[^"]*"/', '', $svg);
        $svg = preg_replace('/height="[^"]*"/', '', $svg);
        
        return $svg;
    }
    
    protected function cleanupAttrs(string $svg): string
    {
        if (!$this->options['cleanupAttrs']) {
            return $svg;
        }
        
        // Remove empty attributes
        $svg = preg_replace('/\s+[a-z-]+=""/', '', $svg);
        
        // Collapse whitespace
        $svg = preg_replace('/\s+/', ' ', $svg);
        
        return $svg;
    }
    
    protected function minifyStyles(string $svg): string
    {
        if (!$this->options['minifyStyles']) {
            return $svg;
        }
        
        // Simple CSS minification
        $svg = preg_replace('/\/\*.*?\*\//s', '', $svg);
        $svg = preg_replace('/\s*{\s*/', '{', $svg);
        $svg = preg_replace('/\s*}\s*/', '}', $svg);
        $svg = preg_replace('/\s*:\s*/', ':', $svg);
        $svg = preg_replace('/\s*;\s*/', ';', $svg);
        $svg = preg_replace('/;}/', '}', $svg);
        
        return $svg;
    }
    
    protected function convertColors(string $svg): string
    {
        if (!$this->options['convertColors']) {
            return $svg;
        }
        
        // Convert named colors to hex
        $colorMap = [
            'black' => '#000',
            'white' => '#fff',
            'red' => '#f00',
            'green' => '#008000',
            'blue' => '#00f',
            'yellow' => '#ff0',
            'cyan' => '#0ff',
            'magenta' => '#f0f',
        ];
        
        foreach ($colorMap as $name => $hex) {
            $svg = preg_replace("/fill=\"{$name}\"/i", "fill=\"{$hex}\"", $svg);
            $svg = preg_replace("/stroke=\"{$name}\"/i", "stroke=\"{$hex}\"", $svg);
        }
        
        return $svg;
    }
    
    protected function cleanupNumericValues(string $svg): string
    {
        if (!$this->options['cleanupNumericValues']) {
            return $svg;
        }
        
        // Remove trailing zeros
        $svg = preg_replace('/(\d+\.\d*?)0+(?=[^\d])/', '$1', $svg);
        $svg = preg_replace('/(\d+)\.0+(?=[^\d])/', '$1', $svg);
        
        return $svg;
    }
    
    protected function removeEmptyAttrs(string $svg): string
    {
        return preg_replace('/\s+[a-z-]+=""', '', $svg);
    }
    
    protected function removeEmptyText(string $svg): string
    {
        return preg_replace('/<text[^>]*>\s*<\/text>/', '', $svg);
    }
    
    protected function collapseWhitespace(string $svg): string
    {
        $svg = preg_replace('/\s+/', ' ', $svg);
        return trim($svg);
    }
}
```

## Icon Performance

### Caching Strategy

```php
<?php

namespace GsmUi\Icons;

class IconCache
{
    protected string $cacheDir;
    protected int $ttl;
    
    public function __construct(string $cacheDir = null, int $ttl = 3600)
    {
        $this->cacheDir = $cacheDir ?? storage_path('app/gsmui/icons');
        $this->ttl = $ttl;
        
        if (!is_dir($this->cacheDir)) {
            mkdir($this->cacheDir, 0755, true);
        }
    }
    
    public function get(string $key): ?string
    {
        $path = $this->cachePath($key);
        
        if (!file_exists($path)) {
            return null;
        }
        
        if (filemtime($path) + $this->ttl < time()) {
            unlink($path);
            return null;
        }
        
        return file_get_contents($path);
    }
    
    public function set(string $key, string $content): void
    {
        $path = $this->cachePath($key);
        file_put_contents($path, $content, LOCK_EX);
    }
    
    public function has(string $key): bool
    {
        return $this->get($key) !== null;
    }
    
    public function clear(): void
    {
        $files = glob($this->cacheDir . '/*.svg');
        foreach ($files as $file) {
            unlink($file);
        }
    }
    
    protected function cachePath(string $key): string
    {
        return $this->cacheDir . '/' . md5($key) . '.svg';
    }
}
```

### Preload Critical Icons

```php
<?php

namespace GsmUi\Icons;

class IconPreloader
{
    protected array $criticalIcons = [];
    protected IconCache $cache;
    
    public function __construct(IconCache $cache)
    {
        $this->cache = $cache;
    }
    
    public function preloadCriticalIcons(): string
    {
        $icons = $this->getCriticalIcons();
        $sprite = $this->buildInlineSprite($icons);
        
        return $sprite;
    }
    
    protected function getCriticalIcons(): array
    {
        $icons = [
            'chevron-down',
            'chevron-up',
            'chevron-left',
            'chevron-right',
            'x-mark',
            'check',
            'magnifying-glass',
            'arrow-path',
        ];
        
        return array_map(function ($name) {
            $cached = $this->cache->get("outline:{$name}");
            if ($cached) {
                return ['name' => $name, 'svg' => $cached];
            }
            return null;
        }, $icons);
    }
    
    protected function buildInlineSprite(array $icons): string
    {
        $sprite = '<svg xmlns="http://www.w3.org/2000/svg" class="hidden">';
        
        foreach ($icons as $icon) {
            if (!$icon) continue;
            
            $svg = $icon['svg'];
            preg_match('/viewBox="([^"]+)"/', $svg, $matches);
            $viewBox = $matches[1] ?? '0 0 24 24';
            
            preg_match('/<svg[^>]*>(.*?)<\/svg>/s', $svg, $matches);
            $content = $matches[1] ?? '';
            
            $sprite .= "<symbol id=\"icon-{$icon['name']}\" viewBox=\"{$viewBox}\">{$content}</symbol>";
        }
        
        $sprite .= '</svg>';
        
        return $sprite;
    }
}
```

### Icon Naming Conventions

Follow these conventions when naming icons:

1. **Use kebab-case**: `chevron-down`, not `chevronDown` or `chevron_down`
2. **Be descriptive**: `arrow-up-tray` not `upload2`
3. **Avoid abbreviations**: `magnifying-glass` not `magn-glass`
4. **Use consistent naming**: All arrows use `arrow-*`, all chevrons use `chevron-*`

## Icon System Best Practices

### Icon Usage Guidelines

#### Do's and Don'ts

#### Do

1. **Use consistent sizing** — Follow the standard size scale (xs, sm, md, lg, xl)
2. **Provide accessible labels** — Add `aria-label` for icon buttons
3. **Use semantic colors** — Match icon colors to their meaning
4. **Optimize SVGs** — Remove unnecessary attributes and code
5. **Cache icon sprites** — Use the built-in sprite system

#### Don't

1. **Don't use raster images** — Always use vector SVGs
2. **Don't mix icon sets** — Stick to one icon family per project
3. **Don't forget `aria-hidden`** — Decorative icons need `aria-hidden="true"`
4. **Don't animate complex SVGs** — Keep animations simple
5. **Don't override icon sizes** — Use the provided size utilities

### Icon Size Guide

| Size | Dimensions | Use Case | Component |
|------|------------|----------|-----------|
| `xs` | 12×12 | Inline text, badges | Badge |
| `sm` | 16×16 | Small buttons, inputs | Input, Button (sm) |
| `md` | 20×20 | **Default**, standard buttons | Button (md) |
| `lg` | 24×24 | Large buttons, cards | Card, Button (lg) |
| `xl` | 32×32 | Hero sections, feature cards | Hero |
| `2xl` | 40×40 | Large illustrations | Empty State |
| `3xl` | 48×48 | Empty states | Empty State |

### Icon Color Guide

| Color | Use Case | CSS Class |
|-------|----------|-----------|
| Primary | Main actions | `text-blue-600` |
| Secondary | Supporting elements | `text-zinc-500` |
| Success | Positive states | `text-emerald-600` |
| Warning | Caution states | `text-amber-600` |
| Danger | Error states | `text-red-600` |
| Info | Information | `text-blue-500` |
| Muted | Decorative | `text-zinc-400` |

### Icon Naming Conventions

| Pattern | Example | Use Case |
|---------|---------|----------|
| `{verb}` | `plus`, `minus`, `check` | Actions |
| `{noun}` | `home`, `user`, `settings` | Objects |
| `{adjective}-{noun}` | `chevron-down`, `arrow-right` | Directional |
| `{noun}-{variant}` | `eye-slash`, `lock-open` | Variants |

### Icon Accessibility Checklist

- [ ] Decorative icons have `aria-hidden="true"`
- [ ] Icon buttons have `aria-label`
- [ ] Status icons have visible text alternatives
- [ ] Icons have sufficient contrast
- [ ] Icons work in high contrast mode
- [ ] Icons are not essential for understanding

### Icon Performance Optimization

1. **Use sprite sheets** — Combine multiple icons into one SVG
2. **Remove unused icons** — Only include icons you actually use
3. **Optimize SVGs** — Remove metadata, comments, and unnecessary attributes
4. **Cache aggressively** — Use browser caching for static icon assets
5. **Lazy load** — Load non-critical icons on demand

### Icon Component Implementation

```php
<?php

namespace GsmUi\Components;

use GsmUi\Support\IconRegistry;
use Illuminate\View\Component;

class Icon extends Component
{
    public string $name;
    public string $variant;
    public string $class;
    public bool $ariaHidden;
    
    protected IconRegistry $registry;
    
    public function __construct(
        string $name = '',
        string $variant = 'outline',
        string $class = 'w-5 h-5',
        bool $ariaHidden = true
    ) {
        $this->name = $name;
        $this->variant = $variant;
        $this->class = $class;
        $this->ariaHidden = $ariaHidden;
        $this->registry = app(IconRegistry::class);
    }
    
    public function render()
    {
        return view('gsmui::components.icon');
    }
    
    public function getSvg(): ?string
    {
        $path = $this->registry->resolve($this->name, $this->variant);
        
        if (!$path || !file_exists($path)) {
            return null;
        }
        
        $svg = file_get_contents($path);
        
        // Clean SVG
        $svg = $this->cleanSvg($svg);
        
        // Add attributes
        $svg = $this->addAttributes($svg);
        
        return $svg;
    }
    
    protected function cleanSvg(string $svg): string
    {
        // Remove XML declaration
        $svg = preg_replace('/<\?xml.*?\?>/', '', $svg);
        
        // Remove dimensions
        $svg = preg_replace('/width=".*?"/', '', $svg);
        $svg = preg_replace('/height=".*?"/', '', $svg);
        
        // Clean whitespace
        $svg = preg_replace('/\s+/', ' ', $svg);
        
        return trim($svg);
    }
    
    protected function addAttributes(string $svg): string
    {
        $attributes = $this->buildAttributes();
        
        return str_replace('<svg', "<svg {$attributes}", $svg);
    }
    
    protected function buildAttributes(): string
    {
        $attributes = [];
        
        $attributes[] = "class=\"{$this->class}\"";
        
        if ($this->ariaHidden) {
            $attributes[] = 'aria-hidden="true"';
        }
        
        return implode(' ', $attributes);
    }
}
```

### Icon Registry Implementation

```php
<?php

namespace GsmUi\Support;

class IconRegistry
{
    protected array $icons = [];
    protected array $aliases = [];
    protected string $defaultVariant = 'outline';
    protected string $iconsPath;
    
    public function __construct(string $iconsPath = null)
    {
        $this->iconsPath = $iconsPath ?? resource_path('views/vendor/gsmui/icons');
        $this->registerBuiltInIcons();
    }
    
    protected function registerBuiltInIcons(): void
    {
        $variants = ['outline', 'solid', 'mini'];
        
        foreach ($variants as $variant) {
            $variantPath = "{$this->iconsPath}/{$variant}";
            
            if (!is_dir($variantPath)) {
                continue;
            }
            
            $files = glob("{$variantPath}/*.svg");
            
            foreach ($files as $file) {
                $name = basename($file, '.svg');
                $this->register($name, $file, $variant);
            }
        }
    }
    
    public function register(string $name, string $path, string $variant = 'outline'): void
    {
        $this->icons["{$variant}:{$name}"] = $path;
    }
    
    public function alias(string $alias, string $target): void
    {
        $this->aliases[$alias] = $target;
    }
    
    public function resolve(string $name, string $variant = null): ?string
    {
        $variant = $variant ?? $this->defaultVariant;
        $key = "{$variant}:{$name}";
        
        // Check direct
        if (isset($this->icons[$key])) {
            return $this->icons[$key];
        }
        
        // Check alias
        if (isset($this->aliases[$name])) {
            return $this->resolve($this->aliases[$name], $variant);
        }
        
        // Fallback to outline
        if ($variant !== 'outline') {
            $outlineKey = "outline:{$name}";
            if (isset($this->icons[$outlineKey])) {
                return $this->icons[$outlineKey];
            }
        }
        
        return null;
    }
    
    public function exists(string $name, string $variant = null): bool
    {
        return $this->resolve($name, $variant) !== null;
    }
    
    public function getAll(string $variant = null): array
    {
        $variant = $variant ?? $this->defaultVariant;
        
        return array_keys(array_filter(
            $this->icons,
            fn($key) => str_starts_with($key, "{$variant}:"),
            ARRAY_FILTER_USE_KEY
        ));
    }
    
    public function getAllVariants(): array
    {
        $variants = [];
        
        foreach ($this->icons as $key => $path) {
            [$variant, $name] = explode(':', $key);
            $variants[$variant][] = $name;
        }
        
        return $variants;
    }
    
    public function stats(): array
    {
        $stats = [];
        
        foreach ($this->icons as $key => $path) {
            [$variant, $name] = explode(':', $key);
            
            if (!isset($stats[$variant])) {
                $stats[$variant] = 0;
            }
            
            $stats[$variant]++;
        }
        
        return $stats;
    }
}
```

### Icon Service Provider

```php
<?php

namespace GsmUi\Providers;

use Illuminate\Support\ServiceProvider;
use GsmUi\Support\IconRegistry;
use GsmUi\Support\IconCollectionManager;

class IconServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        $this->app->singleton(IconRegistry::class, function ($app) {
            return new IconRegistry(
                $app->make('gsmui.icons.path')
            );
        });
        
        $this->app->singleton(IconCollectionManager::class, function ($app) {
            return new IconCollectionManager();
        });
    }
    
    public function boot(): void
    {
        $this->registerPublishes();
        $this->registerComponent();
    }
    
    protected function registerPublishes(): void
    {
        $this->publishes([
            __DIR__ . '/../../resources/icons' => resource_path('views/vendor/gsmui/icons'),
        ], 'gsmui-icons');
    }
    
    protected function registerComponent(): void
    {
        Blade::anonymousComponentPath(__DIR__ . '/../../resources/views/components', 'gsmui');
    }
}
```
