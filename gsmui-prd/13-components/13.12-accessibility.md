# 13.12 Accessibility — GsmUi Accessibility Specification

## Overview

All GsmUi components must meet **WCAG 2.1 Level AA** compliance as a minimum baseline. This document defines the accessibility requirements, patterns, and testing procedures.

## Core Requirements

### 1. Keyboard Navigation

| Component | Keys | Behavior |
|-----------|------|----------|
| Button | `Enter`, `Space` | Activate |
| Link | `Enter` | Navigate |
| Input | `Tab` | Focus |
| Select | `Enter`/`Space` | Open, `ArrowUp/Down` | Navigate, `Enter` | Select, `Escape` | Close |
| Menu | `ArrowUp/Down` | Navigate, `Enter` | Activate, `Escape` | Close |
| Tabs | `ArrowLeft/Right` | Navigate, `Enter`/`Space` | Activate, `Home/End` | First/Last |
| Dialog | `Tab` | Trap focus, `Escape` | Close |
| Slider | `ArrowLeft/Right` | Decrease/Increase, `Home/End` | Min/Max, `PageUp/PageDown` | Large step |
| Tree | `ArrowRight` | Expand, `ArrowLeft` | Collapse, `ArrowUp/Down` | Navigate |

### 2. Focus Management

```css
/* Focus visible - only show on keyboard focus */
*:focus-visible {
    outline: 2px solid var(--gsmui-color-primary);
    outline-offset: 2px;
}

/* Remove default focus for mouse users */
*:focus:not(:focus-visible) {
    outline: none;
}

/* Custom focus for non-native elements */
[role="button"]:focus-visible,
[role="tab"]:focus-visible,
[role="menuitem"]:focus-visible,
[role="option"]:focus-visible {
    outline: 2px solid var(--gsmui-color-primary);
    outline-offset: 2px;
}
```

### 3. ARIA Attributes

#### Required ARIA by Component Type

| Component | Role | Required ARIA |
|-----------|------|---------------|
| Button | `button` (native) | `aria-disabled`, `aria-busy`, `aria-pressed` (toggle) |
| Link | `link` (native) | `aria-disabled`, `aria-current` |
| Input | `textbox` (native) | `aria-label`/`aria-labelledby`, `aria-describedby`, `aria-invalid`, `aria-required`, `aria-readonly`, `aria-disabled` |
| Select | `combobox` | `aria-haspopup="listbox"`, `aria-expanded`, `aria-controls`, `aria-activedescendant` |
| Checkbox | `checkbox` (native) | `aria-checked`, `aria-describedby` |
| Radio | `radio` (native) | `aria-checked`, `aria-required` |
| Switch | `switch` | `aria-checked`, `aria-describedby` |
| Slider | `slider` | `aria-valuemin`, `aria-valuemax`, `aria-valuenow`, `aria-orientation` |
| Tabs | `tablist`/`tab`/`tabpanel` | `aria-selected`, `aria-controls`, `aria-labelledby` |
| Dialog | `dialog` | `aria-modal="true"`, `aria-labelledby`, `aria-describedby` |
| Menu | `menu`/`menuitem` | `aria-haspopup`, `aria-expanded` |
| Tooltip | `tooltip` | `role="tooltip"`, `aria-describedby` |
| Toast | `status`/`alert` | `role="status"`, `aria-live="polite"` / `role="alert"`, `aria-live="assertive"` |
| Progress | `progressbar` | `aria-valuemin`, `aria-valuemax`, `aria-valuenow`, `aria-label` |
| Accordion | `region` | `aria-expanded`, `aria-controls` |

## Advanced Accessibility Patterns

### Focus Trap Implementation

```javascript
class FocusTrap {
    constructor(element, options = {}) {
        this.element = element;
        this.options = {
            initialFocus: null,
            returnFocus: true,
            escapeDeactivates: true,
            clickOutsideDeactivates: true,
            ...options,
        };
        
        this.previouslyFocusedElement = null;
        this.active = false;
        this.abortController = null;
    }
    
    activate() {
        if (this.active) return;
        
        this.previouslyFocusedElement = document.activeElement;
        this.active = true;
        
        this.element.setAttribute('data-focus-trap', 'active');
        
        // Setup event listeners
        this.abortController = new AbortController();
        
        document.addEventListener('keydown', this.handleKeyDown.bind(this), {
            signal: this.abortController.signal,
        });
        
        this.element.addEventListener('click', this.handleClick.bind(this), {
            signal: this.abortController.signal,
        });
        
        // Focus initial element
        const initialFocus = this.options.initialFocus || this.getFirstFocusableElement();
        if (initialFocus) {
            initialFocus.focus();
        }
    }
    
    deactivate() {
        if (!this.active) return;
        
        this.active = false;
        this.element.removeAttribute('data-focus-trap');
        
        this.abortController?.abort();
        this.abortController = null;
        
        // Return focus
        if (this.options.returnFocus && this.previouslyFocusedElement) {
            this.previouslyFocusedElement.focus();
        }
    }
    
    handleKeyDown(e) {
        if (e.key === 'Escape' && this.options.escapeDeactivates) {
            this.deactivate();
            return;
        }
        
        if (e.key !== 'Tab') return;
        
        const focusableElements = this.getFocusableElements();
        const firstElement = focusableElements[0];
        const lastElement = focusableElements[focusableElements.length - 1];
        
        if (e.shiftKey) {
            if (document.activeElement === firstElement) {
                e.preventDefault();
                lastElement.focus();
            }
        } else {
            if (document.activeElement === lastElement) {
                e.preventDefault();
                firstElement.focus();
            }
        }
    }
    
    handleClick(e) {
        if (this.options.clickOutsideDeactivates && !this.element.contains(e.target)) {
            this.deactivate();
        }
    }
    
    getFocusableElements() {
        const selector = [
            'a[href]',
            'button:not([disabled])',
            'input:not([disabled])',
            'select:not([disabled])',
            'textarea:not([disabled])',
            '[tabindex]:not([tabindex="-1"])',
        ].join(', ');
        
        return Array.from(this.element.querySelectorAll(selector));
    }
    
    getFirstFocusableElement() {
        return this.getFocusableElements()[0] || null;
    }
}
```

### Roving Tabindex

```javascript
class RovingTabindex {
    constructor(container, itemSelector) {
        this.container = container;
        this.itemSelector = itemSelector;
        this.items = [];
        this.currentIndex = 0;
        
        this.init();
    }
    
    init() {
        this.items = Array.from(this.container.querySelectorAll(this.itemSelector));
        
        // Set tabindex attributes
        this.items.forEach((item, index) => {
            item.setAttribute('tabindex', index === this.currentIndex ? '0' : '-1');
        });
        
        // Event listeners
        this.container.addEventListener('keydown', this.handleKeyDown.bind(this));
    }
    
    handleKeyDown(e) {
        switch (e.key) {
            case 'ArrowRight':
            case 'ArrowDown':
                e.preventDefault();
                this.moveFocus(1);
                break;
                
            case 'ArrowLeft':
            case 'ArrowUp':
                e.preventDefault();
                this.moveFocus(-1);
                break;
                
            case 'Home':
                e.preventDefault();
                this.setFocusToIndex(0);
                break;
                
            case 'End':
                e.preventDefault();
                this.setFocusToIndex(this.items.length - 1);
                break;
        }
    }
    
    moveFocus(delta) {
        const newIndex = this.currentIndex + delta;
        
        if (newIndex < 0) {
            this.setFocusToIndex(this.items.length - 1);
        } else if (newIndex >= this.items.length) {
            this.setFocusToIndex(0);
        } else {
            this.setFocusToIndex(newIndex);
        }
    }
    
    setFocusToIndex(index) {
        this.items[this.currentIndex].setAttribute('tabindex', '-1');
        this.currentIndex = index;
        this.items[this.currentIndex].setAttribute('tabindex', '0');
        this.items[this.currentIndex].focus();
    }
    
    getActiveItem() {
        return this.items[this.currentIndex];
    }
    
    destroy() {
        this.items.forEach(item => item.removeAttribute('tabindex'));
        this.container.removeEventListener('keydown', this.handleKeyDown.bind(this));
    }
}
```

### Live Region Manager

```javascript
class LiveRegionManager {
    constructor() {
        this.politeRegion = this.createRegion('polite');
        this.assertiveRegion = this.createRegion('assertive');
        this.statusRegion = this.createRegion('polite', 'status');
        this.alertRegion = this.createRegion('assertive', 'alert');
    }
    
    createRegion(live, role = null) {
        const region = document.createElement('div');
        region.setAttribute('aria-live', live);
        region.setAttribute('aria-atomic', 'true');
        region.classList.add('sr-only');
        
        if (role) {
            region.setAttribute('role', role);
        }
        
        document.body.appendChild(region);
        return region;
    }
    
    announce(message, priority = 'polite') {
        const region = priority === 'assertive' ? this.assertiveRegion : this.politeRegion;
        
        // Clear first to ensure re-announcement
        region.textContent = '';
        
        // Use setTimeout to ensure DOM update
        setTimeout(() => {
            region.textContent = message;
        }, 50);
    }
    
    announceStatus(message) {
        this.statusRegion.textContent = '';
        setTimeout(() => {
            this.statusRegion.textContent = message;
        }, 50);
    }
    
    announceAlert(message) {
        this.alertRegion.textContent = '';
        setTimeout(() => {
            this.alertRegion.textContent = message;
        }, 50);
    }
    
    clear() {
        this.politeRegion.textContent = '';
        this.assertiveRegion.textContent = '';
        this.statusRegion.textContent = '';
        this.alertRegion.textContent = '';
    }
    
    destroy() {
        this.politeRegion.remove();
        this.assertiveRegion.remove();
        this.statusRegion.remove();
        this.alertRegion.remove();
    }
}
```

## Accessibility Best Practices

### WCAG 2.1 Compliance Checklist

#### Perceivable

- [ ] All images have alt text
- [ ] Videos have captions
- [ ] Color is not the only way to convey information
- [ ] Text contrast meets minimum ratio (4.5:1 for normal text)
- [ ] UI component contrast meets minimum ratio (3:1)
- [ ] Content can be resized up to 200% without loss

#### Operable

- [ ] All functionality available via keyboard
- [ ] No keyboard traps
- [ ] Skip navigation links provided
- [ ] Page titles are descriptive
- [ ] Focus order is logical
- [ ] Focus indicators are visible
- [ ] No seizure-inducing content

#### Understandable

- [ ] Page language is specified
- [ ] Labels and instructions are provided
- [ ] Error messages are clear and helpful
- [ ] Form fields have proper labels
- [ ] Consistent navigation

#### Robust

- [ ] HTML is valid
- [ ] ARIA attributes are used correctly
- [ ] Content works with screen readers
- [ ] Custom components have proper roles

### Common Accessibility Patterns

#### Skip Navigation

```html
<a href="#main-content" class="sr-only focus:not-sr-only focus:absolute focus:z-50 focus:p-4 focus:bg-white focus:text-blue-600">
    Skip to main content
</a>

<main id="main-content" tabindex="-1">
    <!-- Main content -->
</main>
```

#### Focus Management

```css
/* Visible focus for keyboard users */
*:focus-visible {
    outline: 2px solid var(--gsmui-color-primary);
    outline-offset: 2px;
}

/* Remove focus for mouse users */
*:focus:not(:focus-visible) {
    outline: none;
}
```

#### Live Regions

```html
<!-- Polite announcements -->
<div role="status" aria-live="polite" class="sr-only">
    3 results found
</div>

<!-- Assertive announcements -->
<div role="alert" aria-live="assertive" class="sr-only">
    Error: Invalid email address
</div>
```

### Focus Management Implementation

```javascript
class FocusManager {
    constructor() {
        this.focusableSelectors = [
            'a[href]',
            'button:not([disabled])',
            'input:not([disabled])',
            'select:not([disabled])',
            'textarea:not([disabled])',
            '[tabindex]:not([tabindex="-1"])',
        ].join(', ');
        
        this.previouslyFocused = null;
        this.currentFocusTrap = null;
    }
    
    getFocusableElements(container = document) {
        return Array.from(container.querySelectorAll(this.focusableSelectors))
            .filter(el => el.offsetParent !== null);
    }
    
    trapFocus(container) {
        this.currentFocusTrap = container;
        this.previouslyFocused = document.activeElement;
        
        const focusable = this.getFocusableElements(container);
        
        if (focusable.length === 0) {
            container.setAttribute('tabindex', '-1');
            container.focus();
            return;
        }
        
        // Set up focus trap
        container.addEventListener('keydown', this.handleFocusTrap);
        
        // Focus first element
        focusable[0].focus();
    }
    
    releaseFocus() {
        if (!this.currentFocusTrap) return;
        
        this.currentFocusTrap.removeEventListener('keydown', this.handleFocusTrap);
        this.currentFocusTrap = null;
        
        if (this.previouslyFocused) {
            this.previouslyFocused.focus();
            this.previouslyFocused = null;
        }
    }
    
    handleFocusTrap = (e) => {
        if (e.key !== 'Tab') return;
        
        const focusable = this.getFocusableElements(this.currentFocusTrap);
        
        if (focusable.length === 0) return;
        
        const first = focusable[0];
        const last = focusable[focusable.length - 1];
        
        if (e.shiftKey) {
            if (document.activeElement === first) {
                e.preventDefault();
                last.focus();
            }
        } else {
            if (document.activeElement === last) {
                e.preventDefault();
                first.focus();
            }
        }
    }
    
    rovingTabindex(container, itemSelector) {
        const items = Array.from(container.querySelectorAll(itemSelector));
        
        items.forEach((item, index) => {
            item.setAttribute('tabindex', index === 0 ? '0' : '-1');
        });
        
        container.addEventListener('keydown', (e) => {
            const currentIndex = items.indexOf(document.activeElement);
            
            let newIndex;
            
            switch (e.key) {
                case 'ArrowRight':
                case 'ArrowDown':
                    e.preventDefault();
                    newIndex = (currentIndex + 1) % items.length;
                    break;
                    
                case 'ArrowLeft':
                case 'ArrowUp':
                    e.preventDefault();
                    newIndex = (currentIndex - 1 + items.length) % items.length;
                    break;
                    
                case 'Home':
                    e.preventDefault();
                    newIndex = 0;
                    break;
                    
                case 'End':
                    e.preventDefault();
                    newIndex = items.length - 1;
                    break;
                    
                default:
                    return;
            }
            
            items[currentIndex].setAttribute('tabindex', '-1');
            items[newIndex].setAttribute('tabindex', '0');
            items[newIndex].focus();
        });
    }
}
```

### Screen Reader Support

```javascript
class ScreenReaderSupport {
    constructor() {
        this.liveRegions = {
            polite: this.createLiveRegion('polite'),
            assertive: this.createLiveRegion('assertive'),
            status: this.createLiveRegion('polite', 'status'),
            alert: this.createLiveRegion('assertive', 'alert'),
        };
    }
    
    createLiveRegion(live, role = null) {
        const region = document.createElement('div');
        region.setAttribute('aria-live', live);
        region.setAttribute('aria-atomic', 'true');
        region.classList.add('sr-only');
        
        if (role) {
            region.setAttribute('role', role);
        }
        
        document.body.appendChild(region);
        return region;
    }
    
    announce(message, priority = 'polite') {
        const region = this.liveRegions[priority];
        
        // Clear first to ensure re-announcement
        region.textContent = '';
        
        setTimeout(() => {
            region.textContent = message;
        }, 50);
    }
    
    announceStatus(message) {
        this.liveRegions.status.textContent = '';
        setTimeout(() => {
            this.liveRegions.status.textContent = message;
        }, 50);
    }
    
    announceAlert(message) {
        this.liveRegions.alert.textContent = '';
        setTimeout(() => {
            this.liveRegions.alert.textContent = message;
        }, 50);
    }
    
    clear() {
        Object.values(this.liveRegions).forEach(region => {
            region.textContent = '';
        });
    }
    
    destroy() {
        Object.values(this.liveRegions).forEach(region => {
            region.remove();
        });
    }
}
```

### Keyboard Navigation Helper

```javascript
class KeyboardNavigation {
    constructor(container, options = {}) {
        this.container = container;
        this.options = {
            itemSelector: ':scope > *',
            orientation: 'vertical',
            wrap: true,
            ...options,
        };
        
        this.items = [];
        this.currentIndex = 0;
        
        this.init();
    }
    
    init() {
        this.items = Array.from(this.container.querySelectorAll(this.options.itemSelector));
        
        this.items.forEach((item, index) => {
            item.setAttribute('tabindex', index === 0 ? '0' : '-1');
        });
        
        this.container.addEventListener('keydown', this.handleKeydown);
    }
    
    handleKeydown = (e) => {
        const isHorizontal = this.options.orientation === 'horizontal';
        
        let newIndex = this.currentIndex;
        
        switch (e.key) {
            case isHorizontal ? 'ArrowRight' : 'ArrowDown':
                e.preventDefault();
                newIndex = this.currentIndex + 1;
                break;
                
            case isHorizontal ? 'ArrowLeft' : 'ArrowUp':
                e.preventDefault();
                newIndex = this.currentIndex - 1;
                break;
                
            case 'Home':
                e.preventDefault();
                newIndex = 0;
                break;
                
            case 'End':
                e.preventDefault();
                newIndex = this.items.length - 1;
                break;
                
            default:
                return;
        }
        
        if (this.options.wrap) {
            if (newIndex < 0) newIndex = this.items.length - 1;
            if (newIndex >= this.items.length) newIndex = 0;
        } else {
            newIndex = Math.max(0, Math.min(newIndex, this.items.length - 1));
        }
        
        this.moveToIndex(newIndex);
    }
    
    moveToIndex(index) {
        this.items[this.currentIndex].setAttribute('tabindex', '-1');
        this.currentIndex = index;
        this.items[this.currentIndex].setAttribute('tabindex', '0');
        this.items[this.currentIndex].focus();
    }
    
    getActiveItem() {
        return this.items[this.currentIndex];
    }
    
    destroy() {
        this.container.removeEventListener('keydown', this.handleKeydown);
        this.items.forEach(item => item.removeAttribute('tabindex'));
    }
}
```
