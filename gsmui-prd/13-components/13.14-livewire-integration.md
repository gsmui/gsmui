# 13.14 Livewire Integration — GsmUi Livewire Specification

## Overview

GsmUi components are built on **Livewire 4** and provide seamless two-way binding, server-side rendering, and real-time reactivity. Every form component supports `wire:model` out of the box.

## Core Integration

### Controllable Trait

```php
<?php

namespace GsmUi\Concerns;

use Livewire\Attributes\Locked;
use Livewire\Component;

trait Controllable
{
    /** @var mixed */
    #[Locked]
    public $value = null;
    
    /** @var mixed */
    #[Locked]
    public $model = null;
    
    /** @var string */
    public string $wireModel = 'model';
    
    /** @var bool */
    public bool $detangle = true;
    
    public function initializeControllable(): void
    {
        $this->syncModel();
    }
    
    public function syncModel(): void
    {
        if ($this->model !== null) {
            $this->value = $this->model;
        }
    }
    
    public function updatedValue($value): void
    {
        if ($this->detangle) {
            $this->detangle();
        }
        $this->dispatch('input', $value);
        $this->dispatch('change', $value);
    }
    
    public function detangle(): void
    {
        $this->model = $this->value;
    }
    
    public function getValueAttribute(): mixed
    {
        return $this->value ?? $this->model;
    }
    
    public function setValueAttribute(mixed $value): void
    {
        $this->value = $value;
    }
}
```

## Component Implementation

### Input Component

```php
<?php

namespace GsmUi\Components\Forms;

use GsmUi\Concerns\Controllable;
use GsmUi\Concerns\Disableable;
use GsmUi\Concerns\Submittable;
use GsmUi\Components\BaseComponent;
use Livewire\Attributes\Locked;

class Input extends BaseComponent
{
    use Controllable, Disableable, Submittable;
    
    #[Locked]
    public string $type = 'text';
    
    #[Locked]
    public string $placeholder = '';
    
    #[Locked]
    public ?string $name = null;
    
    #[Locked]
    public bool $required = false;
    
    #[Locked]
    public bool $autofocus = false;
    
    #[Locked]
    public bool $autocomplete = false;
    
    #[Locked]
    public ?string $inputmode = null;
    
    #[Locked]
    public ?string $pattern = null;
    
    #[Locked]
    public ?int $minlength = null;
    
    #[Locked]
    public ?int $maxlength = null;
    
    #[Locked]
    public ?float $min = null;
    
    #[Locked]
    public ?float $max = null;
    
    #[Locked]
    public ?float $step = null;
    
    #[Locked]
    public bool $invalid = false;
    
    #[Locked]
    public ?string $errorMessage = null;
    
    public function mount(): void
    {
        $this->initializeControllable();
        $this->id = $this->id ?? 'gsmui-input-' . uniqid();
    }
    
    public function updatedValue($value): void
    {
        // Clear error on user input
        if ($this->invalid) {
            $this->invalid = false;
            $this->errorMessage = null;
        }
        
        parent::updatedValue($value);
    }
    
    public function getValidationRules(): array
    {
        $rules = [];
        
        if ($this->required) {
            $rules[] = 'required';
        }
        
        if ($this->type === 'email') {
            $rules[] = 'email';
        }
        
        if ($this->minlength) {
            $rules[] = "min:{$this->minlength}";
        }
        
        if ($this->maxlength) {
            $rules[] = "max:{$this->maxlength}";
        }
        
        if ($this->pattern) {
            $rules[] = "regex:{$this->pattern}";
        }
        
        if ($this->min !== null) {
            $rules[] = "min:{$this->min}";
        }
        
        if ($this->max !== null) {
            $rules[] = "max:{$this->max}";
        }
        
        return $rules;
    }
    
    public function render(): \Illuminate\Contracts\View\View
    {
        return view('gsmui::components.forms.input.index');
    }
}
```

### Select Component

```php
<?php

namespace GsmUi\Components\Forms;

use GsmUi\Concerns\Controllable;
use GsmUi\Concerns\Disableable;
use GsmUi\Components\BaseComponent;
use Livewire\Attributes\Locked;

class Select extends BaseComponent
{
    use Controllable, Disableable;
    
    #[Locked]
    public array $options = [];
    
    #[Locked]
    public bool $multiple = false;
    
    #[Locked]
    public bool $searchable = false;
    
    #[Locked]
    public ?string $placeholder = null;
    
    #[Locked]
    public bool $clearable = false;
    
    #[Locked]
    public ?string $optionValue = 'value';
    
    #[Locked]
    public ?string $optionLabel = 'label';
    
    #[Locked]
    public ?string $optionGroup = 'children';
    
    #[Locked]
    public bool $allowCustom = false;
    
    #[Locked]
    public bool $open = false;
    
    public function mount(): void
    {
        $this->initializeControllable();
        $this->normalizeOptions();
    }
    
    public function updatedValue($value): void
    {
        $this->syncSelectedOptions();
        parent::updatedValue($value);
    }
    
    public function updatedOptions(): void
    {
        $this->normalizeOptions();
        $this->syncSelectedOptions();
    }
    
    protected function normalizeOptions(): void
    {
        $this->options = collect($this->options)->map(function ($option) {
            if (is_string($option)) {
                return ['value' => $option, 'label' => $option];
            }
            if (is_array($option) && !isset($option['value'])) {
                return ['value' => $option[$this->optionValue ?? 'value'], 'label' => $option[$this->optionLabel ?? 'label']];
            }
            return $option;
        })->toArray();
    }
    
    protected function syncSelectedOptions(): void
    {
        // Ensure selected options exist in options list
    }
    
    public function toggleOpen(): void
    {
        if ($this->isDisabled()) return;
        $this->open = !$this->open;
        $this->dispatchEvent('open-change', ['open' => $this->open]);
    }
    
    public function selectOption($value): void
    {
        if ($this->isDisabled()) return;
        
        if ($this->multiple) {
            $values = (array) $this->value;
            if (in_array($value, $values)) {
                $values = array_diff($values, [$value]);
            } else {
                $values[] = $value;
            }
            $this->value = $values;
        } else {
            $this->value = $value;
            $this->open = false;
        }
        
        $this->dispatchEvent('select', [
            'value' => $this->value,
            'label' => $this->getOptionLabel($value),
            'item' => $this->getOption($value),
            'multiple' => $this->multiple,
        ]);
    }
    
    public function clear(): void
    {
        if ($this->isDisabled() || !$this->clearable) return;
        $this->value = $this->multiple ? [] : null;
        $this->dispatchEvent('clear');
    }
    
    public function render(): \Illuminate\Contracts\View\View
    {
        return view('gsmui::components.forms.select.index');
    }
}
```

## Wire:Model Modifiers

### Supported Modifiers

| Modifier | Description | Example |
|----------|-------------|---------|
| `.lazy` | Sync on blur/change | `wire:model.lazy="email"` |
| `.debounce.300ms` | Debounce updates | `wire:model.debounce.300ms="search"` |
| `.defer` | Defer to next batch | `wire:model.defer="name"` |
| `.live` | Immediate (default) | `wire:model.live="query"` |

### Modifier Implementation

```php
trait HandlesWireModelModifiers
{
    protected array $wireModelModifiers = [];
    
    public function setWireModelModifiers(array $modifiers): void
    {
        $this->wireModelModifiers = $modifiers;
        
        $this->debounce = $modifiers['debounce'] ?? null;
        $this->throttle = $modifiers['throttle'] ?? null;
        $this->lazy = isset($modifiers['lazy']);
        $this->defer = isset($modifiers['defer']);
    }
    
    protected function applyModifiers(): void
    {
        if ($this->debounce) {
            $this->dispatch('wire:model:debounce', $this->debounce);
        }
        if ($this->throttle) {
            $this->dispatch('wire:model:throttle', $this->throttle);
        }
    }
}
```

### Form Component

```php
<?php

namespace GsmUi\Components\Forms;

use GsmUi\Components\BaseComponent;
use Livewire\Attributes\Locked;

class Form extends BaseComponent
{
    #[Locked]
    public ?string $action = null;
    
    #[Locked]
    public string $method = 'POST';
    
    #[Locked]
    public bool $novalidate = false;
    
    #[Locked]
    public array $data = [];
    
    #[Locked]
    public array $errors = [];
    
    public function submit(): void
    {
        $this->validate($this->getRules());
        $this->dispatch('submit', [
            'data' => $this->data,
            'action' => $this->action,
            'method' => $this->method,
        ]);
    }
    
    public function resetForm(): void
    {
        $this->data = [];
        $this->errors = [];
        $this->dispatch('reset');
    }
    
    protected function getRules(): array
    {
        // Collect rules from child components
        return [];
    }
    
    public function render(): \Illuminate\Contracts\View\View
    {
        return view('gsmui::components.forms.form.index');
    }
}
```

## Offline Support

### Persist State

```php
trait PersistsState
{
    public function persistState(): void
    {
        $this->dispatch('persist:state', [
            'component' => $this->componentName(),
            'state' => $this->getPersistableState(),
        ]);
    }
    
    public function restoreState(array $state): void
    {
        foreach ($state as $key => $value) {
            if (property_exists($this, $key)) {
                $this->$key = $value;
            }
        }
    }
    
    protected function getPersistableState(): array
    {
        return [
            'value' => $this->value,
            'open' => $this->open ?? false,
            // ... other state
        ];
    }
}
```

### JavaScript Persistence

```javascript
// Auto-save to localStorage
document.addEventListener('livewire:load', () => {
    Livewire.hook('message.sent', (message) => {
        if (message.type === 'persist:state') {
            localStorage.setItem(`gsmui:${message.payload.component}`, JSON.stringify(message.payload.state));
        }
    });
    
    Livewire.hook('initial-page-load', () => {
        document.querySelectorAll('[data-gsmui-persist]').forEach(el => {
            const key = el.getAttribute('data-gsmui-persist');
            const saved = localStorage.getItem(`gsmui:${key}`);
            if (saved) {
                Livewire.emit('restore:state', { component: key, state: JSON.parse(saved) });
            }
        });
    });
});
```

## Wire:Model Patterns

### Advanced wire:model Usage

GsmUi extends wire:model with component-specific behaviors for seamless two-way binding.

### Standard wire:model Patterns

```blade
{{-- Live binding (default, fires on every keystroke) --}}
<x-gsmui:input wire:model="form.name" label="Name" />

{{-- Lazy binding (fires on blur/change) --}}
<x-gsmui:input wire:model.lazy="form.email" label="Email" />

{{-- Debounced binding (fires after delay) --}}
<x-gsmui:input wire:model.debounce.500ms="search.query" label="Search" />

{{-- Deferred binding (fires on next commit cycle) --}}
<x-gsmui:input wire:model.defer="form.notes" label="Notes" />

{{-- Throttled binding (fires at most once per interval) --}}
<x-gsmui:input wire:model.throttle.1000ms="realtime.position" label="Position" />
```


## Livewire Lifecycle Hooks

### Component Lifecycle Integration

GsmUi components hook into Livewire's lifecycle for initialization, updates, and cleanup.

### Lifecycle Methods

```php
<?php

namespace GsmUi\Concerns;

trait InteractsWithLivewireLifecycle
{
    public function bootGsmuiComponent(): void
    {
        // Called once when component first mounts
        $this->initializeGsmuiState();
    }

    public function mountGsmuiComponent(): void
    {
        // Called on initial mount
        $this->setDefaults();
        $this->validateInitialProps();
    }

    public function hydrateGsmuiComponent(): void
    {
        // Called when component is hydrated from storage
        $this->restoreState();
    }

    public function dehydrateGsmuiComponent(): void
    {
        // Called when component is serialized for storage
        $this->persistState();
    }

    public function renderGsmuiComponent(): void
    {
        // Called before each render
        $this->prepareViewData();
    }

    public function updatedGsmuiComponent(string $property, mixed $value): void
    {
        // Called after any property is updated
        $this->onPropertyUpdated($property, $value);
    }

    public function updatingGsmuiComponent(string $property, mixed $value): void
    {
        // Called before a property is updated (can cancel)
        $this->onPropertyUpdating($property, $value);
    }
}
```

### Lifecycle Hooks in Components

```php
<?php

namespace GsmUi\Components\Forms;

use GsmUi\Components\BaseComponent;
use GsmUi\Concerns\Controllable;
use Livewire\Attributes\Locked;

class Select extends BaseComponent
{
    use Controllable;

    #[Locked]
    public array $options = [];

    #[Locked]
    public bool $open = false;

    public function mount(): void
    {
        parent::mount();
        $this->normalizeOptions();
        $this->syncSelectedOptions();
    }

    public function hydrate(): void
    {
        parent::hydrate();
        $this->normalizeOptions();
    }

    public function dehydrate(): array
    {
        $data = parent::dehydrate();
        $data['gsmui_state'] = [
            'open' => $this->open,
            'value' => $this->value,
        ];
        return $data;
    }

    public function updatedOptions(): void
    {
        $this->normalizeOptions();
        $this->syncSelectedOptions();
        $this->dispatch('options-changed', options: $this->options);
    }

    public function updatedOpen(bool $value): void
    {
        $this->dispatch('open-change', open: $value);
        if ($value) {
            $this->dispatch('gsmui:focus-search');
        }
    }

    protected function normalizeOptions(): void
    {
        $this->options = collect($this->options)->map(function ($option) {
            if (is_string($option)) {
                return ['value' => $option, 'label' => $option];
            }
            return $option;
        })->toArray();
    }
}
```

## Dispatch Events

### Event Patterns

### Component Events

```php
<?php

namespace GsmUi\Components\Forms;

use GsmUi\Components\BaseComponent;

class Input extends BaseComponent
{
    public function updatedValue(mixed $value): void
    {
        // Emit both input and change to match native behavior
        $this->dispatch('input', value: $value);
        $this->dispatch('change', value: $value);

        // Clear validation error on input
        if ($this->invalid) {
            $this->invalid = false;
            $this->errorMessage = null;
        }
    }

    public function focus(): void
    {
        $this->dispatch('focus');
    }

    public function blur(): void
    {
        $this->dispatch('blur');
    }

    public function clear(): void
    {
        $this->value = '';
        $this->dispatch('input', value: '');
        $this->dispatch('change', value: '');
        $this->dispatch('clear');
    }
}
```

```php
trait PersistsState
```

### JavaScript Persistence Layer

```javascript
class GsmuiPersistence {
    constructor(options = {}) {
        this.strategy = options.strategy || 'localStorage';
        this.prefix = options.prefix || 'gsmui:';
        this.ttl = options.ttl || null; // null = forever
    }

    save(key, state) {
        const data = {
            state,
            timestamp: Date.now(),
        };

        switch (this.strategy) {
            case 'localStorage':
                localStorage.setItem(this.prefix + key, JSON.stringify(data));
                break;
            case 'sessionStorage':
                sessionStorage.setItem(this.prefix + key, JSON.stringify(data));
                break;
            case 'indexedDB':
                this.saveToIndexedDB(key, data);
                break;
        }
    }

    load(key) {
        let raw;
        switch (this.strategy) {
            case 'localStorage':
                raw = localStorage.getItem(this.prefix + key);
                break;
            case 'sessionStorage':
                raw = sessionStorage.getItem(this.prefix + key);
                break;
            case 'indexedDB':
                return this.loadFromIndexedDB(key);
        }

        if (!raw) return null;

        const data = JSON.parse(raw);
        if (this.ttl && Date.now() - data.timestamp > this.ttl) {
            this.clear(key);
            return null;
        }

        return data.state;
    }

    clear(key) {
        if (key) {
            localStorage.removeItem(this.prefix + key);
        } else {
            Object.keys(localStorage)
                .filter(k => k.startsWith(this.prefix))
                .forEach(k => localStorage.removeItem(k));
        }
    }

    async saveToIndexedDB(key, data) {
        const db = await this.openDB();
        const tx = db.transaction('gsmui', 'readwrite');
        tx.objectStore('gsmui').put({ key: this.prefix + key, ...data });
    }

    async loadFromIndexedDB(key) {
        const db = await this.openDB();
        const tx = db.transaction('gsmui', 'readonly');
        const result = await tx.objectStore('gsmui').get(this.prefix + key);
        if (!result || (this.ttl && Date.now() - result.timestamp > this.ttl)) {
            return null;
        }
        return result.state;
    }

    openDB() {
        return new Promise((resolve) => {
            const request = indexedDB.open('gsmui', 1);
            request.onupgradeneeded = () => {
                request.result.createObjectStore('gsmui', { keyPath: 'key' });
            };
            request.onsuccess = () => resolve(request.result);
        });
    }
}

window.GsmuiPersistence = new GsmuiPersistence();
```

## Optimistic Updates

### Optimistic UI Patterns

```php
<?php

namespace App\Livewire;

use Livewire\Component;

class TodoList extends Component
{
    public array $todos = [];
    public string $newTodo = '';

    public function addTodo(): void
    {
        $todo = [
            'id' => uniqid(),
            'text' => $this->newTodo,
            'completed' => false,
        ];

        // Optimistic update: add immediately
        $this->todos[] = $todo;
        $this->newTodo = '';

        try {
            // Then persist to server
            $savedTodo = $this->todoService->create($todo);
            // Update with server-assigned ID
            $this->todos = collect($this->todos)
                ->map(fn($t) => $t['id'] === $todo['id'] ? $savedTodo : $t)
                ->toArray();
        } catch (\Exception $e) {
            // Rollback on failure
            $this->todos = collect($this->todos)
                ->filter(fn($t) => $t['id'] !== $todo['id'])
                ->toArray();
            $this->dispatch('gsmui:toast', [
                'message' => 'Failed to add todo',
                'variant' => 'error',
            ]);
        }
    }

    public function toggleTodo(string $id): void
    {
        $index = collect($this->todos)->.findIndex(fn($t) => $t['id'] === $id);
        if ($index === null) return;

        // Optimistic update
        $this->todos[$index]['completed'] = !$this->todos[$index]['completed'];

        try {
            $this->todoService->toggle($id);
        } catch (\Exception $e) {
            // Rollback
            $this->todos[$index]['completed'] = !$this->todos[$index]['completed'];
            $this->dispatch('gsmui:toast', [
                'message' => 'Failed to update todo',
                'variant' => 'error',
            ]);
        }
    }

    public function deleteTodo(string $id): void
    {
        $deletedTodo = collect($this->todos)->firstWhere('id', $id);
        if (!$deletedTodo) return;

        // Optimistic update: remove immediately
        $this->todos = collect($this->todos)->filter(fn($t) => $t['id'] !== $id)->toArray();

        try {
            $this->todoService->delete($id);
        } catch (\Exception $e) {
            // Rollback: re-add the todo
            $this->todos[] = $deletedTodo;
            $this->dispatch('gsmui:toast', [
                'message' => 'Failed to delete todo',
                'variant' => 'error',
            ]);
        }
    }
}
```

### Optimistic Select Component

```php
<?php

namespace GsmUi\Concerns;

trait HasOptimisticUpdates
{
    public bool $optimistic = false;
    public array $pendingUpdates = [];

    public function optimisticUpdate(string $property, mixed $value): void
    {
        $previousValue = $this->$property;
        $this->$property = $value;
        $this->pendingUpdates[$property] = $previousValue;

        $this->dispatch('optimistic:update', [
            'property' => $property,
            'value' => $value,
        ]);
    }

    public function confirmUpdate(string $property): void
    {
        unset($this->pendingUpdates[$property]);
    }

    public function rollbackUpdate(string $property): void
    {
        if (isset($this->pendingUpdates[$property])) {
            $this->$property = $this->pendingUpdates[$property];
            unset($this->pendingUpdates[$property]);
        }
    }

    public function rollbackAllUpdates(): void
    {
        foreach ($this->pendingUpdates as $property => $value) {
            $this->$property = $value;
        }
        $this->pendingUpdates = [];
    }
}
```

```php
    }

    public function loadData(): void
    {
        // Lazy load heavy data
        $this->stats = $this->fetchStats();
        $this->recentActivity = $this->fetchActivity();
        $this->initialized = true;
    }

    public function render()
    {
        return view('livewire.dashboard', [
            'stats' => $this->stats,
            'recentActivity' => $this->recentActivity,
            'initialized' => $this->initialized,
        ]);
    }
}
```

### Polling with State Management

```php
<?php

namespace App\Livewire;

use Livewire\Component;

class LiveFeed extends Component
{
    public int $pollInterval = 5;
    public bool $polling = true;
    public int $lastUpdate = 0;

    public function fetchUpdates(): void
    {
        if (!$this->polling) return;

        $newItems = $this->getNewItemsSince($this->lastUpdate);
        if (!empty($newItems)) {
            $this->items = array_merge($newItems, $this->items);
            $this->lastUpdate = time();
        }
    }

    public function setPollInterval(int $seconds): void
    {
        $this->pollInterval = max(1, min(60, $seconds));
    }

    public function togglePolling(): void
    {
        $this->polling = !$this->polling;
    }
}
```
