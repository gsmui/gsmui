**Project:** GsmUi
**Domain:** Blog + CMS (Shared)
**Version:** 1.0.0

# Purpose

This document defines the database schema for the Blog domain, shared with CMS. Blog posts, CMS pages, and announcements live in a single `content_entries` table discriminated by `content_type`. Categories and tags are shared taxonomies. Revision history enables rollback; a media table supports embedded images, video, and attachments.

References: [06.01-design-principles.md](./06.01-design-principles.md) | [06.02-naming-conventions.md](./06.02-naming-conventions.md) | [06.03-common-columns.md](./06.03-common-columns.md) | [06.04-relationships.md](./06.04-relationships.md)

# Tables

## 1. content_entries

`content_type` ENUM (`blog_post`, `cms_page`, `announcement`). `slug` unique per `content_type`. Status transitions: `draft` → `scheduled` → `published` → `archived`. `metadata` JSON for SEO. `search_vector` TEXT for fulltext. Soft-deletable.

### Migration

```php
Schema::create('content_entries', function (Blueprint $table) {
    $table->id();
    $table->enum('content_type', ['blog_post', 'cms_page', 'announcement']);
    $table->foreignId('author_id')->constrained('users')->cascadeOnDelete();
    $table->string('slug', 255);
    $table->string('title', 255);
    $table->string('excerpt', 500)->nullable();
    $table->longText('content');
    $table->enum('status', ['draft', 'scheduled', 'published', 'archived'])->default('draft');
    $table->boolean('is_featured')->default(false);
    $table->dateTime('published_at')->nullable();
    $table->dateTime('scheduled_at')->nullable();
    $table->unsignedInteger('view_count')->default(0);
    $table->unsignedInteger('like_count')->default(0);
    $table->unsignedInteger('comment_count')->default(0);
    $table->json('metadata')->nullable();
    $table->text('search_vector')->nullable();
    $table->timestamps(3);
    $table->softDeletes();

    $table->unique(['content_type', 'slug']);
    $table->index(['author_id', 'content_type']);
    $table->index(['status', 'published_at']);
    $table->index(['content_type', 'status', 'is_featured']);
    $table->index('scheduled_at');
    $table->fullText('search_vector');
});
```

### Model

```php
class ContentEntry extends Model
{
    use SoftDeletes;

    protected $fillable = [
        'content_type', 'author_id', 'slug', 'title', 'excerpt',
        'content', 'status', 'is_featured', 'published_at',
        'scheduled_at', 'view_count', 'like_count', 'comment_count',
        'metadata', 'search_vector',
    ];

    protected function casts(): array
    {
        return [
            'is_featured' => 'boolean',
            'view_count' => 'integer', 'like_count' => 'integer', 'comment_count' => 'integer',
            'metadata' => 'array', 'published_at' => 'datetime', 'scheduled_at' => 'datetime',
        ];
    }

    public function author(): BelongsTo { return $this->belongsTo(User::class); }
    public function categories(): BelongsToMany { return $this->belongsToMany(ContentCategory::class, 'content_category'); }
    public function tags(): BelongsToMany { return $this->belongsToMany(ContentTag::class, 'content_tag'); }
    public function revisions(): HasMany { return $this->hasMany(ContentEntryRevision::class); }
    public function media(): HasMany { return $this->hasMany(ContentMedia::class); }
}
```

**Relationships:** `belongsTo` **User** (`author_id`); `belongsToMany` **ContentCategory** via `content_category`; `belongsToMany` **ContentTag** via `content_tag`; `hasMany` **ContentEntryRevision** and **ContentMedia**.

# Index Strategy

### content_entries

```sql
-- Query pattern: "Blog posts by author"
CREATE INDEX content_entries_author_type 
ON content_entries(author_id, content_type, status, published_at DESC);

-- Query pattern: "Published posts of type"
CREATE INDEX content_entries_type_status_featured 
ON content_entries(content_type, status, is_featured, published_at DESC);

-- Query pattern: "Scheduled posts for cron"
CREATE INDEX content_entries_scheduled 
ON content_entries(scheduled_at, status) WHERE status = 'scheduled';

-- Query pattern: "Search content"
CREATE FULLTEXT INDEX content_entries_search 
ON content_entries(search_vector, title);

-- Query pattern: "Popular posts"
CREATE INDEX content_entries_type_views 
ON content_entries(content_type, view_count DESC);
```

```sql
-- Query pattern: "Active categories of type"
CREATE INDEX content_categories_type_active
```

# Data Access Patterns

### Blog Post Management

```php
<?php

namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;
use App\Domains\Blog\Models\ContentEntryRevision;

class BlogService
{
    public function createPost(array $data): ContentEntry
    {
        $post = ContentEntry::create(array_merge($data, [
            'content_type' => 'blog_post',
            'slug' => $this->generateSlug($data['title']),
            'search_vector' => $this->buildSearchVector($data['title'], $data['content']),
        ]));

        if (isset($data['categories'])) {
            $post->categories()->sync($data['categories']);
        }

        if (isset($data['tags'])) {
            $post->tags()->sync($data['tags']);
        }

        $this->createRevision($post);

        event(new BlogPostCreated($post));

        return $post;
    }

    public function publishPost(ContentEntry $post): void
    {
        $post->update([
            'status' => 'published',
            'published_at' => now(),
        ]);

        $this->createRevision($post);

        event(new BlogPostPublished($post));
    }

    public function schedulePost(ContentEntry $post, \DateTimeInterface $scheduledAt): void
    {
        $post->update([
            'status' => 'scheduled',
            'scheduled_at' => $scheduledAt,
        ]);

        event(new BlogPostScheduled($post, $scheduledAt));
    }

    public function getPublishedPosts(int $page = 1, int $perPage = 15): \Illuminate\Contracts\Pagination\LengthAwarePaginator
    {
        return ContentEntry::where('content_type', 'blog_post')
            ->where('status', 'published')
            ->whereNotNull('published_at')
            ->with(['author', 'categories', 'tags'])
            ->orderBy('published_at', 'desc')
            ->paginate($perPage, ['*'], 'page', $page);
    }

    public function getFeaturedPosts(int $limit = 5): \Illuminate\Support\Collection
    {
        return ContentEntry::where('content_type', 'blog_post')
            ->where('status', 'published')
            ->where('is_featured', true)
            ->with(['author', 'categories'])
            ->orderBy('published_at', 'desc')
            ->limit($limit)
            ->get();
    }

    public function incrementViewCount(ContentEntry $post): void
    {
        $post->increment('view_count');
    }

    private function generateSlug(string $title): string
    {
        $slug = \Illuminate\Support\Str::slug($title);
        $original = $slug;
        $counter = 1;

        while (ContentEntry::where('slug', $slug)->exists()) {
            $slug = "{$original}-{$counter}";
            $counter++;
        }

        return $slug;
    }

    private function buildSearchVector(string $title, string $content): string
    {
        return strip_tags($title . ' ' . $content);
    }

    private function createRevision(ContentEntry $post): void
    {
        $lastRevision = $post->revisions()->max('revision_number') ?? 0;

        $post->revisions()->create([
            'title' => $post->title,
            'content' => $post->content,
            'excerpt' => $post->excerpt,
            'revision_number' => $lastRevision + 1,
            'created_by' => auth()->id(),
        ]);
    }
}
```

### Revision Service

```php
<?php

namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;
use App\Domains\Blog\Models\ContentEntryRevision;

class RevisionService
{
    public function getRevisions(ContentEntry $entry, int $limit = 50): \Illuminate\Support\Collection
    {
        return $entry->revisions()
            ->with('creator')
            ->orderBy('revision_number', 'desc')
            ->limit($limit)
            ->get();
    }

    public function restoreRevision(ContentEntry $entry, int $revisionNumber): ContentEntry
    {
        $revision = $entry->revisions()
            ->where('revision_number', $revisionNumber)
            ->firstOrFail();

        $entry->update([
            'title' => $revision->title,
            'content' => $revision->content,
            'excerpt' => $revision->excerpt,
        ]);

        $this->createRevision($entry, $revision->content, $revision->title);

        event(new EntryRestored($entry, $revision));

        return $entry;
    }

    public function compareRevisions(ContentEntry $entry, int $from, int $to): array
    {
        $fromRevision = $entry->revisions()->where('revision_number', $from)->firstOrFail();
        $toRevision = $entry->revisions()->where('revision_number', $to)->firstOrFail();

        return [
            'from' => $fromRevision,
            'to' => $toRevision,
            'diff' => $this->computeDiff($fromRevision->content, $toRevision->content),
        ];
    }

    private function computeDiff(string $old, string $new): string
    {
        $oldLines = explode("\n", $old);
        $newLines = explode("\n", $new);
        $diff = [];
        $maxLines = max(count($oldLines), count($newLines));

        for ($i = 0; $i < $maxLines; $i++) {
            $oldLine = $oldLines[$i] ?? null;
            $newLine = $newLines[$i] ?? null;

            if ($oldLine === $newLine) {
                $diff[] = "  {$oldLine}";
            } else {
                if ($oldLine !== null) $diff[] = "- {$oldLine}";
                if ($newLine !== null) $diff[] = "+ {$newLine}";
            }
        }

        return implode("\n", $diff);
    }

    private function createRevision(ContentEntry $entry, string $content, string $title): void
    {
        $lastRevision = $entry->revisions()->max('revision_number') ?? 0;

        $entry->revisions()->create([
            'title' => $title,
            'content' => $content,
            'revision_number' => $lastRevision + 1,
            'created_by' => auth()->id(),
        ]);
    }
}
```

### Search Service

```php
<?php

namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;

class BlogSearchService
{
    public function search(string $query, string $type = null, int $limit = 50): \Illuminate\Support\Collection
    {
        $search = ContentEntry::where('status', 'published')
            ->where(function ($q) use ($query) {
                $q->where('title', 'LIKE', "%{$query}%")
                  ->orWhere('content', 'LIKE', "%{$query}%")
                  ->orWhere('excerpt', 'LIKE', "%{$query}%");
            });

        if ($type) {
            $search->where('content_type', $type);
        }

        return $search->with(['author', 'categories', 'tags'])
            ->orderBy('published_at', 'desc')
            ->limit($limit)
            ->get();
    }

    public function rebuildSearchVectors(string $type = null): void
    {
        $query = ContentEntry::query();
        if ($type) {
            $query->where('content_type', $type);
        }

        $query->each(function (ContentEntry $entry) {
            $entry->update([
                'search_vector' => strip_tags($entry->title . ' ' . $entry->content),
            ]);
        });
    }
}
```

### Media Service

```php
<?php

namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;
use App\Domains\Blog\Models\ContentMedia;
use Illuminate\Http\UploadedFile;

class MediaService
{
    public function uploadMedia(ContentEntry $entry, UploadedFile $file, string $type = 'image', ?string $altText = null): ContentMedia
    {
        $path = $file->store("content/{$entry->id}", 'public');
        $maxSort = $entry->media()->max('sort_order') ?? 0;

        return $entry->media()->create([
            'type' => $type,
            'url' => "/storage/{$path}",
            'alt_text' => $altText ?? $file->getClientOriginalName(),
            'sort_order' => $maxSort + 1,
        ]);
    }

    public function reorderMedia(ContentEntry $entry, array $mediaIds): void
    {
        foreach ($mediaIds as $index => $mediaId) {
            ContentMedia::where('id', $mediaId)
                ->where('content_entry_id', $entry->id)
                ->update(['sort_order' => $index]);
        }
    }

    public function deleteMedia(ContentMedia $media): void
    {
        // Delete file from storage
        $path = str_replace('/storage/', '', $media->url);
        \Storage::disk('public')->delete($path);

        $media->delete();
    }
}
```

```php
<?php

namespace App\Domains\Blog\Observers;

use App\Domains\Blog\Models\ContentEntry;

class ContentEntryObserver
{
    public function created(ContentEntry $entry): void
    {
        \App\Domains\Admin\Models\AuditLog::create([
            'user_id' => $entry->author_id,
            'action' => 'content.created',
            'entity_type' => ContentEntry::class,

                'slug' => $entry->slug,

        ]);
    }

    public function updated(ContentEntry $entry): void
    {
        if ($entry->isDirty('status')) {
            \App\Domains\Admin\Models\AuditLog::create([
                'user_id' => $entry->author_id,
                'action' => "content.{$entry->status}",
                'entity_type' => ContentEntry::class,
                'entity_id' => $entry->id,
                'old_values' => ['status' => $entry->getOriginal('status')],
                'new_values' => ['status' => $entry->status],
            ]);
        }
    }

    public function deleted(ContentEntry $entry): void
    {
        \App\Domains\Admin\Models\AuditLog::create([
            'user_id' => auth()->id(),
            'action' => 'content.deleted',
            'entity_type' => ContentEntry::class,
            'entity_id' => $entry->id,
            'old_values' => ['title' => $entry->title, 'type' => $entry->content_type],
        ]);
    }
}
```

```php
<?php

namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;

class BlogSeoService
{
    public function generateSeoData(ContentEntry $entry): array
    {
        $metadata = $entry->metadata ?? [];

        return [
            'meta_title' => $metadata['meta_title'] ?? $this->generateMetaTitle($entry),
            'meta_description' => $metadata['meta_description'] ?? $this->generateMetaDescription($entry),
            'og_title' => $metadata['og_title'] ?? $entry->title,
            'og_description' => $metadata['og_description'] ?? $entry->excerpt,
            'og_image' => $metadata['og_image'] ?? $this->getFeaturedImage($entry),
            'og_type' => 'article',
            'twitter_card' => 'summary_large_image',
            'canonical_url' => $metadata['canonical_url'] ?? $this->getCanonicalUrl($entry),
            'robots' => $metadata['robots'] ?? 'index,follow',
            'schema_type' => 'Article',
        ];
    }

    public function generateSchema.org(ContentEntry $entry): array
    {
        return [
            '@context' => 'https://schema.org',
            '@type' => 'Article',
            'headline' => $entry->title,
            'description' => $entry->excerpt,
            'author' => [
                '@type' => 'Person',
                'name' => $entry->author->name ?? 'Unknown',
            ],
            'datePublished' => $entry->published_at?->toIso8601String(),
            'dateModified' => $entry->updated_at->toIso8601String(),
            'image' => $this->getFeaturedImage($entry),
            'url' => $this->getCanonicalUrl($entry),
        ];
    }

    private function generateMetaTitle(ContentEntry $entry): string
    {
        $title = $entry->title;
        if (strlen($title) > 57) {
            $title = substr($title, 0, 57) . '...';
        }
        return $title . ' | ' . config('app.name');
    }

    private function generateMetaDescription(ContentEntry $entry): string
    {
        $description = $entry->excerpt ?? strip_tags(substr($entry->content, 0, 160));
        return substr($description, 0, 157) . '...';
    }

    private function getFeaturedImage(ContentEntry $entry): ?string
    {
        $media = $entry->media()->where('type', 'image')->first();
        return $media?->url;
    }

    private function getCanonicalUrl(ContentEntry $entry): string
    {
        return url("/blog/{$entry->slug}");
    }
}
```

```php
namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;
use SimpleXMLElement;

class BlogSitemapService
{
    public function generate(): string
    {
        $xml = new SimpleXMLElement('<?xml version="1.0" encoding="UTF-8"?><urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9"/>');

        $posts = ContentEntry::where('content_type', 'blog_post')
            ->where('status', 'published')
            ->whereNotNull('published_at')
            ->orderBy('published_at', 'desc')
            ->get();

        foreach ($posts as $post) {
            $url = $xml->addChild('url');
            $url->addChild('loc', url("/blog/{$post->slug}"));
            $url->addChild('lastmod', $post->updated_at->format('Y-m-d'));
            $url->addChild('changefreq', 'weekly');
            $url->addChild('priority', $post->is_featured ? '0.9' : '0.7');
        }

        return $xml->asXML();
    }

    public function generateForSearchEngines(): string
    {
        $xml = $this->generate();
        
        // Store sitemap
        \Storage::disk('public')->put('sitemap.xml', $xml);
        
        return $xml;
    }
}
```

```php
namespace App\Domains\Blog\Services;

class BlogRssService
{
    public function generate(): string
    {
        $xml = new SimpleXMLElement('<?xml version="1.0" encoding="UTF-8"?><rss version="2.0" xmlns:atom="http://www.w3.org/2005/Atom"/>');
        
        $channel = $xml->addChild('channel');
        $channel->addChild('title', config('app.name') . ' Blog');
        $channel->addChild('link', url('/blog'));
        $channel->addChild('description', 'Latest blog posts from ' . config('app.name'));
        $channel->addChild('language', 'en-us');
        $channel->addChild('lastBuildDate', now()->toRfc2822String());
        $channel->addChild('atom:link', '', 'http://www.w3.org/2005/Atom');
        $channel->addAttribute('href', url('/feed.xml'));
        $channel->addAttribute('rel', 'self');
        $channel->addAttribute('type', 'application/rss+xml');

        $posts = ContentEntry::where('content_type', 'blog_post')
            ->where('status', 'published')
            ->whereNotNull('published_at')
            ->orderBy('published_at', 'desc')
            ->limit(50)
            ->with('author')
            ->get();

        foreach ($posts as $post) {
            $item = $channel->addChild('item');
            $item->addChild('title', htmlspecialchars($post->title));
            $item->addChild('link', url("/blog/{$post->slug}"));
            $item->addChild('description', htmlspecialchars($post->excerpt ?? strip_tags(substr($post->content, 0, 500))));
            $item->addChild('pubDate', $post->published_at->toRfc2822String());
            $item->addChild('guid', url("/blog/{$post->slug}"));
            
            if ($post->author) {
                $item->addChild('author', htmlspecialchars($post->author->email));
            }

            foreach ($post->categories as $category) {
                $item->addChild('category', htmlspecialchars($category->name));
            }
        }

        return $xml->asXML();
    }
}
```

## Content Analytics Service

```php
<?php

namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;
use Illuminate\Support\Facades\DB;

class ContentAnalyticsService
{
    public function getPopularPosts(int $days = 30, int $limit = 10): \Illuminate\Support\Collection
    {
        return ContentEntry::where('content_type', 'blog_post')
            ->where('status', 'published')
            ->where('published_at', '>=', now()->subDays($days))
            ->orderBy('view_count', 'desc')
            ->limit($limit)
            ->with('author')
            ->get();
    }

    public function getEngagementMetrics(ContentEntry $post): array
    {
        return [
            'views' => $post->view_count,
            'likes' => $post->like_count,
            'comments' => $post->comment_count,
            'engagement_rate' => $this->calculateEngagementRate($post),
            'avg_time_on_page' => $this->getAvgTimeOnPage($post),
        ];
    }

    public function getAuthorStats(int $authorId, int $days = 30): array
    {
        return ContentEntry::where('author_id', $authorId)
            ->where('content_type', 'blog_post')
            ->where('created_at', '>=', now()->subDays($days))
            ->selectRaw('
                COUNT(*) as total_posts,
                SUM(view_count) as total_views,
                SUM(like_count) as total_likes,
                SUM(comment_count) as total_comments,
                AVG(view_count) as avg_views_per_post
            ')
            ->first()
            ->toArray();
    }

    public function getContentPerformanceReport(int $days = 30): array
    {
        $posts = ContentEntry::where('content_type', 'blog_post')
            ->where('status', 'published')
            ->where('published_at', '>=', now()->subDays($days))
            ->get();

        return [
            'total_posts' => $posts->count(),
            'total_views' => $posts->sum('view_count'),
            'total_likes' => $posts->sum('like_count'),
            'total_comments' => $posts->sum('comment_count'),
            'avg_views' => $posts->avg('view_count'),
            'avg_likes' => $posts->avg('like_count'),
            'top_performer' => $posts->sortByDesc('view_count')->first(),
            'engagement_trend' => $this->getEngagementTrend($days),
        ];
    }

    private function calculateEngagementRate(ContentEntry $post): float
    {
        if ($post->view_count === 0) return 0;
        
        $engagements = $post->like_count + $post->comment_count;
        return round(($engagements / $post->view_count) * 100, 2);
    }

    private function getAvgTimeOnPage(ContentEntry $post): ?float
    {
        return DB::table('analytics_page_views')
            ->where('url', 'LIKE', "%/{$post->slug}")
            ->whereNotNull('dwell_time')
            ->avg('dwell_time');
    }

    private function getEngagementTrend(int $days): array
    {
        return ContentEntry::where('content_type', 'blog_post')
            ->where('status', 'published')
            ->where('published_at', '>=', now()->subDays($days))
            ->selectRaw('
                DATE(published_at) as date,
                SUM(view_count) as views,
                SUM(like_count) as likes,
                SUM(comment_count) as comments
            ')
            ->groupBy('date')
            ->orderBy('date')
            ->get()
            ->toArray();
    }
}
```

```php
<?php

namespace App\Domains\Blog\Services;

use App\Domains\Blog\Models\ContentEntry;

class ContentModerationService
{
    public function submitForReview(ContentEntry $entry): void
    {
        $entry->update(['status' => 'draft']);
        
        \App\Domains\Admin\Models\ModerationQueue::create([
            'content_type' => 'content_entry',
            'content_id' => $entry->id,
            'status' => 'pending',
            'submitted_by' => $entry->author_id,
        ]);

        event(new ContentSubmittedForReview($entry));
    }

    public function approve(ContentEntry $entry, int $moderatorId, ?string $notes = null): void
    {
        $entry->update(['status' => 'published', 'published_at' => now()]);
        
        \App\Domains\Admin\Models\ModerationQueue::where('content_type', 'content_entry')
            ->where('content_id', $entry->id)
            ->update([
                'status' => 'approved',
                'moderated_by' => $moderatorId,
                'moderation_notes' => $notes,
                'moderated_at' => now(),
            ]);

        event(new ContentApproved($entry, $moderatorId));
    }

    public function reject(ContentEntry $entry, int $moderatorId, string $reason): void
    {
        $entry->update(['status' => 'draft']);
        
        \App\Domains\Admin\Models\ModerationQueue::where('content_type', 'content_entry')
            ->where('content_id', $entry->id)
            ->update([
                'status' => 'rejected',
                'moderated_by' => $moderatorId,
                'moderation_notes' => $reason,
                'moderated_at' => now(),
            ]);

        event(new ContentRejected($entry, $moderatorId, $reason));
    }
}
```

## Spam Detection Service

```php
<?php

namespace App\Domains\Blog\Services;

class SpamDetectionService
{
    private array $spamPatterns = [
        '/\b(buy|click here|free money|act now)\b/i',
        '/https?:\/\/[^\s]+/i',
        '/\b(viagra|cialis|casino)\b/i',
    ];

    public function isSpam(string $content): bool
    {
        foreach ($this->spamPatterns as $pattern) {
            if (preg_match($pattern, $content)) {
                return true;
            }
        }

        return false;
    }

    public function getSpamScore(string $content): float
    {
        $score = 0;
        $wordCount = str_word_count($content);

        // Check for excessive links
        preg_match_all('/https?:\/\/[^\s]+/', $content, $links);
        $linkRatio = count($links[0]) / max($wordCount, 1);
        if ($linkRatio > 0.3) $score += 0.5;

        // Check for spam patterns
        foreach ($this->spamPatterns as $pattern) {
            if (preg_match($pattern, $content)) {
                $score += 0.3;
            }
        }

        // Check for ALL CAPS
        if (preg_match('/^[A-Z\s!]{20,}$/', $content)) {
            $score += 0.4;
        }

        return min($score, 1.0);
    }
}
```

```php
<?php

namespace Database\Factories\Domains\Blog;

use App\Domains\Blog\Models\BlogPost;
use App\Models\User;
use Illuminate\Database\Eloquent\Factories\Factory;

class BlogPostFactory extends Factory
{
    protected $model = BlogPost::class;

    public function definition(): array
    {
        return [
            'author_id' => User::factory(),
            'title' => $this->faker->sentence(6),
            'slug' => $this->faker->unique()->slug(3),
            'excerpt' => $this->faker->paragraph(),
            'body_markdown' => $this->faker->paragraphs(5, true),
            'body_html' => '<p>' . $this->faker->paragraphs(5, true) . '</p>',
            'status' => $this->faker->randomElement(['draft', 'published', 'archived']),
            'featured_image' => $this->faker->optional(60)->imageUrl(1200, 630),
            'reading_time' => $this->faker->numberBetween(1, 15),
            'metadata' => [
                'description' => $this->faker->optional()->sentence(),
                'keywords' => $this->faker->optional()->words(5),
            ],
        ];
    }

    public function published(): static
    {
        return $this->state(fn (array $attributes) => [
            'status' => 'published',
            'published_at' => $this->faker->dateTimeBetween('-1 year', 'now'),
        ]);
    }

    public function draft(): static
    {
        return $this->state(['status' => 'draft']);
    }

    public function withCategories(int $count = 2): static
    {
        return $this->afterCreating(function (BlogPost $post) use ($count) {
            $categories = \App\Domains\Blog\Models\BlogCategory::factory()
                ->count($count)
                ->create();

            $post->categories()->attach($categories->pluck('id'));
        });
    }

    public function withTags(int $count = 3): static
    {
        return $this->afterCreating(function (BlogPost $post) use ($count) {
            $tags = \App\Domains\Blog\Models\BlogTag::factory()
                ->count($count)
                ->create();

            $post->tags()->attach($tags->pluck('id'));
        });
    }
}
```

## Blog Seeder

```php
<?php

namespace Database\Seeders;

use App\Domains\Blog\Models\BlogPost;
use App\Domains\Blog\Models\BlogCategory;
use App\Domains\Blog\Models\BlogTag;
use App\Models\User;
use Illuminate\Database\Seeder;

class BlogSeeder extends Seeder
{
    public function run(): void
    {
        $users = User::factory()->count(3)->create();

        $categories = BlogCategory::factory()->count(8)->create();
        $tags = BlogTag::factory()->count(15)->create();

        BlogPost::factory()
            ->count(25)
            ->published()
            ->create([
                'author_id' => $users->random()->id,
            ])
            ->each(function (BlogPost $post) use ($categories, $tags) {
                $post->categories()->attach(
                    $categories->random(rand(1, 3))->pluck('id')
                );
                $post->tags()->attach(
                    $tags->random(rand(1, 5))->pluck('id')
                );
            });
    }
}
```
