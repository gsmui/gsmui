# 10.22 - Dynamic Permissions

**Project:** GsmUi
**Domain:** Authorization & RBAC
**Version:** 1.0.0

---

## Purpose

This document defines the **dynamic permission system** — the ability to create, modify, and remove permissions at runtime without code changes. Dynamic permissions support custom roles for enterprise customers, feature-specific permissions, and time-limited access grants. They extend the static permission set defined in [10.04-permission-groups.md](./10.04-permission-groups.md) with runtime flexibility.

Static permissions are seeded at deploy time and never change. Dynamic permissions are created by administrators at runtime and must be validated, cached, and audited.

## Dynamic Permission Creation

### Permission Model

```php
// app/Models/DynamicPermission.php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;
use Illuminate\Database\Eloquent\Builder;

class DynamicPermission extends Model
{
    protected $fillable = [
        'name',
        'display_name',
        'description',
        'guard_name',
        'group',
        'created_by',
        'organization_id',
        'is_system',
        'is_active',
        'version',
        'deprecated_at',
        'replaced_by',
    ];

    protected $casts = [
        'is_system' => 'boolean',
        'is_active' => 'boolean',
        'deprecated_at' => 'datetime',
    ];

    protected static function booted(): void
    {
        static::creating(function (DynamicPermission $permission) {
            if (! $permission->guard_name) {
                $permission->guard_name = config('permission.defaults.guard', 'web');
            }
            if (! $permission->version) {
                $permission->version = 1;
            }
        });
    }

    public function roles(): BelongsToMany
    {
        return $this->belongsToMany(Role::class, 'dynamic_role_has_permissions');
    }

    public function createdBy(): BelongsTo
    {
        return $this->belongsTo(User::class, 'created_by');
    }

    public function organization(): BelongsTo
    {
        return $this->belongsTo(Organization::class);
    }

    public function scopeActive(Builder $query): Builder
    {
        return $query->where('is_active', true);
    }

    public function scopeNonSystem(Builder $query): Builder
    {
        return $query->where('is_system', false);
    }

    public function scopeForOrganization(Builder $query, ?int $organizationId): Builder
    {
        if ($organizationId) {
            return $query->where('organization_id', $organizationId)
                ->orWhereNull('organization_id');
        }
        return $query->whereNull('organization_id');
    }
}
```

### Permission Creation Service

```php
// app/Services/DynamicPermissionService.php

namespace App\Services;

use App\Models\DynamicPermission;
use App\Models\User;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;
use InvalidArgumentException;

class DynamicPermissionService
{
    protected PermissionValidator $validator;

    public function __construct(PermissionValidator $validator)
    {
        $this->validator = $validator;
    }

    public function create(
        string $name,
        string $displayName,
        ?string $description = null,
        ?string $group = null,
        ?User $createdBy = null,
        ?int $organizationId = null,
    ): DynamicPermission {
        // Validate the permission name
        $this->validator->validateName($name);

        // Check for uniqueness
        if ($this->permissionExists($name, $organizationId)) {
            throw new InvalidArgumentException(
                "Permission '{$name}' already exists"
            );
        }

        // Check if it conflicts with a system permission
        if ($this->conflictsWithSystem($name)) {
            throw new InvalidArgumentException(
                "Permission '{$name}' conflicts with a system permission"
            );
        }

        $permission = DynamicPermission::create([
            'name' => $name,
            'display_name' => $displayName,
            'description' => $description,
            'group' => $group,
            'created_by' => $createdBy?->id,
            'organization_id' => $organizationId,
            'is_system' => false,
            'is_active' => true,
        ]);

        // Clear permission cache
        $this->clearCache();

        // Log the creation
        if ($createdBy) {
            activity()
                ->performedOn($permission)
                ->causedBy($createdBy)
                ->withProperties([
                    'name' => $name,
                    'group' => $group,
                ])
                ->event('dynamic_permission_created')
                ->log('Dynamic permission created');
        }

        return $permission;
    }

    public function update(
        DynamicPermission $permission,
        array $attributes,
        ?User $updatedBy = null,
    ): DynamicPermission {
        if ($permission->is_system) {
            throw new InvalidArgumentException(
                "Cannot modify system permission: {$permission->name}"
            );
        }

        $oldAttributes = $permission->only(array_keys($attributes));

        $permission->update($attributes);

        $this->clearCache();

        if ($updatedBy) {
            activity()
                ->performedOn($permission)
                ->causedBy($updatedBy)
                ->withProperties([
                    'old' => $oldAttributes,
                    'new' => $attributes,
                ])
                ->event('dynamic_permission_updated')
                ->log('Dynamic permission updated');
        }

        return $permission->refresh();
    }

    public function deactivate(
        DynamicPermission $permission,
        ?User $deactivatedBy = null,
    ): DynamicPermission {
        if ($permission->is_system) {
            throw new InvalidArgumentException(
                "Cannot deactivate system permission: {$permission->name}"
            );
        }

        $permission->update([
            'is_active' => false,
            'deprecated_at' => now(),
        ]);

        // Remove from all roles
        $permission->roles()->detach();

        $this->clearCache();

        if ($deactivatedBy) {
            activity()
                ->performedOn($permission)
                ->causedBy($deactivatedBy)
                ->event('dynamic_permission_deactivated')
                ->log('Dynamic permission deactivated');
        }

        return $permission->refresh();
    }

    public function replace(
        DynamicPermission $oldPermission,
        string $newName,
        string $newDisplayName,
        ?User $replacedBy = null,
    ): DynamicPermission {
        DB::beginTransaction();

        try {
            // Create the new permission
            $newPermission = $this->create(
                name: $newName,
                displayName: $newDisplayName,
                description: $oldPermission->description,
                group: $oldPermission->group,
                createdBy: $replacedBy,
                organizationId: $oldPermission->organization_id,
            );

            // Migrate roles from old to new
            $roles = $oldPermission->roles()->get();
            foreach ($roles as $role) {
                $role->givePermissionTo($newPermission);
            }

            // Mark old permission as deprecated
            $oldPermission->update([
                'is_active' => false,
                'deprecated_at' => now(),
                'replaced_by' => $newPermission->id,
            ]);

            $oldPermission->roles()->detach();

            $this->clearCache();

            if ($replacedBy) {
                activity()
                    ->performedOn($oldPermission)
                    ->causedBy($replacedBy)
                    ->withProperties([
                        'old_permission' => $oldPermission->name,
                        'new_permission' => $newPermission->name,
                        'roles_migrated' => $roles->pluck('name')->toArray(),
                    ])
                    ->event('dynamic_permission_replaced')
                    ->log('Dynamic permission replaced');
            }

            DB::commit();

            return $newPermission;
        } catch (\Exception $e) {
            DB::rollBack();
            throw $e;
        }
    }

    protected function permissionExists(string $name, ?int $organizationId): bool
    {
        $query = DynamicPermission::where('name', $name);

        if ($organizationId) {
            $query->where(function ($q) use ($organizationId) {
                $q->where('organization_id', $organizationId)
                    ->orWhereNull('organization_id');
            });
        } else {
            $query->whereNull('organization_id');
        }

        return $query->exists();
    }

    protected function conflictsWithSystem(string $name): bool
    {
        // Check against Spatie's static permissions
        return \Spatie\Permission\Models\Permission::where('name', $name)->exists();
    }

    protected function clearCache(): void
    {
        DynamicPermission::forgetCache();
        Cache::tags(['dynamic_permissions'])->flush();
    }
}
```

### Group Management Service

```php
// app/Services/PermissionGroupService.php

namespace App\Services;

use App\Models\PermissionGroup;
use App\Models\DynamicPermission;
use App\Models\User;
use InvalidArgumentException;

class PermissionGroupService
{
    public function create(
        string $name,
        string $description,
        ?User $createdBy = null,
        ?int $organizationId = null,
    ): PermissionGroup {
        $slug = \Str::slug($name);

        // Check uniqueness within organization
        $existing = PermissionGroup::where('slug', $slug);

        if ($organizationId) {
            $existing->where(function ($q) use ($organizationId) {
                $q->where('organization_id', $organizationId)
                    ->orWhereNull('organization_id');
            });
        } else {
            $existing->whereNull('organization_id');
        }

        if ($existing->exists()) {
            throw new InvalidArgumentException(
                "Permission group '{$name}' already exists"
            );
        }

        $group = PermissionGroup::create([
            'name' => $name,
            'description' => $description,
            'slug' => $slug,
            'organization_id' => $organizationId,
            'created_by' => $createdBy?->id,
            'is_system' => false,
            'sort_order' => PermissionGroup::max('sort_order') + 1,
        ]);

        if ($createdBy) {
            activity()
                ->performedOn($group)
                ->causedBy($createdBy)
                ->event('permission_group_created')
                ->log('Permission group created');
        }

        return $group;
    }

    public function addPermissions(
        PermissionGroup $group,
        array $permissionIds,
    ): PermissionGroup {
        if ($group->is_system) {
            throw new InvalidArgumentException(
                "Cannot modify system permission group"
            );
        }

        // Validate all permissions exist
        $permissions = DynamicPermission::whereIn('id', $permissionIds)->get();

        if ($permissions->count() !== count($permissionIds)) {
            throw new InvalidArgumentException(
                "Some permissions not found"
            );
        }

        // Check organization scope
        if ($group->organization_id) {
            $orgPermissions = $permissions->filter(
                fn ($p) => $p->organization_id === $group->organization_id
                    || is_null($p->organization_id)
            );

            if ($orgPermissions->count() !== $permissions->count()) {
                throw new InvalidArgumentException(
                    "Some permissions belong to a different organization"
                );
            }
        }

        $group->permissions()->syncWithoutDetaching(
            $permissionIds->mapWithKeys(fn ($id) => [
                $id => ['sort_order' => $group->permissions()->count()],
            ])->toArray()
        );

        return $group->load('permissions');
    }

    public function removePermission(
        PermissionGroup $group,
        DynamicPermission $permission,
    ): PermissionGroup {
        if ($group->is_system) {
            throw new InvalidArgumentException(
                "Cannot modify system permission group"
            );
        }

        $group->permissions()->detach($permission->id);

        return $group->load('permissions');
    }

    public function getPermissionsForGroup(PermissionGroup $group): array
    {
        return $group->permissions()
            ->active()
            ->pluck('name')
            ->toArray();
    }
}
```

### Custom Role Service

```php
// app/Services/CustomRoleService.php

namespace App\Services;

use App\Models\Role;
use App\Models\DynamicPermission;
use App\Models\User;
use App\Models\Organization;
use InvalidArgumentException;

class CustomRoleService
{
    protected DynamicPermissionService $permissionService;

    public function __construct(DynamicPermissionService $permissionService)
    {
        $this->permissionService = $permissionService;
    }

    public function create(
        string $name,
        string $description,
        ?User $createdBy = null,
        ?Organization $organization = null,
    ): Role {
        // Validate name
        $this->validateRoleName($name);

        // Check uniqueness
        $query = Role::where('name', $name);

        if ($organization) {
            $query->where(function ($q) use ($organization) {
                $q->where('organization_id', $organization->id)
                    ->orWhereNull('organization_id');
            });
        } else {
            $query->whereNull('organization_id');
        }

        if ($query->exists()) {
            throw new InvalidArgumentException(
                "Role '{$name}' already exists"
            );
        }

        $role = Role::create([
            'name' => $name,
            'guard_name' => 'web',
            'description' => $description,
            'organization_id' => $organization?->id,
            'is_system' => false,
        ]);

        if ($createdBy) {
            activity()
                ->performedOn($role)
                ->causedBy($createdBy)
                ->withProperties([
                    'name' => $name,
                    'organization_id' => $organization?->id,
                ])
                ->event('custom_role_created')
                ->log('Custom role created');
        }

        return $role;
    }

    public function syncPermissions(
        Role $role,
        array $permissionNames,
    ): Role {
        if ($role->is_system) {
            throw new InvalidArgumentException(
                "Cannot modify system role: {$role->name}"
            );
        }

        // Resolve permission names to IDs
        $permissionIds = [];

        foreach ($permissionNames as $name) {
            // Try static permissions first
            $staticPermission = \Spatie\Permission\Models\Permission::where('name', $name)->first();

            if ($staticPermission) {
                $permissionIds[] = $staticPermission->id;
                continue;
            }

            // Try dynamic permissions
            $dynamicPermission = DynamicPermission::where('name', $name)
                ->active()
                ->first();

            if ($dynamicPermission) {
                $permissionIds[] = $dynamicPermission->id;
                continue;
            }

            throw new InvalidArgumentException(
                "Permission '{$name}' not found"
            );
        }

        // Sync permissions
        $role->syncPermissions($permissionNames);

        return $role->load('permissions');
    }

    public function assignToUser(
        User $user,
        Role $role,
        ?User $assignedBy = null,
    ): void {
        if ($role->organization_id) {
            // Verify user is a member of the organization
            if (! $user->organizations()
                ->where('id', $role->organization_id)
                ->exists()) {
                throw new InvalidArgumentException(
                    "User is not a member of the required organization"
                );
            }
        }

        $user->assignRole($role);

        if ($assignedBy) {
            activity()
                ->performedOn($user)
                ->causedBy($assignedBy)
                ->withProperties([
                    'role' => $role->name,
                    'organization_id' => $role->organization_id,
                ])
                ->event('custom_role_assigned')
                ->log('Custom role assigned');
        }
    }

    protected function validateRoleName(string $name): void
    {
        if (empty(trim($name))) {
            throw new InvalidArgumentException('Role name cannot be empty');
        }

        if (strlen($name) > 128) {
            throw new InvalidArgumentException('Role name cannot exceed 128 characters');
        }

        // Reserved names
        $reserved = ['admin', 'super_admin', 'guest', 'user', 'moderator'];
        if (in_array(strtolower($name), $reserved)) {
            throw new InvalidArgumentException(
                "Role name '{$name}' is reserved"
            );
        }
    }
}
```

## Validation Rules

### Permission Name Validation

```php
// app/Services/PermissionValidator.php

namespace App\Services;

use InvalidArgumentException;

class PermissionValidator
{
    protected const NAME_PATTERN = '/^[a-z][a-z0-9_]*\.[a-z][a-z0-9_]*$/';
    protected const MAX_NAME_LENGTH = 255;
    protected const RESERVED_PREFIXES = ['system.', 'internal.'];

    public function validateName(string $name): void
    {
        if (empty(trim($name))) {
            throw new InvalidArgumentException('Permission name cannot be empty');
        }

        if (strlen($name) > self::MAX_NAME_LENGTH) {
            throw new InvalidArgumentException(
                "Permission name cannot exceed " . self::MAX_NAME_LENGTH . " characters"
            );
        }

        if (! preg_match(self::NAME_PATTERN, $name)) {
            throw new InvalidArgumentException(
                "Permission name must follow resource.action format (e.g., themes.create). " .
                "Got: {$name}"
            );
        }

        foreach (self::RESERVED_PREFIXES as $prefix) {
            if (str_starts_with($name, $prefix)) {
                throw new InvalidArgumentException(
                    "Permission name cannot use reserved prefix: {$prefix}"
                );
            }
        }
    }

    public function validateDisplayName(string $displayName): void
    {
        if (empty(trim($displayName))) {
            throw new InvalidArgumentException('Display name cannot be empty');
        }

        if (strlen($displayName) > 255) {
            throw new InvalidArgumentException('Display name cannot exceed 255 characters');
        }
    }

    public function validateGroup(string $group): void
    {
        if (empty(trim($group))) {
            throw new InvalidArgumentException('Group name cannot be empty');
        }

        if (strlen($group) > 128) {
            throw new InvalidArgumentException('Group name cannot exceed 128 characters');
        }
    }
}
```

### Validation Summary

| Rule | Description | Example |
|---|---|---|
| Format | Must be `resource.action` | `themes.create` ✅, `manage` ❌ |
| Uniqueness | Unique within guard + organization | Two orgs can have same name |
| No system prefix | Cannot use `system.` or `internal.` | `system.admin` ❌ |
| No reserved names | Cannot override core permissions | `admin.access` ❌ |
| Length | Max 255 characters | — |
| Character set | Lowercase alphanumeric + underscores | `my_feature.read` ✅ |

```php
// app/Observers/DynamicPermissionObserver.php

namespace App\Observers;

use App\Models\DynamicPermission;

class DynamicPermissionObserver
{
    public function created(DynamicPermission $permission): void
    {
        $this->createVersion($permission, 'created');
    }

    public function updated(DynamicPermission $permission): void
    {
        if ($permission->wasChanged()) {
            $permission->increment('version');

            $this->createVersion($permission, 'updated');
        }

    {
        $this->createVersion($permission, 'deleted');

            'version' => $permission->version,
            'name' => $permission->name,
            'display_name' => $permission->display_name,
            'description' => $permission->description,
            'group' => $permission->group,
            'is_active' => $permission->is_active,
            'changed_by' => auth()->id(),
            'change_type' => $changeType,
        ]);
    }
}
```

## Caching Implications

Dynamic permission changes must clear the permission cache to prevent stale authorization decisions.

### Cache Strategy

```php
// app/Services/DynamicPermissionCache.php

namespace App\Services;

use App\Models\DynamicPermission;
use App\Models\User;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Redis;

class DynamicPermissionCache
{
    protected const CACHE_PREFIX = 'dynamic_perm:';
    protected const CACHE_TTL = 3600; // 1 hour

    public function getForUser(User $user): array
    {
        $cacheKey = self::CACHE_PREFIX . "user:{$user->id}";

        return Cache::remember($cacheKey, self::CACHE_TTL, function () use ($user) {
            return $this->resolvePermissions($user);
        });
    }

    public function hasPermission(User $user, string $permission): bool
    {
        $permissions = $this->getForUser($user);

        return in_array($permission, $permissions);
    }

    public function clearForUser(User $user): void
    {
        Cache::forget(self::CACHE_PREFIX . "user:{$user->id}");
    }

    public function clearAll(): void
    {
        // Clear all dynamic permission caches
        $keys = Redis::keys(self::CACHE_PREFIX . '*');

        if (! empty($keys)) {
            Redis::del($keys);
        }

        // Also clear Spatie's cache
        DynamicPermission::forgetCache();
    }

    public function clearForOrganization(int $organizationId): void
    {
        // Get all users in the organization
        $userIds = \App\Models\OrganizationMember::where('organization_id', $organizationId)
            ->pluck('user_id')
            ->toArray();

        foreach ($userIds as $userId) {
            $this->clearForUser(\App\Models\User::find($userId));
        }
    }

    protected function resolvePermissions(User $user): array
    {
        $permissions = [];

        // Get dynamic permissions from roles
        foreach ($user->roles as $role) {
            $rolePermissions = $role->dynamicPermissions()
                ->active()
                ->pluck('name')
                ->toArray();

            $permissions = array_merge($permissions, $rolePermissions);
        }

        // Get direct dynamic permissions
        $directPermissions = $user->dynamicPermissions()
            ->active()
            ->pluck('name')
            ->toArray();

        $permissions = array_merge($permissions, $directPermissions);

        return array_unique($permissions);
    }
}
```

## Runtime Permission Generation

Enterprise administrators can configure systems that automatically generate permissions at runtime based on resource creation, organizational changes, or integration events. This eliminates the need for manual permission management as the platform scales.

### Resource-Based Permission Generation

When new resources are created (e.g., projects, workspaces, departments), corresponding permissions can be auto-generated to control access.

```php
// app/Services/RuntimePermissionGenerator.php

namespace App\Services;

use App\Models\DynamicPermission;
use App\Models\User;
use Illuminate\Support\Str;

class RuntimePermissionGenerator
{
    /**
     * Generate permissions for a newly created resource.
     *
     * For a resource like "project" with slug "my-app", this generates:
     * - projects.my-app.view
     * - projects.my-app.update
     * - projects.my-app.delete
     * - projects.my-app.manage_members
     * - projects.my-app.manage_settings
     * - projects.my-app.deploy
     * - projects.my-app.view_audit
     */
    public function generateForResource(
        string $resourceType,
        string $resourceSlug,
        string $resourceName,
        User $createdBy,
        ?int $organizationId = null,
    ): array {
        $generatedPermissions = [];
        $actions = $this->getActionsForResource($resourceType);

        foreach ($actions as $action => $actionConfig) {
            $permissionName = "{$resourceType}.{$resourceSlug}.{$action}";
            $displayName = "{$actionConfig['label']} {$resourceName}";

            $permission = DynamicPermission::create([
                'name' => $permissionName,
                'display_name' => $displayName,
                'description' => $actionConfig['description'] ?? "Permission to {$action} {$resourceName}",
                'group' => "{$resourceType}_generated",
                'created_by' => $createdBy->id,
                'organization_id' => $organizationId,
                'is_system' => false,
                'is_active' => true,
            ]);

            $generatedPermissions[] = $permission;
        }

        // Grant all permissions to the creator
        $creatorRole = $this->ensureCreatorRole($resourceType, $organizationId);
        foreach ($generatedPermissions as $permission) {
            $creatorRole->givePermissionTo($permission);
        }

        return $generatedPermissions;
    }

    /**
     * Generate team-scoped permissions when a team is created.
     */
    public function generateForTeam(
        \App\Models\Team $team,
        User $createdBy,
    ): array {
        $slug = Str::slug($team->name);
        $actions = ['view', 'update', 'manage_members', 'manage_resources', 'manage_settings', 'view_audit'];

        $permissions = [];
        foreach ($actions as $action) {
            $name = "team.{$slug}.{$action}";
            $permission = DynamicPermission::create([
                'name' => $name,
                'display_name' => ucfirst(str_replace('_', ' ', $action)) . ' — ' . $team->name,
                'group' => 'teams_generated',
                'created_by' => $createdBy->id,
                'organization_id' => $team->organization_id,
            ]);
            $permissions[] = $permission;
        }

        return $permissions;
    }

    /**
     * Generate subscription-tier permissions when a plan changes.
     */
    public function generateForSubscriptionTier(
        User $user,
        string $newTier,
        string $oldTier,
    ): array {
        $tierActions = config("subscription_permissions.{$newTier}", []);
        $oldTierActions = config("subscription_permissions.{$oldTier}", []);

        $newPermissions = array_diff($tierActions, $oldActions ?? []);
        $revokedPermissions = array_diff($oldActions ?? [], $tierActions);

        $generated = [];

        foreach ($newPermissions as $action) {
            $name = "subscription.{$newTier}.{$action}";
            $permission = DynamicPermission::where('name', $name)->first();

            if ($permission) {
                $user->givePermissionTo($permission);
                $generated[] = $permission;
            }
        }

        foreach ($revokedPermissions as $action) {
            $name = "subscription.{$oldTier}.{$action}";
            $user->revokePermissionTo($name);
        }

        return $generated;
    }

    /**
     * Clean up permissions for a deleted resource.
     */
    public function cleanupForResource(
        string $resourceType,
        string $resourceSlug,
    ): int {
        $pattern = "{$resourceType}.{$resourceSlug}.%";

        $permissions = DynamicPermission::where('name', 'LIKE', $pattern)
            ->where('is_system', false)
            ->get();

        foreach ($permissions as $permission) {
            $permission->roles()->detach();
            $permission->delete();
        }

        return $permissions->count();
    }

    protected function getActionsForResource(string $resourceType): array
    {
        return config("generated_permissions.{$resourceType}", [
            'view' => [
                'label' => 'View',
                'description' => 'View this resource',
            ],
            'update' => [
                'label' => 'Edit',
                'description' => 'Edit this resource',
            ],
            'delete' => [
                'label' => 'Delete',
                'description' => 'Delete this resource',
            ],
        ]);
    }

    protected function ensureCreatorRole(string $resourceType, ?int $organizationId): \Spatie\Permission\Models\Role
    {
        $roleName = "{$resourceType}_owner";

        $role = \Spatie\Permission\Models\Role::firstOrCreate([
            'name' => $roleName,
            'guard_name' => 'web',
        ]);

        return $role;
    }
}
```

## Contextual Permissions

Contextual permissions are permissions that are only active under specific conditions — time of day, IP address, device type, geographic location, or specific request context. They extend the base permission system with situational access control.

### Contextual Permission Model

```php
// app/Models/ContextualPermission.php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class ContextualPermission extends Model
{
    protected $fillable = [
        'user_id',
        'permission',
        'context_type',
        'context_config',
        'is_active',
        'created_by',
        'organization_id',
        'expires_at',
    ];

    protected $casts = [
        'is_active' => 'boolean',
        'context_config' => 'array',
        'expires_at' => 'datetime',
    ];

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function createdBy()
    {
        return $this->belongsTo(User::class, 'created_by');
    }

    public function isActiveNow(): bool
    {
        if (! $this->is_active) {
            return false;
        }

        if ($this->expires_at && $this->expires_at->isPast()) {
            return false;
        }

        return $this->evaluateContext();
    }

    protected function evaluateContext(): bool
    {
        return match ($this->context_type) {
            'time_range' => $this->evaluateTimeRange(),
            'ip_whitelist' => $this->evaluateIpWhitelist(),
            'device_type' => $this->evaluateDeviceType(),
            'geo_location' => $this->evaluateGeoLocation(),
            'header_match' => $this->evaluateHeaderMatch(),
            'composite' => $this->evaluateComposite(),
            default => true,
        };
    }

    protected function evaluateTimeRange(): bool
    {
        $config = $this->context_config;
        $now = now();
        $days = $config['days'] ?? null;

        if ($days && ! in_array($now->isoDayOfWeek, $days)) {
            return false;
        }

        if (isset($config['start_time']) && $now->format('H:i') < $config['start_time']) {
            return false;
        }

        if (isset($config['end_time']) && $now->format('H:i') > $config['end_time']) {
            return false;
        }

        return true;
    }

    protected function evaluateIpWhitelist(): bool
    {
        $allowedIps = $this->context_config['ips'] ?? [];
        $currentIp = request()->ip();

        foreach ($allowedIps as $allowedIp) {
            if (str_contains($allowedIp, '/')) {
                if ($this->ipInCidr($currentIp, $allowedIp)) {
                    return true;
                }
            } elseif ($currentIp === $allowedIp) {
                return true;
            }
        }

        return false;
    }

    protected function evaluateDeviceType(): bool
    {
        $allowedTypes = $this->context_config['types'] ?? ['desktop'];
        $userAgent = request()->userAgent();

        $isMobile = preg_match('/Mobile|Android|iPhone/i', $userAgent);
        $isTablet = preg_match('/iPad|Tablet/i', $userAgent);

        $currentType = match (true) {
            $isMobile => 'mobile',
            $isTablet => 'tablet',
            default => 'desktop',
        };

        return in_array($currentType, $allowedTypes);
    }

    protected function evaluateGeoLocation(): bool
    {
        $allowedCountries = $this->context_config['countries'] ?? [];
        $currentCountry = request()->header('CF-IPCountry') ?? null;

        if (! $currentCountry) {
            return false;
        }

        return in_array($currentCountry, $allowedCountries);
    }

    protected function evaluateHeaderMatch(): bool
    {
        $headers = $this->context_config['headers'] ?? [];

        foreach ($headers as $key => $expectedValue) {
            $actualValue = request()->header($key);
            if ($actualValue !== $expectedValue) {
                return false;
            }
        }

        return true;
    }

    protected function evaluateComposite(): bool
    {
        $operator = $this->context_config['operator'] ?? 'and';
        $conditions = $this->context_config['conditions'] ?? [];

        $results = [];
        foreach ($conditions as $condition) {
            $cp = new self([
                'context_type' => $condition['type'],
                'context_config' => $condition['config'],
                'is_active' => true,
            ]);
            $results[] = $cp->isActiveNow();
        }

        if ($operator === 'and') {
            return ! in_array(false, $results);
        } else {
            return in_array(true, $results);
        }
    }

    protected function ipInCidr(string $ip, string $cidr): bool
    {
        [$subnet, $mask] = explode('/', $cidr);
        return (ip2long($ip) & ~((1 << (32 - $mask)) - 1)) === ip2long($subnet);
    }
}
```

### Contextual Permission Service

```php
// app/Services/ContextualPermissionService.php

namespace App\Services;

use App\Models\ContextualPermission;
use App\Models\User;

class ContextualPermissionService
{
    public function grant(
        User $user,
        string $permission,
        string $contextType,
        array $contextConfig,
        ?User $grantedBy = null,
        ?\Carbon\Carbon $expiresAt = null,
    ): ContextualPermission {
        return ContextualPermission::create([
            'user_id' => $user->id,
            'permission' => $permission,
            'context_type' => $contextType,
            'context_config' => $contextConfig,
            'is_active' => true,
            'created_by' => $grantedBy?->id,
            'expires_at' => $expiresAt,
        ]);
    }

    public function hasPermission(User $user, string $permission): bool
    {
        $contextualPerms = ContextualPermission::where('user_id', $user->id)
            ->where('permission', $permission)
            ->where('is_active', true)
            ->where(function ($q) {
                $q->whereNull('expires_at')
                    ->orWhere('expires_at', '>', now());
            })
            ->get();

        foreach ($contextualPerms as $cp) {
            if ($cp->isActiveNow()) {
                return true;
            }
        }

        return false;
    }

    public function getAllActivePermissions(User $user): array
    {
        $contextualPerms = ContextualPermission::where('user_id', $user->id)
            ->where('is_active', true)
            ->where(function ($q) {
                $q->whereNull('expires_at')
                    ->orWhere('expires_at', '>', now());
            })
            ->get();

        $activePermissions = [];

        foreach ($contextualPerms as $cp) {
            if ($cp->isActiveNow()) {
                $activePermissions[] = $cp->permission;
            }
        }

        return array_unique($activePermissions);
    }

    public function revoke(ContextualPermission $permission, ?User $revokedBy = null): void
    {
        $permission->update(['is_active' => false]);

        activity()
            ->performedOn($permission)
            ->causedBy($revokedBy)
            ->event('contextual_permission_revoked')
            ->log('Contextual permission revoked');
    }
}
```

```php
// app/Services/PermissionTemplateService.php

namespace App\Services;

use App\Models\PermissionTemplate;
use App\Models\User;

use InvalidArgumentException;

class PermissionTemplateService
{
    public function create(
        string $name,

        ?string $category = null,

        ?int $organizationId = null,

        foreach ($permissionNames as $permName) {
            $exists = \Spatie\Permission\Models\Permission::where('name', $permName)->exists()
                || \App\Models\DynamicPermission::where('name', $permName)->active()->exists();

            if (! $exists) {
                throw new InvalidArgumentException("Permission '{$permName}' does not exist");
            }
        }

        $template = PermissionTemplate::create([
            'name' => $name,
            'description' => $description,
            'slug' => \Str::slug($name),
            'category' => $category,
            'permissions' => $permissionNames,
            'is_system' => false,
            'is_active' => true,
            'organization_id' => $organizationId,
            'created_by' => $createdBy?->id,
            'version' => 1,
        ]);

        if ($createdBy) {
            activity()
                ->performedOn($template)
                ->causedBy($createdBy)
                ->event('permission_template_created')
                ->log('Permission template created');
        }

        return $template;
    }

    public function applyToUser(
        PermissionTemplate $template,
        User $user,
        ?User $appliedBy = null,
    ): void {
        foreach ($template->permissions as $permName) {
            $user->givePermissionTo($permName);
        }

        $template->increment('usage_count');

        activity()
            ->performedOn($template)
            ->causedBy($appliedBy)
            ->withProperties([
                'user_id' => $user->id,
                'permissions' => $template->permissions,
            ])
            ->event('permission_template_applied')
            ->log('Permission template applied to user');
    }

    public function applyToRole(
        PermissionTemplate $template,
        Role $role,
        ?User $appliedBy = null,
    ): void {
        foreach ($template->permissions as $permName) {
            $role->givePermissionTo($permName);
        }

        $template->increment('usage_count');

        activity()
            ->performedOn($template)
            ->causedBy($appliedBy)
            ->withProperties([
                'role_id' => $role->id,
                'permissions' => $template->permissions,
            ])
            ->event('permission_template_applied')
            ->log('Permission template applied to role');
    }

    public function bulkApply(
        PermissionTemplate $template,
        array $userIds,
        ?User $appliedBy = null,
    ): array {
        $results = ['applied' => 0, 'failed' => 0, 'errors' => []];

        foreach ($userIds as $userId) {
            try {
                $user = User::findOrFail($userId);
                $this->applyToUser($template, $user, $appliedBy);
                $results['applied']++;
            } catch (\Exception $e) {
                $results['failed']++;
                $results['errors'][] = [
                    'user_id' => $userId,
                    'error' => $e->getMessage(),
                ];
            }
        }

        return $results;
    }

    public function getTemplatesForCategory(string $category, ?int $organizationId = null): array
    {
        return PermissionTemplate::active()
            ->forCategory($category)
            ->where(function ($q) use ($organizationId) {
                $q->whereNull('organization_id')
                    ->orWhere('organization_id', $organizationId);
            })
            ->orderBy('usage_count', 'desc')
            ->get()
            ->toArray();
    }

    public function getPopularTemplates(int $limit = 10): array
    {
        return PermissionTemplate::active()
            ->orderBy('usage_count', 'desc')
            ->limit($limit)
            ->get()
            ->toArray();
    }
```

### Bulk Permission Service

```php
// app/Services/BulkPermissionService.php

namespace App\Services;

use App\Models\User;

use App\Models\DynamicPermission;
use Illuminate\Support\Facades\DB;

class BulkPermissionService

    /**
     * Grant a permission to multiple users at once.
     */
    public function bulkGrantToUsers(
        string $permissionName,
        array $userIds,
        ?User $grantedBy = null,
    ): array {
        $results = ['success' => 0, 'failed' => 0, 'errors' => []];

        DB::beginTransaction();

        try {
            foreach ($userIds as $userId) {
                try {
                    $user = User::findOrFail($userId);

                        $user->givePermissionTo($permissionName);
                    }

                    $results['success']++;
                } catch (\Exception $e) {
                    $results['failed']++;
                    $results['errors'][] = [
                        'user_id' => $userId,
                        'error' => $e->getMessage(),
                    ];
                }
            }

            DB::commit();
        } catch (\Exception $e) {
            DB::rollBack();
            throw $e;
        }

        if ($grantedBy) {
            activity()
                ->causedBy($grantedBy)
                ->withProperties([
                    'permission' => $permissionName,
                    'user_count' => $results['success'],
                    'failed_count' => $results['failed'],
                ])
                ->event('bulk_permission_granted')
                ->log("Bulk permission granted: {$permissionName} to {$results['success']} users");
        }

        return $results;
```

```php
// app/Jobs/BulkApplyPermissionTemplate.php

namespace App\Jobs;

use App\Models\PermissionTemplate;
use App\Models\User;
use App\Services\PermissionTemplateService;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;

class BulkApplyPermissionTemplate implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $timeout = 300;
    public int $tries = 3;

    public function __construct(
        public int $templateId,
        public array $userIds,
        public ?int $appliedBy,
    ) {}

    public function handle(PermissionTemplateService $service): void
    {
        $template = PermissionTemplate::findOrFail($this->templateId);

        $operationId = "bulk_{$this->templateId}_" . time();

        cache()->put("bulk_operation:{$operationId}", [
            'status' => 'running',
            'total' => count($this->userIds),
            'processed' => 0,
            'started_at' => now()->toIso8601String(),
        ], 3600);

        $results = ['applied' => 0, 'failed' => 0, 'errors' => []];

        foreach (array_chunk($this->userIds, 100) as $chunk) {
            foreach ($chunk as $userId) {
                try {
                    $user = User::findOrFail($userId);
                    $service->applyToUser($template, $user);
                    $results['applied']++;
                } catch (\Exception $e) {
                    $results['failed']++;
                    $results['errors'][] = [
                        'user_id' => $userId,
                        'error' => $e->getMessage(),
                    ];
                }
            }

            // Update progress
            cache()->put("bulk_operation:{$operationId}", [
                'status' => 'running',
                'total' => count($this->userIds),
                'processed' => $results['applied'] + $results['failed'],
                'started_at' => now()->subSeconds($this->timeout)->toIso8601String(),
            ], 3600);
        }

        cache()->put("bulk_operation:{$operationId}", [
            'status' => 'completed',
            'total' => count($this->userIds),
            'processed' => $results['applied'] + $results['failed'],
            'results' => $results,
            'completed_at' => now()->toIso8601String(),
        ], 3600);
    }
}
```

### Migration Service

```php
// app/Services/PermissionMigrationService.php

namespace App\Services;

use App\Models\PermissionMigration;
use App\Models\DynamicPermission;
use Illuminate\Support\Facades\DB;

class PermissionMigrationService
{
    /**
     * Create a rename migration (old → new).
     */
    public function createRenameMigration(
        string $oldPermission,
        string $newPermission,
        ?int $createdBy = null,
    ): PermissionMigration {
        $affectedUsers = $this->countUsersWithPermission($oldPermission);
        $affectedRoles = $this->countRolesWithPermission($oldPermission);

        return PermissionMigration::create([
            'name' => "rename:{$oldPermission}→{$newPermission}",
            'old_permission' => $oldPermission,
            'new_permission' => $newPermission,
            'migration_type' => 'rename',
            'status' => 'pending',
            'affected_users_count' => $affectedUsers,
            'affected_roles_count' => $affectedRoles,
            'created_by' => $createdBy,
        ]);
    }

    /**
     * Create a split migration (old → multiple new).
     */
    public function createSplitMigration(
        string $oldPermission,
        array $newPermissions,
        ?int $createdBy = null,
    ): PermissionMigration {
        return PermissionMigration::create([
            'name' => "split:{$oldPermission}→" . implode(',', $newPermissions),
            'old_permission' => $oldPermission,
            'new_permission' => implode(',', $newPermissions),
            'migration_type' => 'split',
            'status' => 'pending',
            'affected_users_count' => $this->countUsersWithPermission($oldPermission),
            'affected_roles_count' => $this->countRolesWithPermission($oldPermission),
            'created_by' => $createdBy,
        ]);
    }

    /**
     * Create a merge migration (multiple old → new).
     */
    public function createMergeMigration(
        array $oldPermissions,
        string $newPermission,
        ?int $createdBy = null,
    ): PermissionMigration {
        return PermissionMigration::create([
            'name' => "merge:" . implode(',', $oldPermissions) . "→{$newPermission}",
            'old_permission' => implode(',', $oldPermissions),
            'new_permission' => $newPermission,
            'migration_type' => 'merge',
            'status' => 'pending',
            'affected_users_count' => 0,
            'affected_roles_count' => 0,
            'created_by' => $createdBy,
        ]);
    }

    /**
     * Execute a migration.
     */
    public function execute(PermissionMigration $migration): void
    {
        $migration->update([
            'status' => 'in_progress',
            'started_at' => now(),
        ]);

        DB::beginTransaction();

        try {
            match ($migration->migration_type) {
                'rename' => $this->executeRename($migration),
                'split' => $this->executeSplit($migration),
                'merge' => $this->executeMerge($migration),
                'deprecated' => $this->executeDeprecation($migration),
            };

            $migration->update([
                'status' => 'completed',
                'completed_at' => now(),
            ]);

            DB::commit();
        } catch (\Exception $e) {
            DB::rollBack();

            $migration->update(['status' => 'failed']);
            throw $e;
        }
    }

    /**
     * Rollback a migration.
     */
    public function rollback(PermissionMigration $migration): void
    {
        DB::beginTransaction();

        try {
            match ($migration->migration_type) {
                'rename' => $this->rollbackRename($migration),
                'split' => $this->rollbackSplit($migration),
                'merge' => $this->rollbackMerge($migration),
                default => null,
            };

            $migration->update([
                'status' => 'rolled_back',
                'rolled_back_at' => now(),
            ]);

            DB::commit();
        } catch (\Exception $e) {
            DB::rollBack();
            throw $e;
        }
    }

    protected function executeRename(PermissionMigration $migration): void
    {
        // Update all users
        $users = $this->getUsersWithPermission($migration->old_permission);
        foreach ($users as $user) {
            $user->revokePermissionTo($migration->old_permission);
            $user->givePermissionTo($migration->new_permission);
        }

        // Update all roles
        $roles = $this->getRolesWithPermission($migration->old_permission);
        foreach ($roles as $role) {
            $role->revokePermissionTo($migration->old_permission);
            $role->givePermissionTo($migration->new_permission);
        }
    }

    protected function rollbackRename(PermissionMigration $migration): void
    {
        $users = $this->getUsersWithPermission($migration->new_permission);
        foreach ($users as $user) {
            $user->revokePermissionTo($migration->new_permission);
            $user->givePermissionTo($migration->old_permission);
        }

        $roles = $this->getRolesWithPermission($migration->new_permission);
        foreach ($roles as $role) {
            $role->revokePermissionTo($migration->new_permission);
            $role->givePermissionTo($migration->old_permission);
        }
    }

    protected function executeSplit(PermissionMigration $migration): void
    {
        $newPermissions = explode(',', $migration->new_permission);

        $users = $this->getUsersWithPermission($migration->old_permission);
        foreach ($users as $user) {
            $user->revokePermissionTo($migration->old_permission);
            foreach ($newPermissions as $newPerm) {
                $user->givePermissionTo(trim($newPerm));
            }
        }
    }

    protected function rollbackSplit(PermissionMigration $migration): void
    {
        $newPermissions = explode(',', $migration->new_permission);

        $users = $this->getUsersWithPermission($migration->old_permission);
        foreach ($users as $user) {
            foreach ($newPermissions as $newPerm) {
                $user->revokePermissionTo(trim($newPerm));
            }
            $user->givePermissionTo($migration->old_permission);
        }
    }

    protected function executeMerge(PermissionMigration $migration): void
    {
        $oldPermissions = explode(',', $migration->old_permission);

        foreach ($oldPermissions as $oldPerm) {
            $users = $this->getUsersWithPermission(trim($oldPerm));
            foreach ($users as $user) {
                $user->revokePermissionTo(trim($oldPerm));
                $user->givePermissionTo($migration->new_permission);
            }
        }
    }

    protected function rollbackMerge(PermissionMigration $migration): void
    {
        $oldPermissions = explode(',', $migration->old_permission);

        $users = $this->getUsersWithPermission($migration->new_permission);
        foreach ($users as $user) {
            $user->revokePermissionTo($migration->new_permission);
            foreach ($oldPermissions as $oldPerm) {
                $user->givePermissionTo(trim($oldPerm));
            }
        }
    }

    protected function executeDeprecation(PermissionMigration $migration): void
    {
        $users = $this->getUsersWithPermission($migration->old_permission);
        foreach ($users as $user) {
            $user->revokePermissionTo($migration->old_permission);
        }

        $roles = $this->getRolesWithPermission($migration->old_permission);
        foreach ($roles as $role) {
            $role->revokePermissionTo($migration->old_permission);
        }
    }

    protected function countUsersWithPermission(string $permission): int
    {
        return DB::table('model_has_permissions')
            ->where('permission_id', function ($q) use ($permission) {
                $q->select('id')->from('permissions')->where('name', $permission);
            })
            ->count();
    }

    protected function countRolesWithPermission(string $permission): int
    {
        return DB::table('role_has_permissions')
            ->where('permission_id', function ($q) use ($permission) {
                $q->select('id')->from('permissions')->where('name', $permission);
            })
            ->count();
    }

    protected function getUsersWithPermission(string $permission): \Illuminate\Support\Collection
    {
        return User::whereHas('permissions', function ($q) use ($permission) {
            $q->where('permissions.name', $permission);
        })->get();
    }

    protected function getRolesWithPermission(string $permission): \Illuminate\Support\Collection
    {
        return \Spatie\Permission\Models\Role::whereHas('permissions', function ($q) use ($permission) {
            $q->where('permissions.name', $permission);
        })->get();
    }
}
```

### Query Optimization Service

```php
// app/Services/PermissionQueryOptimizer.php

namespace App\Services;

use App\Models\DynamicPermission;
use App\Models\User;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class PermissionQueryOptimizer
{
    protected const CACHE_TTL = 600; // 10 minutes
    protected const BATCH_SIZE = 100;

    /**
     * Pre-warm the permission cache for all active users.
     */
    public function prewarmCache(): int
    {
        $userIdChunks = User::where('is_active', true)
            ->pluck('id')
            ->chunk(self::BATCH_SIZE);

        $count = 0;

        foreach ($userIdChunks as $chunk) {
            foreach ($chunk as $userId) {
                $user = User::find($userId);
                if ($user) {
                    $this->resolveAndCache($user);
                    $count++;
                }
            }
        }

        return $count;
    }

    /**
     * Resolve all permissions for a user and cache the result.
     */
    public function resolveAndCache(User $user): array
    {
        $cacheKey = "permissions:user:{$user->id}";

        return Cache::remember($cacheKey, self::CACHE_TTL, function () use ($user) {
            return $this->resolvePermissions($user);
        });
    }

    /**
     * Resolve permissions from multiple sources efficiently.
     */
    protected function resolvePermissions(User $user): array
    {
        $permissions = [];

        // 1. Static permissions via Spatie (cached internally)
        $staticPerms = $user->getAllPermissions()->pluck('name')->toArray();
        $permissions = array_merge($permissions, $staticPerms);

        // 2. Dynamic permissions (single query for all roles)
        $dynamicPerms = DB::table('dynamic_role_has_permissions')
            ->join('dynamic_permissions', 'dynamic_permissions.id', '=', 'dynamic_role_has_permissions.dynamic_permission_id')
            ->whereIn('dynamic_role_has_permissions.role_id', $user->roles->pluck('id'))
            ->where('dynamic_permissions.is_active', true)
            ->pluck('dynamic_permissions.name')
            ->toArray();

        $permissions = array_merge($permissions, $dynamicPerms);

        // 3. Direct dynamic permissions
        $directDynamic = DB::table('model_has_dynamic_permissions')
            ->join('dynamic_permissions', 'dynamic_permissions.id', '=', 'model_has_dynamic_permissions.dynamic_permission_id')
            ->where('model_has_dynamic_permissions.user_id', $user->id)
            ->where('dynamic_permissions.is_active', true)
            ->pluck('dynamic_permissions.name')
            ->toArray();

        $permissions = array_merge($permissions, $directDynamic);

        return array_unique($permissions);
    }

    /**
     * Invalidate cache for a specific user.
     */
    public function invalidateUser(User $user): void
    {
        Cache::forget("permissions:user:{$user->id}");
    }

    /**
     * Bulk invalidate cache for multiple users.
     */
    public function bulkInvalidate(array $userIds): void
    {
        $keys = array_map(fn ($id) => "permissions:user:{$id}", $userIds);
        Cache::delete($keys);
    }

    /**
     * Invalidate cache for all users in an organization.
     */
    public function invalidateOrganization(int $organizationId): int
    {
        $userIds = DB::table('organization_members')
            ->where('organization_id', $organizationId)
            ->pluck('user_id')
            ->toArray();

        $this->bulkInvalidate($userIds);

        return count($userIds);
    }

    /**
     * Get permission statistics for monitoring.
     */
    public function getStats(): array
    {
        return [
            'total_dynamic_permissions' => DynamicPermission::count(),
            'active_dynamic_permissions' => DynamicPermission::active()->count(),
            'total_permission_templates' => \App\Models\PermissionTemplate::count(),
            'total_contextual_permissions' => \App\Models\ContextualPermission::where('is_active', true)->count(),
            'permission_migrations_pending' => \App\Models\PermissionMigration::where('status', 'pending')->count(),
        ];
    }
}
```

### Scheduled Optimization

```php
// app/Console/Kernel.php

protected function schedule(Schedule $schedule): void
{
    // Pre-warm permission cache every 30 minutes
    $schedule->call(function () {
        $count = app(PermissionQueryOptimizer::class)->prewarmCache();
        \Log::info("Permission cache prewarmed for {$count} users");
    })->everyThirtyMinutes();

    // Clean up expired contextual permissions
    $schedule->call(function () {
        $deleted = \App\Models\ContextualPermission::where('expires_at', '<', now())
            ->where('is_active', false)
            ->delete();
        if ($deleted > 0) {
            \Log::info("Cleaned up {$deleted} expired contextual permissions");
        }
    })->daily();

    // Clean up old completed permission migrations
    $schedule->call(function () {
        $deleted = \App\Models\PermissionMigration::whereIn('status', ['completed', 'rolled_back'])
            ->where('completed_at', '<', now()->subDays(90))
            ->delete();
        if ($deleted > 0) {
            \Log::info("Archived {$deleted} old permission migrations");
        }
    })->weekly();
}
```

```php
// tests/Feature/DynamicPermissions/RuntimePermissionTest.php

use App\Models\User;
use App\Services\RuntimePermissionGenerator;
use App\Services\ContextualPermissionService;
use App\Services\PermissionTemplateService;
use Tests\TestCase;

class RuntimePermissionTest extends TestCase
{
    public function test_generate_project_permissions(): void
    {
        $generator = app(RuntimePermissionGenerator::class);
        $user = User::factory()->create();

        $permissions = $generator->generateForResource(
            resourceType: 'project',
            resourceSlug: 'my-app',
            resourceName: 'My App',
            createdBy: $user,
        );

        $this->assertNotEmpty($permissions);
        $this->assertStringStartsWith('project.my-app.', $permissions[0]->name);
    }

    public function test_contextual_permission_respects_time_range(): void
    {
        $service = app(ContextualPermissionService::class);
        $user = User::factory()->create();

        $service->grant(
            user: $user,
            permission: 'reports.generate',
            contextType: 'time_range',
            contextConfig: [
                'start_time' => '09:00',
                'end_time' => '17:00',
                'days' => [1, 2, 3, 4, 5],
            ],
        );

        // The permission is granted, but whether it's active depends on current time
        $this->assertDatabaseHas('contextual_permissions', [
            'user_id' => $user->id,
            'permission' => 'reports.generate',
        ]);
    }

    public function test_permission_template_apply(): void
    {
        $templateService = app(PermissionTemplateService::class);
        $user = User::factory()->create();

        // Create a role with some permissions
        $role = \Spatie\Permission\Models\Role::create(['name' => 'test_editor']);
        $role->givePermissionTo(['blog.create', 'blog.update']);

        $template = $templateService->create(
            name: 'Editor Plus',
            description: 'Editor permissions with documentation access',
            permissionNames: ['blog.create', 'blog.update', 'documentation.create'],
        );

        $templateService->applyToUser($template, $user);

        $this->assertTrue($user->hasPermissionTo('documentation.create'));
        $this->assertEquals(1, $template->fresh()->usage_count);
    }

    public function test_bulk_permission_grant(): void
    {
        $service = app(BulkPermissionService::class);
        $users = User::factory()->count(5)->create();

        $results = $service->bulkGrantToUsers(
            'reports.view',
            $users->pluck('id')->toArray(),
        );

        $this->assertEquals(5, $results['success']);
        $this->assertEquals(0, $results['failed']);

        foreach ($users as $user) {
            $this->assertTrue($user->fresh()->hasPermissionTo('reports.view'));
        }
    }

    public function test_permission_rename_migration(): void
    {
        $migrationService = app(PermissionMigrationService::class);
        $user = User::factory()->create();
        $user->givePermissionTo('old.permission');

        $migration = $migrationService->createRenameMigration(
            oldPermission: 'old.permission',
            newPermission: 'new.permission',
        );

        $migrationService->execute($migration);

        $this->assertTrue($user->fresh()->hasPermissionTo('new.permission'));
        $this->assertFalse($user->fresh()->hasPermissionTo('old.permission'));
    }

    public function test_permission_migration_rollback(): void
    {
        $migrationService = app(PermissionMigrationService::class);
        $user = User::factory()->create();
        $user->givePermissionTo('old.permission');

        $migration = $migrationService->createRenameMigration(
            oldPermission: 'old.permission',
            newPermission: 'new.permission',
        );

        $migrationService->execute($migration);
        $this->assertTrue($user->fresh()->hasPermissionTo('new.permission'));

        $migrationService->rollback($migration->fresh());
        $this->assertTrue($user->fresh()->hasPermissionTo('old.permission'));
        $this->assertFalse($user->fresh()->hasPermissionTo('new.permission'));
    }
}
```
