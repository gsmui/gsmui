# 10.19 — MCP Permissions

**Project:** GsmUi
**Domain:** Authorization & RBAC
**Version:** 1.0.0

## Purpose

This document defines authorization rules for the **MCP (Model Context Protocol) server domain**. MCP enables AI-powered tool execution for theme generation, component building, and other automated workflows. MCP permissions govern server configuration, tool execution, log access, and API key management.

MCP authorization must balance accessibility (enabling AI-assisted workflows) with security (sandboxing tool execution, preventing unauthorized API key access, and auditing all tool invocations). MCP tools can execute arbitrary code, so authorization is critical.

## Scope

MCP permissions govern:

- Viewing MCP server configurations
- Configuring MCP servers (own servers)
- Executing MCP tools
- Managing all MCP servers (admin)
- Viewing MCP execution logs
- Managing MCP API keys
- Sandbox security for tool execution
- Rate limiting per user/server

## MCP Policy

```php
<?php

namespace App\Policies;

use App\Models\User;
use App\Models\McpServer;
use App\Models\McpExecutionLog;
use Illuminate\Auth\Access\HandlesAuthorization;

class McpPolicy
{
    use HandlesAuthorization;

    public function viewAny(User $user): bool
    {
        return $user->hasRole(['pro_user', 'creator', 'admin']);
    }

    public function view(User $user, McpServer $server): bool
    {
        if ($user->hasRole('admin')) {
            return true;
        }

        return $user->id === $server->user_id
            && $user->hasRole(['pro_user', 'creator']);
    }

    public function create(User $user): bool
    {
        return $user->hasRole(['pro_user', 'creator', 'admin']);
    }

    public function configure(User $user, McpServer $server): bool
    {
        if ($user->hasRole('admin')) {
            return true;
        }

        return $user->id === $server->user_id;
    }

    public function execute(User $user, McpServer $server): bool
    {
        if (! $server->is_active) {
            return false;
        }

        if ($user->hasRole('admin')) {
            return true;
        }

        return $user->id === $server->user_id
            && $user->hasRole(['pro_user', 'creator']);
    }

    public function delete(User $user, McpServer $server): bool
    {
        if ($user->hasRole('admin')) {
            return true;
        }

        return $user->id === $server->user_id
            && $user->hasRole(['pro_user', 'creator']);
    }

    public function viewLogs(User $user, McpServer $server): bool
    {
        if ($user->hasRole('admin')) {
            return true;
        }

        return $user->id === $server->user_id;
    }

    public function manageApiKeys(User $user): bool
    {
        return $user->hasRole('admin');
    }
}
```

## MCP Execution Authorization Flow

```text
User requests MCP tool execution
    ↓
Check: mcp.execute permission
    ↓
Check: MCP server is active
    ↓
Check: User owns server (or is admin)
    ↓
Check: Rate limit (per user/server)
    ↓
Validate input parameters
    ↓
Allocate sandbox container
    ↓
Execute tool in isolated environment
    ↓
Capture output
    ↓
Release container
    ↓
Log execution (user, server, tool, duration, status)
    ↓
Return result to user
```

### Execution Policy

```php
class McpExecutionPolicy
{
    public function execute(User $user, string $toolName, array $params): bool
    {
        $server = $this->resolveServer($toolName);

        if (! $server) {
            return false;
        }

        // Check server-level permission
        if (! $user->can('execute', $server)) {
            return false;
        }

        // Check rate limit
        if ($this->rateLimiter->tooManyRequests($user, $server)) {
            return false;
        }

        // Check tool-specific permissions
        return $this->toolPermissions->isAllowed($user, $toolName);
    }
}
```

## MCP Server Configuration

```php
class McpServerController extends Controller
{
    public function store(StoreMcpServerRequest $request): JsonResponse
    {
        $this->authorize('create', McpServer::class);

        $server = McpServer::create([
            'user_id' => $request->user()->id,
            'name' => $request->name,
            'type' => $request->type,
            'endpoint' => $request->endpoint,
            'config' => $request->config,
            'api_key' => encrypt($request->api_key),
            'is_active' => true,
        ]);

        activity()
            ->performedOn($server)
            ->log('mcp.server.created');

        return response()->json($server, 201);
    }

    public function update(UpdateMcpServerRequest $request, McpServer $server): JsonResponse
    {
        $this->authorize('configure', $server);

        $server->update($request->validated());

        activity()
            ->performedOn($server)
            ->log('mcp.server.updated');

        return response()->json($server);
    }

    public function execute(ExecuteMcpToolRequest $request, McpServer $server): JsonResponse
    {
        $this->authorize('execute', $server);

        $result = $this->mcpService->execute(
            $server,
            $request->tool,
            $request->params
        );

        activity()
            ->performedOn($server)
            ->withProperties([
                'tool' => $request->tool,
                'duration_ms' => $result->duration_ms,
                'status' => $result->status,
            ])
            ->log('mcp.tool.executed');

        return response()->json($result);
    }
}
```

### Rate Limiter Implementation

```php
class McpRateLimiter
{
    private const LIMITS = [
        'pro_user' => ['per_minute' => 10, 'concurrent' => 2, 'daily' => 200],
        'creator' => ['per_minute' => 20, 'concurrent' => 5, 'daily' => 500],
        'admin' => ['per_minute' => 50, 'concurrent' => 10, 'daily' => -1],
    ];

    public function tooManyRequests(User $user, McpServer $server): bool
    {
        $limits = $this->getLimits($user);

        // Check per-minute rate
        $minuteKey = "mcp:rate:minute:{$user->id}:{$server->id}";
        if (Cache::get($minuteKey, 0) >= $limits['per_minute']) {
            return true;
        }

        // Check concurrent executions
        $concurrentKey = "mcp:concurrent:{$user->id}";
        if (Cache::get($concurrentKey, 0) >= $limits['concurrent']) {
            return true;
        }

        // Check daily limit
        if ($limits['daily'] !== -1) {
            $dailyKey = "mcp:rate:daily:{$user->id}";
            if (Cache::get($dailyKey, 0) >= $limits['daily']) {
                return true;
            }
        }

        return false;
    }

    public function recordExecution(User $user, McpServer $server): void
    {
        $minuteKey = "mcp:rate:minute:{$user->id}:{$server->id}";
        $concurrentKey = "mcp:concurrent:{$user->id}";
        $dailyKey = "mcp:rate:daily:{$user->id}";

        Cache::increment($minuteKey);
        Cache::increment($concurrentKey);
        Cache::increment($dailyKey);

        // Auto-decrement concurrent after execution completes
        Cache::decrement($concurrentKey, 300); // 5-minute TTL
    }
}
```

```php
class McpServerConfigService
{
    public function updateConfig(User $user, McpServer $server, array $config): void
    {
        if (! $user->can('configure', $server)) {
            throw new AuthorizationException('Cannot configure this server');
        }

        $validated = $this->validateConfig($config, $server->type);

        if (! $validated) {
            throw new ValidationException('Invalid configuration');
        }

        $server->update([
            'config' => $config,
            'last_configured_at' => now(),
        ]);

        activity()
            ->performedOn($server)
            ->withProperties([
                'config_keys' => array_keys($config),
            ])
            ->event('mcp.server.configured')
            ->log('MCP server configuration updated');

        // Clear execution cache
        Cache::forget("mcp:server:{$server->id}:config");
    }
}
```

### Execution Analytics

```php
class McpAnalyticsService
{
    public function getToolUsageStats(int $days = 30): Collection
    {
        return DB::table('mcp_execution_logs')
            ->where('executed_at', '>=', now()->subDays($days))
            ->groupBy('tool_name')
            ->select([
                'tool_name',
                DB::raw('COUNT(*) as total_executions'),
                DB::raw('SUM(CASE WHEN status = "success" THEN 1 ELSE 0 END) as success_count'),
                DB::raw('SUM(CASE WHEN status = "error" THEN 1 ELSE 0 END) as error_count'),
                DB::raw('AVG(duration_ms) as avg_duration_ms'),
                DB::raw('AVG(output_size_bytes) as avg_output_size'),
            ])
            ->get();
    }

    public function getUserUsageStats(User $user, int $days = 30): array
    {
        $logs = McpExecutionLog::where('user_id', $user->id)
            ->where('executed_at', '>=', now()->subDays($days))
            ->get();

        return [
            'total_executions' => $logs->count(),
            'success_rate' => $logs->where('status', 'success')->count() / max($logs->count(), 1) * 100,
            'avg_duration_ms' => $logs->avg('duration_ms'),
            'total_duration_ms' => $logs->sum('duration_ms'),
            'tools_used' => $logs->pluck('tool_name')->unique()->count(),
            'most_used_tool' => $logs->groupBy('tool_name')
                ->map(fn($group) => $group->count())
                ->sortDesc()
                ->keys()
                ->first(),
        ];
    }

    public function getErrorRateByTool(int $days = 30): Collection
    {
        return DB::table('mcp_execution_logs')
            ->where('executed_at', '>=', now()->subDays($days))
            ->groupBy('tool_name')
            ->select([
                'tool_name',
                DB::raw('COUNT(*) as total'),
                DB::raw('SUM(CASE WHEN status IN ("error", "timeout", "killed") THEN 1 ELSE 0 END) as errors'),
                DB::raw('ROUND(SUM(CASE WHEN status IN ("error", "timeout", "killed") THEN 1 ELSE 0 END) / COUNT(*) * 100, 2) as error_rate'),
            ])
            ->get();
    }
}
```

### ACL Resolution

```php
class McpAclResolver
{
    public function resolve(User $user, McpServer $server, string $permission): bool
    {
        // Check role-based permission first
        if ($user->can($permission, $server)) {
            return true;
        }

        // Check explicit ACL grant
        $acl = McpAcl::where('server_id', $server->id)
            ->where('user_id', $user->id)
            ->where('permission', $permission)
            ->where(function ($q) {
                $q->whereNull('expires_at')
                  ->orWhere('expires_at', '>', now());
            })
            ->first();

        return $acl !== null;
    }

    public function grant(User $grantor, User $grantee, McpServer $server, string $permission, ?Carbon $expiresAt = null): void
    {
        if (! $grantor->can('manage', $server)) {
            throw new AuthorizationException('Cannot grant MCP permissions');
        }

        McpAcl::updateOrCreate(
            [
                'server_id' => $server->id,
                'user_id' => $grantee->id,
                'permission' => $permission,
            ],
            [
                'granted_by' => $grantor->id,
                'expires_at' => $expiresAt,
            ]
        );

        activity()
            ->performedOn($server)
            ->withProperties([
                'grantee_id' => $grantee->id,
                'permission' => $permission,
                'expires_at' => $expiresAt?->toISOString(),
            ])
            ->event('mcp.acl.granted')
            ->log('MCP permission granted');
    }

    public function revoke(User $revoker, User $grantee, McpServer $server, string $permission): void
    {
        if (! $revoker->can('manage', $server)) {
            throw new AuthorizationException('Cannot revoke MCP permissions');
        }

        McpAcl::where('server_id', $server->id)
            ->where('user_id', $grantee->id)
            ->where('permission', $permission)
            ->delete();

        activity()
            ->performedOn($server)
            ->withProperties([
                'grantee_id' => $grantee->id,
                'permission' => $permission,
            ])
            ->event('mcp.acl.revoked')
            ->log('MCP permission revoked');
    }
}
```

|---|---|---|

| Average response time | < 10 seconds | Yes |

### Quality Gate Service

```php
class McpQualityGate
{
    public function passAllGates(): array
    {
        return [
            'reliability' => $this->checkReliability(),
            'response_time' => $this->checkResponseTime(),
            'error_rate' => $this->checkErrorRate(),
            'uptime' => $this->checkUptime(),
            'security' => $this->checkSecurity(),
            'rate_limiting' => $this->checkRateLimiting(),
            'sandbox' => $this->checkSandbox(),
        ];
    }

    public function canOperate(): bool
    {
        $gates = $this->passAllGates();

        return ! in_array(false, $gates);
    }

    private function checkReliability(): bool
    {
        $successRate = DB::table('mcp_execution_logs')
            ->where('executed_at', '>=', now()->subDays(7))
            ->selectRaw('SUM(CASE WHEN status = "success" THEN 1 ELSE 0 END) / COUNT(*) * 100 as success_rate')
            ->value('success_rate');

        return $successRate >= 99;
    }

    private function checkResponseTime(): bool
    {
        $avgTime = DB::table('mcp_execution_logs')
            ->where('executed_at', '>=', now()->subDays(7))
            ->avg('duration_ms');

        return $avgTime <= 10000;
    }
}
```
