# 10.23 - Feature Flags

**Project:** GsmUi
**Domain:** Authorization & RBAC
**Version:** 1.0.0

---

## Purpose

This document defines the **feature flag system** that controls feature visibility and access based on subscription tier, role, or experimental status. Feature flags work alongside RBAC (see [10.02-rbac-architecture.md](./10.02-rbac-architecture.md)) but are not replacements for it. RBAC answers *"Can this user do this?"* while feature flags answer *"Is this feature available to this user?"*

Feature flags enable gradual rollouts, A/B testing, subscription-gated features, and experimental access without code deployments.

### Feature Flag Service

```php
// app/Services/FeatureFlagService.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Config;

class FeatureFlagService
{
    protected const CACHE_TTL = 300; // 5 minutes

    public function __construct(
        protected FeatureFlagResolver $resolver,
    ) {}

    /**
     * Check if a feature is enabled for a user.
     */
    public function isEnabled(User $user, string $flag): bool
    {
        $cacheKey = "feature_flag:{$user->id}:{$flag}";

        return Cache::remember($cacheKey, self::CACHE_TTL, function () use ($user, $flag) {
            return $this->resolver->resolve($user, $flag);
        });
    }

    /**
     * Check if a feature is enabled (uses auth()->user() implicitly).
     */
    public function active(string $flag): bool
    {
        $user = auth()->user();

        if (! $user) {
            return $this->isEnabledForGuest($flag);
        }

        return $this->isEnabled($user, $flag);
    }

    /**
     * Get all enabled flags for a user.
     */
    public function getEnabledFlags(User $user): array
    {
        $flags = Config::get('features', []);

        $enabled = [];

        foreach (array_keys($flags) as $flag) {
            if ($this->isEnabled($user, $flag)) {
                $enabled[] = $flag;
            }
        }

        return $enabled;
    }

    /**
     * Check if a feature is enabled for unauthenticated users.
     */
    protected function isEnabledForGuest(string $flag): bool
    {
        $config = Config::get("features.{$flag}");

        if (! $config) {
            return false;
        }

        // Only environment flags are available to guests
        if ($config['type'] === 'environment') {
            return in_array(app()->environment(), $config['environments']);
        }

        return false;
    }

    /**
     * Clear cache for a user's feature flags.
     */
    public function clearCacheForUser(User $user): void
    {
        $flags = Config::get('features', []);

        foreach (array_keys($flags) as $flag) {
            Cache::forget("feature_flag:{$user->id}:{$flag}");
        }
    }

    /**
     * Clear all feature flag caches.
     */
    public function clearAllCache(): void
    {
        Cache::tags(['feature_flags'])->flush();
    }
}
```

### Feature Flag Resolver

```php
// app/Services/FeatureFlagResolver.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\Config;

class FeatureFlagResolver
{
    public function resolve(User $user, string $flag): bool
    {
        $config = Config::get("features.{$flag}");

        if (! $config) {
            return false;
        }

        return match ($config['type']) {
            'subscription' => $this->resolveSubscription($user, $config),
            'role' => $this->resolveRole($user, $config),
            'percentage' => $this->resolvePercentage($user, $flag, $config),
            'environment' => $this->resolveEnvironment($config),
            'custom' => $this->resolveCustom($user, $config),
            default => false,
        };
    }

    protected function resolveSubscription(User $user, array $config): bool
    {
        $plan = $user->subscription?->plan;

        if (! $plan) {
            return false;
        }

        return in_array($plan, $config['plans']);
    }

    protected function resolveRole(User $user, array $config): bool
    {
        return $user->hasAnyRole($config['roles']);
    }

    protected function resolvePercentage(User $user, string $flag, array $config): bool
    {
        // Deterministic: same user always gets same result
        $hash = hash('sha256', "{$flag}:{$user->id}");
        $bucket = hexdec(substr($hash, 0, 8)) % 100;

        return $bucket < $config['percentage'];
    }

    protected function resolveEnvironment(array $config): bool
    {
        return in_array(app()->environment(), $config['environments']);
    }

    protected function resolveCustom(User $user, array $config): bool
    {
        if (isset($config['callback']) && is_callable($config['callback'])) {
            return call_user_func($config['callback'], $user);
        }

        return false;
    }
}
```

## Integration with RBAC

Feature flags are checked **AFTER** RBAC. Both checks must pass for feature access.

### Authorization Flow

```text
User requests feature
    ↓
RBAC Check: Does user have required permission?
    ↓ (if denied)
Return 403 Forbidden
    ↓ (if passed)
Feature Flag Check: Is feature enabled for this user?
    ↓ (if disabled)
Return 404 Not Found or Feature Gated Page
    ↓ (if enabled)
Grant Access
```

### Combined Check Service

```php
// app/Services/FeatureAccessService.php

namespace App\Services;

use App\Models\User;
use Illuminate\Auth\Access\AuthorizationException;

class FeatureAccessService
{
    public function __construct(
        protected FeatureFlagService $featureFlags,
    ) {}

    /**
     * Check if a user can access a feature.
     * Combines RBAC and feature flag checks.
     */
    public function canAccess(
        User $user,
        string $permission,
        string $featureFlag,
    ): bool {
        // 1. RBAC check
        if (! $user->hasPermissionTo($permission)) {
            return false;
        }

        // 2. Feature flag check
        if (! $this->featureFlags->isEnabled($user, $featureFlag)) {
            return false;
        }

        return true;
    }

    /**
     * Authorize access or throw exception.
     */
    public function authorize(
        User $user,
        string $permission,
        string $featureFlag,
    ): void {
        if (! $this->canAccess($user, $permission, $featureFlag)) {
            // Determine which check failed
            if (! $user->hasPermissionTo($permission)) {
                throw new AuthorizationException(
                    'You do not have permission to access this feature.'
                );
            }

            throw new FeatureGatedException(
                'This feature is not available for your subscription plan.',
                $featureFlag,
            );
        }
    }

    /**
     * Check feature access in Blade/Livewire.
     */
    public function check(string $permission, string $featureFlag): bool
    {
        $user = auth()->user();

        if (! $user) {
            return false;
        }

        return $this->canAccess($user, $permission, $featureFlag);
    }
}
```

### Feature Gated Exception

```php
// app/Exceptions/FeatureGatedException.php

namespace App\Exceptions;

use Exception;

class FeatureGatedException extends Exception
{
    protected string $featureFlag;

    public function __construct(string $message, string $featureFlag)
    {
        parent::__construct($message);
        $this->featureFlag = $featureFlag;
    }

    public function getFeatureFlag(): string
    {
        return $this->featureFlag;
    }

    public function render($request)
    {
        if ($request->expectsJson()) {
            return response()->json([
                'message' => $this->getMessage(),
                'feature_flag' => $this->featureFlag,
                'upgrade_url' => route('billing.upgrade'),
            ], 403);
        }

        return redirect()->route('features.gated', [
            'feature' => $this->featureFlag,
        ]);
    }
}
```

### Feature Gate Middleware

```php
// app/Http/Middleware/FeatureGate.php

namespace App\Http\Middleware;

use App\Services\FeatureFlagService;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class FeatureGate
{
    public function __construct(
        protected FeatureFlagService $featureFlags,
    ) {}

    public function handle(Request $request, Closure $next, string $flag): Response
    {
        $user = $request->user();

        if (! $user || ! $this->featureFlags->isEnabled($user, $flag)) {
            if ($request->expectsJson()) {
                return response()->json([
                    'message' => 'This feature is not available.',
                    'feature_flag' => $flag,
                ], 403);
            }

            return redirect()->route('features.gated', ['feature' => $flag]);
        }

        return $next($request);
    }
}
```

## Admin UI for Feature Flags

### Database-Stored Flags

```php
// app/Models/FeatureFlag.php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Cache;

class FeatureFlag extends Model
{
    protected $fillable = [
        'name',
        'description',
        'is_enabled',
        'type',
        'config',
        'rollout_percentage',
        'allowed_plans',
        'allowed_roles',
        'environments',
    ];

    protected $casts = [
        'is_enabled' => 'boolean',
        'config' => 'array',
        'allowed_plans' => 'array',
        'allowed_roles' => 'array',
        'environments' => 'array',
    ];

    public function isEnabledForUser(?User $user): bool
    {
        if (! $this->is_enabled) {
            return false;
        }

        if (! $user) {
            return in_array(app()->environment(), $this->environments ?? []);
        }

        // Check plan restriction
        if ($this->allowed_plans && ! empty($this->allowed_plans)) {
            $plan = $user->subscription?->plan;
            if (! $plan || ! in_array($plan, $this->allowed_plans)) {
                return false;
            }
        }

        // Check role restriction
        if ($this->allowed_roles && ! empty($this->allowed_roles)) {
            if (! $user->hasAnyRole($this->allowed_roles)) {
                return false;
            }
        }

        // Check rollout percentage
        if ($this->rollout_percentage !== null && $this->rollout_percentage < 100) {
            $hash = hash('sha256', "{$this->name}:{$user->id}");
            $bucket = hexdec(substr($hash, 0, 8)) % 100;
            if ($bucket >= $this->rollout_percentage) {
                return false;
            }
        }

        return true;
    }

    public static function clearCache(): void
    {
        Cache::tags(['feature_flags'])->flush();
    }
}
```

### Boolean Flag Service

```php
// app/Services/BooleanFlagService.php

namespace App\Services;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Config;

class BooleanFlagService
{
    protected const CACHE_TTL = 300;

    /**
     * Check if a boolean flag is enabled.
     */
    public function isEnabled(string $flag): bool
    {
        $cacheKey = "bool_flag:{$flag}";

        return Cache::remember($cacheKey, self::CACHE_TTL, function () use ($flag) {
            $config = Config::get("features-boolean.{$flag}");

            if (! $config) {
                return false;
            }

            // Check database override
            $dbValue = $this->getDatabaseOverride($flag);
            if ($dbValue !== null) {
                return $dbValue;
            }

            // Check environment override
            $envKey = 'FEATURE_' . strtoupper(str_replace('.', '_', $flag));
            if (env($envKey) !== null) {
                return env($envKey) === 'true';
            }

            return $config['default'] ?? false;
        });
    }

    /**
     * Toggle a boolean flag.
     */
    public function toggle(string $flag, ?\App\Models\User $toggledBy = null): bool
    {
        $currentValue = $this->isEnabled($flag);
        $newValue = ! $currentValue;

        $this->setOverride($flag, $newValue);

        if ($toggledBy) {
            activity()
                ->withProperties([
                    'flag' => $flag,
                    'old_value' => $currentValue,
```

## Percentage-Based Feature Flags

Percentage flags enable gradual rollouts where a configurable percentage of users see the new feature. The allocation is deterministic — the same user always gets the same result for a given percentage.

### Percentage Flag Implementation

```php
// app/Services/PercentageFlagService.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\Cache;

class PercentageFlagService
{
    /**
     * Check if a user falls within the rollout percentage.
     * Uses deterministic hashing for consistency.
     */
    public function isUserInRollout(
        User $user,
        string $flag,
        int $percentage,
    ): bool {
        $cacheKey = "pct_flag:{$flag}:{$user->id}:{$percentage}";

        return Cache::remember($cacheKey, 3600, function () use ($user, $flag, $percentage) {
            $hash = hash('sha256', "{$flag}:{$user->id}");
            $bucket = hexdec(substr($hash, 0, 8)) % 100;

            return $bucket < $percentage;
        });
    }

    /**
     * Calculate the distribution of users across rollout buckets.
     */
    public function getDistribution(int $totalUsers, int $percentage): array
    {
        $inRollout = round($totalUsers * ($percentage / 100));
        $notInRollout = $totalUsers - $inRollout;

        return [
            'total' => $totalUsers,
            'in_rollout' => (int) $inRollout,
            'not_in_rollout' => (int) $notInRollout,
            'percentage' => $percentage,
            'actual_percentage' => $totalUsers > 0
                ? round(($inRollout / $totalUsers) * 100, 2)
                : 0,
        ];
    }

    /**
     * Gradually increase rollout percentage.
     */
    public function gradualRollout(
        string $flag,
        array $steps = [5, 10, 25, 50, 75, 100],
        int $stepDurationMinutes = 60,
    ): array {
        $currentStep = Cache::get("rollout_step:{$flag}", 0);
        $lastStepTime = Cache::get("rollout_step_time:{$flag}");

        if ($lastStepTime && now()->diffInMinutes($lastStepTime) < $stepDurationMinutes) {
            return [
                'status' => 'waiting',
                'current_percentage' => $steps[$currentStep] ?? 100,
                'next_step_at' => $lastStepTime->addMinutes($stepDurationMinutes),
            ];
        }

        $nextStep = $currentStep + 1;

        if ($nextStep >= count($steps)) {
            return [
                'status' => 'completed',
                'current_percentage' => 100,
            ];
        }

        $newPercentage = $steps[$nextStep];

        Cache::put("rollout_step:{$flag}", $nextStep, now()->addDays(7));
        Cache::put("rollout_step_time:{$flag}", now(), now()->addDays(7));

        activity()
            ->withProperties([
                'flag' => $flag,
                'old_percentage' => $steps[$currentStep] ?? 0,
                'new_percentage' => $newPercentage,
            ])
            ->event('rollout_percentage_changed')
            ->log("Rollout for '{$flag}' increased to {$newPercentage}%");

        return [
            'status' => 'advanced',
            'previous_percentage' => $steps[$currentStep] ?? 0,
            'current_percentage' => $newPercentage,
            'next_step_at' => now()->addMinutes($stepDurationMinutes),
        ];
    }
}
```

```php
// app/Models/TargetingRule.php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class TargetingRule extends Model
{
    protected $fillable = [
        'feature_flag_id',
        'name',
        'description',
        'priority',
        'is_active',
        'conditions',
        'variant', // control, treatment_a, treatment_b, etc.
        'percentage',
        'user_segment',
    ];

    protected $casts = [
        'is_active' => 'boolean',
        'conditions' => 'array',
        'percentage' => 'integer',
        'user_segment' => 'array',
        'priority' => 'integer',
    ];

    public function featureFlag()
    {
        return $this->belongsTo(FeatureFlag::class);
    }

    public function evaluate(\App\Models\User $user): bool
    {
        if (! $this->is_active) {
            return false;
        }

        // Check segment membership
        if ($this->user_segment) {
            if (! $this->matchesSegment($user)) {
                return false;
            }
        }

        // Check conditions
        if ($this->conditions) {
            foreach ($this->conditions as $condition) {
                if (! $this->evaluateCondition($condition, $user)) {
                    return false;
                }
            }
        }

        // Check percentage
        if ($this->percentage !== null && $this->percentage < 100) {
            $hash = hash('sha256', "{$this->feature_flag_id}:{$this->id}:{$user->id}");
            $bucket = hexdec(substr($hash, 0, 8)) % 100;
            if ($bucket >= $this->percentage) {
                return false;
            }
        }

        return true;
    }

    protected function matchesSegment(\App\Models\User $user): bool
    {
        $segment = $this->user_segment;

        return match ($segment['type'] ?? null) {
            'plan' => in_array($user->subscription?->plan, $segment['values'] ?? []),
            'role' => $user->hasAnyRole($segment['values'] ?? []),
            'created_after' => $user->created_at->gte($segment['date']),
            'created_before' => $user->created_at->lte($segment['date']),
            'id_in' => in_array($user->id, $segment['values'] ?? []),
            'id_not_in' => ! in_array($user->id, $segment['values'] ?? []),
            'email_domain' => in_array(
                strtolower(substr(strrchr($user->email, '@'), 1)),
                $segment['values'] ?? []
            ),
            default => true,
        };
    }

    protected function evaluateCondition(array $condition, \App\Models\User $user): bool
    {
        $attribute = $condition['attribute'] ?? '';
        $operator = $condition['operator'] ?? 'equals';
        $value = $condition['value'] ?? null;

        $actualValue = match ($attribute) {
            'plan' => $user->subscription?->plan,
            'role' => $user->roles->pluck('name')->toArray(),
            'email' => $user->email,
            'created_at' => $user->created_at,
            'id' => $user->id,
            default => $user->{$attribute} ?? null,
        };

        return match ($operator) {
            'equals' => $actualValue == $value,
            'not_equals' => $actualValue != $value,
            'in' => in_array($actualValue, (array) $value),
            'not_in' => ! in_array($actualValue, (array) $value),
            'gt' => $actualValue > $value,
            'gte' => $actualValue >= $value,
            'lt' => $actualValue < $value,
            'lte' => $actualValue <= $value,
            'contains' => str_contains((string) $actualValue, (string) $value),
            'starts_with' => str_starts_with((string) $actualValue, (string) $value),
            default => false,
        };
    }
}
```

```php
// app/Services/AbTestService.php

namespace App\Services;

use App\Models\AbTest;
use App\Models\AbTestMetric;
use App\Models\User;

class AbTestService
{
    public function create(
        string $name,
        string $description,
        array $variants,
        string $primaryMetric,
        ?array $secondaryMetrics = null,
        ?array $targetAudience = null,
    ): AbTest {
        $featureFlag = app(FeatureFlagService::class);

        return AbTest::create([
            'name' => $name,
            'description' => $description,
            'status' => 'draft',
            'variants' => $variants,
            'primary_metric' => $primaryMetric,
            'secondary_metrics' => $secondaryMetrics,
            'target_audience' => $targetAudience,
            'confidence_level' => 0.95,
        ]);
    }

    public function start(AbTest $test): void
    {
        $test->update([
            'status' => 'running',
            'started_at' => now(),
        ]);

        activity()
            ->performedOn($test)
            ->event('ab_test_started')
            ->log("A/B test '{$test->name}' started");
    }

    public function assignVariant(AbTest $test, User $user): string
    {
        $cacheKey = "ab_test:{$test->id}:{$user->id}";

        return cache()->remember($cacheKey, null, function () use ($test, $user) {
            $hash = hash('sha256', "{$test->id}:{$user->id}");
            $bucket = hexdec(substr($hash, 0, 8)) % 100;

            $cumulative = 0;
            foreach ($test->variants as $variant) {
                $cumulative += $variant['percentage'] ?? 50;
                if ($bucket < $cumulative) {
                    return $variant['name'];
                }
            }

            return $test->variants[0]['name'];
        });
    }

    public function recordMetric(
        AbTest $test,
        User $user,
        string $metricName,
        float $value,
    ): void {
        $variant = $this->assignVariant($test, $user);

        AbTestMetric::create([
            'ab_test_id' => $test->id,
            'user_id' => $user->id,
            'variant' => $variant,
            'metric_name' => $metricName,
            'metric_value' => $value,
        ]);

        // Update running totals in cache
        $key = "ab_metric:{$test->id}:{$variant}:{$metricName}";
        $stats = cache()->get($key, ['sum' => 0, 'count' => 0]);
        $stats['sum'] += $value;
        $stats['count']++;
        cache()->put($key, $stats);
    }

    public function getResults(AbTest $test): array
    {
        $results = [];

        foreach ($test->variants as $variant) {
            $metrics = AbTestMetric::where('ab_test_id', $test->id)
                ->where('variant', $variant['name'])
                ->get();

            $results[$variant['name']] = [
                'sample_size' => $metrics->count(),
                'primary_metric' => $this->calculateMetricStats(
                    $metrics->where('metric_name', $test->primary_metric)
                ),
                'secondary_metrics' => collect($test->secondary_metrics ?? [])
                    ->mapWithKeys(fn ($m) => [$m => $this->calculateMetricStats(
                        $metrics->where('metric_name', $m)
                    )])
                    ->toArray(),
            ];
        }

        return [
            'test' => $test->name,
            'status' => $test->status,
            'started_at' => $test->started_at,
            'variants' => $results,
            'winner' => $this->determineWinner($results, $test->primary_metric),
        ];
    }

    public function stop(AbTest $test, ?string $winningVariant = null): void
    {
        $results = $this->getResults($test);

        $winner = $winningVariant ?? $this->determineWinner($results, $test->primary_metric);

        $test->update([
            'status' => 'completed',
            'ended_at' => now(),
            'winning_variant' => $winner,
            'results' => $results,
        ]);

        activity()
            ->performedOn($test)
            ->withProperties([
                'winner' => $winner,
                'results' => $results,
            ])
            ->event('ab_test_completed')
            ->log("A/B test '{$test->name}' completed. Winner: {$winner}");
    }

    protected function calculateMetricStats($metrics): array
    {
        if ($metrics->isEmpty()) {
            return ['mean' => 0, 'count' => 0, 'sum' => 0, 'std_dev' => 0];
        }

        $values = $metrics->pluck('metric_value');
        $mean = $values->avg();
        $count = $values->count();
        $variance = $count > 1
            ? $values->map(fn ($v) => pow($v - $mean, 2))->sum() / ($count - 1)
            : 0;

        return [
            'mean' => round($mean, 4),
            'count' => $count,
            'sum' => round($values->sum(), 4),
            'std_dev' => round(sqrt($variance), 4),
        ];
    }

    protected function determineWinner(array $results, string $primaryMetric): ?string
    {
        $bestVariant = null;
        $bestMean = -INF;

        foreach ($results as $variant => $data) {
            $mean = $data['primary_metric']['mean'] ?? 0;
            if ($mean > $bestMean) {
                $bestMean = $mean;
                $bestVariant = $variant;
            }
        }

        return $bestVariant;
    }
}
```

## Gradual Rollout Strategies

Gradual rollouts reduce risk by progressively enabling features to larger user groups.

### Rollout Strategy Service

```php
// app/Services/GradualRolloutService.php

namespace App\Services;

use App\Models\FeatureFlag;
use App\Models\User;
use Illuminate\Support\Facades\Cache;

class GradualRolloutService
{
    /**
     * Canary rollout: Start with internal users, then expand.
     */
    public function canaryRollout(
        FeatureFlag $flag,
        User $user,
        array $config = [],
    ): bool {
        $stages = $config['stages'] ?? [
            ['type' => 'role', 'values' => ['admin', 'super_admin'], 'label' => 'Internal'],
            ['type' => 'plan', 'values' => ['enterprise'], 'label' => 'Enterprise'],
            ['type' => 'percentage', 'value' => 10, 'label'] => '10%'],
            ['type' => 'percentage', 'value' => 25, 'label'] => '25%'],
            ['type' => 'percentage', 'value' => 50, 'label'] => '50%'],
            ['type' => 'percentage', 'value' => 100, 'label'] => 'Everyone'],
        ];

        $currentStage = Cache::get("canary_stage:{$flag->id}", 0);

        if ($currentStage >= count($stages)) {
            return true;
        }

        $stage = $stages[$currentStage];

        return match ($stage['type']) {
            'role' => $user->hasAnyRole($stage['values']),
            'plan' => in_array($user->subscription?->plan, $stage['values']),
            'percentage' => $this->checkPercentage($user, $flag->id, $stage['value']),
            default => false,
        };
    }

    /**
     * Advance the canary rollout to the next stage.
     */
    public function advanceCanary(FeatureFlag $flag): array
    {
        $currentStage = Cache::get("canary_stage:{$flag->id}", 0);
        $nextStage = $currentStage + 1;

        Cache::put("canary_stage:{$flag->id}", $nextStage);

        activity()
            ->performedOn($flag)
            ->withProperties([
                'old_stage' => $currentStage,
                'new_stage' => $nextStage,
            ])
            ->event('canary_rollout_advanced')
            ->log("Canary rollout for '{$flag->name}' advanced to stage {$nextStage}");

        return [
            'previous_stage' => $currentStage,
            'current_stage' => $nextStage,
        ];
    }

    /**
     * Ring-based rollout: Expand by geographic or organizational rings.
     */
    public function ringRollout(
        FeatureFlag $flag,
        User $user,
        array $rings,
    ): bool {
        foreach ($rings as $ring) {
            if ($this->matchesRing($ring, $user)) {
                return true;
            }
        }

        return false;
    }

    protected function checkPercentage(User $user, int $flagId, int $percentage): bool
    {
        $hash = hash('sha256', "{$flagId}:{$user->id}");
        $bucket = hexdec(substr($hash, 0, 8)) % 100;

        return $bucket < $percentage;
    }

    protected function matchesRing(array $ring, User $user): bool
    {
        return match ($ring['type'] ?? null) {
            'organization' => $user->current_organization_id == $ring['id'],
            'region' => ($user->settings['region'] ?? null) === $ring['value'],
            'domain' => in_array(
                strtolower(substr(strrchr($user->email, '@'), 1)),
                $ring['domains'] ?? []
            ),
            default => false,
        };
    }
```

## Kill Switches

Kill switches allow instant disabling of features or entire systems in case of emergencies.

### Kill Switch Implementation

```php
// app/Services/KillSwitchService.php

namespace App\Services;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class KillSwitchService
{
    /**
     * Check if a system or feature is killed.
     */
    public function isKilled(string $system): bool
    {
        // Check Redis first for fastest response
        $killed = Cache::get("kill_switch:{$system}");

        if ($killed !== null) {
            return $killed;
        }

        // Check database
        $killed = DB::table('kill_switches')
            ->where('system', $system)
            ->where('is_active', true)
            ->where(function ($q) {
                $q->whereNull('expires_at')
                    ->orWhere('expires_at', '>', now());
            })
            ->exists();

        Cache::put("kill_switch:{$system}", $killed, 60);

        return $killed;
    }

    /**
     * Activate a kill switch.
     */
    public function activate(
        string $system,
        string $reason,
        ?int $activatedBy = null,
        ?\Carbon\Carbon $expiresAt = null,
    ): void {
        DB::table('kill_switches')->updateOrInsert(
            ['system' => $system, 'is_active' => true],
            [
                'reason' => $reason,
                'activated_by' => $activatedBy,
                'activated_at' => now(),
                'expires_at' => $expiresAt,
                'updated_at' => now(),
            ]
        );

        Cache::put("kill_switch:{$system}", true, 60);

        activity()
            ->withProperties([
                'system' => $system,
                'reason' => $reason,
                'expires_at' => $expiresAt?->toIso8601String(),
            ])
            ->event('kill_switch_activated')
            ->log("Kill switch activated for '{$system}': {$reason}");
    }

    /**
     * Deactivate a kill switch.
     */
    public function deactivate(string $system, ?int $deactivatedBy = null): void
    {
        DB::table('kill_switches')
            ->where('system', $system)
            ->where('is_active', true)
            ->update([
                'is_active' => false,
                'deactivated_by' => $deactivatedBy,
                'deactivated_at' => now(),
                'updated_at' => now(),
            ]);

        Cache::put("kill_switch:{$system}", false, 60);

        activity()
            ->withProperties([
                'system' => $system,
            ])
            ->event('kill_switch_deactivated')
            ->log("Kill switch deactivated for '{$system}'");
    }

    /**
     * Get all active kill switches.
     */
    public function getActiveKillSwitches(): array
    {
        return DB::table('kill_switches')
            ->where('is_active', true)
            ->where(function ($q) {
                $q->whereNull('expires_at')
                    ->orWhere('expires_at', '>', now());
            })
            ->get()
            ->toArray();
    }

    /**
     * Emergency: Kill all non-essential systems.
     */
    public function emergencyKillAll(?int $activatedBy = null): array
    {
        $systems = [
            'ai_features',
            'marketplace',
            'notifications',
            'email_service',
            'third_party_integrations',
            'playground',
            'blog',
            'documentation',
        ];

        $killed = [];

        foreach ($systems as $system) {
            if (! $this->isKilled($system)) {
                $this->activate($system, 'Emergency kill-all activated', $activatedBy);
                $killed[] = $system;
            }
        }

        return $killed;
    }
}
```

```php
namespace App\Http\Middleware;

use App\Services\KillSwitchService;
use Closure;
use Illuminate\Http\Request;

class KillSwitchCheck
{
    public function __construct(
        protected KillSwitchService $killSwitch,
    ) {}

    public function handle(Request $request, Closure $next, string $system)
    {
        if ($this->killSwitch->isKilled($system)) {
            if ($request->expectsJson()) {
                return response()->json([
                    'message' => 'This feature is temporarily unavailable.',
                    'system' => $system,
                ], 503);
            }

            return response()->view('errors.503', [
                'system' => $system,
            ], 503);
        }

        return $next($request);
    }
}
```

## Flag Lifecycle Management

Feature flags go through a lifecycle from creation to retirement, and proper lifecycle management prevents flag accumulation and technical debt.

### Flag Lifecycle States

```text
created
  ↓ (enabled)
active
  ↓ (gradual rollout)
rolling_out
  ↓ (fully enabled)
enabled
  ↓ (disabled)
disabled
  ↓ (scheduled for removal)
deprecated
  ↓ (removed from code)
archived
```

### Lifecycle Service

```php
// app/Services/FlagLifecycleService.php

namespace App\Services;

use App\Models\FeatureFlag;
use Carbon\Carbon;

class FlagLifecycleService
{
    /**
     * Check all flags for lifecycle violations.
     */
    public function auditFlags(): array
    {
        $flags = FeatureFlag::all();
        $issues = [];

        foreach ($flags as $flag) {
            $flagIssues = $this->auditFlag($flag);
            if (! empty($flagIssues)) {
                $issues[$flag->name] = $flagIssues;
            }
        }

        return [
            'total_flags' => $flags->count(),
            'flags_with_issues' => count($issues),
            'issues' => $issues,
        ];
    }

    protected function auditFlag(FeatureFlag $flag): array
    {
        $issues = [];
        $config = config("features.{$flag->name}");

        // Check for flags that have been active too long
        if ($flag->is_enabled && $flag->updated_at->diffInDays(now()) > 90) {
            $issues[] = [
                'type' => 'long_running',
                'message' => "Flag '{$flag->name}' has been enabled for over 90 days. Consider enabling permanently or removing.",
            ];
        }

        // Check for deprecated flags still in use
        if ($flag->deprecated_at && $flag->deprecated_at->diffInDays(now()) > 30) {
            $issues[] = [
                'type' => 'deprecated_stale',
                'message' => "Flag '{$flag->name}' was deprecated over 30 days ago. Remove from code.",
            ];
        }

        // Check for flags with no description
        if (empty($flag->description)) {
            $issues[] = [
                'type' => 'missing_description',
                'message' => "Flag '{$flag->name}' has no description.",
            ];
        }

        return $issues;
    }

    /**
     * Auto-deprecate flags that have been enabled for too long.
     */
    public function autoDeprecate(int $daysThreshold = 180): array
    {
        $flags = FeatureFlag::where('is_enabled', true)
            ->where('updated_at', '<', now()->subDays($daysThreshold))
            ->get();

        $deprecated = [];

        foreach ($flags as $flag) {
            $flag->update([
                'deprecated_at' => now(),
            ]);

            $deprecated[] = $flag->name;

            activity()
                ->performedOn($flag)
                ->event('flag_auto_deprecated')
                ->log("Flag '{$flag->name}' auto-deprecated after {$daysThreshold} days of being enabled");
        }

        return $deprecated;
    }

    /**
     * Archive deprecated flags.
     */
    public function archiveDeprecated(int $deprecationDays = 60): array
    {
        $flags = FeatureFlag::whereNotNull('deprecated_at')
            ->where('deprecated_at', '<', now()->subDays($deprecationDays))
            ->get();

        $archived = [];

        foreach ($flags as $flag) {
            $flag->update([
                'is_enabled' => false,
            ]);

            $archived[] = $flag->name;
        }

        return $archived;
    }

    /**
     * Generate a flag inventory report.
     */
    public function getInventory(): array
    {
        $flags = FeatureFlag::all();

        return [
            'total' => $flags->count(),
            'enabled' => $flags->where('is_enabled', true)->count(),
            'disabled' => $flags->where('is_enabled', false)->count(),
            'deprecated' => $flags->whereNotNull('deprecated_at')->count(),
            'by_type' => [
                'subscription' => $flags->filter(fn ($f) => ($f->type ?? '') === 'subscription')->count(),
                'role' => $flags->filter(fn ($f) => ($f->type ?? '') === 'role')->count(),
                'percentage' => $flags->filter(fn ($f) => ($f->type ?? '') === 'percentage')->count(),
                'environment' => $flags->filter(fn ($f) => ($f->type ?? '') === 'environment')->count(),
            ],
            'flags' => $flags->map(fn ($f) => [
                'name' => $f->name,
                'enabled' => $f->is_enabled,
                'type' => $f->type,
                'description' => $f->description,
                'updated_at' => $f->updated_at->toIso8601String(),
                'deprecated' => $f->deprecated_at !== null,
            ])->toArray(),
        ];
    }
}
```

```php
// app/Services/FeatureFlagAnalytics.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\DB;

class FeatureFlagAnalytics
{
    /**
     * Record a flag evaluation.
     */
    public function recordEvaluation(
        User $user,
        string $flag,
        bool $result,
        string $reason,
    ): void {
        $key = "flag_eval:{$flag}:" . now()->format('Y-m-d-H');

        DB::table('feature_flag_analytics')->insert([
            'user_id' => $user->id,
            'flag_name' => $flag,
            'result' => $result,
            'reason' => $reason,
            'evaluated_at' => now(),
        ]);

        // Update hourly aggregate
        $aggregateKey = "flag_stats:{$flag}:" . now()->format('Y-m-d-H');
        $stats = cache()->get($aggregateKey, ['true' => 0, 'false' => 0]);
        $stats[$result ? 'true' : 'false']++;
        cache()->put($aggregateKey, $stats, 86400);
    }

    /**
     * Record a flag evaluation with outcome tracking.
     */
    public function recordOutcome(
        User $user,
        string $flag,
        bool $flagResult,
        string $outcome,
        ?float $value = null,
    ): void {
        DB::table('feature_flag_outcomes')->insert([
            'user_id' => $user->id,
            'flag_name' => $flag,
            'flag_result' => $flagResult,
            'outcome' => $outcome,
            'value' => $value,
            'recorded_at' => now(),
        ]);
    }

    /**
     * Get flag performance summary.
     */
    public function getFlagSummary(string $flag, int $days = 30): array
    {
        $start = now()->subDays($days);

        $evaluations = DB::table('feature_flag_analytics')
            ->where('flag_name', $flag)
            ->where('evaluated_at', '>=', $start)
            ->select(
                'result',
                DB::raw('COUNT(*) as count'),
                DB::raw('DATE(evaluated_at) as date')
            )
            ->groupBy('result', 'date')
            ->get();

        $outcomes = DB::table('feature_flag_outcomes')
            ->where('flag_name', $flag)
            ->where('recorded_at', '>=', $start)
            ->select(
                'outcome',
                'flag_result',
                DB::raw('COUNT(*) as count'),
                DB::raw('AVG(value) as avg_value')
            )
            ->groupBy('outcome', 'flag_result')
            ->get();

        return [
            'flag' => $flag,
            'period_days' => $days,
            'total_evaluations' => $evaluations->sum('count'),
            'evaluations_enabled' => $evaluations->where('result', true)->sum('count'),
            'evaluations_disabled' => $evaluations->where('result', false)->sum('count'),
            'daily_breakdown' => $evaluations->groupBy('date')->map(fn ($records) => [
                'true' => $records->where('result', true)->sum('count'),
                'false' => $records->where('result', false)->sum('count'),
            ])->toArray(),
            'outcomes' => $outcomes->toArray(),
        ];
    }

    /**
     * Compare outcomes between enabled and disabled groups.
     */
    public function compareOutcomes(string $flag, string $outcome): array
    {
        $enabled = DB::table('feature_flag_outcomes')
            ->where('flag_name', $flag)
            ->where('flag_result', true)
            ->where('outcome', $outcome)
            ->select(DB::raw('COUNT(*) as count'), DB::raw('AVG(value) as avg_value'))
            ->first();

        $disabled = DB::table('feature_flag_outcomes')
            ->where('flag_name', $flag)
            ->where('flag_result', false)
            ->where('outcome', $outcome)
            ->select(DB::raw('COUNT(*) as count'), DB::raw('AVG(value) as avg_value'))
            ->first();

        return [
            'flag' => $flag,
            'outcome' => $outcome,
            'enabled_group' => [
                'count' => $enabled->count ?? 0,
                'avg_value' => round($enabled->avg_value ?? 0, 4),
            ],
            'disabled_group' => [
                'count' => $disabled->count ?? 0,
                'avg_value' => round($disabled->avg_value ?? 0, 4),
            ],
            'lift' => $this->calculateLift(
                $enabled->avg_value ?? 0,
                $disabled->avg_value ?? 0,
            ),
        ];
    }

    protected function calculateLift(float $enabled, float $disabled): ?float
    {
        if ($disabled == 0) {
            return null;
        }

        return round((($enabled - $disabled) / $disabled) * 100, 2);
    }
}
```
