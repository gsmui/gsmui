## Security Features

### Password Change

Requires current password confirmation:

```php
class ChangePassword extends Component
{
    public string $current_password = '';
    public string $password = '';
    public string $password_confirmation = '';

    protected array $rules = [
        'current_password' => ['required', 'string'],
        'password' => ['required', 'string', 'min:8', 'confirmed'],
    ];

    public function changePassword(): void
    {
        $this->validate();

        if (! Hash::check($this->current_password, Auth::user()->password)) {
            $this->addError('current_password', 'Current password is incorrect.');
            return;
        }

        Auth::user()->forceFill([
            'password' => Hash::make($this->password),
        ])->save();

        // Revoke other sessions
        Auth::user()->sessions()
            ->where('id', '!=', session()->getId())
            ->delete();

        event(new PasswordChanged(Auth::user()));
    }
}
```

## Session Management Service

### Complete Session Service

```php
// app/Services/UserSessionService.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Session;

class UserSessionService
{
    /**
     * Get all active sessions for a user.
     */
    public function getActiveSessions(User $user): array
    {
        $sessions = DB::table('sessions')
            ->where('user_id', $user->id)
            ->get();

        $currentSessionId = Session::getId();

        return $sessions->map(function ($session) use ($currentSessionId) {
            $payload = json_decode($session->payload, true);
            $lastActivity = $session->last_activity;
            $isCurrent = $session->id === $currentSessionId;

            return [
                'id' => $session->id,
                'ip_address' => $payload['ip_address'] ?? 'Unknown',
                'user_agent' => $payload['user_agent'] ?? 'Unknown',
                'browser' => $this->parseBrowser($payload['user_agent'] ?? ''),
                'os' => $this->parseOS($payload['user_agent'] ?? ''),
                'is_current' => $isCurrent,
                'last_active' => $lastActivity
                    ? now()->subSeconds(now()->timestamp - $lastActivity)->diffForHumans()
                    : 'Unknown',
                'last_active_at' => $lastActivity
                    ? now()->subSeconds(now()->timestamp - $lastActivity)
                    : null,
            ];
        })->toArray();
    }

    /**
     * Revoke a specific session.
     */
    public function revokeSession(User $user, string $sessionId): bool
    {
        if ($sessionId === Session::getId()) {
            return false; // Cannot revoke current session
        }

        DB::table('sessions')
            ->where('id', $sessionId)
            ->where('user_id', $user->id)
            ->delete();

        activity('session-management')
            ->performedOn($user)
            ->withProperties(['session_id' => $sessionId])
            ->log('Session revoked');

        return true;
    }

    /**
     * Revoke all sessions except the current one.
     */
    public function revokeAllOtherSessions(User $user): int
    {
        $currentSessionId = Session::getId();

        $revoked = DB::table('sessions')
            ->where('user_id', $user->id)
            ->where('id', '!=', $currentSessionId)
            ->delete();

        activity('session-management')
            ->performedOn($user)
            ->withProperties(['revoked_count' => $revoked])
            ->log('All other sessions revoked');

        return $revoked;
    }

    /**
     * Parse browser name from user agent.
     */
    protected function parseBrowser(string $userAgent): string
    {
        $browsers = [
            'Edge' => '/Edg[e]?\/(\d+)/i',
            'Chrome' => '/Chrome\/(\d+)/i',
            'Firefox' => '/Firefox\/(\d+)/i',
            'Safari' => '/Version\/(\d+).*Safari/i',
            'Opera' => '/(?:Opera|OPR)\/(\d+)/i',
            'MSIE' => '/MSIE (\d+)/i',
            'Trident' => '/Trident\/.*rv:(\d+)/i',
        ];

        foreach ($browsers as $name => $pattern) {
            if (preg_match($pattern, $userAgent, $matches)) {
                return $name . ' ' . ($matches[1] ?? '');
            }
        }

        return 'Unknown Browser';
    }

    /**
     * Parse OS name from user agent.
     */
    protected function parseOS(string $userAgent): string
    {
        $oses = [
            'Windows 11' => '/Windows NT 10.0.*build\/(\d{5,})/i',
            'Windows 10' => '/Windows NT 10\.0/i',
            'Windows 8' => '/Windows NT 6\.3/i',
            'Windows 7' => '/Windows NT 6\.1/i',
            'macOS' => '/Mac OS X (\d+[._]\d+[._]\d+)/i',
            'Linux' => '/Linux/i',
            'Android' => '/Android (\d+\.\d+)/i',
            'iOS' => '/iPhone.*OS (\d+_\d+)/i',
        ];

        foreach ($oses as $name => $pattern) {
            if (preg_match($pattern, $userAgent)) {
                return $name;
            }
        }

        return 'Unknown OS';
    }
}
```

## Password Change Service

### Password Change with Security

```php
// app/Services/PasswordChangeService.php

namespace App\Services;

use App\Models\User;
use App\Events\PasswordChanged;
use Illuminate\Support\Facades\Hash;

class PasswordChangeService
{
    protected AdminPasswordPolicyService $policyService;

    public function __construct(AdminPasswordPolicyService $policyService)
    {
        $this->policyService = $policyService;
    }

    /**
     * Change user password with full security checks.
     */
    public function changePassword(
        User $user,
        string $currentPassword,
        string $newPassword,
        bool $revokeOtherSessions = true
    ): array {
        // Verify current password
        if (! Hash::check($currentPassword, $user->password)) {
            return [
                'success' => false,
                'errors' => ['current_password' => ['The current password is incorrect.']],
            ];
        }

        // Check password strength
        $strengthErrors = $this->policyService->validatePasswordStrength($newPassword);
        if (! empty($strengthErrors)) {
            return [
                'success' => false,
                'errors' => ['password' => $strengthErrors],
            ];
        }

        // Check password reuse
        if ($this->policyService->isPasswordReused($user, $newPassword)) {
            return [
                'success' => false,
                'errors' => ['password' => ['This password was recently used. Please choose a different one.']],
            ];
        }

        // Update password
        $user->forceFill([
            'password' => Hash::make($newPassword),
            'password_changed_at' => now(),
        ])->save();

        // Store in history
        $this->policyService->storePasswordHistory($user);

        // Revoke other sessions if requested
        if ($revokeOtherSessions) {
            app(UserSessionService::class)->revokeAllOtherSessions($user);
        }

        // Fire event
        event(new PasswordChanged($user));

        // Log the change
        activity('password-change')
            ->performedOn($user)
            ->withProperties([
                'revoke_sessions' => $revokeOtherSessions,
                'ip' => request()->ip(),
            ])
            ->log('Password changed');

        return [
            'success' => true,
            'message' => 'Password changed successfully.',
        ];
    }
}
```

## Email Change Service

### Email Change with Verification

```php
// app/Services/EmailChangeService.php

namespace App\Services;

use App\Models\User;
use App\Events\EmailChangeRequested;
use App\Events\EmailChanged;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;

class EmailChangeService
{
    /**
     * Request an email change.
     */
    public function requestChange(
        User $user,
        string $newEmail,
        string $password
    ): array {
        // Verify password
        if (! Hash::check($password, $user->password)) {
            return [
                'success' => false,
                'errors' => ['password' => ['The password is incorrect.']],
            ];
        }

        // Check if email is already taken
        if (User::where('email', $newEmail)->where('id', '!=', $user->id)->exists()) {
            return [
                'success' => false,
                'errors' => ['new_email' => ['This email is already in use.']],
            ];
        }

        // Check if it's the same email
        if ($newEmail === $user->email) {
            return [
                'success' => false,
                'errors' => ['new_email' => ['This is your current email address.']],
            ];
        }

        // Generate verification token
        $token = Str::random(60);

        // Store pending change
        DB::table('email_change_requests')->updateOrInsert(
            ['user_id' => $user->id],
            [
                'new_email' => $newEmail,
                'token' => $token,
                'created_at' => now(),
                'expires_at' => now()->addHours(24),
            ]
        );

        // Send verification email
        event(new EmailChangeRequested($user, $newEmail, $token));

        activity('email-change')
            ->performedOn($user)
            ->withProperties([
                'new_email' => $newEmail,
                'ip' => request()->ip(),
            ])
            ->log('Email change requested');

        return [
            'success' => true,
            'message' => 'Verification email sent to ' . $newEmail,
        ];
    }

    /**
     * Confirm email change with token.
     */
    public function confirmChange(string $token): array
    {
        $request = DB::table('email_change_requests')
            ->where('token', $token)
            ->where('expires_at', '>', now())
            ->first();

        if (! $request) {
            return [
                'success' => false,
                'message' => 'Invalid or expired verification token.',
            ];
        }

        $user = User::find($request->user_id);
        $oldEmail = $user->email;

        $user->forceFill([
            'email' => $request->new_email,
            'email_verified_at' => null,
        ])->save();

        // Delete the request
        DB::table('email_change_requests')
            ->where('id', $request->id)
            ->delete();

        // Send verification to new email
        $user->sendEmailVerificationNotification();

        // Notify old email
        \Mail::to($oldEmail)->send(
            new \App\Mail\EmailChangedNotification($oldEmail, $request->new_email)
        );

        event(new EmailChanged($user, $oldEmail, $request->new_email));

        activity('email-change')
            ->performedOn($user)
            ->withProperties([
                'old_email' => $oldEmail,
                'new_email' => $request->new_email,
            ])
            ->log('Email change confirmed');

        return [
            'success' => true,
            'message' => 'Email changed successfully. Please verify your new email.',
        ];
    }
}
```

## Account Deletion Service

### Account Deletion with Grace Period

```php
// app/Services/AccountDeletionService.php

namespace App\Services;

use App\Models\User;
use App\Events\AccountDeletionRequested;
use App\Events\AccountDeleted;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\DB;

class AccountDeletionService
{
    /**
     * Request account deletion (starts 30-day grace period).
     */
    public function requestDeletion(
        User $user,
        string $password,
        string $reason = ''
    ): array {
        // Verify password
        if (! Hash::check($password, $user->password)) {
            return [
                'success' => false,
                'errors' => ['password' => ['The password is incorrect.']],
            ];
        }

        // Prevent admin/super_admin from deleting themselves
        if ($user->hasRole(['admin', 'super_admin'])) {
            return [
                'success' => false,
                'errors' => ['password' => ['Admin accounts cannot be self-deleted. Contact another admin.']],
            ];
        }

        // Calculate impact
        $impact = $this->calculateImpact($user);

        // Schedule deletion
        $user->forceFill([
            'deleted_at' => now(),
            'deletion_scheduled_at' => now()->addDays(30),
        ])->save();

        // Store deletion request
        DB::table('account_deletions')->insert([
            'user_id' => $user->id,
            'reason' => $reason,
            'scheduled_for' => now()->addDays(30),
            'impact_data' => json_encode($impact),
            'created_at' => now(),
        ]);

        event(new AccountDeletionRequested($user, $impact));

        activity('account-deletion')
            ->performedOn($user)
            ->withProperties([
                'impact' => $impact,
                'reason' => $reason,
            ])
            ->log('Account deletion requested (30-day grace period)');

        return [
            'success' => true,
            'message' => 'Account deletion scheduled. You have 30 days to cancel.',
            'impact' => $impact,
            'scheduled_for' => now()->addDays(30)->format('Y-m-d'),
        ];
    }

    /**
     * Cancel account deletion.
     */
    public function cancelDeletion(User $user): array
    {
        if (! $user->deleted_at) {
            return [
                'success' => false,
                'message' => 'No pending deletion found.',
            ];
        }

        $user->forceFill([
            'deleted_at' => null,
            'deletion_scheduled_at' => null,
        ])->save();

        DB::table('account_deletions')
            ->where('user_id', $user->id)
            ->whereNull('completed_at')
            ->update(['cancelled_at' => now()]);

        activity('account-deletion')
            ->performedOn($user)
            ->log('Account deletion cancelled');

        return [
            'success' => true,
            'message' => 'Account deletion cancelled successfully.',
        ];
    }

    /**
     * Permanently delete an account (called by scheduler after 30 days).
     */
    public function permanentlyDelete(User $user): void
    {
        // Anonymize data instead of hard delete
        $user->forceFill([
            'name' => 'Deleted User',
            'email' => "deleted_{$user->id}@deleted.local",
            'password' => 'deleted',
            'email_verified_at' => null,
            'two_factor_secret' => null,
            'two_factor_recovery_codes' => null,
        ])->save();

        // Soft-delete related data
        $user->themes()->delete();
        $user->components()->delete();
        $user->orders()->update(['status' => 'cancelled']);
        $user->tokens()->delete();
        $user->sessions()->delete();

        // Remove roles
        $user->syncRoles([]);

        // Mark deletion as completed
        DB::table('account_deletions')
            ->where('user_id', $user->id)
            ->update(['completed_at' => now()]);

        event(new AccountDeleted($user));

        activity('account-deletion')
            ->withProperties(['user_id' => $user->id])
            ->log('Account permanently deleted');
    }

    /**
     * Calculate impact of account deletion.
     */
    protected function calculateImpact(User $user): array
    {
        return [
            'themes' => $user->themes()->count(),
            'components' => $user->components()->count(),
            'orders' => $user->orders()->count(),
            'tokens' => $user->tokens()->count(),
            'active_subscription' => $user->hasActiveSubscription(),
            'pending_orders' => $user->orders()->where('status', 'pending')->count(),
            'earnings_balance' => $user->earnings_balance ?? 0,
        ];
    }
}
```

```php
// app/Services/LoginHistoryService.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\DB;

class LoginHistoryService
{
    /**
     * Record a login event.
     */
    public function recordLogin(
        User $user,
        bool $success,
        ?string $failureReason = null
    ): void {
        DB::table('login_history')->insert([
            'user_id' => $user->id,
            'success' => $success,
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
            'browser' => $this->parseBrowser(request()->userAgent()),
            'os' => $this->parseOS(request()->userAgent()),
            'location' => $this->getLocation(request()->ip()),
            'failure_reason' => $failureReason,
            'created_at' => now(),
        ]);
    }

    /**
     * Get login history for a user.
     */
    public function getLoginHistory(User $user, int $limit = 20): array
    {
        return DB::table('login_history')
            ->where('user_id', $user->id)
            ->orderBy('created_at', 'desc')
            ->limit($limit)
            ->get()
            ->toArray();
    }

    /**
     * Get failed login attempts.
     */
    public function getFailedAttempts(
        User $user,
        int $hours = 24
    ): int {
        return DB::table('login_history')
            ->where('user_id', $user->id)
            ->where('success', false)
            ->where('created_at', '>=', now()->subHours($hours))
            ->count();
    }

    /**
     * Check for suspicious login patterns.
     */
    public function detectSuspiciousActivity(User $user): array
    {
        $recentLogins = DB::table('login_history')
            ->where('user_id', $user->id)
            ->where('created_at', '>=', now()->subHours(24))
            ->get();

        $uniqueIps = $recentLogins->pluck('ip_address')->unique()->count();
        $uniqueLocations = $recentLogins->pluck('location')->unique()->count();
        $failedCount = $recentLogins->where('success', false)->count();

        return [
            'unique_ips_24h' => $uniqueIps,
            'unique_locations_24h' => $uniqueLocations,
            'failed_attempts_24h' => $failedCount,
            'is_suspicious' => $uniqueIps > 5 || $uniqueLocations > 3 || $failedCount > 10,
        ];
    }

    /**
     * Get login statistics for admin dashboard.
     */
    public function getLoginStats(int $days = 30): array
    {
        $startDate = now()->subDays($days);

        return [
            'total_logins' => DB::table('login_history')
                ->where('created_at', '>=', $startDate)
                ->count(),
            'successful_logins' => DB::table('login_history')
                ->where('created_at', '>=', $startDate)
                ->where('success', true)
                ->count(),
            'failed_logins' => DB::table('login_history')
                ->where('created_at', '>=', $startDate)
                ->where('success', false)
                ->count(),
            'unique_users' => DB::table('login_history')
                ->where('created_at', '>=', $startDate)
                ->distinct('user_id')
                ->count(),
            'unique_ips' => DB::table('login_history')
                ->where('created_at', '>=', $startDate)
                ->distinct('ip_address')
                ->count(),
            'by_browser' => DB::table('login_history')
                ->where('created_at', '>=', $startDate)
                ->select('browser', DB::raw('count(*) as count'))
                ->groupBy('browser')
                ->pluck('count', 'browser')
                ->toArray(),
            'by_os' => DB::table('login_history')
                ->where('created_at', '>=', $startDate)
                ->select('os', DB::raw('count(*) as count'))
                ->groupBy('os')
                ->pluck('count', 'os')
                ->toArray(),
        ];
    }
}
```

## Security Notification Service

### Notification Manager

```php
// app/Services/SecurityNotificationService.php

namespace App\Services;

use App\Models\User;

class SecurityNotificationService
{
    /**
     * Send new login notification.
     */
    public function notifyNewLogin(User $user, string $ip, string $userAgent): void
    {
        if (! $user->hasSetting('notify_new_login', true)) {
            return;
        }

        $user->notify(
            new \App\Notifications\NewLoginNotification($ip, $userAgent)
        );
    }

    /**
     * Send password changed notification.
     */
    public function notifyPasswordChanged(User $user, bool $initiatedByUser = true): void
    {
        $user->notify(
            new \App\Notifications\PasswordChangedNotification($initiatedByUser)
        );
    }

    /**
     * Send 2FA status change notification.
     */
    public function notify2FAChanged(User $user, bool $enabled): void
    {
        $user->notify(
            new \App\Notifications\TwoFactorChangedNotification($enabled)
        );
    }

    /**
     * Send suspicious activity notification.
     */
    public function notifySuspiciousActivity(User $user, array $details): void
    {
        $user->notify(
            new \App\Notifications\SuspiciousActivityNotification($details)
        );
    }

    /**
     * Send email change notification.
     */
    public function notifyEmailChanged(User $user, string $oldEmail, string $newEmail): void
    {
        $user->notify(
            new \App\Notifications\EmailChangedNotification($oldEmail, $newEmail)
        );

        // Also notify old email
        \Mail::to($oldEmail)->send(
            new \App\Mail\EmailChangedNotification($oldEmail, $newEmail)
        );
    }

    /**
     * Send account deletion notification.
     */
    public function notifyAccountDeletionScheduled(User $user, string $scheduledDate): void
    {
        $user->notify(
            new \App\Notifications\AccountDeletionScheduledNotification($scheduledDate)
        );
    }

    /**
     * Check user notification preferences.
     */
    public function shouldNotify(User $user, string $type): bool
    {
        $preferences = $user->notification_preferences ?? [];

        return $preferences[$type] ?? true;
    }
}
```

```php
use App\Services\UserSessionService;
use App\Services\PasswordChangeService;
use App\Services\LoginHistoryService;
use Illuminate\Support\Facades\Auth;

class SecuritySettings extends Component
{
    // Password change
    public string $current_password = '';
    public string $new_password = '';
    public string $new_password_confirmation = '';

    // Active sessions
    public array $sessions = [];
    public ?string $sessionToRevoke = null;
    public bool $showRevokeAll = false;

    // Login history
    public array $loginHistory = [];
    public array $loginStats = [];

    // 2FA
    public bool $twoFactorEnabled = false;

    // Notification preferences
    public bool $notifyNewLogin = true;
    public bool $notifyPasswordChange = true;
    public bool $notify2FAChange = true;
    public bool $notifySuspiciousActivity = true;

    protected UserSessionService $sessionService;
    protected PasswordChangeService $passwordService;
    protected LoginHistoryService $loginHistoryService;

    protected array $rules = [
        'current_password' => ['required', 'string'],
        'new_password' => ['required', 'string', 'min:8', 'confirmed'],
    ];

    public function boot(
        UserSessionService $sessionService,
        PasswordChangeService $passwordService,
        LoginHistoryService $loginHistoryService
    ): void {
        $this->sessionService = $sessionService;
        $this->passwordService = $passwordService;
        $this->loginHistoryService = $loginHistoryService;
    }

    public function mount(): void
    {
        $user = Auth::user();
        $this->twoFactorEnabled = $user->two_factor_secret !== null;
        $this->loadSessions();
        $this->loadLoginHistory();
        $this->loadNotificationPreferences();
    }

    public function changePassword(): void
    {
        $this->validate();

        $result = $this->passwordService->changePassword(
            Auth::user(),
            $this->current_password,
            $this->new_password,
            true
        );

        if ($result['success']) {
            $this->current_password = '';
            $this->new_password = '';
            $this->new_password_confirmation = '';
            session()->flash('success', 'Password changed successfully. Other sessions have been revoked.');
        } else {
            foreach ($result['errors'] as $field => $messages) {
                foreach ($messages as $message) {
                    $this->addError($field, $message);
                }
            }
        }
    }

    public function loadSessions(): void
    {
        $this->sessions = $this->sessionService->getActiveSessions(Auth::user());
    }

    public function revokeSession(string $sessionId): void
    {
        $this->sessionService->revokeSession(Auth::user(), $sessionId);
        $this->loadSessions();
        session()->flash('success', 'Session revoked successfully.');
    }

    public function confirmRevokeAll(): void
    {
        $this->showRevokeAll = true;
    }

    public function revokeAllSessions(): void
    {
        $revoked = $this->sessionService->revokeAllOtherSessions(Auth::user());
        $this->loadSessions();
        $this->showRevokeAll = false;
        session()->flash('success', "Revoked {$revoked} other sessions.");
    }

    public function loadLoginHistory(): void
    {
        $this->loginHistory = $this->loginHistoryService->getLoginHistory(Auth::user(), 10);
        $this->loginStats = $this->loginHistoryService->getLoginStats(30);
    }

    public function loadNotificationPreferences(): void
    {
        $user = Auth::user();
        $prefs = $user->notification_preferences ?? [];

        $this->notifyNewLogin = $prefs['notify_new_login'] ?? true;
        $this->notifyPasswordChange = $prefs['notify_password_change'] ?? true;
        $this->notify2FAChange = $prefs['notify_2fa_change'] ?? true;
        $this->notifySuspiciousActivity = $prefs['notify_suspicious_activity'] ?? true;
    }

    public function saveNotificationPreferences(): void
    {
        Auth::user()->update([
            'notification_preferences' => [
                'notify_new_login' => $this->notifyNewLogin,
                'notify_password_change' => $this->notifyPasswordChange,
                'notify_2fa_change' => $this->notify2FAChange,
                'notify_suspicious_activity' => $this->notifySuspiciousActivity,
            ],
        ]);

        session()->flash('success', 'Notification preferences saved.');
    }

    public function render()
    {
        return view('livewire.settings.security-settings', [
            'user' => Auth::user(),
        ]);
    }
}
```

## Security Audit Log

### Database Schema

```sql
CREATE TABLE security_audit_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    event VARCHAR(255) NOT NULL,
    ip_address VARCHAR(45) NULL,
    user_agent TEXT NULL,
    properties JSON NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX (user_id, created_at),
    INDEX (user_id, event)
);
```

### Audit Log Service

```php
// app/Services/SecurityAuditService.php

namespace App\Services;

use App\Models\SecurityAuditLog;
use Illuminate\Support\Facades\Request;

class SecurityAuditService
{
    public function log(string $event, array $properties = []): SecurityAuditLog
    {
        $user = auth()->user();

        return SecurityAuditLog::create([
            'user_id' => $user?->id,
            'event' => $event,
            'ip_address' => Request::ip(),
            'user_agent' => Request::userAgent(),
            'properties' => $properties,
        ]);
    }

    public function getRecentLogs(int $userId, int $limit = 50): array
    {
        return SecurityAuditLog::where('user_id', $userId)
            ->orderByDesc('created_at')
            ->limit($limit)
            ->get()
            ->map(fn ($log) => [
                'id' => $log->id,
                'event' => $log->event,
                'description' => $this->describeEvent($log->event),
                'ip_address' => $log->ip_address,
                'properties' => $log->properties,
                'created_at' => $log->created_at->diffForHumans(),
            ])
            ->toArray();
    }

    public function getLogsByEvent(string $event, int $limit = 20): array
    {
        return SecurityAuditLog::where('user_id', auth()->id())
            ->where('event', $event)
            ->orderByDesc('created_at')
            ->limit($limit)
            ->get()
            ->toArray();
    }

    public function getLoginHistory(int $userId, int $days = 30): array
    {
        return SecurityAuditLog::where('user_id', $userId)
            ->whereIn('event', ['login.success', 'login.failed'])
            ->where('created_at', '>=', now()->subDays($days))
            ->orderByDesc('created_at')
            ->get()
            ->toArray();
    }

    public function getSuspiciousActivity(int $userId): array
    {
        $recentFailures = SecurityAuditLog::where('user_id', $userId)
            ->where('event', 'login.failed')
            ->where('created_at', '>=', now()->subHours(24))
            ->count();

        $uniqueIps = SecurityAuditLog::where('user_id', $userId)
            ->where('event', 'login.success')
            ->where('created_at', '>=', now()->subDays(7))
            ->distinct('ip_address')
            ->count();

        return [
            'failed_attempts_24h' => $recentFailures,
            'unique_ips_7d' => $uniqueIps,
            'high_risk' => $recentFailures > 10 || $uniqueIps > 5,
        ];
    }

    private function describeEvent(string $event): string
    {
        return match ($event) {
            'password.changed' => 'You changed your password',
            'password.reset.requested' => 'You requested a password reset',
            'password.reset.completed' => 'You reset your password',
            'email.changed' => 'You changed your email address',
            'email.verified' => 'You verified your email address',
            '2fa.enabled' => 'You enabled two-factor authentication',
            '2fa.disabled' => 'You disabled two-factor authentication',
            '2fa.recovery_codes.regenerated' => 'You regenerated recovery codes',
            'login.success' => 'You logged in successfully',
            'login.failed' => 'A login attempt failed',
            'login.locked' => 'Your account was temporarily locked',
            'session.revoked' => 'You revoked a device session',
            'session.revoked_all' => 'You revoked all other sessions',
            'device.trusted' => 'You trusted a device',
            'device.untrusted' => 'You untrusted a device',
            'api_token.created' => 'You created an API token',
            'api_token.revoked' => 'You revoked an API token',
            'mcp_token.created' => 'You created an MCP token',
            'mcp_token.revoked' => 'You revoked an MCP token',
            'email_change.initiated' => 'You initiated an email change',
            'email_change.cancelled' => 'You cancelled an email change',
            'account.deletion.requested' => 'You requested account deletion',
            'account.deletion.cancelled' => 'You cancelled account deletion',
            'notification.preferences.changed' => 'You updated notification preferences',
            'recovery.codes.used' => 'You used a recovery code to log in',
            default => 'Unknown event',
        };
    }
}
```

```php
// app/Livewire/SecurityAuditLogComponent.php

namespace App\Livewire;

use App\Services\SecurityAuditService;
use Livewire\Component;
use Livewire\WithPagination;

class SecurityAuditLogComponent extends Component
{
    use WithPagination;

    public string $eventFilter = '';
    public int $daysFilter = 30;
    public bool $showDetails = false;

    protected SecurityAuditService $auditService;

    public function boot(SecurityAuditService $auditService): void
    {
        $this->auditService = $auditService;
    }

    public function getLogsProperty(): array
    {
        $query = \App\Models\SecurityAuditLog::where('user_id', auth()->id());

        if ($this->eventFilter) {
            $query->where('event', $this->eventFilter);
        }

        if ($this->daysFilter) {
            $query->where('created_at', '>=', now()->subDays($this->daysFilter));
        }

        return $query->orderByDesc('created_at')
            ->paginate(20);
    }

    public function getSuspiciousProperty(): array
    {
        return $this->auditService->getSuspiciousActivity(auth()->id());
    }

    public function clearFilters(): void
    {
        $this->eventFilter = '';
        $this->daysFilter = 30;
    }

    public function render()
    {
        return view('livewire.security-audit-log');
    }
}
```

```php
// app/Services/SessionService.php

namespace App\Services;

use App\Models\UserSession;
use Illuminate\Support\Facades\Session;
use Jenssegers\Agent\Agent;

class SessionService
{
    public function trackSession(\App\Models\User $user): UserSession
    {
        $agent = new Agent();
        $userAgent = request()->userAgent();

        $agent->setUserAgent($userAgent);

        return UserSession::updateOrCreate(
            [
                'user_id' => $user->id,
                'session_id' => Session::getId(),
            ],
            [
                'ip_address' => request()->ip(),
                'user_agent' => $userAgent,
                'browser' => $agent->browser(),
                'os' => $agent->platform(),
                'is_current' => true,
                'last_active_at' => now(),
            ]
        );
    }

    public function getActiveSessions(int $userId): array
    {
        $this->clearExpiredSessions($userId);

        return UserSession::where('user_id', $userId)
            ->orderByDesc('is_current')
            ->orderByDesc('last_active_at')
            ->get()
            ->map(fn ($session) => $session->toArray())
            ->toArray();
    }

    public function revokeSession(int $userId, int $sessionId): bool
    {
        $session = UserSession::where('user_id', $userId)
            ->where('id', $sessionId)
            ->first();

        if (! $session) {
            return false;
        }

        if ($session->is_current) {
            return false;
        }

        // Invalidate the Laravel session
        Session::where('id', $session->session_id)->delete();

        $session->delete();

        activity('session')
            ->performedOn(auth()->user())
            ->withProperties([
                'session_id' => $sessionId,
                'browser' => $session->browser,
                'os' => $session->os,
            ])
            ->log('Session revoked');

        return true;
    }

    public function revokeAllSessions(int $userId, ?string $exceptSessionId = null): int
    {
        $query = UserSession::where('user_id', $userId)
            ->where('is_current', false);

        if ($exceptSessionId) {
            $query->where('session_id', '!=', $exceptSessionId);
        }

        $sessions = $query->get();
        $count = $sessions->count();

        // Delete Laravel sessions
        $sessionIds = $sessions->pluck('session_id')->toArray();
        Session::whereIn('id', $sessionIds)->delete();

        // Delete device records
        $sessions->each->delete();

        activity('session')
            ->performedOn(auth()->user())
            ->withProperties(['revoked_count' => $count])
            ->log('All sessions revoked');

        return $count;
    }

    public function updateLastActivity(int $userId): void
    {
        UserSession::where('user_id', $userId)
            ->where('session_id', Session::getId())
            ->update(['last_active_at' => now()]);
    }

    public function clearExpiredSessions(int $userId, int $maxAgeDays = 7): void
    {
        UserSession::where('user_id', $userId)
            ->where('last_active_at', '<', now()->subDays($maxAgeDays))
            ->delete();
    }

    public function getSessionsByLocation(int $userId): array
    {
        return UserSession::where('user_id', $userId)
            ->selectRaw('ip_address, COUNT(*) as count, MIN(created_at) as first_seen, MAX(last_active_at) as last_active')
            ->groupBy('ip_address')
            ->orderByDesc('last_active')
            ->get()
            ->toArray();
    }

    public function getConcurrentSessions(int $userId): int
    {
        return UserSession::where('user_id', $userId)
            ->where('last_active_at', '>=', now()->subMinutes(15))
            ->count();
    }
}
```

```php
// app/Services/EmailChangeService.php

namespace App\Services;

use App\Models\EmailChangeRequest;
use App\Models\User;
use App\Notifications\EmailChangeVerification;
use App\Notifications\EmailChangePending;
use Illuminate\Support\Str;

class EmailChangeService
{
    public function initiate(User $user, string $newEmail): EmailChangeRequest
    {
        // Check if new email is already taken
        if (User::where('email', $newEmail)->exists()) {
            throw new \Exception('Email address is already in use.');
        }

        // Delete any existing pending requests
        EmailChangeRequest::where('user_id', $user->id)
            ->whereNull('verified_at')
            ->delete();

        $verificationToken = Str::random(64);
        $confirmationToken = Str::random(64);

        $request = EmailChangeRequest::create([
            'user_id' => $user->id,
            'old_email' => $user->email,
            'new_email' => $newEmail,
            'verification_token' => hash('sha256', $verificationToken),
            'confirmation_token' => hash('sha256', $confirmationToken),
            'expires_at' => now()->addHours(24),
        ]);

        // Send verification to new email
        $user->notify(new EmailChangeVerification($newEmail, $verificationToken));

        // Notify old email about the change request
        $user->notify(new EmailChangePending($newEmail));

        activity('email_change')
            ->performedOn($user)
            ->withProperties(['new_email' => $newEmail])
            ->log('Email change initiated');

        return $request;
    }

    public function verify(User $user, string $token): bool
    {
        $request = EmailChangeRequest::where('user_id', $user->id)
            ->whereNull('verified_at')
            ->where('expires_at', '>', now())
            ->first();

        if (! $request) {
            return false;
        }

        if (hash_equals($request->verification_token, hash('sha256', $token))) {
            $request->update(['verified_at' => now()]);
            return true;
        }

        return false;
    }

    public function confirm(User $user, string $token): bool
    {
        $request = EmailChangeRequest::where('user_id', $user->id)
            ->whereNull('completed_at')
            ->where('verified_at', '!=', null)
            ->where('expires_at', '>', now())
            ->first();

        if (! $request) {
            return false;
        }

        if (hash_equals($request->confirmation_token, hash('sha256', $token))) {
            $user->update(['email' => $request->new_email]);
            $user->markEmailAsVerified();

            $request->update(['completed_at' => now()]);

            activity('email_change')
                ->performedOn($user)
                ->withProperties([
                    'old_email' => $request->old_email,
                    'new_email' => $request->new_email,
                ])
                ->log('Email change completed');

            return true;
        }

        return false;
    }

    public function cancel(User $user): bool
    {
        $deleted = EmailChangeRequest::where('user_id', $user->id)
            ->whereNull('completed_at')
            ->delete();

        if ($deleted) {
            activity('email_change')
                ->performedOn($user)
                ->log('Email change cancelled');
        }

        return $deleted > 0;
    }

    public function getPendingRequest(User $user): ?EmailChangeRequest
    {
        return EmailChangeRequest::where('user_id', $user->id)
            ->whereNull('completed_at')
            ->where('expires_at', '>', now())
            ->first();
    }
}
```

```php
namespace App\Services;

use App\Models\AccountDeletionRequest;
use App\Models\User;
use App\Notifications\AccountDeletionConfirmation;
use App\Notifications\AccountDeletionGracePeriod;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;

class AccountDeletionService
{
    public const GRACE_PERIOD_DAYS = 30;

    public function requestDeletion(User $user, string $password): void
    {
        if (! Hash::check($password, $user->password)) {
            throw new \Exception('Incorrect password.');
        }

        // Cancel any existing pending deletion
        AccountDeletionRequest::where('user_id', $user->id)
            ->whereNull('deleted_at')
            ->delete();

        $request = AccountDeletionRequest::create([
            'user_id' => $user->id,
            'grace_period_ends_at' => now()->addDays(self::GRACE_PERIOD_DAYS),
        ]);

        $user->notify(new AccountDeletionConfirmation(self::GRACE_PERIOD_DAYS));

        activity('account')
            ->performedOn($user)
            ->withProperties(['grace_days' => self::GRACE_PERIOD_DAYS])
            ->log('Account deletion requested');
    }

    public function cancelDeletion(User $user): bool
    {
        $deleted = AccountDeletionRequest::where('user_id', $user->id)
            ->whereNull('deleted_at')
            ->delete();

        if ($deleted) {
            $user->notify(new AccountDeletionGracePeriod('cancelled'));

            activity('account')
                ->performedOn($user)
                ->log('Account deletion cancelled');
        }

        return $deleted > 0;
    }

    public function processExpiredDeletions(): int
    {
        $expiredRequests = AccountDeletionRequest::whereNull('deleted_at')
            ->where('grace_period_ends_at', '<=', now())
            ->get();

        $count = 0;

        foreach ($expiredRequests as $request) {
            $user = User::find($request->user_id);

            if ($user) {
                $this->deleteUser($user);
                $request->update(['deleted_at' => now()]);
                $count++;
            }
        }

        return $count;
    }

    private function deleteUser(User $user): void
    {
        DB::transaction(function () use ($user) {
            // Revoke all tokens
            $user->tokens()->delete();

            // Delete related data
            $user->sessions()->delete();
            $user->devices()->delete();
            $user->securityAuditLogs()->delete();
            $user->emailChangeRequests()->delete();
            $user->apiTokens()->delete();

            // Delete file uploads
            Storage::disk('private')->deleteDirectory("users/{$user->id}");

            // Anonymize instead of hard delete to preserve referential integrity
            $user->update([
                'name' => 'Deleted User',
                'email' => "deleted_{$user->id}@removed.example",
                'password' => null,
                'email_verified_at' => null,
                'deleted_at' => now(),
            ]);
        });

        activity('account')
            ->withProperties(['user_id' => $user->id])
            ->log('Account deleted');
    }
}
```

```php
// app/Services/SecurityHealthService.php

namespace App\Services;

use App\Models\User;

class SecurityHealthService
{
    public function calculateScore(User $user): array
    {
        $checks = [
            'password_strength' => $this->checkPasswordStrength($user),
            '2fa_enabled' => $this->check2FA($user),
            'email_verified' => $this->checkEmailVerified($user),
            'recent_password_change' => $this->checkRecentPasswordChange($user),
            'no_suspicious_activity' => $this->checkSuspiciousActivity($user),
            'recovery_options' => $this->checkRecoveryOptions($user),
            'session_hygiene' => $this->checkSessionHygiene($user),
        ];

        $passed = collect($checks)->filter(fn ($check) => $check['passed'])->count();
        $total = count($checks);
        $score = round(($passed / $total) * 100);

        return [
            'score' => $score,
            'label' => $this->getScoreLabel($score),
            'checks' => $checks,
        ];
    }

    private function checkPasswordStrength(User $user): array
    {
        $lastChange = $user->password_changed_at ?? $user->created_at;
        $daysSinceChange = $lastChange->diffInDays(now());

        return [
            'label' => 'Strong password',
            'passed' => $user->password && strlen($user->password) >= 60, // bcrypt hash
            'recommendation' => 'Use a strong, unique password',
        ];
    }

    private function check2FA(User $user): array
    {
        return [
            'label' => 'Two-factor authentication',
            'passed' => $user->hasTwoFactorEnabled(),
            'recommendation' => 'Enable 2FA for extra security',
        ];
    }

    private function checkEmailVerified(User $user): array
    {
        return [
            'label' => 'Email verified',
            'passed' => $user->hasVerifiedEmail(),
            'recommendation' => 'Verify your email address',
        ];
    }

    private function checkRecentPasswordChange(User $user): array
    {
        $lastChange = $user->password_changed_at ?? $user->created_at;
        $daysSince = $lastChange->diffInDays(now());

        return [
            'label' => 'Recent password change',
            'passed' => $daysSince <= 90,
            'recommendation' => 'Change your password every 90 days',
        ];
    }

    private function checkSuspiciousActivity(User $user): array
    {
        $auditService = app(SecurityAuditService::class);
        $suspicious = $auditService->getSuspiciousActivity($user->id);

        return [
            'label' => 'No suspicious activity',
            'passed' => ! $suspicious['high_risk'],
            'recommendation' => 'Review recent login activity',
        ];
    }

    private function checkRecoveryOptions(User $user): array
    {
        return [
            'label' => 'Recovery options configured',
            'passed' => $user->hasTwoFactorEnabled() && $user->recovery_codes !== null,
            'recommendation' => 'Save your recovery codes in a safe place',
        ];
    }

    private function checkSessionHygiene(User $user): array
    {
        $activeSessions = $user->sessions()
            ->where('last_active_at', '>=', now()->subHours(24))
            ->count();

        return [
            'label' => 'Session hygiene',
            'passed' => $activeSessions <= 3,
            'recommendation' => 'Revoke unused sessions',
        ];
    }

    private function getScoreLabel(int $score): string
    {
        return match (true) {
            $score >= 90 => 'Excellent',
            $score >= 70 => 'Good',
            $score >= 50 => 'Fair',
            default => 'Needs attention',
        };
    }
}
```

## Password History Enforcement

```php
// app/Policies/PasswordHistoryPolicy.php

namespace App\Policies;

use App\Models\User;
use Illuminate\Support\Facades\Hash;

class PasswordHistoryPolicy
{
    public function changePassword(User $user, string $newPassword): bool
    {
        $historyCount = config('security.password.history_count', 5);

        $recentPasswords = $user->passwordHistory()
            ->orderByDesc('created_at')
            ->limit($historyCount)
            ->get();

        foreach ($recentPasswords as $record) {
            if (Hash::check($newPassword, $record->password_hash)) {
                return false;
            }
        }

        return true;
    }

    public function recordPassword(User $user, string $password): void
    {
        $user->passwordHistory()->create([
            'password_hash' => $password,
        ]);

        // Keep only last 10 entries
        $user->passwordHistory()
            ->orderByDesc('created_at')
            ->skip(10)
            ->delete();
    }
}
```
