# 09.15 — Authorization

## Purpose

Define the authorization architecture for GsmUi, covering the relationship between authentication, policies, gates, roles, and permissions.

## Authorization Layers

```text
Request
    ↓
Authentication (who are you?)
    ↓
Authorization (what can you do?)
    ↓
Layer 1: Gates (global ability checks)
    ↓
Layer 2: Policies (resource-specific checks)
    ↓
Layer 3: Roles & Permissions (RBAC)
    ↓
Access Granted or Denied
```

## Authorization Methods

| Method | Scope | Example |
|---|---|---|
| Gates | Global abilities | `Gate::define('admin', fn() => ...)` |
| Policies | Resource-specific | `ThemePolicy@update(User $user, Theme $theme)` |
| Roles | Role-based | `$user->hasRole('admin')` |
| Permissions | Granular | `$user->hasPermissionTo('themes:create')` |

## Authorization Flow

```text
User makes request
    ↓
Authenticated? → No → Redirect to login
    ↓ Yes
Gate check (e.g., is_admin)
    ↓ Pass
Policy check (e.g., can update this theme?)
    ↓ Pass
Permission check (e.g., has themes:update?)
    ↓ Pass
Role check (e.g., has admin role?)
    ↓ Pass
Access Granted
```

| Scenario | Response |
|---|---|
| Not authenticated | 401 + redirect to login |
| Gate denies | 403 Forbidden |
| Policy denies | 403 Forbidden |
| Role missing | 403 Forbidden |
| Permission missing | 403 Forbidden |

## Authorization Service

```php
// app/Services/AuthorizationService.php

class AuthorizationService
{
    public function check(User $user, string $ability, $model = null): bool
    {
        if ($model) {
            return $user->can($ability, $model);
        }

        return $user->can($ability);
    }

    public function authorize(User $user, string $ability, $model = null): void
    {
        abort_unless($this->check($user, $ability, $model), 403);
    }
}
```

## Best Practices

1. Always use policies for resource authorization
2. Use gates for global abilities (admin, supporter)
3. Combine roles and permissions for RBAC
4. Don't put authorization logic in controllers — use policies
5. Use middleware for route-level authorization
6. Use `@can` directives in Blade for UI-level authorization
7. Always return 403 for unauthorized access (not 401)
8. Log authorization failures for security monitoring

### Gates

#### Defining Gates

```php
// app/Providers/AuthServiceProvider.php

namespace App\Providers;

use Illuminate\Foundation\Support\Providers\AuthServiceProvider as ServiceProvider;
use Illuminate\Support\Facades\Gate;

class AuthServiceProvider extends ServiceProvider
{
    protected $policies = [
        \App\Models\Theme::class => \App\Policies\ThemePolicy::class,
        \App\Models\Component::class => \App\Policies\ComponentPolicy::class,
        \App\Models\Template::class => \App\Policies\TemplatePolicy::class,
        \App\Models\Product::class => \App\Policies\ProductPolicy::class,
    ];

    public function boot(): void
    {
        $this->registerPolicies();

        // Define gates
        Gate::define('admin', function ($user) {
            return $user->hasRole('admin');
        });

        Gate::define('super-admin', function ($user) {
            return $user->hasRole('super-admin');
        });

        Gate::define('supporter', function ($user) {
            return $user->hasRole('supporter') || $user->hasRole('admin');
        });

        Gate::define('creator', function ($user) {
            return $user->hasRole('creator') || $user->hasRole('admin');
        });

        Gate::define('manage-users', function ($user) {
            return $user->hasPermissionTo('users:manage');
        });

        Gate::define('manage-themes', function ($user) {
            return $user->hasPermissionTo('themes:manage');
        });

        Gate::define('manage-components', function ($user) {
            return $user->hasPermissionTo('components:manage');
        });

        Gate::define('manage-marketplace', function ($user) {
            return $user->hasPermissionTo('marketplace:manage');
        });

        Gate::define('access-billing', function ($user) {
            return $user->hasPermissionTo('billing:access');
        });

        // Before/After hooks
        Gate::before(function ($user, $ability) {
            if ($user->hasRole('super-admin')) {
                return true;
            }
        });

        Gate::after(function ($user, $ability, $result) {
            if ($result === true) {
                activity()
                    ->performedOn($user)
                    ->withProperties(['ability' => $ability])
                    ->log("Gate check passed: {$ability}");
            }
        });
    }
}
```

### Policies

#### Policy Structure

```php
// app/Policies/ThemePolicy.php

namespace App\Policies;

use App\Models\Theme;
use App\Models\User;
use Illuminate\Auth\Access\HandlesAuthorization;

class ThemePolicy
{
    use HandlesAuthorization;

    /**
     * Determine whether the user can view any themes.
     */
    public function viewAny(User $user): bool
    {
        return true; // All authenticated users can view themes
    }

    /**
     * Determine whether the user can view the theme.
     */
    public function view(User $user, Theme $theme): bool
    {
        return $user->id === $theme->user_id
            || $user->hasRole('admin')
            || $user->hasRole('supporter');
    }

    /**
     * Determine whether the user can create themes.
     */
    public function create(User $user): bool
    {
        return $user->hasRole('creator')
            || $user->hasRole('admin')
            || $user->hasPermissionTo('themes:create');
    }

    /**
     * Determine whether the user can update the theme.
     */
    public function update(User $user, Theme $theme): bool
    {
        return ($user->id === $theme->user_id && $user->hasPermissionTo('themes:update'))
            || $user->hasRole('admin')
            || $user->hasRole('supporter');
    }

    /**
     * Determine whether the user can delete the theme.
     */
    public function delete(User $user, Theme $theme): bool
    {
        return ($user->id === $theme->user_id && $user->hasPermissionTo('themes:delete'))
            || $user->hasRole('admin');
    }

    /**
     * Determine whether the user can restore the theme.
     */
    public function restore(User $user, Theme $theme): bool
    {
        return $user->hasRole('admin');
    }

    /**
     * Determine whether the user can force delete the theme.
     */
    public function forceDelete(User $user, Theme $theme): bool
    {
        return $user->hasRole('super-admin');
    }

    /**
     * Determine whether the user can publish the theme.
     */
    public function publish(User $user, Theme $theme): bool
    {
        return ($user->id === $theme->user_id && $theme->status === 'review')
            || $user->hasRole('admin');
    }

    /**
     * Determine whether the user can feature the theme.
     */
    public function feature(User $user, Theme $theme): bool
    {
        return $user->hasRole('admin');
    }

    /**
     * Get the allowed actions for the user.
     */
    public function before(User $user, string $ability): ?bool
    {
        if ($user->hasRole('super-admin')) {
            return true;
        }

        return null;
    }
}
```

```php
// app/Http/Middleware/RoleMiddleware.php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, string ...$roles)
    {
        if (! $request->user() || ! $request->user()->hasAnyRole($roles)) {
            abort(403, 'Unauthorized. Required role: ' . implode(' or ', $roles));
        }

        return $next($request);
    }
}
```

## Authorization Service

```php
// app/Services/AuthorizationService.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\Gate;

class AuthorizationService
{
    /**
     * Check if user has ability.
     */
    public function check(User $user, string $ability, $model = null): bool
    {
        if ($model) {
            return $user->can($ability, $model);
        }

        return $user->can($ability);
    }

    /**
     * Authorize user ability or abort.
     */
    public function authorize(User $user, string $ability, $model = null): void
    {
        abort_unless($this->check($user, $ability, $model), 403);
    }

    /**
     * Get all abilities for a user.
     */
    public function getUserAbilities(User $user): array
    {
        $abilities = [];

        // Check role-based abilities
        $abilities['admin'] = $user->hasRole('admin');
        $abilities['super-admin'] = $user->hasRole('super-admin');
        $abilities['supporter'] = $user->hasRole('supporter');
        $abilities['creator'] = $user->hasRole('creator');

        // Check permission-based abilities
        $abilities['manage-users'] = $user->hasPermissionTo('users:manage');
        $abilities['manage-themes'] = $user->hasPermissionTo('themes:manage');
        $abilities['manage-components'] = $user->hasPermissionTo('components:manage');
        $abilities['manage-marketplace'] = $user->hasPermissionTo('marketplace:manage');
        $abilities['access-billing'] = $user->hasPermissionTo('billing:access');

        return $abilities;
    }

    /**
     * Check multiple abilities at once.
     */
    public function checkAny(User $user, array $abilities, $model = null): bool
    {
        foreach ($abilities as $ability) {
            if ($this->check($user, $ability, $model)) {
                return true;
            }
        }

        return false;
    }

    /**
     * Check all abilities.
     */
    public function checkAll(User $user, array $abilities, $model = null): bool
    {
        foreach ($abilities as $ability) {
            if (! $this->check($user, $ability, $model)) {
                return false;
            }
        }

        return true;
    }

    /**
     * Get allowed actions for a model.
     */
    public function getAllowedActions(User $user, $model): array
    {
        $actions = ['view', 'create', 'update', 'delete', 'restore', 'forceDelete'];
        $allowed = [];

        foreach ($actions as $action) {
            if ($user->can($action, $model)) {
                $allowed[] = $action;
            }
        }

        return $allowed;
    }
}
```

### Role Hierarchy

```php
// app/Services/RoleHierarchy.php

namespace App\Services;

class RoleHierarchy
{
    protected array $hierarchy = [
        'super-admin' => ['admin', 'supporter', 'creator', 'user'],
        'admin' => ['supporter', 'creator', 'user'],
        'supporter' => ['user'],
        'creator' => ['user'],
        'user' => [],
    ];

    /**
     * Check if role has hierarchy.
     */
    public function hasRole(string $userRole, string $requiredRole): bool
    {
        if ($userRole === $requiredRole) {
            return true;
        }

        return in_array($requiredRole, $this->hierarchy[$userRole] ?? []);
    }

    /**
     * Get all roles that a role can manage.
     */
    public function getManageableRoles(string $role): array
    {
        return $this->hierarchy[$role] ?? [];
    }

    /**
     * Get role level (higher = more permissions).
     */
    public function getLevel(string $role): int
    {
        return match($role) {
            'super-admin' => 100,
            'admin' => 80,
            'supporter' => 60,
            'creator' => 40,
            'user' => 20,
            'guest' => 10,
            default => 0,
        };
    }
}
```

```php
// app/Services/PermissionRegistry.php

namespace App\Services;

class PermissionRegistry
{
    public array $categories = [
        'users' => [
            'users:view',
            'users:create',
            'users:update',
            'users:delete',
            'users:manage',
        ],
        'themes' => [
            'themes:view',
            'themes:create',
            'themes:update',
            'themes:delete',
            'themes:manage',
            'themes:publish',
            'themes:feature',
        ],
        'components' => [
            'components:view',
            'components:create',
            'components:update',
            'components:delete',
            'components:manage',
        ],
        'templates' => [
            'templates:view',
            'templates:create',
            'templates:update',
            'templates:delete',
            'templates:manage',
        ],
        'marketplace' => [
            'marketplace:view',
            'marketplace:create',
            'marketplace:update',
            'marketplace:delete',
            'marketplace:manage',
            'marketplace:publish',
        ],
        'billing' => [
            'billing:view',
            'billing:manage',
            'billing:access',
        ],
        'admin' => [
            'admin:access',
            'admin:manage',
            'admin:settings',
        ],
    ];

    /**
     * Get all permissions.
     */
    public function getAll(): array
    {
        return collect($this->categories)
            ->flatMap(fn($permissions) => $permissions)
            ->toArray();
    }

    /**
     * Get permissions by category.
     */
    public function getByCategory(string $category): array
    {
        return $this->categories[$category] ?? [];
    }

    /**
     * Get categories.
     */
    public function getCategories(): array
    {
        return array_keys($this->categories);
    }

    /**
```

```php
// app/Services/AuthorizationService.php

namespace App\Services;

use App\Models\User;

class AuthorizationService
{
    public function abilities(User $user, array $abilities, $resource = null): array
    {
        $results = [];
        foreach ($abilities as $ability) {
            $results[$ability] = $resource
                ? $user->can($ability, $resource)
                : $user->can($ability);
        }
        return $results;
    }

    public function getAllPermissions(User $user): array
    {
        $rolePermissions = $user->getAllPermissions()->pluck('name')->toArray();
        $directPermissions = $user->getDirectPermissions()->pluck('name')->toArray();
        return array_unique(array_merge($rolePermissions, $directPermissions));
    }

    public function canBulkAction(User $user, string $action, string $resourceType): bool
    {
        return match($action) {
            'delete' => $user->hasRole(['admin', 'super-admin']),
            'update' => $user->can("update-any.{$resourceType}"),
            'publish' => $user->can("publish-any.{$resourceType}"),
            default => false,
        };
    }

    public function filterByPermission($query, User $user, string $permission)
    {
        if ($user->hasRole(['admin', 'super-admin'])) {
            return $query;
        }

        return $query->where(function ($q) use ($user, $permission) {
            $q->where('user_id', $user->id)
              ->orWhereHas('permissions', fn($pq) => $pq->where('name', $permission));
        });
    }
```

```php
namespace App\Services;

class RoleHierarchy
{
    protected array $hierarchy = [
        'super-admin' => ['admin', 'supporter', 'creator', 'user'],
        'admin' => ['supporter', 'creator', 'user'],
        'supporter' => ['user'],
        'creator' => ['user'],
        'user' => [],

     * Check if role has hierarchy.
     */
    public function hasRole(string $userRole, string $requiredRole): bool
    {
        if ($userRole === $requiredRole) {
            return true;

        return in_array($requiredRole, $this->hierarchy[$userRole] ?? []);
    }

    /**
     * Get all roles that a role can manage.

    public function getManageableRoles(string $role): array
    {
        return $this->hierarchy[$role] ?? [];
    }

    /**
     * Get role level (higher = more permissions).

    public function getLevel(string $role): int
    {
        return match($role) {
            'super-admin' => 100,
            'admin' => 80,
            'supporter' => 60,
            'creator' => 40,
            'user' => 20,
            'guest' => 10,
            default => 0,
        };
```

```php
     * Get permissions by category.
     */
    public function getByCategory(string $category): array
    {
        return $this->categories[$category] ?? [];
    }

    /**
     * Get categories.
     */
    public function getCategories(): array
    {
        return array_keys($this->categories);
    }

    /**
     * Check if permission exists.
     */
    public function exists(string $permission): bool
    {
        return in_array($permission, $this->getAll());
    }
```

### Centralized Authorization Service

```php
// app/Services/AuthorizationService.php

namespace App\Services;

use App\Models\User;

class AuthorizationService
{
    public function abilities(User $user, array $abilities, $resource = null): array
    {
        $results = [];
        foreach ($abilities as $ability) {
            $results[$ability] = $resource
                ? $user->can($ability, $resource)
                : $user->can($ability);
        }
        return $results;
    }

    public function getAllPermissions(User $user): array
    {
        $rolePermissions = $user->getAllPermissions()->pluck('name')->toArray();
        $directPermissions = $user->getDirectPermissions()->pluck('name')->toArray();
        return array_unique(array_merge($rolePermissions, $directPermissions));
    }

    public function canBulkAction(User $user, string $action, string $resourceType): bool
    {
        return match($action) {
            'delete' => $user->hasRole(['admin', 'super-admin']),
            'update' => $user->can("update-any.{$resourceType}"),
            'publish' => $user->can("publish-any.{$resourceType}"),
            default => false,
        };
    }

    public function filterByPermission($query, User $user, string $permission)
    {
        if ($user->hasRole(['admin', 'super-admin'])) {
            return $query;
        }

        return $query->where(function ($q) use ($user, $permission) {
            $q->where('user_id', $user->id)
              ->orWhereHas('permissions', fn($pq) => $pq->where('name', $permission));
        });
    }
}
```

```php
// app/Policies/ThemePolicy.php

namespace App\Policies;

use App\Models\Theme;
use App\Models\User;

class ThemePolicy
{
    public function viewAny(User $user): bool
    {
        return $user->hasPermissionTo('themes:view') || $user->hasRole('admin');
    }

    public function view(User $user, Theme $theme): bool
    {
        if ($user->hasRole(['admin', 'super-admin'])) {
            return true;
        }
        if ($user->hasPermissionTo('themes:manage')) {
            return true;
        }
        if ($theme->team_id && $user->teams()->where('teams.id', $theme->team_id)->exists()) {
            return $user->hasPermissionTo('themes:view');
        }
        return $theme->user_id === $user->id && $user->hasPermissionTo('themes:view');
    }

    public function create(User $user): bool
    {
        return $user->hasPermissionTo('themes:create') && $user->hasVerifiedEmail();
    }

    public function update(User $user, Theme $theme): bool
    {
        if ($user->hasRole('admin')) {
            return true;
        }
        if ($theme->team_id && $user->teams()->where('teams.id', $theme->team_id)->exists()) {
            return $user->hasPermissionTo('themes:update');
        }
        return $theme->user_id === $user->id && $user->hasPermissionTo('themes:update');
    }

    public function delete(User $user, Theme $theme): bool
    {
        if ($user->hasRole('admin')) {
            return true;
        }
        return $theme->user_id === $user->id && $user->hasPermissionTo('themes:delete');
    }

    public function forceDelete(User $user, Theme $theme): bool
    {
        return $user->hasRole('super-admin');
    }
}
```
