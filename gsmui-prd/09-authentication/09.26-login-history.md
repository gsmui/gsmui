## Login Events

| Event | Data |
|---|---|
| `LoginSuccessful` | user_id, ip, user_agent, timestamp |
| `LoginFailed` | email, ip, user_agent, reason, timestamp |
| `LoginFromNewDevice` | user_id, device info, ip, timestamp |
| `LoginFromNewLocation` | user_id, location, ip, timestamp |

## Security Integration

| Check | Source | Action |
|---|---|---|
| Multiple failures | LoginHistory | Rate limit + alert |
| New device | UserDevices | Email notification |
| New location | LoginHistory | Email notification |
| Suspicious IP | LoginHistory | Flag for review |

## Data Retention

| Data | Retention | Action |
|---|---|---|
| Successful logins | 90 days | Archive then delete |
| Failed logins | 30 days | Delete |
| Suspicious logins | 1 year | Archive |

```php
        public readonly string $ipAddress,
        public readonly string $userAgent,
        public readonly ?string $deviceFingerprint,
        public readonly ?string $sessionId,

    {
```

### Login Event Dispatcher

```php
// app/Services/LoginEventDispatcher.php

class LoginEventDispatcher
{
    public function __construct(
        private LoginHistoryService $historyService,
        private SecurityEventDispatcher $securityDispatcher,
        private NotificationService $notificationService,
    ) {}

    public function dispatchSuccessful(LoginSuccessful $event): void
    {
        // Record in login history
        $this->historyService->recordLogin(
            user: $event->user,
            success: true,
            context: [
                'ip' => $event->ipAddress,
                'user_agent' => $event->userAgent,
                'device_fingerprint' => $event->deviceFingerprint,
                'session_id' => $event->sessionId,
                'login_method' => $event->loginMethod,
                'two_factor_used' => $event->twoFactorUsed,
                'two_factor_method' => $event->twoFactorMethod,
                'is_new_device' => $event->isNewDevice,
                'is_new_location' => $event->isNewLocation,
                'risk_score' => $event->riskScore,
            ]
        );

        // Dispatch security events if needed
        if ($event->isNewDevice) {
            $this->securityDispatcher->handleNewDeviceLogin($event);
        }

        if ($event->isNewLocation) {
            $this->securityDispatcher->handleNewLocationLogin($event);
        }

        if ($event->riskScore >= 60) {
            $this->securityDispatcher->handleSuspiciousLogin($event);
        }

        // Send notifications
        if ($event->isNewDevice || $event->isNewLocation) {
            $this->notificationService->notifyNewLogin($event->user, $event);
        }
    }

    public function dispatchFailed(LoginFailed $event): void
    {
        // Find or create user reference
        $user = User::where('email', $event->email)->first();

        // Record in login history
        $this->historyService->recordLogin(
            user: $user,
            success: false,
            failureReason: $event->failureReason,
            context: [
                'ip' => $event->ipAddress,
                'user_agent' => $event->userAgent,
                'device_fingerprint' => $event->deviceFingerprint,
                'failure_category' => $event->failureCategory,
            ]
        );

        // Update failure counters
        $this->updateFailureCounters($event);

        // Check for brute force
        if ($event->failureCountForIp >= 5) {
            $this->securityDispatcher->handleBruteForce($event);
        }

        if ($event->failureCountForEmail >= 10) {
            $this->securityDispatcher->handleCredentialStuffing($event);
        }
    }

    private function updateFailureCounters(LoginFailed $event): void
    {
        $ipKey = "login_failures:ip:{$event->ipAddress}";
        $emailKey = "login_failures:email:{$event->email}";

        Cache::increment($ipKey);
        Cache::increment($emailKey);

        Cache::put($ipKey, Cache::get($ipKey, 0), now()->addMinutes(15));
        Cache::put($emailKey, Cache::get($emailKey, 0), now()->addMinutes(15));
    }
}
```

## Enhanced Login History Service

### Service Implementation

```php
// app/Services/LoginHistoryService.php

class LoginHistoryService
{
    private const SUSPICIOUS_FAILURE_THRESHOLD = 5;
    private const NEW_DEVICE_WINDOW_DAYS = 90;
    private const NEW_LOCATION_WINDOW_DAYS = 30;

    public function recordLogin(
        ?User $user,
        bool $success,
        ?string $failureReason = null,
        array $context = [],
    ): LoginHistory {
        $request = request();
        $userAgent = $context['user_agent'] ?? $request->userAgent();
        $ip = $context['ip'] ?? $request->ip();

        $parsed = $this->parseUserAgent($userAgent);
        $geo = $this->getGeoLocation($ip);
        $deviceFingerprint = $context['device_fingerprint'] ?? null;

        $isNewDevice = $deviceFingerprint
            ? $this->isNewDevice($user, $deviceFingerprint)
            : false;

        $isNewLocation = $geo
            ? $this->isNewLocation($user, $geo)
            : false;

        $riskScore = 0;
        if ($user) {
            $scorer = app(LoginRiskScorer::class);
            $riskScore = $scorer->calculateScore($user, $request);
        }

        $loginHistory = LoginHistory::create([
            'user_id' => $user?->id,
            'email' => $user?->email ?? $context['email'] ?? '',
            'successful' => $success,
            'ip_address' => $ip,
            'user_agent' => $userAgent,
            'browser' => $parsed['browser'],
            'browser_version' => $parsed['browser_version'],
            'os' => $parsed['os'],
            'os_version' => $parsed['os_version'],
            'device' => $parsed['device'],
            'device_type' => $parsed['device_type'],
            'location_country' => $geo['country_code'] ?? null,
            'location_region' => $geo['region'] ?? null,
            'location_city' => $geo['city'] ?? null,
            'location_latitude' => $geo['latitude'] ?? null,
            'location_longitude' => $geo['longitude'] ?? null,
            'location_timezone' => $geo['timezone'] ?? null,
            'is_vpn' => $this->isVpn($ip),
            'is_proxy' => $this->isProxy($ip),
            'is_tor' => $this->isTor($ip),
            'is_known_device' => ! $isNewDevice,
            'risk_score' => $riskScore,
            'failure_reason' => $failureReason,
            'failure_category' => $this->categorizeFailure($failureReason),
            'two_factor_used' => $context['two_factor_used'] ?? false,
            'two_factor_method' => $context['two_factor_method'] ?? null,
            'login_method' => $context['login_method'] ?? 'password',
            'session_id' => $context['session_id'] ?? null,
            'device_fingerprint' => $deviceFingerprint,
            'is_new_device' => $isNewDevice,
            'is_new_location' => $isNewLocation,
            'blocked_by_rate_limit' => $context['blocked_by_rate_limit'] ?? false,
            'blocked_by_ip_blocklist' => $context['blocked_by_ip_blocklist'] ?? false,
            'metadata' => $context['metadata'] ?? null,
        ]);

        // Update aggregates
        if ($user) {
            $this->updateAggregates($user, $success, $riskScore);
        }

        return $loginHistory;
    }

    public function getSuspiciousLogins(User $user, int $days = 7): Collection
    {
        return LoginHistory::where('user_id', $user->id)
            ->where('created_at', '>=', now()->subDays($days))
            ->where(function ($query) {
                $query->where('successful', false)
                    ->orWhere('risk_score', '>=', 60)
                    ->orWhere('is_new_device', true)
                    ->orWhere('is_new_location', true)
                    ->orWhere('is_vpn', true)
                    ->orWhere('is_tor', true);
            })
            ->orderByDesc('created_at')
            ->get();
    }

    public function getLoginHistory(
        User $user,
        int $perPage = 25,
        ?string $filter = null,
    ): LengthAwarePaginator {
        $query = LoginHistory::where('user_id', $user->id);

        if ($filter === 'successful') {
            $query->where('successful', true);
        } elseif ($filter === 'failed') {
            $query->where('successful', false);
        } elseif ($filter === 'new_device') {
            $query->where('is_new_device', true);
        } elseif ($filter === 'new_location') {
            $query->where('is_new_location', true);
        } elseif ($filter === 'suspicious') {
            $query->where('risk_score', '>=', 60);
        }

        return $query->orderByDesc('created_at')->paginate($perPage);
    }

    public function getFailedLoginAttempts(string $email, int $minutes = 15): int
    {
        return LoginHistory::where('email', $email)
            ->where('successful', false)
            ->where('created_at', '>=', now()->subMinutes($minutes))
            ->count();
    }

    public function getFailedLoginAttemptsByIp(string $ip, int $minutes = 15): int
    {
        return LoginHistory::where('ip_address', $ip)
            ->where('successful', false)
            ->where('created_at', '>=', now()->subMinutes($minutes))
            ->count();
    }

    public function hasSuspiciousActivity(User $user): bool
    {
        $recentFailures = $this->getFailedLoginAttempts($user->email, 60);
        $recentSuspicious = LoginHistory::where('user_id', $user->id)
            ->where('created_at', '>=', now()->subHours(24))
            ->where('risk_score', '>=', 80)
            ->count();

        return $recentFailures >= self::SUSPICIOUS_FAILURE_THRESHOLD
            || $recentSuspicious > 0;
    }

    public function getLoginStatistics(User $user, int $days = 30): array
    {
        $startDate = now()->subDays($days);

        $stats = LoginHistory::where('user_id', $user->id)
            ->where('created_at', '>=', $startDate)
            ->selectRaw('
                COUNT(*) as total,
                SUM(CASE WHEN successful = 1 THEN 1 ELSE 0 END) as successful,
                SUM(CASE WHEN successful = 0 THEN 1 ELSE 0 END) as failed,
                COUNT(DISTINCT ip_address) as unique_ips,
                COUNT(DISTINCT device_fingerprint) as unique_devices,
                COUNT(DISTINCT location_country) as unique_countries,
                AVG(risk_score) as avg_risk_score,
                MAX(risk_score) as max_risk_score
            ')
            ->first();

        $byDevice = LoginHistory::where('user_id', $user->id)
            ->where('created_at', '>=', $startDate)
            ->where('successful', true)
            ->selectRaw('device_type, COUNT(*) as count')
            ->groupBy('device_type')
            ->pluck('count', 'device_type')
            ->toArray();

        $byCountry = LoginHistory::where('user_id', $user->id)
            ->where('created_at', '>=', $startDate)
            ->where('successful', true)
            ->selectRaw('location_country, COUNT(*) as count')
            ->groupBy('location_country')
            ->orderByDesc('count')
            ->limit(10)
            ->pluck('count', 'location_country')
            ->toArray();

        $byHour = LoginHistory::where('user_id', $user->id)
            ->where('created_at', '>=', $startDate)
            ->where('successful', true)
            ->selectRaw('HOUR(created_at) as hour, COUNT(*) as count')
            ->groupBy('hour')
            ->pluck('count', 'hour')
            ->toArray();

        return [
            'summary' => $stats,
            'by_device_type' => $byDevice,
            'by_country' => $byCountry,
            'by_hour' => $byHour,
            'period_days' => $days,
        ];
    }

    private function isNewDevice(User $user, string $fingerprint): bool
    {
        if (! $user) return false;

        return ! LoginHistory::where('user_id', $user->id)
            ->where('device_fingerprint', $fingerprint)
            ->where('successful', true)
            ->where('created_at', '>=', now()->subDays(self::NEW_DEVICE_WINDOW_DAYS))
            ->exists();
    }

    private function isNewLocation(User $user, array $geo): bool
    {
        if (! $user) return false;

        return ! LoginHistory::where('user_id', $user->id)
            ->where('location_country', $geo['country_code'] ?? '')
            ->where('successful', true)
            ->where('created_at', '>=', now()->subDays(self::NEW_LOCATION_WINDOW_DAYS))
            ->exists();
    }

    private function parseUserAgent(string $userAgent): array
    {
        $parser = app(UserAgentParser::class);
        return $parser->parse($userAgent);
    }

    private function getGeoLocation(string $ip): ?array
    {
        $service = app(GeoLocationService::class);
        return $service->lookup($ip);
    }

    private function isVpn(string $ip): bool
    {
        return app(ThreatIntelligenceService::class)->isVpn($ip);
    }

    private function isProxy(string $ip): bool
    {
        return app(ThreatIntelligenceService::class)->isProxy($ip);
    }

    private function isTor(string $ip): bool
    {
        return app(ThreatIntelligenceService::class)->isTor($ip);
    }

    private function categorizeFailure(?string $reason): ?string
    {
        return match (true) {
            str_contains($reason ?? '', 'password') => 'invalid_password',
            str_contains($reason ?? '', 'email') => 'invalid_email',
            str_contains($reason ?? '', 'locked') => 'account_locked',
            str_contains($reason ?? '', 'suspended') => 'account_suspended',
            str_contains($reason ?? '', 'verified') => 'email_not_verified',
            str_contains($reason ?? '', '2fa') => '2fa_invalid',
            str_contains($reason ?? '', 'rate') => 'rate_limited',
            default => 'unknown',
        };
    }

    private function updateAggregates(User $user, bool $success, int $riskScore): void
    {
        $date = now()->toDateString();

        LoginHistoryAggregate::updateOrCreate(
            ['user_id' => $user->id, 'date' => $date],
            [
                'total_logins' => DB::raw('total_logins + 1'),
                'successful_logins' => $success ? DB::raw('successful_logins + 1') : DB::raw('successful_logins'),
                'failed_logins' => ! $success ? DB::raw('failed_logins + 1') : DB::raw('failed_logins'),
                'avg_risk_score' => DB::raw("(avg_risk_score * (total_logins) + {$riskScore}) / (total_logins + 1)"),
                'max_risk_score' => DB::raw("GREATEST(max_risk_score, {$riskScore})"),
            ]
        );
    }
}
```

## User Agent Parsing

### Parser Implementation

```php
// app/Services/UserAgentParser.php

class UserAgentParser
{
    private const BROWSERS = [
        'Chrome' => '/Chrome\/([0-9.]+)/',
        'Firefox' => '/Firefox\/([0-9.]+)/',
        'Safari' => '/Version\/([0-9.]+).*Safari/',
        'Edge' => '/Edg\/([0-9.]+)/',
        'Opera' => '/Opera\/([0-9.]+)/',
        'Samsung' => '/SamsungBrowser\/([0-9.]+)/',
        'UC Browser' => '/UCBrowser\/([0-9.]+)/',
    ];

    private const OS = [
        'Windows' => '/Windows NT ([0-9.]+)/',
        'macOS' => '/Mac OS X ([0-9._]+)/',
        'Linux' => '/Linux/',
        'Android' => '/Android ([0-9.]+)/',
        'iOS' => '/iPhone OS ([0-9_]+)/',
        'Chrome OS' => '/CrOS/',
    ];

    private const DEVICES = [
        'iPhone' => '/iPhone/',
        'iPad' => '/iPad/',
        'Android Phone' => '/Android.*Mobile/',
        'Android Tablet' => '/Android(?!.*Mobile)/',
        'Windows Phone' => '/Windows Phone/',
        'BlackBerry' => '/BlackBerry/',
    ];

    public function parse(string $userAgent): array
    {
        $browser = $this->matchPattern($userAgent, self::BROWSERS);
        $os = $this->matchPattern($userAgent, self::OS);
        $device = $this->matchPattern($userAgent, self::DEVICES);

        $deviceType = $this->determineDeviceType($userAgent, $device);

        return [
            'browser' => $browser['name'] ?? 'Unknown',
            'browser_version' => $browser['version'] ?? null,
            'os' => $os['name'] ?? 'Unknown',
            'os_version' => $os['version'] ?? null,
            'device' => $device ?? 'Unknown',
            'device_type' => $deviceType,
            'raw' => $userAgent,
        ];
    }

    private function matchPattern(string $userAgent, array $patterns): ?array
    {
        foreach ($patterns as $name => $pattern) {
            if (preg_match($pattern, $userAgent, $matches)) {
                return [
                    'name' => $name,
                    'version' => $matches[1] ?? null,
                ];
            }
        }
        return null;
    }

    private function determineDeviceType(string $ua, ?string $device): string
    {
        if (in_array($device, ['iPhone', 'Android Phone', 'Windows Phone', 'BlackBerry'])) {
            return 'mobile';
        }

        if ($device === 'iPad' || $device === 'Android Tablet') {
            return 'tablet';
        }

        if (preg_match('/bot|crawler|spider|curl|wget|python/i', $ua)) {
            return 'bot';
        }

        return 'desktop';
    }
```

```php
// app/Services/GeoLocationService.php

class GeoLocationService
{
    private const CACHE_TTL = 86400; // 24 hours

    public function lookup(string $ip): ?array
    {
        if ($this->isPrivateIp($ip)) {
            return $this->getLocalhostGeo();
        }

        $cached = Cache::get("geo:{$ip}");
        if ($cached) {
            return $cached;
        }

        $geo = $this->performLookup($ip);

        if ($geo) {
            Cache::put("geo:{$ip}", $geo, self::CACHE_TTL);
        }

        return $geo;
    }

    private function performLookup(string $ip): ?array
    {
        try {
            $reader = new Reader(storage_path('app/geoip/GeoLite2-City.mmdb'));
            $record = $reader->city($ip);

            return [
                'country_code' => $record->country->isoCode,
                'country_name' => $record->country->name,
                'region' => $record->mostSpecificSubdivision->name,
                'city' => $record->city->name,
                'latitude' => $record->location->latitude,
                'longitude' => $record->location->longitude,
                'timezone' => $record->location->timeZone,
            ];
        } catch (\Exception $e) {
            return $this->fallbackLookup($ip);
        }
    }

    private function fallbackLookup(string $ip): ?array
    {
        try {
            $response = Http::timeout(5)
                ->get("http://ip-api.com/json/{$ip}");

            if ($response->successful()) {
                $data = $response->json();
                return [
                    'country_code' => $data['countryCode'],
                    'country_name' => $data['country'],
                    'region' => $data['regionName'],
                    'city' => $data['city'],
                    'latitude' => $data['lat'],
                    'longitude' => $data['lon'],
                    'timezone' => $data['timezone'],
                ];
            }
        } catch (\Exception $e) {
            // Silently fail
        }

        return null;
    }

    private function isPrivateIp(string $ip): bool
    {
        return filter_var($ip, FILTER_VALIDATE_IP, FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE) === false;
    }

    private function getLocalhostGeo(): array
    {
        return [
            'country_code' => 'US',
            'country_name' => 'United States',
            'region' => 'Local',
            'city' => 'Localhost',
            'latitude' => 0,
            'longitude' => 0,
            'timezone' => 'UTC',
        ];
    }

    public function getDistance(float $lat1, float $lon1, float $lat2, float $lon2): float
    {
        $earthRadius = 6371;

        $dLat = deg2rad($lat2 - $lat1);
        $dLon = deg2rad($lon2 - $lon1);

        $a = sin($dLat / 2) * sin($dLat / 2)
            + cos(deg2rad($lat1)) * cos(deg2rad($lat2))
            * sin($dLon / 2) * sin($dLon / 2);

        $c = 2 * atan2(sqrt($a), sqrt(1 - $a));

        return $earthRadius * $c;
    }
}
```

```php
// app/Jobs/CleanupLoginHistory.php

class CleanupLoginHistory implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $timeout = 300;
    public int $tries = 3;

    public function handle(): void
    {
        $this->cleanupFailedLogins();
        $this->cleanupSuccessfulLogins();
        $this->cleanupSuspiciousLogins();
        $this->cleanupAggregates();
        $this->cleanupOrphanedRecords();
        $this->logCleanupStats();
    }

    private function cleanupFailedLogins(): void
    {
        $deleted = LoginHistory::where('successful', false)
            ->where('created_at', '<', now()->subDays(30))
            ->delete();

        Log::info("Cleaned up {$deleted} failed login records");
    }

    private function cleanupSuccessfulLogins(): void
    {
        $deleted = LoginHistory::where('successful', true)
            ->where('created_at', '<', now()->subDays(90))
            ->delete();

        Log::info("Cleaned up {$deleted} successful login records");
    }

    private function cleanupSuspiciousLogins(): void
    {
        $deleted = LoginHistory::where('risk_score', '>=', 60)
            ->where('created_at', '<', now()->subYear())
            ->delete();

        Log::info("Cleaned up {$deleted} suspicious login records");
    }

    private function cleanupAggregates(): void
    {
        LoginHistoryAggregate::where('date', '<', now()->subMonths(6))
            ->delete();
    }

    private function cleanupOrphanedRecords(): void
    {
        $deleted = LoginHistory::whereNull('user_id')
            ->where('created_at', '<', now()->subDays(7))
            ->delete();

        Log::info("Cleaned up {$deleted} orphaned login records");
    }

    private function logCleanupStats(): void
    {
        $remaining = LoginHistory::count();
        Log::info("Login history cleanup complete. Records remaining: {$remaining}");
    }
```

```php
                    'is_new_device' => $login->is_new_device,
                    'is_new_location' => $login->is_new_location,
                ];
            }
        }

        return $anomalies;
    }
```

```php
// app/Services/BotDetectionService.php

class BotDetectionService
{
    private const BOT_PATTERNS = [
        '/bot/i', '/crawler/i', '/spider/i', '/scraper/i',
        '/curl/i', '/wget/i', '/python/i', '/java/i',
        '/go-http/i', '/php/i', '/ruby/i',
    ];

    public function isBot(Request $request): bool
    {
        $userAgent = $request->userAgent();

        if (empty($userAgent)) {
            return true;
        }

        foreach (self::BOT_PATTERNS as $pattern) {
            if (preg_match($pattern, $userAgent)) {
                return true;
            }
        }

        if ($this->isHeadlessBrowser($request)) {
            return true;
        }

        if ($this->hasUnnaturalTiming($request)) {
            return true;
        }

        return false;
    }

    private function isHeadlessBrowser(Request $request): bool
    {
        $ua = $request->userAgent();

        if (str_contains($ua, 'HeadlessChrome')) return true;
        if (str_contains($ua, 'PhantomJS')) return true;
        if (str_contains($ua, 'SlimerJS')) return true;

        $navigatorProps = $request->header('Sec-CH-UA-Platform');
        if (empty($navigatorProps)) return true;

        return false;
    }

    private function hasUnnaturalTiming(Request $request): bool
    {
        $ip = $request->ip();
        $key = "login_timing:{$ip}";

        $times = Cache::get($key, []);
        $times[] = microtime(true);

        // Keep last 10 attempts
        $times = array_slice($times, -10);

        Cache::put($key, $times, now()->addMinutes(5));

        if (count($times) < 3) return false;

        $intervals = [];
        for ($i = 1; $i < count($times); $i++) {
            $intervals[] = $times[$i] - $times[$i - 1];
        }

        // Check for suspiciously regular intervals (bot behavior)
        if (count($intervals) >= 3) {
            $avgInterval = array_sum($intervals) / count($intervals);
            $variance = 0;
            foreach ($intervals as $interval) {
                $variance += pow($interval - $avgInterval, 2);
            }
            $variance /= count($intervals);
            $stdDev = sqrt($variance);

            // If standard deviation is very low, likely a bot
            if ($stdDev < 0.1 && $avgInterval < 2.0) {
                return true;
            }
        }

        return false;
    }
}
```

### Geographic Anomaly Detection

```php
// app/Services/GeographicAnomalyDetector.php

class GeographicAnomalyDetector
{
    private const MAX_TRAVEL_SPEED_KMH = 900;
    private const MIN_DISTANCE_KM = 100;

    public function detectAnomalies(User $user, float $newLat, float $newLng, Carbon $loginTime): array
    {
        $lastLogins = LoginHistory::where('user_id', $user->id)
            ->where('successful', true)
            ->whereNotNull('location_latitude')
            ->where('created_at', '>=', now()->subDays(30))
            ->orderByDesc('created_at')
            ->limit(5)
            ->get();

        $anomalies = [];

        foreach ($lastLogins as $lastLogin) {
            $distance = $this->haversine(
                $lastLogin->location_latitude,
                $lastLogin->location_longitude,
                $newLat,
                $newLng
            );

            if ($distance < self::MIN_DISTANCE_KM) {
                continue;
            }

            $timeDiffHours = $loginTime->diffInMinutes($lastLogin->created_at) / 60;

            if ($timeDiffHours < 0.01) {
                $timeDiffHours = 0.01;
            }

            $speed = $distance / $timeDiffHours;

            if ($speed > self::MAX_TRAVEL_SPEED_KMH) {
                $anomalies[] = [
                    'type' => 'impossible_travel',
                    'distance_km' => round($distance, 2),
                    'speed_kmh' => round($speed, 2),
                    'time_diff_hours' => round($timeDiffHours, 2),
                    'from_location' => [
                        'lat' => $lastLogin->location_latitude,
                        'lng' => $lastLogin->location_longitude,
                        'city' => $lastLogin->location_city,
                        'country' => $lastLogin->location_country,
                    ],
                    'to_location' => [
                        'lat' => $newLat,
                        'lng' => $newLng,
                    ],
                    'last_login_at' => $lastLogin->created_at,
                    'severity' => $this->calculateSeverity($speed, $distance),
                ];
            }
        }

        return $anomalies;
    }

    private function calculateSeverity(float $speed, float $distance): string
    {
        if ($speed > 5000 || $distance > 10000) return 'critical';
        if ($speed > 2000 || $distance > 5000) return 'high';
        if ($speed > self::MAX_TRAVEL_SPEED_KMH) return 'medium';
        return 'low';
    }

    private function haversine(float $lat1, float $lon1, float $lat2, float $lon2): float
    {
        $earthRadius = 6371;
        $dLat = deg2rad($lat2 - $lat1);
        $dLon = deg2rad($lon2 - $lon1);

        $a = sin($dLat / 2) * sin($dLat / 2)
            + cos(deg2rad($lat1)) * cos(deg2rad($lat2))
            * sin($dLon / 2) * sin($dLon / 2);

        $c = 2 * atan2(sqrt($a), sqrt(1 - $a));

        return $earthRadius * $c;
    }
}
```

## Rate Limiting Integration

### Login History Rate Limits

| Action | Limit | Window | Key |
|---|---|---|---|
| View login history | 60 requests | Per minute | User ID |
| Export login history | 5 requests | Per hour | User ID |
| Clear login history | 1 request | Per day | User ID |
| Delete single record | 10 requests | Per minute | User ID |
| View statistics | 30 requests | Per minute | User ID |

### Rate Limit Implementation

```php
// app/Http/Middleware/LoginHistoryRateLimit.php

class LoginHistoryRateLimit
{
    public function handle(Request $request, Closure $next): mixed
    {
        $user = $request->user();
        $action = $request->route()->getActionMethod();

        $limits = [
            'index' => ['max' => 60, 'minutes' => 1],
            'export' => ['max' => 5, 'minutes' => 60],
            'clear' => ['max' => 1, 'minutes' => 1440],
            'destroy' => ['max' => 10, 'minutes' => 1],
            'stats' => ['max' => 30, 'minutes' => 1],
        ];

        $limit = $limits[$action] ?? ['max' => 60, 'minutes' => 1];
        $key = "login_history:{$action}:{$user->id}";

        if (RateLimiter::tooManyAttempts($key, $limit['max'])) {
            $seconds = RateLimiter::availableIn($key);

            return response()->json([
                'message' => 'Too many requests. Please try again later.',
                'retry_after' => $seconds,
            ], 429);
        }

        RateLimiter::hit($key, $limit['minutes'] * 60);

        return $next($request);
    }
}
```

## Event Listeners

### Login History Event Listeners

```php
// app/Listeners/RecordLoginListener.php

class RecordLoginListener
{
    public function __construct(
        private LoginHistoryService $historyService,
    ) {}

    public function handleLoginSuccessful(LoginSuccessful $event): void
    {
        $this->historyService->recordLogin(
            user: $event->user,
            success: true,
            context: [
                'ip' => $event->ipAddress,
                'user_agent' => $event->userAgent,
                'device_fingerprint' => $event->deviceFingerprint,
                'session_id' => $event->sessionId,
                'login_method' => $event->loginMethod,
                'two_factor_used' => $event->twoFactorUsed,
                'two_factor_method' => $event->twoFactorMethod,
                'is_new_device' => $event->isNewDevice,
                'is_new_location' => $event->isNewLocation,
                'risk_score' => $event->riskScore,
            ]
        );
    }

    public function handleLoginFailed(LoginFailed $event): void
    {
        $user = User::where('email', $event->email)->first();

        $this->historyService->recordLogin(
            user: $user,
            success: false,
            failureReason: $event->failureReason,
            context: [
                'ip' => $event->ipAddress,
                'user_agent' => $event->userAgent,
                'device_fingerprint' => $event->deviceFingerprint,
                'failure_category' => $event->failureCategory,
            ]
        );
    }
}

// app/Listeners/SendLoginNotificationListener.php

class SendLoginNotificationListener
{
    public function handleLoginSuccessful(LoginSuccessful $event): void
    {
        $user = $event->user;
        $preferences = UserSecurityPreference::forUser($user);

        if (! $preferences->email_new_device && ! $event->isNewDevice) {
            return;
        }

        if (! $preferences->email_new_location && ! $event->isNewLocation) {
            return;
        }

        if ($event->isNewDevice || $event->isNewLocation) {
            Notification::send($user, new NewLoginNotification(
                user: $user,
                ipAddress: $event->ipAddress,
                userAgent: $event->userAgent,
                isNewDevice: $event->isNewDevice,
                isNewLocation: $event->isNewLocation,
            ));
        }
    }
}

// app/Listeners/UpdateDeviceTrustListener.php

class UpdateDeviceTrustListener
{
    public function handleLoginSuccessful(LoginSuccessful $event): void
    {
        if (! $event->isNewDevice) {
            return;
        }

        $trustedDeviceService = app(TrustedDeviceService::class);

        if ($trustedDeviceService->isTrusted($event->user, request())) {
            return;
        }

        // Auto-trust device if risk score is low
        if ($event->riskScore < 30) {
            $trustedDeviceService->trust($event->user, request());
        }
    }
}
```
