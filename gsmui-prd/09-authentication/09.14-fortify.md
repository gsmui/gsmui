# 09.14 — Fortify

## Purpose

Define the Laravel Fortify configuration and integration for GsmUi, covering all authentication features, views, and customizations.

## Fortify Overview

Fortify provides the authentication scaffolding for GsmUi:

* Registration
* Login
* Password Reset
* Email Verification
* Password Confirmation
* Two-Factor Authentication

```php
public function create(Request $request): User
{
    Validator::make($request->all(), [
        'name' => ['required', 'string', 'max:255'],
        'email' => ['required', 'string', 'email', 'max:255', 'unique:users'],
```

### Fortify Service Provider

```php
// app/Providers/FortifyServiceProvider.php

namespace App\Providers;

use App\Actions\Fortify\CreateNewUser;
use App\Actions\Fortify\ResetUserPassword;
use App\Actions\Fortify\UpdateUserPassword;
use App\Actions\Fortify\UpdateUserProfileInformation;
use Illuminate\Cache\RateLimiting\Limit;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\ServiceProvider;
use Laravel\Fortify\Fortify;

class FortifyServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        //
    }

    public function boot(): void
    {
        $this->configureRateLimiting();

        $this->configureViews();

        $this->configureActions();

        $this->configureLogin();
    }

    protected function configureRateLimiting(): void
    {
        RateLimiter::for('login', function (Request $request) {
            return Limit::perMinute(5)
                ->by($request->email . $request->ip());
        });

        RateLimiter::for('two-factor', function (Request $request) {
            return Limit::perMinute(5)
                ->by($request->email . $request->ip());
        });

        RateLimiter::for('password-reset', function (Request $request) {
            return Limit::perMinute(3)
                ->by($request->email);
        });
    }

    protected function configureViews(): void
    {
        Fortify::viewPath(viewPath: resource_path('views/auth'));

        Fortify::loginView(fn() => view('auth.login'));
        Fortify::registerView(fn() => view('auth.register'));
        Fortify::requestPasswordResetLinkView(fn() => view('auth.forgot-password'));
        Fortify::resetPasswordView(fn($request) => view('auth.reset-password', ['request' => $request]));
        Fortify::confirmPasswordView(fn() => view('auth.confirm-password'));
        Fortify::twoFactorChallengeView(fn() => view('auth.two-factor-challenge'));
        Fortify::verifyEmailView(fn() => view('auth.verify-email'));
        Fortify::profileInformationView(fn() => view('auth.profile-information'));
        Fortify::updatePasswordView(fn() => view('auth.update-password'));
    }

    protected function configureActions(): void
    {
        Fortify::createUsersUsing(CreateNewUser::class);
        Fortify::updateUserProfileInformationUsing(UpdateUserProfileInformation::class);
        Fortify::updateUserPasswordsUsing(UpdateUserPassword::class);
        Fortify::resetUserPasswordsUsing(ResetUserPassword::class);
    }

    protected function configureLogin(): void
    {
        Fortify::authenticateUsing(function (Request $request) {
            $user = \App\Models\User::where('email', $request->email)->first();

            if ($user && \Hash::check($request->password, $user->password)) {
                return $user;
            }

            return null;
        });
    }
}
```

```php
// app/Actions/Fortify/CreateNewUser.php

namespace App\Actions\Fortify;

use App\Models\User;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Validator;
use Illuminate\Validation\Rules\Password;
use Laravel\Fortify\Contracts\CreatesNewUsers;

class CreateNewUser implements CreatesNewUsers
{
    public function create(array $input): User
    {
        Validator::make($input, [
            'name' => ['required', 'string', 'max:255'],
            'email' => ['required', 'string', 'email', 'max:255', 'unique:users'],
            'password' => ['required', 'string', 'confirmed', Password::min(8)->mixedCase()->numbers()->symbols()],
            'terms' => ['accepted'],
        ])->validate();

        $user = User::create([
            'name' => $input['name'],
            'email' => $input['email'],
            'password' => Hash::make($input['password']),
        ]);

        $user->assignRole('user');

        activity()
            ->performedOn($user)
            ->withProperties([
                'ip' => request()->ip(),
                'user_agent' => request()->userAgent(),
            ])
            ->log('User registered');

        return $user;
    }
}
```

### Update User Password (Extended)

```php
// app/Actions/Fortify/UpdateUserPassword.php

namespace App\Actions\Fortify;

use App\Models\User;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Validator;
use Illuminate\Validation\Rules\Password;
use Laravel\Fortify\Contracts\UpdatesUserPasswords;

class UpdateUserPassword implements UpdatesUserPasswords
{
    public function update(User $user, array $input): void
    {
        Validator::make($input, [
            'current_password' => ['required', 'string'],
            'password' => ['required', 'string', 'confirmed', Password::min(8)->mixedCase()->numbers()->symbols()],
        ])->validate();

        if (! Hash::check($input['current_password'], $user->password)) {
            throw \Illuminate\Validation\ValidationException::withMessages([
                'current_password' => ['The provided password does not match your current password.'],
            ]);
        }

        $user->forceFill([
            'password' => Hash::make($input['password']),
        ])->save();

        // Revoke all other sessions on password change
        $user->sessions()
            ->where('id', '!=', session()->getId())
            ->delete();

        // Revoke all API tokens
        $user->tokens()->delete();

        activity()
            ->performedOn($user)
            ->withProperties([
                'ip' => request()->ip(),
            ])
            ->log('Password changed');
    }
}
```

### Update Profile Information (Extended)

```php
// app/Actions/Fortify/UpdateUserProfileInformation.php

namespace App\Actions\Fortify;

use App\Models\User;
use Illuminate\Support\Facades\Validator;
use Laravel\Fortify\Contracts\UpdatesUserProfileInformation;

class UpdateUserProfileInformation implements UpdatesUserProfileInformation
{
    public function update(User $user, array $input): void
    {
        Validator::make($input, [
            'name' => ['required', 'string', 'max:255'],
            'email' => ['required', 'string', 'email', 'max:255', 'unique:users,email,' . $user->id],
            'avatar' => ['nullable', 'image', 'max:2048'],
        ])->validate();

        if (isset($input['avatar'])) {
            $path = $input['avatar']->store('avatars', 'public');
            $user->update(['avatar' => $path]);
        }

        if ($input['email'] !== $user->email) {
            $user->forceFill([
                'name' => $input['name'],
                'email' => $input['email'],
                'email_verified_at' => null,
            ])->save();

            $user->sendEmailVerificationNotification();

            activity()
                ->performedOn($user)
                ->withProperties([
                    'old_email' => $user->email,
                    'new_email' => $input['email'],
                ])
                ->log('Email changed (verification pending)');
        } else {
            $user->forceFill([
                'name' => $input['name'],
            ])->save();

            activity()
                ->performedOn($user)
                ->log('Profile information updated');
        }
    }
}
```

### Reset User Password (Extended)

```php
// app/Actions/Fortify/ResetUserPassword.php

namespace App\Actions\Fortify;

use App\Models\User;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Validator;
use Illuminate\Validation\Rules\Password;
use Laravel\Fortify\Contracts\ResetsUserPasswords;

class ResetUserPassword implements ResetsUserPasswords
{
    public function reset(User $user, array $input): void
    {
        Validator::make($input, [
            'password' => ['required', 'string', 'confirmed', Password::min(8)->mixedCase()->numbers()->symbols()],
        ])->validate();

        $user->forceFill([
            'password' => Hash::make($input['password']),
        ])->save();

        // Revoke all sessions except current
        $user->sessions()
            ->where('id', '!=', session()->getId())
            ->delete();

        // Revoke all API tokens
        $user->tokens()->delete();

        activity()
            ->performedOn($user)
            ->withProperties([
                'ip' => request()->ip(),
            ])
            ->log('Password reset');
    }
}
```

## Two-Factor Authentication

### 2FA Setup Flow

```text
1. User goes to Settings → Security → Two-Factor Authentication
2. User clicks "Enable Two-Factor Authentication"
3. Password confirmation required
4. QR code displayed (scan with authenticator app)
5. User enters code from authenticator app
6. 2FA enabled
7. Recovery codes displayed (save these!)
```

### 2FA Livewire Component

```php
// app/Livewire/Settings/TwoFactorAuthentication.php

namespace App\Livewire\Settings;

use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;
use Livewire\Component;

class TwoFactorAuthentication extends Component
{
    public bool $showingConfirmation = false;
    public ?string $qrCode = null;
    public ?string $recoveryCodes = null;
    public string $code = '';
    public string $password = '';
    public bool $enabled = false;
    public bool $showingRecoveryCodes = false;

    public function mount(): void
    {
        $this->enabled = Auth::user()->two_factor_secret !== null;
    }

    public function enableTwoFactorAuthentication(): void
    {
        if (! Hash::check($this->password, Auth::user()->password)) {
            $this->addError('password', 'The provided password does not match your current password.');
            return;
        }

        $user = Auth::user();
        $user->forceFill([
            'two_factor_secret' => encrypt($user->createTwoFactorAuth()->secret()),
            'two_factor_recovery_codes' => encrypt(json_encode($user->createTwoFactorAuth()->generateRecoveryCodes())),
        ])->save();

        $this->showingConfirmation = true;
        $this->qrCode = $user->twoFactorAuth()->qrCodeSvg();
        $this->recoveryCodes = json_decode($user->two_factor_recovery_codes, true);

        $this->reset(['password']);
    }

    public function confirmTwoFactorAuthentication(): void
    {
        $user = Auth::user();

        if (! $user->twoFactorAuth()->verify($this->code)) {
            $this->addError('code', 'The provided code is invalid.');
            return;
        }

        $user->forceFill([
            'two_factor_confirmed_at' => now(),
        ])->save();

        $this->showingConfirmation = false;
        $this->enabled = true;

        $this->reset(['code']);

        session()->flash('status', 'Two-factor authentication has been enabled.');
    }

    public function disableTwoFactorAuthentication(): void
    {
        if (! Hash::check($this->password, Auth::user()->password)) {
            $this->addError('password', 'The provided password does not match your current password.');
            return;
        }

        Auth::user()->forceFill([
            'two_factor_secret' => null,
            'two_factor_recovery_codes' => null,
            'two_factor_confirmed_at' => null,
        ])->save();

        $this->enabled = false;
        $this->reset(['password']);

        session()->flash('status', 'Two-factor authentication has been disabled.');
    }

    public function regenerateRecoveryCodes(): void
    {
        $user = Auth::user();
        $user->forceFill([
            'two_factor_recovery_codes' => encrypt(json_encode($user->twoFactorAuth()->generateRecoveryCodes())),
        ])->save();

        $this->recoveryCodes = json_decode($user->two_factor_recovery_codes, true);
        $this->showingRecoveryCodes = true;
    }

    public function render(): \Illuminate\View\View
    {
        return view('livewire.settings.two-factor-authentication');
    }
}
```

```php
// app/Http/Middleware/EnsureTwoFactorEnabled.php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureTwoFactorEnabled
{
    public function handle(Request $request, Closure $next): Response
    {
        $user = $request->user();

        if ($user && ! $user->two_factor_secret && ! $request->is('user/confirm-password', 'two-factor-challenge', 'two-factor-setup')) {
            return redirect()->route('two-factor.setup')
                ->with('status', 'Two-factor authentication is required.');
        }

        return $next($request);
    }
}
```

```php
// app/Http/Middleware/EnsureEmailIsVerified.php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureEmailIsVerified
{
    public function handle(Request $request, Closure $next): Response
    {
        $user = $request->user();

        if ($user && ! $user->hasVerifiedEmail()) {
            if ($request->expectsJson()) {
                return response()->json(['message' => 'Email not verified.'], 403);
            }

            return redirect()->route('verification.notice');
        }

        return $next($request);
    }
}
```

```php
    return DB::transaction(function () use ($input) {
        $existing = User::where('email', $input['email'])->first();

        if ($existing) {
            throw ValidationException::withMessages([
                'email' => ['An account with this email already exists.'],
            ]);
        }

        return User::create([
            'name' => $input['name'],
            'email' => $input['email'],
            'password' => Hash::make($input['password']),
            'email_verified_at' => now(),
        ]);
    });
}
```

### Two-Factor Recovery Code Manager

```php
// app/Services/RecoveryCodeManager.php

namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;

class RecoveryCodeManager
{
    /**
     * Generate new recovery codes.
     */
    public function generate(User $user): array
    {
        $codes = collect();

        for ($i = 0; $i < 8; $i++) {
            $codes->push(Str::upper(Str::random(4) . '-' . Str::random(4)));
        }

        $hashedCodes = $codes->map(fn($code) => Hash::make($code))->toArray();

        $user->update([
            'two_factor_recovery_codes' => json_encode($hashedCodes),
        ]);

        return $codes->toArray();
    }

    /**
     * Use a recovery code.
     */
    public function useCode(User $user, string $code): bool
    {
        $hashedCodes = json_decode($user->two_factor_recovery_codes, true);

        if (! $hashedCodes) {
            return false;
        }

        foreach ($hashedCodes as $index => $hashedCode) {
            if (Hash::check($code, $hashedCode)) {
                unset($hashedCodes[$index]);
                $user->update([
                    'two_factor_recovery_codes' => json_encode(array_values($hashedCodes)),
                ]);
                return true;
            }
        }

        return false;
    }

    /**
     * Get remaining recovery codes count.
     */
    public function remainingCount(User $user): int
    {
        $codes = json_decode($user->two_factor_recovery_codes, true);
        return is_array($codes) ? count($codes) : 0;
    }
}
```

```php
namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\DB;

class UserActivityTracker
{
    /**
     * Track user activity.
     */
    public function track(User $user, string $activity, array $metadata = []): void
    {
        DB::table('user_activity_log')->insert([
            'user_id' => $user->id,
            'activity' => $activity,
            'metadata' => json_encode($metadata),
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
            'created_at' => now(),
        ]);
    }

    /**
     * Get user activity log.
     */
    public function getLog(User $user, int $limit = 50): array
    {
        return DB::table('user_activity_log')
            ->where('user_id', $user->id)
            ->orderByDesc('created_at')
            ->limit($limit)
            ->get()
            ->toArray();
    }

    /**
     * Get last activity for a user.
     */
    public function getLastActivity(User $user): ?object
    {
        return DB::table('user_activity_log')
            ->where('user_id', $user->id)
            ->orderByDesc('created_at')
            ->first();
    }
}
```

```php
// app/Services/PasswordBreachChecker.php

namespace App\Services;

use Illuminate\Support\Facades\Http;

class PasswordBreachChecker
{
    protected string $apiUrl = 'https://api.pwnedpasswords.com/range/';

    /**
     * Check if password has been breached.
     */
    public function isBreached(string $password): bool
    {
        $sha1 = strtoupper(sha1($password));
        $prefix = substr($sha1, 0, 5);
        $suffix = substr($sha1, 5);

        try {
            $response = Http::timeout(3)->get($this->apiUrl . $prefix);

            if ($response->successful()) {
                $hashes = explode("\r\n", $response->body());

                foreach ($hashes as $hash) {
                    [$hashSuffix, $count] = explode(':', $hash);
                    if ($hashSuffix === $suffix) {
                        return (int) $count > 0;
                    }
                }
            }
        } catch (\Exception $e) {
            // Fail open - don't block password changes if API is down
        }

        return false;
    }
```
