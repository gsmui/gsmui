## Detailed Suspicious Activity Detection

### Device Fingerprinting

GsmUi generates a composite device fingerprint from multiple browser attributes to identify returning devices with high confidence. The fingerprint includes browser version, operating system, screen resolution, timezone, installed fonts hash, canvas fingerprint, and WebGL renderer string. Each fingerprint is stored as a SHA-256 hash in the database to avoid storing raw PII.

### Impossible Travel Detection

When a user logs in from a new location, GsmUi calculates the distance between the last known location and the new location using the Haversine formula, then compares the elapsed time between logins. If the calculated speed exceeds 900 km/h (commercial aircraft speed), the login is flagged as suspicious.

```php
// app/Services/ImpossibleTravelDetector.php

class ImpossibleTravelDetector
{
    private const MAX_SPEED_KMH = 900;

    public function isImpossibleTravel(User $user, float $lat, float $lng): bool
    {
        $lastLogin = LoginHistory::where('user_id', $user->id)
            ->where('successful', true)
            ->whereNotNull('latitude')
            ->orderByDesc('created_at')
            ->first();

        if (! $lastLogin) {
            return false;
        }

        $distanceKm = $this->haversine(
            $lastLogin->latitude, $lastLogin->longitude,
            $lat, $lng
        );

        $timeHours = now()->diffInMinutes($lastLogin->created_at) / 60;

        if ($timeHours < 0.01) {
            return true;
        }

        $speedKmh = $distanceKm / $timeHours;

        return $speedKmh > self::MAX_SPEED_KMH;
    }

    private function haversine(float $lat1, float $lon1, float $lat2, float $lon2): float
    {
        $earthRadius = 6371;
        $dLat = deg2rad($lat2 - $lat1);
        $dLon = deg2rad($lon2 - $lon1);

        $a = sin($dLat / 2) * sin($dLat / 2)
            + cos(deg2rad($lat1)) * cos(deg2rad($lat2))
            * sin($dLon / 2) * sin($dLon / 2);

        $c = 2 * atan2(sqrt($a), sqrt(1 - $a));

        return $earthRadius * $c;
    }
}
```

### Risk Scoring

Each login attempt receives a composite risk score from 0 to 100. The score is calculated from multiple weighted factors and determines the authentication flow required.

```php
// app/Services/LoginRiskScorer.php

class LoginRiskScorer
{
    private const WEIGHTS = [
        'ip_known' => -20,
        'ip_country_match' => -15,
        'ip_frequent' => -10,
        'device_known' => -25,
        'device_fingerprint_match' => -20,
        'time_normal' => -5,
        'location_possible' => -10,
        'failed_recent' => 20,
        'vpn_detected' => 15,
        'tor_detected' => 30,
        'proxy_detected' => 10,
        'new_browser' => 5,
        'new_os' => 10,
        'geo_velocity_impossible' => 40,
        'credential_leaked' => 25,
    ];

    public function calculateScore(User $user, Request $request): int
    {
        $score = 50; // Base score

        $factors = [
            $this->isKnownIp($user, $request->ip()) ? 'ip_known' : null,
            $this->isCountryMatch($user, $request->ip()) ? 'ip_country_match' : null,
            $this->isFrequentIp($user, $request->ip()) ? 'ip_frequent' : null,
            $this->isKnownDevice($user, $request) ? 'device_known' : null,
            $this->isNormalTime($user) ? 'time_normal' : null,
            $this->isLocationPossible($user, $request->ip()) ? 'location_possible' : null,
            $this->hasRecentFailures($user) ? 'failed_recent' : null,
            $this->isVpn($request->ip()) ? 'vpn_detected' : null,
            $this->isTor($request->ip()) ? 'tor_detected' : null,
            $this->isNewBrowser($user, $request) ? 'new_browser' : null,
            $this->isNewOs($user, $request) ? 'new_os' : null,
        ];

        foreach ($factors as $factor) {
            if ($factor && isset(self::WEIGHTS[$factor])) {
                $score += self::WEIGHTS[$factor];
            }
        }

        return max(0, min(100, $score));
    }

    public function getAuthRequirement(int $score): string
    {
        if ($score >= 80) return 'block';
        if ($score >= 60) return 'step_up_2fa';
        if ($score >= 40) return 'email_verification';
        return 'allow';
    }
}
```

```php
// app/Services/CredentialStuffingDetector.php

class CredentialStuffingDetector
{
    private const UNIQUE_EMAIL_THRESHOLD = 50;
    private const TIME_WINDOW_MINUTES = 10;

    public function detect(string $ip): bool
    {
        $uniqueEmails = Cache::remember(
            "credential_stuffing:{$ip}",
            $this->timeWindow(),
            fn () => collect()
        );

        $uniqueEmails->push(Str::random(10) . '@' . Str::random(5));

        if ($uniqueEmails->count() > self::UNIQUE_EMAIL_THRESHOLD) {
            $this->blockIp($ip);
            $this->alertAdmin($ip);
            return true;
        }

        return false;
    }

    private function blockIp(string $ip): void
    {
        Cache::put("ip_blocked:{$ip}", true, now()->addHours(24));
    }

    private function alertAdmin(string $ip): void
    {
        Mail::to(config('app.admin_email'))
            ->send(new CredentialStuffingAlertEmail($ip));
    }
}
```

### Session Binding Implementation

```php
// app/Http/Middleware/SessionBinding.php

class SessionBinding
{
    public function handle(Request $request, Closure $next): mixed
    {
        $binding = [
            'ip' => $request->ip(),
            'user_agent' => Str::limit($request->userAgent(), 200),
        ];

        $storedBinding = Session::get('binding_hash');

        if ($storedBinding && $storedBinding !== $this->hashBinding($binding)) {
            Auth::logout();
            Session::invalidate();
            Session::regenerateToken();

            activity('security')
                ->event('session.hijack_attempt')
                ->withProperties([
                    'ip' => $request->ip(),
                    'user_agent' => $request->userAgent(),
                ])
                ->log('Possible session hijacking detected');

            return redirect()->route('login')
                ->with('error', 'Session invalidated for security reasons.');
        }

        if (! $storedBinding) {
            Session::put('binding_hash', $this->hashBinding($binding));
        }

        return $next($request);
    }

    private function hashBinding(array $binding): string
    {
        return hash('sha256', json_encode($binding));
    }
}
```

### Have I Been Pwned Integration

```php
// app/Services/PasswordBreachChecker.php

class PasswordBreachChecker
{
    private const HIBP_API = 'https://api.pwnedpasswords.com/range/';

    public function isPasswordBreached(string $password): bool
    {
        $sha1 = strtoupper(sha1($password));
        $prefix = substr($sha1, 0, 5);
        $suffix = substr($sha1, 5);

        $response = Http::get(self::HIBP_API . $prefix);

        if ($response->failed()) {
            return false; // Fail open - don't block registration
        }

        $hashes = explode("\r\n", $response->body());

        foreach ($hashes as $hash) {
            [$hashSuffix, $count] = explode(':', $hash);
            if ($hashSuffix === $suffix && (int) $count > 0) {
                return true;
            }
        }

        return false;
    }

    public function getBreachCount(string $password): int
    {
        $sha1 = strtoupper(sha1($password));
        $prefix = substr($sha1, 0, 5);
        $suffix = substr($sha1, 5);

        $response = Http::get(self::HIBP_API . $prefix);

        if ($response->failed()) {
            return 0;
        }

        $hashes = explode("\r\n", $response->body());

        foreach ($hashes as $hash) {
            [$hashSuffix, $count] = explode(':', $hash);
            if ($hashSuffix === $suffix) {
                return (int) $count;
            }
        }

        return 0;
    }
}
```

### Password Strength Algorithm

```php
// app/Services/PasswordStrengthAnalyzer.php

class PasswordStrengthAnalyzer
{
    public const LEVELS = [
        'very_weak' => ['min' => 0, 'color' => 'red'],
        'weak' => ['min' => 15, 'color' => 'orange'],
        'fair' => ['min' => 30, 'color' => 'yellow'],
        'strong' => ['min' => 50, 'color' => 'green'],
        'very_strong' => ['min' => 70, 'color' => 'emerald'],
    ];

    public function analyze(string $password): array
    {
        $score = 0;
        $feedback = [];

        // Length
        $score += min(30, strlen($password) * 2);

        // Character variety
        if (preg_match('/[a-z]/', $password)) $score += 5;
        if (preg_match('/[A-Z]/', $password)) $score += 10;
        if (preg_match('/[0-9]/', $password)) $score += 10;
        if (preg_match('/[^a-zA-Z0-9]/', $password)) $score += 15;

        // Patterns (penalize)
        if (preg_match('/(.)\1{2,}/', $password)) {
            $score -= 10;
            $feedback[] = 'Avoid repeated characters';
        }

        if (preg_match('/^[a-zA-Z]+$/', $password)) {
            $score -= 10;
            $feedback[] = 'Add numbers or symbols';
        }

        if (preg_match('/^[0-9]+$/', $password)) {
            $score -= 20;
            $feedback[] = 'Don\'t use only numbers';
        }

        // Common passwords check
        if ($this->isCommonPassword($password)) {
            $score = 0;
            $feedback[] = 'This is a commonly used password';
        }

        $score = max(0, min(100, $score));
        $level = $this->getLevel($score);

        return [
            'score' => $score,
            'level' => $level,
            'color' => self::LEVELS[$level]['color'],
            'feedback' => $feedback,
        ];
    }

    private function getLevel(int $score): string
    {
        foreach (array_reverse(self::LEVELS, true) as $level => $config) {
            if ($score >= $config['min']) {
                return $level;
            }
        }
        return 'very_weak';
    }

    private function isCommonPassword(string $password): bool
    {
        $common = Cache::remember('common_passwords', 3600, function () {
            return collect(explode("\n", file_get_contents(
                resource_path('passwords/common.txt')
            )))
            ->map(fn ($p) => strtolower(trim($p)))
            ->toArray();
        });

        return in_array(strtolower($password), $common);
    }
}
```

```php
// app/Services/PasswordExpiryService.php

class PasswordExpiryService
{
    public function isExpired(User $user): bool
    {
        if (! $user->password_changed_at) {
            return true;
        }

        return $user->password_changed_at
            ->addDays($this->getExpiryDays())
            ->isPast();
    }

    public function daysUntilExpiry(User $user): int
    {
        if (! $user->password_changed_at) {
            return 0;
        }

        $expiry = $user->password_changed_at->addDays($this->getExpiryDays());

        return max(0, now()->diffInDays($expiry, false));
    }

    public function shouldWarn(User $user): bool
    {
        $daysLeft = $this->daysUntilExpiry($user);
        return $daysLeft <= 7 && $daysLeft > 0;
    }

    private function getExpiryDays(): int
    {
        return (int) config('security.password_expiry_days', 90);
    }
}
```

```php
        'email_verified' => ['weight' => 20, 'label' => 'Email verified'],
        'password_strong' => ['weight' => 15, 'label' => 'Strong password'],
        'password_recent' => ['weight' => 10, 'label' => 'Password changed recently'],
        '2fa_enabled' => ['weight' => 25, 'label' => 'Two-factor authentication enabled'],
        'recovery_codes' => ['weight' => 5, 'label' => 'Recovery codes available'],
        'recovery_email' => ['weight' => 5, 'label' => 'Recovery email added'],
        'no_shared_devices' => ['weight' => 5, 'label' => 'No shared devices'],
        'recent_activity_reviewed' => ['weight' => 5, 'label' => 'Recent activity reviewed'],
        'api_tokens_reviewed' => ['weight' => 5, 'label' => 'API tokens reviewed'],
        'passkey_enrolled' => ['weight' => 5, 'label' => 'Passkey enrolled'],
    ];

    public function calculate(User $user): array
    {
        $score = 0;
        $checks = [];

        foreach (self::CHECKS as $key => $config) {
            $passed = $this->check($key, $user);
            $score += $passed ? $config['weight'] : 0;
            $checks[] = [
                'key' => $key,
                'label' => $config['label'],
                'passed' => $passed,
                'weight' => $config['weight'],
            ];
        }

        return [
            'score' => $score,
            'max' => 100,
            'level' => $this->getLevel($score),
            'checks' => $checks,
        ];
    }

    private function check(string $key, User $user): bool
    {
        return match ($key) {
            'email_verified' => $user->hasVerifiedEmail(),
            'password_strong' => $this->isStrongPassword($user),
            'password_recent' => $this->isRecentPassword($user),
            '2fa_enabled' => $user->two_factor_secret !== null,
            'recovery_codes' => $user-> recovery_codes()->count() > 0,
            'recovery_email' => $user->recovery_email !== null,
            'no_shared_devices' => $this->noSharedDevices($user),
            'recent_activity_reviewed' => $this->wasRecentlyReviewed($user),
            'api_tokens_reviewed' => $this->tokensReviewed($user),
            'passkey_enrolled' => $user->passkeys()->count() > 0,
            default => false,
        };
    }

    private function getLevel(int $score): string
    {
        return match (true) {
            $score >= 90 => 'excellent',
            $score >= 70 => 'good',
            $score >= 50 => 'fair',
            $score >= 30 => 'poor',
            default => 'critical',
        };
    }

    private function isStrongPassword(User $user): bool
    {
        $analyzer = app(PasswordStrengthAnalyzer::class);
        $result = $analyzer->analyze($user->password ?? '');
        return in_array($result['level'], ['strong', 'very_strong']);
    }

    private function isRecentPassword(User $user): bool
    {
        if (! $user->password_changed_at) return false;
        return $user->password_changed_at->diffInDays(now()) < 90;
    }

    private function noSharedDevices(User $user): bool
    {
        return $user->devices()->where('is_shared', true)->count() === 0;
    }

    private function wasRecentlyReviewed(User $user): bool
    {
        return Cache::has("security_reviewed:{$user->id}");
    }

    private function tokensReviewed(User $user): bool
    {
        return Cache::has("tokens_reviewed:{$user->id}");
    }
}
```

```php
// app/Services/RecoveryCodeManager.php

class RecoveryCodeManager
{
    private const CODE_COUNT = 10;
    private const CODE_LENGTH = 10;

    public function generate(User $user): array
    {
        $codes = [];
        for ($i = 0; $i < self::CODE_COUNT; $i++) {
            $codes[] = $this->generateCode();
        }

        $hashedCodes = array_map(
            fn ($code) => hash('sha256', $code),
            $codes
        );

        $user->update(['recovery_codes' => $hashedCodes]);

        return $codes; // Return plaintext once, never stored
    }

    public function use(User $user, string $code): bool
    {
        $hashedCode = hash('sha256', $code);
        $codes = $user->recovery_codes ?? [];

        $index = array_search($hashedCode, $codes);
        if ($index === false) {
            return false;
        }

        unset($codes[$index]);
        $user->update(['recovery_codes' => array_values($codes)]);

        activity('security')
            ->event('recovery_code.used')
            ->performedOn($user)
            ->log('Recovery code used');

        return true;
    }

    public function remaining(User $user): int
    {
        return count($user->recovery_codes ?? []);
    }

    public function shouldRegenerate(User $user): bool
    {
        return $this->remaining($user) < 3;
    }

    private function generateCode(): string
    {
        $chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
        $code = '';
        for ($i = 0; $i < self::CODE_LENGTH; $i++) {
            $code .= $chars[random_int(0, strlen($chars) - 1)];
        }
        return substr($code, 0, 5) . '-' . substr($code, 5);
    }
}
```

## Security Automation Rules

### Rule Engine

GsmUi includes a configurable rule engine that executes automated responses to security events. Rules are evaluated in order and can be enabled or disabled per-deployment.

```php
// app/Security/AutomationRule.php

class AutomationRule
{
    public function __construct(
        public string $name,
        public string $description,
        public string $trigger,
        public array $conditions,
        public array $actions,
        public bool $enabled = true,
        public int $priority = 0,
    ) {}

    public function evaluate(array $eventData): bool
    {
        foreach ($this->conditions as $condition) {
            if (! $this->evaluateCondition($condition, $eventData)) {
                return false;
            }
        }
        return true;
    }

    private function evaluateCondition(array $condition, array $data): bool
    {
        $field = $data[$condition['field']] ?? null;
        $operator = $condition['operator'];
        $value = $condition['value'];

        return match ($operator) {
            'equals' => $field === $value,
            'not_equals' => $field !== $value,
            'greater_than' => $field > $value,
            'less_than' => $field < $value,
            'contains' => str_contains($field, $value),
            'in' => in_array($field, $value),
            'not_in' => ! in_array($field, $value),
            default => false,
        };
    }
}
```

```php
// app/Services/ThreatIntelligenceService.php

class ThreatIntelligenceService
{
    public function getIpReputation(string $ip): array
    {
        $cached = Cache::get("ip_reputation:{$ip}");
        if ($cached) {
            return $cached;
        }

        $results = [];

        if (config('services.abuseipdb.enabled')) {
            $results['abuseipdb'] = $this->queryAbuseIpDb($ip);
        }

        if (config('services.virustotal.enabled')) {
            $results['virustotal'] = $this->queryVirusTotal($ip);
        }

        $reputation = [
            'ip' => $ip,
            'is_malicious' => $this->isMalicious($results),
            'is_proxy' => $this->isProxy($results),
            'is_vpn' => $this->isVpn($results),
            'is_tor' => $this->isTor($results),
            'is_bot' => $this->isBot($results),
            'confidence' => $this->calculateConfidence($results),
            'sources' => $results,
            'checked_at' => now()->toIso8601String(),
        ];

        Cache::put("ip_reputation:{$ip}", $reputation, now()->addHours(24));

        return $reputation;
    }

    private function isMalicious(array $results): bool
    {
        $abuseScore = $results['abuseipdb']['confidence_score'] ?? 0;
        $vtScore = $results['virustotal']['malicious'] ?? 0;
        return $abuseScore > 75 || $vtScore > 2;
    }

    private function calculateConfidence(array $results): int
    {
        $sources = count(array_filter($results));
        $total = count($results);
        return $total > 0 ? (int) (($sources / $total) * 100) : 0;
    }
}
```

### Incident Response Automation

```php
// app/Security/IncidentResponder.php

class IncidentResponder
{
    public function handleIncident(SecurityEvent $event): void
    {
        $severity = $event->severity;
        $incident = Incident::create([
            'title' => $event->getTitle(),
            'severity' => $severity,
            'status' => 'open',
            'description' => $event->getDescription(),
            'affected_users' => $event->getAffectedUserIds(),
            'evidence' => $event->getEvidence(),
        ]);

        match ($severity) {
            'critical' => $this->handleCritical($incident, $event),
            'high' => $this->handleHigh($incident, $event),
            'medium' => $this->handleMedium($incident, $event),
            'low' => $this->handleLow($incident, $event),
            default => null,
        };

        // Notify security team
        $this->notifyTeam($incident);

        // Log to audit trail
        activity('security')
            ->event('incident.created')
            ->withProperties($incident->toArray())
            ->log("Incident #{$incident->id} created: {$incident->title}");
    }

    private function handleCritical(Incident $incident, SecurityEvent $event): void
    {
        // Immediate containment
        foreach ($event->getAffectedUserIds() as $userId) {
            $user = User::find($userId);
            if ($user) {
                $user->forceLogout();
                $user->suspend('Security incident #' . $incident->id);
            }
        }

        // Block attacking IPs
        foreach ($event->getAttackingIps() as $ip) {
            Cache::put("ip_blocked:{$ip}", true, now()->addDays(7));
        }

        // Revoke all API tokens for affected users
        foreach ($event->getAffectedUserIds() as $userId) {
            PersonalAccessToken::where('user_id', $userId)->delete();
        }

        // Page on-call security engineer
        $this->pageOnCall($incident);
    }

    private function handleHigh(Incident $incident, SecurityEvent $event): void
    {
        // Temporary lockout
        foreach ($event->getAffectedUserIds() as $userId) {
            $user = User::find($userId);
            if ($user) {
                $user->temporaryLockout(now()->addHours(1));
            }
        }

        // Send alert to security lead
        Mail::to(config('security.alert_email'))
            ->send(new IncidentAlertEmail($incident));
    }

    private function notifyTeam(Incident $incident): void
    {
        $team = config('security.incident_team');

        if ($incident->severity === 'critical' || $incident->severity === 'high') {
            Notification::send($team, new IncidentNotification($incident));
        }
    }

    private function pageOnCall(Incident $incident): void
    {
        // Integration with PagerDuty, OpsGenie, etc.
        if (config('services.pagerduty.enabled')) {
            app(PagerDutyService::class)->createIncident($incident);
        }
    }
}
```

### Account Lockout Strategy

| Threshold | Duration | Escalation |
|---|---|---|
| 5 failed logins | 15 minutes | Email alert |
| 10 failed logins | 1 hour | Admin notification |
| 20 failed logins | 24 hours | Account review |
| 50 failed logins | Permanent | Manual unlock required |

```php
// app/Services/AccountLockoutService.php

class AccountLockoutService
{
    private const LOCKOUT_THRESHOLDS = [
        5 => 900,        // 15 minutes
        10 => 3600,      // 1 hour
        20 => 86400,     // 24 hours
        50 => -1,        // Permanent (manual unlock)
    ];

    public function recordFailure(User $user, string $ip): void
    {
        $failures = Cache::increment("login_failures:{$user->id}");

        if (isset(self::LOCKOUT_THRESHOLDS[$failures])) {
            $duration = self::LOCKOUT_THRESHOLDS[$failures];

            if ($duration === -1) {
                $user->lockPermanently();
                $this->notifyAdmin($user, 'permanent_lockout');
            } else {
                Cache::put(
                    "account_locked:{$user->id}",
                    true,
                    now()->addSeconds($duration)
                );
                $this->notifyUser($user, $duration);
            }

            activity('security')
                ->event('account.locked_out')
                ->performedOn($user)
                ->withProperties([
                    'failures' => $failures,
                    'duration' => $duration,
                    'ip' => $ip,
                ])
                ->log("Account locked out after {$failures} failures");
        }
    }

    public function isLocked(User $user): bool
    {
        return Cache::has("account_locked:{$user->id}") || $user->is_permanently_locked;
    }

    public function getRemainingLockoutTime(User $user): int
    {
        return Cache::store('redis')->ttl("account_locked:{$user->id}");
    }

    public function unlock(User $user, string $reason = 'manual'): void
    {
        Cache::forget("account_locked:{$user->id}");
        Cache::forget("login_failures:{$user->id}");
        $user->is_permanently_locked = false;
        $user->save();

        activity('security')
            ->event('account.unlocked')
            ->performedOn($user)
            ->withProperties(['reason' => $reason])
            ->log("Account unlocked: {$reason}");
    }

    public function resetFailures(User $user): void
    {
        Cache::forget("login_failures:{$user->id}");
    }

    private function notifyUser(User $user, int $duration): void
    {
        $minutes = (int) ceil($duration / 60);
        Notification::send($user, new AccountLockedNotification($minutes));
    }

    private function notifyAdmin(User $user, string $reason): void
    {
        Mail::to(config('app.admin_email'))
            ->send(new AccountLockoutAlertEmail($user, $reason));
    }
}
```

```php
// app/Services/TrustedDeviceService.php

class TrustedDeviceService
{
    private const MAX_TRUSTED_DEVICES = 10;
    private const TRUST_DURATION_DAYS = 30;

    public function trust(User $user, Request $request): TrustedDevice
    {
        $activeDevices = $user->trustedDevices()
            ->where('is_revoked', false)
            ->count();

        if ($activeDevices >= self::MAX_TRUSTED_DEVICES) {
            $oldest = $user->trustedDevices()
                ->where('is_revoked', false)
                ->orderBy('last_used_at')
                ->first();
            $oldest?->update(['is_revoked' => true]);
        }

        return TrustedDevice::create([
            'user_id' => $user->id,
            'name' => $this->getDeviceName($request),
            'device_fingerprint' => $this->getFingerprint($request),
            'browser' => $this->getBrowser($request),
            'os' => $this->getOs($request),
            'ip_address' => $request->ip(),
            'location' => $this->getLocation($request->ip()),
            'last_used_at' => now(),
            'expires_at' => now()->addDays(self::TRUST_DURATION_DAYS),
        ]);
    }

    public function isTrusted(User $user, Request $request): bool
    {
        $fingerprint = $this->getFingerprint($request);

        return $user->trustedDevices()
            ->where('device_fingerprint', $fingerprint)
            ->where('is_revoked', false)
            ->where('expires_at', '>', now())
            ->exists();
    }

    public function revokeAll(User $user): int
    {
        return $user->trustedDevices()
            ->where('is_revoked', false)
            ->update(['is_revoked' => true]);
    }

    private function getFingerprint(Request $request): string
    {
        $data = [
            $request->userAgent(),
            $request->header('Sec-CH-UA-Platform'),
            $request->header('Sec-CH-UA-Mobile'),
        ];
        return hash('sha256', implode('|', $data));
    }

    private function getDeviceName(Request $request): string
    {
        $browser = $this->getBrowser($request);
        $os = $this->getOs($request);
        return "{$browser} on {$os}";
    }
}
```

```php
class ApiTokenSecurityService
{
    public function createSecureToken(
        User $user,
        string $name,
        array $scopes,
        ?array $allowedIps = null,
        ?Carbon $expiresAt = null,
    ): NewAccessToken {
        $token = $user->createToken($name, $scopes);

        if ($allowedIps) {
            $token->accessToken->update([
                'allowed_ips' => $allowedIps,
            ]);
        }

        if ($expiresAt) {

                'expires_at' => $expiresAt,
            ]);
        }

        activity('security')
            ->event('api.token_created')
            ->performedOn($user)
            ->withProperties([
                'token_name' => $name,
                'scopes' => $scopes,
                'ip' => request()->ip(),
            ])
            ->log("API token created: {$name}");

        return $token;
    }

    public function validateTokenRequest(AccessToken $token, Request $request): bool
    {
        if ($token->is_revoked) return false;
        if ($token->expires_at && $token->expires_at->isPast()) return false;

        if ($token->allowed_ips && ! empty($token->allowed_ips)) {
            if (! in_array($request->ip(), $token->allowed_ips)) {
                return false;
            }
        }

        return true;
    }

    {
        $token->update(['last_used_at' => now()]);

                    'token_id' => $token->id,
                    'endpoint' => $request->path(),
                    'method' => $request->method(),
                    'ip' => $request->ip(),
                ])
                ->log('API token used');
        }
    }
```

```php
class SecurityHealthCheck implements HealthCheck
{
    public function name(): string
    {
        return 'security';
    }

    public function run(): Result
    {
        $issues = [];

        // Check for accounts with weak passwords
        $weakPasswords = DB::table('users')
            ->where('password_changed_at', '<', now()->subDays(90))
            ->count();

        if ($weakPasswords > 0) {
            $issues[] = "{$weakPasswords} users have passwords older than 90 days";
        }

        // Check for users without 2FA
        $noMfa = DB::table('users')
            ->whereNull('two_factor_secret')
            ->where('is_admin', true)
            ->count();

        if ($noMfa > 0) {
            $issues[] = "{$noMfa} admin users without 2FA";
        }

        // Check for expired tokens
        $expiredTokens = DB::table('personal_access_tokens')
            ->where('expires_at', '<', now())
            ->count();

        if ($expiredTokens > 100) {
            $issues[] = "{$expiredTokens} expired API tokens need cleanup";
        }

        // Check for locked accounts
        $lockedAccounts = DB::table('users')
            ->where('is_suspended', true)
            ->count();

        if ($lockedAccounts > 10) {
            $issues[] = "{$lockedAccounts} accounts currently suspended";
        }

        if (empty($issues)) {
            return Result::ok()->withMetaData(['message' => 'All security checks passed']);
        }

        return Result::warning()->withMetaData(['issues' => $issues]);
    }
}
```
