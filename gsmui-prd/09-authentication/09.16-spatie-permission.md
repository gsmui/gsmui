# 09.16 — Spatie Permission

## Purpose

Define the Spatie Permission package integration for GsmUi, covering roles, permissions, caching, and management.

## Installation

```bash
composer require spatie/laravel-permission
php artisan vendor:publish --provider="Spatie\Permission\PermissionServiceProvider"
php artisan migrate
```

## Configuration

```php
// config/permission.php

return [
    'models' => [
```

## Caching

Spatie Permission caches roles and permissions in Redis:

```php
// Cache is automatically managed
// Clear cache when roles/permissions change

Role::flushCache();
Permission::flushCache();
```

## Commands

```bash
# List all roles
php artisan permission:list-roles

# List all permissions
php artisan permission:list-permissions

# Create role
php artisan permission:create-role admin

# Create permission
php artisan permission:create-permission themes:create

# Assign role to user
php artisan permission:assign-role {user} admin

# Sync roles
php artisan permission:sync-roles {user} admin,editor
```

### Database Tables

#### Roles Table

```sql
CREATE TABLE roles (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    guard_name VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    UNIQUE KEY (name, guard_name)
);
```

#### Permissions Table

```sql
CREATE TABLE permissions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    guard_name VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    UNIQUE KEY (name, guard_name)
);
```

#### Model Has Roles Table

```sql
CREATE TABLE model_has_roles (
    role_id BIGINT UNSIGNED NOT NULL,
    model_type VARCHAR(255) NOT NULL,
    model_id BIGINT UNSIGNED NOT NULL,
    PRIMARY KEY (role_id, model_id, model_type),
    KEY model_has_roles_model_id_model_type_index (model_id, model_type),
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE
);
```

#### Model Has Permissions Table

```sql
CREATE TABLE model_has_permissions (
    permission_id BIGINT UNSIGNED NOT NULL,
    model_type VARCHAR(255) NOT NULL,
    model_id BIGINT UNSIGNED NOT NULL,
    PRIMARY KEY (permission_id, model_id, model_type),
    KEY model_has_permissions_model_id_model_type_index (model_id, model_type),
    FOREIGN KEY (permission_id) REFERENCES permissions(id) ON DELETE CASCADE
);
```

#### Role Has Permissions Table

```sql
CREATE TABLE role_has_permissions (
    role_id BIGINT UNSIGNED NOT NULL,
    permission_id BIGINT UNSIGNED NOT NULL,
    PRIMARY KEY (role_id, permission_id),
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE,
    FOREIGN KEY (permission_id) REFERENCES permissions(id) ON DELETE CASCADE
);
```

## User Model Integration

```php
// app/Models/User.php

namespace App\Models;

use Illuminate\Contracts\Auth\MustVerifyEmail;
use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;
use Spatie\Permission\Traits\HasRoles;

class User extends Authenticatable implements MustVerifyEmail
{
    use HasRoles, Notifiable;

    protected $fillable = [
        'name',
        'email',
        'password',
        'avatar',
        'email_verified_at',
        'two_factor_secret',
        'two_factor_recovery_codes',
        'two_factor_confirmed_at',
    ];

    protected $hidden = [
        'password',
        'remember_token',
        'two_factor_secret',
        'two_factor_recovery_codes',
    ];

    protected $casts = [
        'email_verified_at' => 'datetime',
        'two_factor_confirmed_at' => 'datetime',
        'password' => 'hashed',
    ];

    /**
     * Initialize roles for new users.
     */
    public function initializeHasRoles(): void
    {
        static::created(function (User $user) {
            $user->assignRole('user');
        });
    }

    /**
     * Check if user is admin.
     */
    public function isAdmin(): bool
    {
        return $this->hasRole('admin');
    }

    /**
     * Check if user is super admin.
     */
    public function isSuperAdmin(): bool
    {
        return $this->hasRole('super-admin');
    }

    /**
     * Check if user is supporter.
     */
    public function isSupporter(): bool
    {
        return $this->hasRole('supporter');
    }

    /**
     * Check if user is creator.
     */
    public function isCreator(): bool
    {
        return $this->hasRole('creator');
    }

    /**
     * Check if user can manage other users.
     */
    public function canManageUsers(): bool
    {
        return $this->hasPermissionTo('users:manage');
    }

    /**
     * Check if user can manage themes.
     */
    public function canManageThemes(): bool
    {
        return $this->hasPermissionTo('themes:manage');
    }

    /**
     * Check if user can manage components.
     */
    public function canManageComponents(): bool
    {
        return $this->hasPermissionTo('components:manage');
    }

    /**
     * Check if user can manage marketplace.
     */
    public function canManageMarketplace(): bool
    {
        return $this->hasPermissionTo('marketplace:manage');
    }

    /**
     * Get user's display role.
     */
    public function getDisplayRoleAttribute(): string
    {
        $roles = $this->getRoleNames();

        if ($roles->contains('super-admin')) {
            return 'Super Admin';
        }

        if ($roles->contains('admin')) {
            return 'Admin';
        }

        if ($roles->contains('supporter')) {
            return 'Supporter';
        }

        if ($roles->contains('creator')) {
            return 'Creator';
        }

        return 'User';
    }
}
```

```php
    /**
     * Define default roles.
     */
    public function defineRoles(): void
    {
        // Super Admin role
        Role::create(['name' => 'super-admin', 'guard_name' => 'web']);

        // Admin role
        Role::create(['name' => 'admin', 'guard_name' => 'web']);

        // Supporter role
        Role::create(['name' => 'supporter', 'guard_name' => 'web']);

        // Creator role
        Role::create(['name' => 'creator', 'guard_name' => 'web']);

        // Pro User role
        Role::create(['name' => 'pro-user', 'guard_name' => 'web']);

        // User role (default)
        Role::create(['name' => 'user', 'guard_name' => 'web']);

        // Guest role
        Role::create(['name' => 'guest', 'guard_name' => 'web']);
    }

    /**
     * Assign permissions to roles.
     */
    public function assignPermissionsToRoles(): void
    {
        // Super Admin - all permissions
        $superAdmin = Role::findByName('super-admin');
        $superAdmin->givePermissionTo(Permission::all());

        // Admin - all except super-admin permissions
        $admin = Role::findByName('admin');
        $admin->givePermissionTo([
            'users:view', 'users:create', 'users:update', 'users:delete',
            'themes:view', 'themes:create', 'themes:update', 'themes:delete', 'themes:manage',
            'components:view', 'components:create', 'components:update', 'components:delete', 'components:manage',
            'templates:view', 'templates:create', 'templates:update', 'templates:delete', 'templates:manage',
            'marketplace:view', 'marketplace:create', 'marketplace:update', 'marketplace:delete', 'marketplace:manage',
            'billing:view', 'billing:manage',
            'admin:access',
        ]);

        // Supporter
        $supporter = Role::findByName('supporter');
        $supporter->givePermissionTo([
            'users:view',
            'themes:view', 'themes:update', 'themes:manage',
            'components:view', 'components:update',

            'billing:view',

            'themes:view',
            'components:view',
```

## Livewire Role Management Component

```php
// app/Livewire/Admin/RoleManager.php

namespace App\Livewire\Admin;

use Livewire\Component;
use Spatie\Permission\Models\Role;
use Spatie\Permission\Models\Permission;

class RoleManager extends Component
{
    public array $roles = [];
    public array $permissions = [];
    public ?string $selectedRoleId = null;
    public string $roleName = '';
    public array $selectedPermissions = [];
    public bool $showCreateModal = false;
    public bool $showEditModal = false;
    public bool $showDeleteModal = false;

    protected array $rules = [
        'roleName' => ['required', 'string', 'max:255'],
        'selectedPermissions' => ['required', 'array'],
    ];

    public function mount(): void
    {
        $this->loadRoles();
        $this->loadPermissions();
    }

    public function loadRoles(): void
    {
        $this->roles = Role::with('permissions')
            ->get()
            ->toArray();
    }

    public function loadPermissions(): void
    {
        $this->permissions = Permission::all()->toArray();
    }

    public function createRole(): void
    {
        $this->validate();

        $role = Role::create([
            'name' => $this->roleName,
            'guard_name' => 'web',
        ]);

        $role->syncPermissions($this->selectedPermissions);

        activity()
            ->withProperties(['role' => $this->roleName])
            ->log('Role created');

        $this->reset(['roleName', 'selectedPermissions']);
        $this->showCreateModal = false;

        $this->loadRoles();

        session()->flash('status', 'Role created successfully.');
    }

    public function editRole(string $roleId): void
    {
        $role = Role::find($roleId);

        if (! $role) {
            $this->addError('role', 'Role not found.');
            return;
        }

        $this->selectedRoleId = $roleId;
        $this->roleName = $role->name;
        $this->selectedPermissions = $role->permissions->pluck('name')->toArray();
        $this->showEditModal = true;
    }

    public function updateRole(): void
    {
        $this->validate();

        $role = Role::find($this->selectedRoleId);

        if (! $role) {
            $this->addError('role', 'Role not found.');
            return;
        }

        $role->update(['name' => $this->roleName]);
        $role->syncPermissions($this->selectedPermissions);

        activity()
            ->withProperties(['role' => $this->roleName])
            ->log('Role updated');

        $this->reset(['selectedRoleId', 'roleName', 'selectedPermissions']);
        $this->showEditModal = false;

        $this->loadRoles();

        session()->flash('status', 'Role updated successfully.');
    }

    public function confirmDeleteRole(string $roleId): void
    {
        $this->selectedRoleId = $roleId;
        $this->showDeleteModal = true;
    }

    public function deleteRole(): void
    {
        $role = Role::find($this->selectedRoleId);

        if (! $role) {
            $this->addError('role', 'Role not found.');
            return;
        }

        // Prevent deleting built-in roles
        if (in_array($role->name, ['super-admin', 'admin', 'user'])) {
            $this->addError('role', 'Cannot delete built-in roles.');
            $this->showDeleteModal = false;
            return;
        }

        $role->delete();

        activity()
            ->withProperties(['role' => $role->name])
            ->log('Role deleted');

        $this->reset(['selectedRoleId']);
        $this->showDeleteModal = false;

        $this->loadRoles();

        session()->flash('status', 'Role deleted successfully.');
    }

    public function cancelModal(): void
    {
        $this->reset(['selectedRoleId', 'roleName', 'selectedPermissions']);
        $this->showCreateModal = false;
        $this->showEditModal = false;
        $this->showDeleteModal = false;
    }

    public function render(): \Illuminate\View\View
    {
        return view('livewire.admin.role-manager');
    }
}
```

```php
// app/Services/PermissionCacheService.php

namespace App\Services;

use Illuminate\Support\Facades\Cache;
use Spatie\Permission\Models\Permission;
use Spatie\Permission\Models\Role;

class PermissionCacheService
{
    /**
     * Clear all permission caches.
     */
    public function flush(): void
    {
        Cache::flush();
        Permission::flushCache();
        Role::flushCache();
    }

    /**
     * Refresh permission caches.
     */
    public function refresh(): void
    {
        Permission::forgetCache();
        Role::forgetCache();
    }

    /**
     * Get cached permissions for user.
     */
    public function getCachedPermissions($userId): array
    {
        return Cache::remember(
            "user_permissions:{$userId}",
            now()->addHour(),
            fn() => auth()->user()->getAllPermissions()->pluck('name')->toArray()
        );
    }

    /**
     * Get cached roles for user.
     */
    public function getCachedRoles($userId): array
    {
        return Cache::remember(
            "user_roles:{$userId}",
            now()->addHour(),
            fn() => auth()->user()->getRoleNames()->toArray()
        );
    }

    /**
     * Clear cache for specific user.
     */
    public function clearUserCache($userId): void
    {
        Cache::forget("user_permissions:{$userId}");
        Cache::forget("user_roles:{$userId}");
    }

    /**
     * Clear cache for all users.
     */
    public function clearAllUserCache(): void
    {
        $userIds = \App\Models\User::pluck('id')->toArray();

        foreach ($userIds as $userId) {
            $this->clearUserCache($userId);
        }
    }
}
```

## Permission Sync Service

### Role-Permission Sync

```php
// app/Services/RolePermissionSyncService.php

namespace App\Services;

use Spatie\Permission\Models\Role;
use Spatie\Permission\Models\Permission;
use Illuminate\Support\Facades\DB;

class RolePermissionSyncService
{
    /**
     * Sync permissions for a role.
     */
    public function syncRolePermissions(Role $role, array $permissions): void
    {
        DB::transaction(function () use ($role, $permissions) {
            // Validate all permissions exist
            $validPermissions = Permission::whereIn('name', $permissions)->pluck('name')->toArray();
            $invalidPermissions = array_diff($permissions, $validPermissions);

            if (! empty($invalidPermissions)) {
                throw new \InvalidArgumentException(
                    "Invalid permissions: " . implode(', ', $invalidPermissions)
                );
            }

            $role->syncPermissions($permissions);

            // Clear permission cache
            app(PermissionRegistrar::class)->forgetCachedPermissions();
        });
    }

    /**
     * Bulk sync permissions for multiple roles.
     */
    public function bulkSync(array $rolePermissions): void
    {
        DB::transaction(function () use ($rolePermissions) {
            foreach ($rolePermissions as $roleName => $permissions) {
                $role = Role::where('name', $roleName)->firstOrFail();
                $this->syncRolePermissions($role, $permissions);
            }
        });
    }

    /**
     * Get role permissions diff.
     */
    public function getRolePermissionsDiff(Role $role, array $newPermissions): array
    {
        $currentPermissions = $role->permissions->pluck('name')->toArray();

        return [
            'added' => array_diff($newPermissions, $currentPermissions),
            'removed' => array_diff($currentPermissions, $newPermissions),
            'unchanged' => array_intersect($currentPermissions, $newPermissions),
        ];
    }
}
```

```php
namespace App\Console\Commands;

use App\Models\User;
use Illuminate\Console\Command;
use Spatie\Permission\Models\Role;

class BulkAssignRole extends Command

                            {role : The role to assign}
                            {--users=* : Specific user emails to assign (optional)}
                            {--all-active : Assign to all active users}';

    protected $description = 'Bulk assign a role to multiple users';

    public function handle(): int
    {
        $roleName = $this->argument('role');
        $role = Role::where('name', $roleName)->first();

        if (! $role) {
            $this->error("Role '{$roleName}' not found.");
            return 1;
        }

        foreach ($users as $user) {
            $user->assignRole($roleName);
            $bar->advance();
        }

        $bar->finish();
        $this->newLine();
        $this->info("Role '{$roleName}' assigned to {$users->count()} users.");

        return 0;
    }
```

```php
// app/Console/Commands/PermissionAudit.php

namespace App\Console\Commands;

use App\Models\User;
use Illuminate\Console\Command;
use Spatie\Permission\Models\Role;
use Spatie\Permission\Models\Permission;

class PermissionAudit extends Command
{
    protected $signature = 'permission:audit
                            {--role=* : Specific roles to audit}
                            {--format=table : Output format (table, json, csv)}';

    protected $description = 'Audit role and permission assignments';

    public function handle(): int
    {
        $roles = $this->option('role')
            ? Role::whereIn('name', $this->option('role'))->get()
            : Role::all();

        $data = $roles->map(function ($role) {
            return [
                'role' => $role->name,
                'permissions' => $role->permissions->pluck('name')->implode(', '),
                'users_count' => $role->users()->count(),
                'guard' => $role->guard_name,
            ];
        });

        match($this->option('format')) {
            'json' => $this->line($data->toJson()),
            'csv' => $this->outputCsv($data->toArray()),
            default => $this->table(['Role', 'Permissions', 'Users', 'Guard'], $data->toArray()),
        };

        return 0;
    }

    protected function outputCsv(array $data): void
    {
        $headers = 'Role,Permissions,Users Count,Guard';
        $this->line($headers);

        foreach ($data as $row) {
            $this->line(implode(',', [
                $row['role'],
                '"' . str_replace('"', '""', $row['permissions']) . '"',
                $row['users_count'],
                $row['guard'],
            ]));
        }
    }
}
```

```php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;

class RequirePermission
{
    public function handle(Request $request, Closure $next, string ...$permissions)
    {
        $user = $request->user();

        if (! $user) {
            return redirect()->route('login');
        }

        foreach ($permissions as $permission) {
            if (! $user->hasPermissionTo($permission)) {
                if ($request->expectsJson()) {
                    return response()->json([
                        'error' => 'forbidden',
                        'message' => "Missing required permission: {$permission}",
                    ], 403);
                }

                abort(403, "Unauthorized. Required permission: {$permission}");
            }
        }

        return $next($request);
    }
}
```

### Role Hierarchy Middleware

```php
// app/Http/Middleware/RequireRoleLevel.php

namespace App\Http\Middleware;

use App\Services\RoleHierarchy;
use Closure;
use Illuminate\Http\Request;

class RequireRoleLevel
{
    public function __construct(
        protected RoleHierarchy $roleHierarchy,
    ) {}

    public function handle(Request $request, Closure $next, int $minLevel): int
    {
        $user = $request->user();

        if (! $user) {
            return redirect()->route('login');
        }

        $userLevel = $this->roleHierarchy->getLevel($user->getRoleNames()->first() ?? 'guest');

        if ($userLevel < $minLevel) {
            if ($request->expectsJson()) {
                return response()->json([
                    'error' => 'insufficient_role_level',
                    'message' => 'Your role does not have sufficient privileges.',
                    'required_level' => $minLevel,
                    'current_level' => $userLevel,
                ], 403);
            }

            abort(403, 'Insufficient role level.');
        }

        return $next($request);
    }
}
```
