# 09.23 — MCP Authentication

## Purpose

Define the authentication system for MCP (Model Context Protocol) clients connecting to GsmUi.

## MCP Authentication Flow

```text
MCP Client sends request with API token
    ↓
Middleware validates token
    ↓
User resolved from token
    ↓
Scope check (token abilities)
    ↓
Rate limit check
    ↓
Request processed
    ↓
Response returned
```

## MCP Authentication Middleware

```php
// app/Http/Middleware/McpAuthMiddleware.php

class McpAuthMiddleware
{
    public function handle(Request $request, Closure $next): Response
    {
        $token = $request->bearerToken();

        if (! $token) {
            return response()->json(['error' => 'Token required'], 401);
        }

        $accessToken = PersonalAccessToken::findToken($token);

        if (! $accessToken) {
            return response()->json(['error' => 'Invalid token'], 401);
        }

        if ($accessToken->expires_at && $accessToken->expires_at->isPast()) {
            return response()->json(['error' => 'Token expired'], 401);
        }

        $user = $accessToken->tokenable;

        if (! $user || $user->isSuspended()) {
            return response()->json(['error' => 'Account suspended'], 403);
        }

        // Log MCP access
        activity('mcp-auth')
            ->performedOn($user)
            ->withProperties([
                'token_name' => $accessToken->name,
                'ip' => $request->ip(),
            ])
            ->log('MCP client authenticated');

        return $next($request);
    }
}
```

## Security Notes

* MCP tokens are regular Sanctum tokens with MCP-specific scopes
* Tokens can be revoked from user dashboard
* All MCP requests logged with user, token, and IP
* Rate limiting per token and per IP
* Token expiration configurable
* Suspended users cannot access MCP

## MCP Token Creation Service

```php
// app/Services/McpTokenService.php

namespace App\Services;

use App\Models\McpTokenMetadata;
use App\Models\User;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Str;

class McpTokenService
{
    public function createToken(
        User $user,
        string $clientName,
        array $scopes,
        array $options = []
    ): array {
        $expiresAt = isset($options['expires_days'])
            ? now()->addDays($options['expires_days'])
            : now()->addDays(90);

        $plainTextToken = $user->createToken(
            name: "mcp:{$clientName}",
            abilities: $scopes,
            expiresAt: $expiresAt
        );

        $token = $user->tokens()->where('name', "mcp:{$clientName}")->first();

        McpTokenMetadata::create([
            'personal_access_token_id' => $token->id,
            'client_name' => $clientName,
            'client_version' => $options['client_version'] ?? null,
            'protocol_version' => $options['protocol_version'] ?? '2024-11-05',
            'permissions_level' => $this->determinePermissionsLevel($scopes),
            'rate_limit_override' => $options['rate_limit'] ?? null,
            'allowed_tools' => $options['allowed_tools'] ?? null,
            'blocked_tools' => $options['blocked_tools'] ?? null,
            'ip_whitelist' => $options['ip_whitelist'] ?? null,
        ]);

        activity('mcp-token')
            ->performedOn($user)
            ->withProperties([
                'client_name' => $clientName,
                'scopes' => $scopes,
                'expires_at' => $expiresAt->toIso8601String(),
            ])
            ->log('MCP token created');

        return [
            'token' => $plainTextToken,
            'token_id' => $token->id,
            'expires_at' => $expiresAt,
            'scopes' => $scopes,
        ];
    }

    public function revokeToken(User $user, int $tokenId): bool
    {
        $token = $user->tokens()
            ->where('id', $tokenId)
            ->where('name', 'like', 'mcp:%')
            ->first();

        if (! $token) {
            return false;
        }

        $clientName = str_replace('mcp:', '', $token->name);

        $token->delete();

        activity('mcp-token')
            ->performedOn($user)
            ->withProperties(['client_name' => $clientName])
            ->log('MCP token revoked');

        return true;
    }

    public function listTokens(User $user): array
    {
        return $user->tokens()
            ->where('name', 'like', 'mcp:%')
            ->get()
            ->map(function ($token) {
                $metadata = $token->metadata;
                return [
                    'id' => $token->id,
                    'client_name' => $metadata?->client_name ?? str_replace('mcp:', '', $token->name),
                    'client_version' => $metadata?->client_version,
                    'protocol_version' => $metadata?->protocol_version,
                    'scopes' => $token->abilities,
                    'permissions_level' => $metadata?->permissions_level,
                    'last_used_at' => $token->last_used_at?->diffForHumans(),
                    'last_ip' => $metadata?->last_ip_address,
                    'request_count' => $metadata?->request_count ?? 0,
                    'error_count' => $metadata?->error_count ?? 0,
                    'expires_at' => $token->expires_at?->diffForHumans(),
                    'is_expired' => $token->expires_at && $token->expires_at->isPast(),
                    'created_at' => $token->created_at->diffForHumans(),
                ];
            })
            ->toArray();
    }

    public function validateToken(\Laravel\Sanctum\PersonalAccessToken $token): array
    {
        $errors = [];

        if ($token->expires_at && $token->expires_at->isPast()) {
            $errors[] = 'Token has expired';
        }

        $user = $token->tokenable;
        if (! $user || method_exists($user, 'isSuspended') && $user->isSuspended()) {
            $errors[] = 'Account is suspended';
        }

        if ($token->metadata && $token->metadata->ip_whitelist) {
            $currentIp = request()->ip();
            if (! in_array($currentIp, $token->metadata->ip_whitelist)) {
                $errors[] = 'IP address not whitelisted';
            }
        }

        return [
            'valid' => empty($errors),
            'errors' => $errors,
        ];
    }

    private function determinePermissionsLevel(array $scopes): string
    {
        if (in_array('mcp:admin', $scopes)) {
            return 'admin';
        }

        if (in_array('mcp:write', $scopes)) {
            return 'write';
        }

        return 'read';
    }
}
```

## Enhanced MCP Authentication Middleware

```php
// app/Http/Middleware/McpAuthMiddleware.php

namespace App\Http\Middleware;

use App\Models\McpTokenMetadata;
use App\Services\McpTokenService;
use Closure;
use Illuminate\Http\Request;
use Illuminate\Http\Response;
use Laravel\Sanctum\PersonalAccessToken;

class McpAuthMiddleware
{
    public function __construct(
        private McpTokenService $tokenService
    ) {}

    public function handle(Request $request, Closure $next): Response
    {
        $token = $request->bearerToken();

        if (! $token) {
            return response()->json([
                'error' => 'unauthorized',
                'message' => 'Bearer token required',
            ], 401);
        }

        $accessToken = PersonalAccessToken::findToken($token);

        if (! $accessToken) {
            return response()->json([
                'error' => 'unauthorized',
                'message' => 'Invalid token',
            ], 401);
        }

        // Validate token (expiration, suspension, IP whitelist)
        $validation = $this->tokenService->validateToken($accessToken);

        if (! $validation['valid']) {
            return response()->json([
                'error' => 'forbidden',
                'message' => implode(', ', $validation['errors']),
            ], 403);
        }

        $user = $accessToken->tokenable;

        // Update token metadata
        $metadata = $accessToken->metadata;
        if ($metadata) {
            $metadata->update([
                'last_ip_address' => $request->ip(),
                'request_count' => $metadata->request_count + 1,
            ]);
        }

        // Check scope for the requested method
        $method = $request->input('method');
        if ($method && ! $this->hasScopeForMethod($accessToken, $method)) {
            return response()->json([
                'error' => 'insufficient_scope',
                'message' => "Token lacks required scope for method: {$method}",
            ], 403);
        }

        // Set authenticated user for the request
        auth()->setUser($user);

        // Log MCP access
        activity('mcp-access')
            ->performedOn($user)
            ->withProperties([
                'token_name' => $accessToken->name,
                'method' => $method,
                'ip' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ])
            ->log('MCP request processed');

        // Inject token and metadata into request for downstream use
        $request->merge([
            '_mcp_token' => $accessToken,
            '_mcp_metadata' => $metadata,
            '_mcp_user' => $user,
        ]);

        return $next($request);
    }

    private function hasScopeForMethod(PersonalAccessToken $token, string $method): bool
    {
        $methodScopeMap = [
            'tools/list' => 'mcp:read',
            'tools/invoke' => 'mcp:write',
            'resources/read' => 'mcp:read',
            'resources/write' => 'mcp:write',
            'prompts/get' => 'mcp:read',
            'sampling/create' => 'mcp:write',
        ];

        $requiredScope = $methodScopeMap[$method] ?? 'mcp:read';

        return in_array($requiredScope, $token->abilities)
            || in_array('mcp:admin', $token->abilities);
    }
}
```

## MCP Scope Validation Middleware

```php
// app/Http/Middleware/McpScopeValidation.php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Http\Response;

class McpScopeValidation
{
    public function handle(Request $request, Closure $next): Response
    {
        $requiredScopes = $this->getRequiredScopes($request);
        $tokenScopes = $request->input('_mcp_token')->abilities ?? [];

        foreach ($requiredScopes as $scope) {
            if (! in_array($scope, $tokenScopes)) {
                return response()->json([
                    'error' => 'insufficient_scope',
                    'message' => "Required scope: {$scope}",
                    'required' => $requiredScopes,
                    'provided' => $tokenScopes,
                ], 403);
            }
        }

        return $next($request);
    }

    private function getRequiredScopes(Request $request): array
    {
        $method = $request->input('method', '');

        $scopeRequirements = [
            'tools/list' => ['mcp:read'],
            'tools/invoke' => ['mcp:write'],
            'resources/read' => ['mcp:read'],
            'resources/write' => ['mcp:write'],
            'resources/list' => ['mcp:read'],
            'prompts/list' => ['mcp:read'],
            'prompts/get' => ['mcp:read'],
            'sampling/create' => ['mcp:write', 'sampling:write'],
            'completion/complete' => ['mcp:read'],
        ];

        return $scopeRequirements[$method] ?? ['mcp:read'];
    }
}
```

```php
// app/Providers/RouteServiceProvider.php (or AppServiceProvider)

use Illuminate\Cache\RateLimiting\Limit;
use Illuminate\Support\Facades\RateLimiter;

RateLimiter::for('mcp', function (Request $request) {
    $token = $request->input('_mcp_token');
    $metadata = $request->input('_mcp_metadata');

    $perMinute = $metadata?->rate_limit_override ?? config('mcp.rate_limits.per_minute', 30);
    $perHour = config('mcp.rate_limits.per_hour', 500);

    return [
        Limit::perMinute($perMinute)->by($token?->id ?? $request->ip()),
        Limit::perHour($perHour)->by($token?->id ?? $request->ip()),
    ];
});

RateLimiter::for('mcp-auth', function (Request $request) {
    return Limit::perMinute(10)->by($request->ip());
});
```

```php
// In MCP Controller or Exception Handler

public function handleRateLimitExceeded(Request $request, string $key)
```

### Tool Authorization Service

```php
// app/Services/McpToolAuthorizationService.php

namespace App\Services;

use App\Models\McpTokenMetadata;
use Illuminate\Http\Request;
use Laravel\Sanctum\PersonalAccessToken;

class McpToolAuthorizationService
{
    public function authorizeToolInvocation(
        PersonalAccessToken $token,
        string $toolName,
        Request $request
    ): array {
        $toolConfig = config("mcp-tools.tools.{$toolName}");

        if (! $toolConfig) {
            return [
                'authorized' => false,
                'error' => 'Tool not found',
                'code' => 'tool_not_found',
            ];
        }

        // Check token scopes against tool requirements
        $requiredScopes = $toolConfig['required_scopes'] ?? ['mcp:read'];
        $tokenScopes = $token->abilities;

        $hasScope = collect($requiredScopes)->every(
            fn ($scope) => in_array($scope, $tokenScopes)
                || in_array('mcp:admin', $tokenScopes)
        );

        if (! $hasScope) {
            return [
                'authorized' => false,
                'error' => 'Insufficient permissions',
                'code' => 'insufficient_scope',
                'required_scopes' => $requiredScopes,
                'provided_scopes' => $tokenScopes,
            ];
        }

        // Check tool-level restrictions from token metadata
        $metadata = $token->metadata;

        if ($metadata) {
            // Check allowed tools whitelist
            if ($metadata->allowed_tools && ! in_array($toolName, $metadata->allowed_tools)) {
                return [
                    'authorized' => false,
                    'error' => 'Tool not in allowed list',
                    'code' => 'tool_not_allowed',
                ];
            }

            // Check blocked tools blacklist
            if ($metadata->blocked_tools && in_array($toolName, $metadata->blocked_tools)) {
                return [
                    'authorized' => false,
                    'error' => 'Tool is blocked',
                    'code' => 'tool_blocked',
                ];
            }
        }

        // Check rate limits for this specific tool
        $rateLimitKey = "mcp-tool:{$token->id}:{$toolName}";
        $toolRateLimit = config("mcp-tools.tools.{$toolName}.rate_limit", 100);

        if (rate_limiter()->tooManyAttempts($rateLimitKey, $toolRateLimit)) {
            $retryAfter = rate_limiter()->availableIn($rateLimitKey);

            return [
                'authorized' => false,
                'error' => 'Tool rate limit exceeded',
                'code' => 'rate_limited',
                'retry_after' => $retryAfter,
            ];
        }

        rate_limiter()->hit($rateLimitKey, 60);

        return [
            'authorized' => true,
            'tool_config' => $toolConfig,
        ];
    }
}
```

## MCP Streaming Authentication

### SSE Authentication

```php
// app/Http/Controllers/McpStreamController.php

namespace App\Http\Controllers;

use App\Services\McpTokenService;
use Illuminate\Http\Request;
use Illuminate\Http\Response;
use Laravel\Sanctum\PersonalAccessToken;
use Symfony\Component\HttpFoundation\StreamedResponse;

class McpStreamController extends Controller
{
    public function stream(Request $request): StreamedResponse
    {
        $token = $request->bearerToken();

        if (! $token) {
            return response()->json(['error' => 'Token required'], 401);
        }

        $accessToken = PersonalAccessToken::findToken($token);

        if (! $accessToken) {
            return response()->json(['error' => 'Invalid token'], 401);
        }

        $validation = app(McpTokenService::class)->validateToken($accessToken);

        if (! $validation['valid']) {
            return response()->json([
                'error' => 'forbidden',
                'message' => implode(', ', $validation['errors']),
            ], 403);
        }

        $user = $accessToken->tokenable;

        return response()->stream(function () use ($request, $user, $accessToken) {
            // Send initial connection event
            echo "event: connected\n";
            echo "data: " . json_encode([
                'user' => $user->id,
                'token_id' => $accessToken->id,
                'capabilities' => $this->getCapabilities($accessToken),
            ]) . "\n\n";
            ob_flush();
            flush();

            // Keep connection alive and process events
            while (true) {
                // Check if client disconnected
                if (connection_aborted()) {
                    break;
                }

                // Check token validity periodically
                if ($accessToken->fresh()->expires_at?->isPast()) {
                    echo "event: error\n";
                    echo "data: " . json_encode(['error' => 'Token expired']) . "\n\n";
                    ob_flush();
                    flush();
                    break;
                }

                // Process any pending events
                $events = $this->getPendingEvents($user);
                foreach ($events as $event) {
                    echo "event: {$event['type']}\n";
                    echo "data: " . json_encode($event['data']) . "\n\n";
```

```php
// app/Exceptions/McpExceptionHandler.php

namespace App\Exceptions;

use Illuminate\Foundation\Exceptions\Handler as ExceptionHandler;
use Illuminate\Http\JsonResponse;
use Throwable;

class McpExceptionHandler extends ExceptionHandler
{
    public function render($request, Throwable $e): JsonResponse
    {
        if (! $request->is('mcp/*')) {
            return parent::render($request, $e);
        }

        $error = match (true) {
            $e instanceof \Symfony\Component\HttpKernel\Exception\UnauthorizedHttpException => [
                'error' => 'unauthorized',
                'code' => -32001,
                'message' => 'Authentication required',
            ],
            $e instanceof \Symfony\Component\HttpKernel\Exception\ForbiddenHttpException => [
                'error' => 'forbidden',
                'code' => -32002,
                'message' => 'Insufficient permissions',
            ],
            $e instanceof \Symfony\Component\HttpKernel\Exception\TooManyRequestsHttpException => [
                'error' => 'rate_limited',
                'code' => -32003,
                'message' => 'Too many requests',
                'retry_after' => $e->getHeaders()['Retry-After'] ?? 60,
            ],
            $e instanceof \Symfony\Component\HttpKernel\Exception\NotFoundHttpException => [
                'error' => 'not_found',
                'code' => -32601,
                'message' => 'Method not found',
            ],
            default => [
                'error' => 'internal_error',
                'code' => -32603,
                'message' => config('app.debug') ? $e->getMessage() : 'Internal error',
            ],
        };

        // Log error
        activity('mcp-error')
            ->withProperties([
                'error' => $error['error'],
                'message' => $e->getMessage(),
                'trace' => config('app.debug') ? $e->getTraceAsString() : null,
            ])
            ->log('MCP error: ' . $error['error']);

        return response()->json([
            'jsonrpc' => '2.0',
            'error' => $error,
            'id' => $request->input('id'),
        ], $this->mapErrorCode($error['code']));
    }

    private function mapErrorCode(int $code): int
    {
        return match (true) {
            $code === -32001 => 401,
            $code === -32002 => 403,
            $code === -32003 => 429,
            $code === -32601 => 404,
            default => 500,
        };
    }
}
```

```php
// app/Services/McpRequestLogger.php

namespace App\Services;

use App\Models\McpRequestLog;
use Illuminate\Http\Request;

class McpRequestLogger
{
    public function logIncoming(Request $request, $token): void
    {
        if (! config('mcp.authentication.log_all_requests')) {
            return;
        }

        McpRequestLog::create([
            'user_id' => $token->tokenable_id,
            'personal_access_token_id' => $token->id,
            'method' => $request->input('method'),
            'params' => $request->input('params'),
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
            'direction' => 'incoming',
        ]);
    }

    public function logOutgoing(Request $request, $response, $token): void
    {
        McpRequestLog::create([
            'user_id' => $token->tokenable_id,
            'personal_access_token_id' => $token->id,
            'method' => $request->input('method'),
            'response' => is_array($response) ? $response : ['result' => $response],
            'ip_address' => $request->ip(),
            'duration_ms' => $this->calculateDuration($request),
            'direction' => 'outgoing',
            'is_error' => isset($response['error']),
        ]);
    }

    public function logError(Request $request, \Throwable $error, $token): void
    {
        McpRequestLog::create([
            'user_id' => $token->tokenable_id,
            'personal_access_token_id' => $token->id,
            'method' => $request->input('method'),
            'error' => [
                'message' => $error->getMessage(),
                'code' => $error->getCode(),
            ],
            'ip_address' => $request->ip(),
            'direction' => 'error',
            'is_error' => true,
        ]);

        // Update error count on token metadata
        $metadata = $token->metadata;
        if ($metadata) {
            $metadata->increment('error_count');
        }
    }

    public function getRequestStats(int $userId, string $period = '24h'): array
    {
        $query = McpRequestLog::where('user_id', $userId);

        $query->when($period === '24h', fn ($q) => $q->where('created_at', '>=', now()->subDay()));
        $query->when($period === '7d', fn ($q) => $q->where('created_at', '>=', now()->subDays(7)));
        $query->when($period === '30d', fn ($q) => $q->where('created_at', '>=', now()->subDays(30)));

        return [
            'total_requests' => $query->count(),
            'successful_requests' => (clone $query)->where('is_error', false)->count(),
            'failed_requests' => (clone $query)->where('is_error', true)->count(),
            'avg_duration_ms' => (clone $query)->avg('duration_ms'),
            'unique_methods' => (clone $query)->distinct('method')->count('method'),
            'unique_tokens' => (clone $query)->distinct('personal_access_token_id')->count('personal_access_token_id'),
        ];
    }

    private function calculateDuration(Request $request): float
    {
        $startTime = $request->server('REQUEST_TIME_FLOAT');
        return $startTime ? (microtime(true) - $startTime) * 1000 : 0;
    }
}
```

```php
            'selectedScopes' => 'required|array|min:1',
            'expiresDays' => 'required|integer|min:1|max:365',
        ]);

        $result = $this->tokenService->createToken(
            user: auth()->user(),
            clientName: $this->clientName,
            scopes: $this->selectedScopes,
            options: ['expires_days' => $this->expiresDays]
        );

        $this->newToken = $result['token'];
        $this->dispatch('token-created', token: $result['token']);

    {
        $this->revokeTokenId = $tokenId;

    }

    public function revokeToken(): void
    {
        if ($this->revokeTokenId) {
            $this->tokenService->revokeToken(auth()->user(), $this->revokeTokenId);
            $this->dispatch('notify', type: 'success', message: 'MCP token revoked');
        }
        $this->revokeTokenId = null;
    }

    public function dismissToken(): void
    {
        $this->newToken = null;
    }

    public function render()
    {
        return view('livewire.mcp-token-manager');
    }
```

```php
// app/Services/McpTokenAnalyticsService.php

namespace App\Services;

use App\Models\McpRequestLog;
use App\Models\McpTokenMetadata;
use App\Models\PersonalAccessToken;
use Illuminate\Support\Carbon;
use Illuminate\Support\Collection;

class McpTokenAnalyticsService
{
    public function getTokenUsageSummary(int $tokenId, string $period = '30d'): array
    {
        $startDate = $this->getStartDate($period);

        $logs = McpRequestLog::where('personal_access_token_id', $tokenId)
            ->where('created_at', '>=', $startDate)
            ->get();

        return [
            'total_requests' => $logs->count(),
            'successful_requests' => $logs->where('is_error', false)->count(),
            'failed_requests' => $logs->where('is_error', true)->count(),
            'success_rate' => $logs->count() > 0
                ? round(($logs->where('is_error', false)->count() / $logs->count()) * 100, 2)
                : 0,
            'avg_response_time_ms' => round($logs->avg('duration_ms') ?? 0, 2),
            'max_response_time_ms' => round($logs->max('duration_ms') ?? 0, 2),
            'min_response_time_ms' => round($logs->min('duration_ms') ?? 0, 2),
            'unique_methods' => $logs->pluck('method')->unique()->toArray(),
            'method_breakdown' => $logs->groupBy('method')->map(fn ($group) => [
                'count' => $group->count(),
                'avg_duration' => round($group->avg('duration_ms') ?? 0, 2),
                'error_rate' => $group->count() > 0
                    ? round(($group->where('is_error', true)->count() / $group->count()) * 100, 2)
                    : 0,
            ])->toArray(),
            'requests_by_day' => $this->groupByDay($logs),
            'errors_by_type' => $logs->where('is_error', true)
                ->pluck('error')
                ->filter()
                ->groupBy('code')
                ->map->count()
                ->toArray(),
        ];
    }

    public function getDailyUsageChart(int $tokenId, int $days = 30): array
    {
        $startDate = Carbon::now()->subDays($days);

        $logs = McpRequestLog::where('personal_access_token_id', $tokenId)
            ->where('created_at', '>=', $startDate)
            ->selectRaw('DATE(created_at) as date, COUNT(*) as total, SUM(CASE WHEN is_error = 1 THEN 1 ELSE 0 END) as errors')
            ->groupBy('date')
            ->orderBy('date')
            ->get();

        $chartData = [];
        for ($i = $days; $i >= 0; $i--) {
            $date = Carbon::now()->subDays($i)->format('Y-m-d');
            $dayData = $logs->firstWhere('date', $date);

            $chartData[] = [
                'date' => $date,
                'total' => $dayData?->total ?? 0,
                'errors' => $dayData?->errors ?? 0,
                'success' => ($dayData?->total ?? 0) - ($dayData?->errors ?? 0),
            ];
        }

        return $chartData;
    }

    public function getTopMethods(int $tokenId, int $limit = 10, string $period = '30d'): array
    {
        $startDate = $this->getStartDate($period);

        return McpRequestLog::where('personal_access_token_id', $tokenId)
            ->where('created_at', '>=', $startDate)
            ->selectRaw('method, COUNT(*) as count, AVG(duration_ms) as avg_duration')
            ->groupBy('method')
            ->orderByDesc('count')
            ->limit($limit)
            ->get()
            ->toArray();
    }

    public function getErrorTrends(int $tokenId, string $period = '30d'): array
    {
        $startDate = $this->getStartDate($period);

        return McpRequestLog::where('personal_access_token_id', $tokenId)
            ->where('is_error', true)
            ->where('created_at', '>=', $startDate)
            ->selectRaw('
                DATE(created_at) as date,
                error->>"$.code" as error_code,
                error->>"$.message" as error_message,
                COUNT(*) as count
            ')
            ->groupBy('date', 'error_code', 'error_message')
            ->orderByDesc('date')
            ->orderByDesc('count')
            ->get()
            ->toArray();
    }

    public function getGeoDistribution(int $tokenId, string $period = '30d'): array
    {
        $startDate = $this->getStartDate($period);

        return McpRequestLog::where('personal_access_token_id', $tokenId)
            ->where('created_at', '>=', $startDate)
            ->selectRaw('ip_address, COUNT(*) as count')
            ->groupBy('ip_address')
            ->orderByDesc('count')
            ->limit(20)
            ->get()
            ->toArray();
    }

    public function getAllTokensSummary(int $userId): array
    {
        $tokens = PersonalAccessToken::where('user_id', $userId)
            ->where('name', 'like', 'mcp:%')
            ->get();

        return $tokens->map(function ($token) {
            $metadata = $token->metadata;
            $recentRequests = McpRequestLog::where('personal_access_token_id', $token->id)
                ->where('created_at', '>=', now()->subDays(30))
                ->count();

            return [
                'id' => $token->id,
                'name' => $metadata?->client_name ?? str_replace('mcp:', '', $token->name),
                'is_expired' => $token->expires_at && $token->expires_at->isPast(),
                'is_active' => $recentRequests > 0,
                'requests_30d' => $recentRequests,
                'last_used' => $token->last_used_at?->diffForHumans(),
                'permissions_level' => $metadata?->permissions_level,
                'error_count' => $metadata?->error_count ?? 0,
            ];
        })->toArray();
    }

    private function groupByDay(Collection $logs): array
    {
        return $logs->groupBy(fn ($log) => $log->created_at->format('Y-m-d'))
            ->map(fn ($group, $date) => [
                'date' => $date,
                'count' => $group->count(),
                'errors' => $group->where('is_error', true)->count(),
            ])
            ->values()
            ->toArray();
    }

    private function getStartDate(string $period): Carbon
    {
        return match ($period) {
            '24h' => now()->subDay(),
            '7d' => now()->subDays(7),
            '30d' => now()->subDays(30),
            '90d' => now()->subDays(90),
            '365d' => now()->subYear(),
            default => now()->subDays(30),
        };
    }
```

```php
// app/Listeners/DetectMcpTokenCompromise.php

namespace App\Listeners;

use App\Models\McpRequestLog;
use App\Models\PersonalAccessToken;
use App\Notifications\McpTokenCompromiseNotification;
use Illuminate\Support\Facades\Cache;

class DetectMcpTokenCompromise
{
    public function handle(object $event): void
    {
        if (! property_exists($event, 'token')) {
            return;
        }

        $token = $event->token;

        // Check for usage from multiple IPs in short time
        $recentIps = McpRequestLog::where('personal_access_token_id', $token->id)
            ->where('created_at', '>=', now()->subMinutes(5))
            ->distinct('ip_address')
            ->pluck('ip_address')
            ->toArray();

        if (count($recentIps) > 3) {
            $cacheKey = "mcp_compromise_alert:{$token->id}";

            if (! Cache::has($cacheKey)) {
                $user = $token->tokenable;

                if ($user) {
                    $user->notify(new McpTokenCompromiseNotification(
                        $token,
                        'Multiple IP addresses detected within 5 minutes'
                    ));
                }

                Cache::put($cacheKey, true, now()->addHours(1));
            }
        }

        // Check for unusual method access patterns
        $recentMethods = McpRequestLog::where('personal_access_token_id', $token->id)
            ->where('created_at', '>=', now()->subMinutes(10))
            ->pluck('method')
            ->unique()
            ->toArray();

        $unusualMethods = array_diff($recentMethods, ['tools/list', 'resources/read', 'prompts/list']);

        if (count($unusualMethods) > 2) {
            $cacheKey = "mcp_compromise_methods:{$token->id}";

            if (! Cache::has($cacheKey)) {
                $user = $token->tokenable;

                if ($user) {
                    $user->notify(new McpTokenCompromiseNotification(
                        $token,
                        'Unusual method access pattern detected'
                    ));
                }

                Cache::put($cacheKey, true, now()->addHours(1));
            }
        }
    }
}
```

### Token Expiry Notification

```php
// app/Listeners/NotifyMcpTokenExpiry.php

namespace App\Listeners;

use App\Models\PersonalAccessToken;
use App\Notifications\McpTokenExpiringNotification;
use Illuminate\Support\Facades\DB;

class NotifyMcpTokenExpiry
{
    public function handle(object $event): void
    {
        // Runs via scheduled job, not event-driven
        $expiringTokens = PersonalAccessToken::where('name', 'like', 'mcp:%')
            ->where('expires_at', '!=', null)
            ->where('expires_at', '>=', now())
            ->where('expires_at', '<=', now()->addDays(7))
            ->get();

        foreach ($expiringTokens as $token) {
            $user = $token->tokenable;

            if (! $user) {
                continue;
            }

            $daysUntilExpiry = now()->diffInDays($token->expires_at);

            // Don't spam — notify at 7 days, 3 days, 1 day
            if (in_array($daysUntilExpiry, [7, 3, 1])) {
                $user->notify(new McpTokenExpiringNotification($token, $daysUntilExpiry));
            }
        }
    }
}
```

## MCP Token Controller

```php
// app/Http/Controllers/McpTokenController.php

namespace App\Http\Controllers;

use App\Services\McpTokenService;
use App\Services\McpTokenAnalyticsService;
use Illuminate\Http\Request;

class McpTokenController extends Controller
{
    public function index()
    {
        return view('settings.mcp-tokens');
    }

    public function analytics(int $tokenId)
    {
        $token = auth()->user()->tokens()
            ->where('id', $tokenId)
            ->where('name', 'like', 'mcp:%')
            ->firstOrFail();

        return view('settings.mcp-token-analytics', [
            'token' => $token,
            'tokenId' => $tokenId,
        ]);
    }
}
```
