# 12.22 — API

**Project:** GsmUi
**Domain:** Subscription Management
**Version:** 1.0.0

---

## Purpose

Define the REST API endpoints for subscription operations — both for internal consumption (admin, user dashboard) and external consumption (API consumers, MCP clients).

## API Architecture

```text
Client (Admin UI / User Dashboard / External API)
    ↓
API Gateway (Laravel Routes)
    ↓
Middleware (auth, sanctum, entitlement)
    ↓
Controllers
    ↓
Services (SubscriptionService, EntitlementService)
    ↓
Domain Model
```

## Authentication

| Endpoint | Auth Method |
|---|---|
| User-facing | `auth:sanctum` |
| Admin-only | `auth:sanctum` + `role:admin` |
| Public (limited) | None |

## Endpoints

### Plans

| Method | Path | Auth | Purpose |
|---|---|---|---|
| GET | `/api/plans` | Public | List active plans |
| GET | `/api/plans/{id}` | Public | Get plan details |
| POST | `/api/plans` | Admin | Create plan |
| PUT | `/api/plans/{id}` | Admin | Update plan |
| DELETE | `/api/plans/{id}` | Admin | Archive plan |

```php
Route::get('/api/plans', [PlanController::class, 'index']);
Route::get('/api/plans/{plan}', [PlanController::class, 'show']);
Route::post('/api/plans', [PlanController::class, 'store'])->middleware('role:admin');
Route::put('/api/plans/{plan}', [PlanController::class, 'update'])->middleware('role:admin');
Route::delete('/api/plans/{plan}', [PlanController::class, 'destroy'])->middleware('role:admin');
```

### Subscriptions

| Method | Path | Auth | Purpose |
|---|---|---|---|
| GET | `/api/subscriptions` | User | List user subscriptions |

### Plan Controller

```php
class PlanController extends Controller
{
    public function index(): JsonResponse
    {
        $plans = Plan::where('is_active', true)
            ->where('is_visible', true)
            ->orderBy('sort_order')
            ->get();

        return response()->json([
            'data' => PlanResource::collection($plans),
        ]);
    }

    public function show(Plan $plan): JsonResponse
    {
        if (!$plan->is_active || !$plan->is_visible) {
            return response()->json(['error' => 'Plan not found'], 404);
        }

        return response()->json([
            'data' => new PlanResource($plan),
        ]);
    }

    public function store(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'slug' => 'required|string|max:255|alpha_dash|unique:plans,slug',
            'description' => 'nullable|string|max:1000',
            'price' => 'required|integer|min:0',
            'currency' => 'required|string|in:usd,eur,gbp',
            'interval' => 'required|in:monthly,yearly',
            'trial_days' => 'required|integer|min:0',
            'max_team_members' => 'required|integer|min:1',
            'max_downloads_per_month' => 'required|integer|min:0',
            'features' => 'array',
            'is_active' => 'boolean',
            'is_visible' => 'boolean',
            'sort_order' => 'required|integer|min:0',
            'stripe_price_id' => 'nullable|string',
        ]);

        $plan = Plan::create($validated);

        PlanCreated::dispatch($plan);

        return response()->json([
            'data' => new PlanResource($plan),
        ], 201);
    }

    public function update(Request $request, Plan $plan): JsonResponse
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'slug' => 'required|string|max:255|alpha_dash|unique:plans,slug,' . $plan->id,
            'description' => 'nullable|string|max:1000',
            'price' => 'required|integer|min:0',
            'currency' => 'required|string|in:usd,eur,gbp',
            'interval' => 'required|in:monthly,yearly',
            'trial_days' => 'required|integer|min:0',
            'max_team_members' => 'required|integer|min:1',
            'max_downloads_per_month' => 'required|integer|min:0',
            'features' => 'array',
            'is_active' => 'boolean',
            'is_visible' => 'boolean',
            'sort_order' => 'required|integer|min:0',
            'stripe_price_id' => 'nullable|string',
        ]);

        $plan->update($validated);

        PlanUpdated::dispatch($plan, $validated);

        return response()->json([
            'data' => new PlanResource($plan),
        ]);
    }

    public function destroy(Plan $plan): JsonResponse
    {
        $plan->update(['is_active' => false]);

        PlanArchived::dispatch($plan);

        return response()->json(['message' => 'Plan archived.']);
    }
}
```

### Subscription Controller

```php
class SubscriptionController extends Controller
{
    public function index(): JsonResponse
    {
        $subscriptions = auth()->user()
            ->subscriptions()
            ->with('plan')
            ->orderByDesc('created_at')
            ->get();

        return response()->json([
            'data' => SubscriptionResource::collection($subscriptions),
        ]);
    }

    public function active(): JsonResponse
    {
        $subscription = auth()->user()->activeSubscription();

        if (!$subscription) {
            return response()->json([
                'error' => 'no_active_subscription',
                'message' => 'No active subscription found.',
            ], 404);
        }

        return response()->json([
            'data' => new SubscriptionResource($subscription->load('plan')),
        ]);
    }

    public function store(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'plan_id' => 'required|exists:plans,id',
            'payment_method_id' => 'nullable|string',
            'coupon_code' => 'nullable|string',
        ]);

        $plan = Plan::findOrFail($validated['plan_id']);
        $user = auth()->user();

        try {
            $subscription = app(SubscriptionService::class)->create(
                user: $user,
                plan: $plan,
                paymentMethodId: $validated['payment_method_id'] ?? null,
                couponCode: $validated['coupon_code'] ?? null,
            );

            return response()->json([
                'data' => new SubscriptionResource($subscription),
            ], 201);
        } catch (PaymentFailedException $e) {
            return response()->json([
                'error' => 'payment_failed',
                'message' => $e->getMessage(),
            ], 422);
        }
    }

    public function show(Subscription $subscription): JsonResponse
    {
        if ($subscription->user_id !== auth()->id()) {
            return response()->json(['error' => 'Unauthorized'], 403);
        }

        return response()->json([
            'data' => new SubscriptionResource($subscription->load(['plan', 'entitlements'])),
        ]);
    }

    public function cancel(Subscription $subscription): JsonResponse
    {
        if ($subscription->user_id !== auth()->id()) {
            return response()->json(['error' => 'Unauthorized'], 403);
        }

        $this->validate($request ?? request(), [
            'reason' => 'nullable|string',
            'feedback' => 'nullable|string',
        ]);

        app(CancellationService::class)->cancel(
            $subscription,
            reason: request('reason'),
            immediate: false,
        );

        return response()->json([
            'data' => new SubscriptionResource($subscription->fresh()),
            'message' => 'Subscription canceled. Access continues until the end of the billing period.',
        ]);
    }

    public function resume(Subscription $subscription): JsonResponse
    {
        if ($subscription->user_id !== auth()->id()) {
            return response()->json(['error' => 'Unauthorized'], 403);
        }

        app(SubscriptionService::class)->resume($subscription);

        return response()->json([
            'data' => new SubscriptionResource($subscription->fresh()),
            'message' => 'Subscription resumed successfully.',
        ]);
    }

    public function swap(Request $request, Subscription $subscription): JsonResponse
    {
        if ($subscription->user_id !== auth()->id()) {
            return response()->json(['error' => 'Unauthorized'], 403);
        }

        $validated = $request->validate([
            'plan_id' => 'required|exists:plans,id',
        ]);

        $newPlan = Plan::findOrFail($validated['plan_id']);

        try {
            if ($newPlan->price > $subscription->plan->price) {
                app(PlanChangeService::class)->upgradePlan($subscription, $newPlan);
            } else {
                app(PlanChangeService::class)->downgradePlan($subscription, $newPlan);
            }

            return response()->json([
                'data' => new SubscriptionResource($subscription->fresh()),
                'message' => 'Plan changed successfully.',
            ]);
        } catch (Exception $e) {
            return response()->json([
                'error' => 'plan_change_failed',
                'message' => $e->getMessage(),
            ], 422);
        }
    }
}
```

```php
class EntitlementController extends Controller
{
    public function index(): JsonResponse
    {
        $user = auth()->user();
        $entitlementService = app(EntitlementService::class);

        $entitlements = $entitlementService->getAll($user);

        return response()->json([

                    'value' => $e->value,

                    'expires_at' => $e->expires_at?->toISOString(),
                ]),
                'features' => $entitlementService->getFeatures($user),
                'limits' => $entitlementService->getLimits($user),
            ],
        ]);
    }

    public function check(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'key' => 'required|string',
        ]);

        $user = auth()->user();
        $entitlementService = app(EntitlementService::class);

        $hasEntitlement = $entitlementService->has($user, $validated['key']);
        $value = $entitlementService->value($user, $validated['key']);

        return response()->json([
            'data' => [
                'key' => $validated['key'],
                'has' => $hasEntitlement,
                'value' => $value,
            ],
        ]);
    }
}
```

```php
class UsageController extends Controller
{
    public function index(): JsonResponse
    {
        $user = auth()->user();
        $usage = app(UsageTrackingService::class)->getUsageSummary($user);

        return response()->json([
            'data' => $usage,
        ]);
    }

    public function show(string $metric): JsonResponse
    {
        $user = auth()->user();
        $subscription = $user->activeSubscription();

        if (!$subscription) {
            return response()->json([
                'error' => 'no_active_subscription',
            ], 404);
        }

        $usageService = app(UsageTrackingService::class);

        $used = $usageService->getPeriodUsage($user, $metric, $subscription);
        $limit = app(EntitlementService::class)->value($user, "limit.{$metric}_per_month");

        return response()->json([
            'data' => [
                'metric' => $metric,
                'used' => $used,
                'limit' => $limit,
                'unlimited' => $limit === null,
                'percentage' => $limit ? round(($used / $limit) * 100, 1) : null,
                'reset_date' => $subscription->current_period_end->toISOString(),
            ],
        ]);
    }
}
```

```php
class AdminSubscriptionController extends Controller
{
    public function index(Request $request): JsonResponse
    {
        $query = Subscription::with(['user', 'plan']);

        if ($request->status) {
            $query->where('status', $request->status);
        }

        if ($request->search) {
            $query->whereHas('user', fn ($q) => $q->where('email', 'like', "%{$request->search}%"));
        }

        $subscriptions = $query->orderByDesc('created_at')->paginate(25);

        return response()->json([
            'data' => SubscriptionResource::collection($subscriptions),
            'meta' => [
                'current_page' => $subscriptions->currentPage(),
                'last_page' => $subscriptions->lastPage(),
                'per_page' => $subscriptions->perPage(),
                'total' => $subscriptions->total(),
            ],
        ]);
    }

    public function update(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'plan_id' => 'sometimes|exists:plans,id',
            'quantity' => 'sometimes|integer|min:1',
        ]);

        $subscription->update($validated);

        return response()->json([
            'data' => new SubscriptionResource($subscription->fresh()),
        ]);
    }

    public function transition(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'status' => 'required|string|in:active,past_due,trialing,canceled,paused,unpaid',
            'reason' => 'required|string|min:3',
        ]);

        app(AdminStateTransition::class)->transition(
            $subscription,
            $validated['status'],
            $validated['reason']
        );

        return response()->json([
            'data' => new SubscriptionResource($subscription->fresh()),
            'message' => 'Subscription status updated.',
        ]);
    }

    public function refund(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'amount' => 'nullable|integer|min:1',
            'reason' => 'required|string|min:3',
        ]);

        app(AdminRefund::class)->refund(
            $subscription,
            $validated['amount'],
            $validated['reason']
        );

        return response()->json(['message' => 'Refund processed.']);
    }

    public function cancel(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'reason' => 'required|string|min:3',
        ]);

        app(AdminStateTransition::class)->transition(
            $subscription,
            'canceled',
            $validated['reason']
        );

        return response()->json(['message' => 'Subscription canceled.']);
    }
}
```

```php
    public function show(User $user): JsonResponse
    {
        $entitlements = app(EntitlementService::class)->getAll($user);

    public function grant(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'user_id' => 'required|exists:users,id',
            'key' => 'required|string',
            'value' => 'required|string',
            'expires_at' => 'nullable|date|after:now',

        $grant = app(GrantEntitlement::class)->grant(

            billable: $user,
            key: EntitlementKey::from($validated['key']),

        );

        return response()->json([
            'data' => [
                'id' => $grant->id,
                'key' => $grant->key,
                'value' => $grant->value,
                'expires_at' => $grant->expires_at?->toISOString(),
            ],
            'message' => 'Entitlement granted successfully.',
        ], 201);
    }

    {
        app(RevokeEntitlement::class)->revoke(
            admin: auth()->user(),
            grant: $grant,
        );

        return response()->json(['message' => 'Entitlement revoked.']);
    }
}
```

```php
class SubscriptionApiExceptionHandler
{
    public function handle(Exception $exception): JsonResponse
    {
        return match (true) {
            $exception instanceof NoActiveSubscriptionException => response()->json([
                'error' => 'no_active_subscription',
                'message' => 'No active subscription found. Please subscribe to access this feature.',
                'suggested_plan' => 'pro',
                'upgrade_url' => route('pricing'),
            ], 403),

            $exception instanceof EntitlementRequiredException => response()->json([
                'error' => 'entitlement_required',
                'message' => "This operation requires the '{$exception->getKey()}' entitlement.",
                'entitlement_key' => $exception->getKey(),
                'plan_required' => $exception->getRequiredPlan(),
                'upgrade_url' => route('billing.change'),
            ], 403),

            $exception instanceof UsageLimitExceededException => response()->json([
                'error' => 'usage_limit_reached',
                'message' => "You have reached your {$exception->getMetric()} limit.",
                'used' => $exception->getUsed(),
                'limit' => $exception->getLimit(),
                'reset_date' => $exception->getResetDate()->toISOString(),
                'upgrade_url' => route('billing.change'),
            ], 403),

            $exception instanceof PaymentFailedException => response()->json([
                'error' => 'payment_failed',
                'message' => $exception->getMessage(),
                'update_payment_url' => route('billing.payment'),
            ], 422),

            $exception instanceof ValidationException => response()->json([
                'error' => 'validation_error',
                'message' => 'The given data was invalid.',
                'errors' => $exception->errors(),
            ], 422),

            $exception instanceof ModelNotFoundException => response()->json([
                'error' => 'not_found',
                'message' => 'The requested resource was not found.',
            ], 404),

            default => response()->json([
                'error' => 'server_error',
                'message' => 'An unexpected error occurred.',
            ], 500),
        };
    }
}
```

```php
class SubscriptionFilter
{
    public function apply(Builder $query, Request $request): Builder
    {
        return $query
            ->when($request->status, fn ($q, $status) => $q->where('status', $status))
            ->when($request->plan_id, fn ($q, $planId) => $q->where('plan_id', $planId))
```

```php
// MetricsMiddleware
class MetricsMiddleware
{
    public function handle(Request $request, Closure $next): Response
    {
        $start = microtime(true);

        $response = $next($request);

        $duration = (microtime(true) - $start) * 1000;

        // Store metrics
        Metric::create([
            'endpoint' => $request->route()?->getName() ?? $request->path(),
            'method' => $request->method(),
            'status_code' => $response->getStatusCode(),
            'duration_ms' => $duration,
            'user_id' => $request->user()?->id,
            'timestamp' => now(),
        ]);

        // Log slow requests
        if ($duration > 500) {
            Log::warning('Slow API request', [
                'path' => $request->path(),
                'duration_ms' => $duration,
            ]);
        }

        $request->attributes->set('response_time_ms', $duration);

        return $response;
    }
}
```

## API CLI & Artisan Commands

### Token Management

```php
// artisan api:token:create
class CreateApiTokenCommand extends Command
{
    protected $signature = 'api:token:create
                            {--user= : User ID}
                            {--name=api-consumer : Token name}
                            {--scopes=* : Token scopes}';

    protected $description = 'Create an API token for a user';

    public function handle(): int
    {
        $user = User::find($this->option('user'));

        if (!$user) {
            $this->error('User not found.');
            return self::FAILURE;
        }

        $token = $user->createToken(
            $this->option('name'),
            $this->option('scopes')
        );

        $this->info("Token created: {$token->plainTextToken}");
        return self::SUCCESS;
    }
}
```

```php
// artisan api:token:revoke-all
class RevokeAllTokensCommand extends Command
{
    protected $signature = 'api:token:revoke-all
                            {--older-than=90 : Revoke tokens older than N days}
                            {--except= : Comma-separated token names to keep}';

    protected $description = 'Bulk revoke API tokens';

    public function handle(): int
    {
        $days = (int) $this->option('older-than');
        $except = explode(',', $this->option('except'));

        $revoked = PersonalAccessToken::where('created_at', '<', now()->subDays($days))
            ->whereNotIn('name', $except)
            ->delete();

        $this->info("Revoked {$revoked} tokens older than {$days} days.");
        return self::SUCCESS;
    }
}
```

### API Metrics Commands

```php
// artisan api:metrics:report
class ApiMetricsReportCommand extends Command
{
    protected $signature = 'api:metrics:report
                            {--period=24h : Report period (24h, 7d, 30d)}';

    protected $description = 'Generate API metrics report';

    public function handle(): int
    {
        $period = $this->option('period');
        $since = now()->sub(CarbonInterval::fromString($period));

        $metrics = Metric::where('created_at', '>=', $since)
            ->selectRaw('endpoint, method, status_code, COUNT(*) as count, AVG(duration_ms) as avg_duration, MAX(duration_ms) as max_duration')
            ->groupBy('endpoint', 'method', 'status_code')
            ->orderByDesc('count')
            ->get();

        $this->table(
            ['Endpoint', 'Method', 'Status', 'Count', 'Avg (ms)', 'Max (ms)'],
            $metrics->toArray()
        );

        return self::SUCCESS;
    }
}
```

## API Migration Guide

### Version Deprecation Policy

| Version | Status | Deprecation Date | Removal Date |
|---|---|---|---|
| v1 | Active | — | — |
| v2 | Beta | — | — |

### Breaking Changes Protocol

1. **Announce** deprecation in API response headers (`Sunset: Sat, 01 Jan 2027`)
2. **Log** usage of deprecated endpoints
3. **Email** affected API consumers 90 days before removal
4. **Return** `410 Gone` with migration URL after removal

### Deprecation Headers

```php
// ResponseMiddleware
class DeprecationHeaders
{
    public function handle(Request $request, Closure $next): Response
    {
        $response = $next($request);

        if ($request->routeIs('v1.*')) {
            $response->headers->set('Deprecation', 'true');
            $response->headers->set('Sunset', 'Sat, 01 Jan 2027 00:00:00 GMT');
            $response->headers->set('Link', '</api/v2>; rel="successor-version"');
        }

        return $response;
    }
}
```
