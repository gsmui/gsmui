**Project:** GsmUi
**Domain:** Subscription Management
**Version:** 1.0.0

## Purpose

Define how the subscription system integrates with payment providers (Stripe, Paddle, Lemon Squeezy). The subscription domain is provider-agnostic — provider IDs are stored but never used in business logic.

## Integration Architecture

```text
Application Layer (domain model, services)
    ↑
Billing Interface (provider-agnostic)
    ↑
Provider Adapters
    ├── StripeAdapter
    ├── PaddleAdapter
    └── LemonSqueezyAdapter
```

## Billing Interface

```php
interface BillingProvider
{
    public function createCustomer(User $user): string;
    public function createSubscription(Subscription $subscription): string;
```

## Provider Adapter

### Stripe

```php
class StripeAdapter implements BillingProvider
{
    public function __construct(
        private StripeClient $stripe,
    ) {}

    public function createCustomer(User $user): string
    {
        $customer = $this->stripe->customers->create([
            'email' => $user->email,
            'name' => $user->name,
            'metadata' => [
                'user_id' => $user->id,
            ],
        ]);

        return $customer->id;
    }

    public function createSubscription(Subscription $subscription): string
    {
        $priceId = $subscription->plan->stripe_price_id;

        $stripeSub = $this->stripe->subscriptions->create([
            'customer' => $subscription->user->stripe_id,
            'items' => [['price' => $priceId]],
            'metadata' => [
                'subscription_id' => $subscription->id,
            ],
            'trial_period_days' => $subscription->plan->trial_days ?: null,
            'payment_behavior' => 'default_incomplete',
            'expand' => ['latest_invoice.payment_intent'],
        ]);

        return $stripeSub->id;
    }

    public function parseWebhook(array $payload): WebhookEvent
    {
        $event = match ($payload['type']) {
            'invoice.paid' => WebhookEvent::InvoicePaid,
            'invoice.payment_failed' => WebhookEvent::InvoicePaymentFailed,
            'customer.subscription.updated' => WebhookEvent::SubscriptionUpdated,
            'customer.subscription.deleted' => WebhookEvent::SubscriptionDeleted,
            default => WebhookEvent::Unknown,
        };

        return new WebhookEvent(
            type: $event,
            provider: 'stripe',
            payload: $payload,
            id: $payload['id'],
        );
    }
}
```

```php
class BillingManager
```

## Provider Switch

Switching providers at the application level:

```php
class ProviderMigration
{
    public function migrate(Subscription $subscription, string $fromProvider, string $toProvider): void
    {
        // Cancel old provider subscription
        $this->billing->provider($fromProvider)->cancelSubscription($subscription);

        // Create in new provider
        $newProviderId = $this->billing->provider($toProvider)
            ->createSubscription($subscription);

        // Store new provider ID
        $idColumn = "{$toProvider}_id";
        $subscription->$idColumn = $newProviderId;
        $subscription->save();

        // Webhook handles remaining sync
    }
}
```

### Payment Method Service

```php
class PaymentMethodService
{
    public function __construct(
        private BillingProvider $billing,
    ) {}

    public function addPaymentMethod(
        User $user,
        string $paymentMethodId,
        bool $setAsDefault = true
    ): PaymentMethod {
        // Attach to customer at provider
        $this->billing->attachPaymentMethod(
            $user,
            $paymentMethodId
        );

        // Retrieve details from provider
        $details = $this->billing->getPaymentMethodDetails($paymentMethodId);

        // If setting as default, unset current default
        if ($setAsDefault) {
            PaymentMethod::where('user_id', $user->id)
                ->where('is_default', true)
                ->update(['is_default' => false]);
        }

        return PaymentMethod::create([
            'user_id' => $user->id,
            'provider' => config('billing.default'),
            'provider_payment_method_id' => $paymentMethodId,
            'type' => $details['type'],
            'brand' => $details['brand'] ?? null,
            'last_four' => $details['last_four'],
            'exp_month' => $details['exp_month'],
            'exp_year' => $details['exp_year'],
            'is_default' => $setAsDefault,
            'billing_address' => $details['billing_address'] ?? null,
        ]);
    }

    public function removePaymentMethod(PaymentMethod $paymentMethod): void
    {
        if ($paymentMethod->is_default) {
            throw new CannotRemoveDefaultPaymentMethodException();
        }

        $this->billing->detachPaymentMethod(
            $paymentMethod->user,
            $paymentMethod->provider_payment_method_id
        );

        $paymentMethod->delete();
    }

    public function setDefault(PaymentMethod $paymentMethod): void
    {
        PaymentMethod::where('user_id', $paymentMethod->user_id)
            ->where('is_default', true)
            ->update(['is_default' => false]);

        $paymentMethod->update(['is_default' => true]);

        $this->billing->setDefaultPaymentMethod(
            $paymentMethod->user,
            $paymentMethod->provider_payment_method_id
        );
    }

    public function getDefault(User $user): ?PaymentMethod
    {
        return $user->paymentMethods()
            ->where('is_default', true)
            ->first();
    }

    public function syncPaymentMethods(User $user): void
    {
        $providerMethods = $this->billing->listPaymentMethods($user);

        foreach ($providerMethods as $method) {
            PaymentMethod::updateOrCreate(
                [
                    'user_id' => $user->id,
                    'provider_payment_method_id' => $method['id'],
                ],
                [
                    'provider' => config('billing.default'),
                    'type' => $method['type'],
                    'brand' => $method['brand'] ?? null,
                    'last_four' => $method['last_four'],
                    'exp_month' => $method['exp_month'],
                    'exp_year' => $method['exp_year'],
                    'is_default' => $method['is_default'] ?? false,
                ]
            );
        }

        // Remove local records that no longer exist at provider
        $providerIds = collect($providerMethods)->pluck('id')->toArray();
        PaymentMethod::where('user_id', $user->id)
            ->where('provider', config('billing.default'))
            ->whereNotIn('provider_payment_method_id', $providerIds)
            ->delete();
    }
}
```

```php
class StripePaymentMethodAdapter
{
    public function __construct(
        private StripeClient $stripe,
    ) {}

    public function attachPaymentMethod(User $user, string $paymentMethodId): void
    {
        $this->stripe->paymentMethods->attach($paymentMethodId, [
            'customer' => $user->stripe_id,
        ]);
    }

    public function detachPaymentMethod(User $user, string $paymentMethodId): void
    {
        $this->stripe->paymentMethods->detach($paymentMethodId);
    }

    public function getPaymentMethodDetails(string $paymentMethodId): array
    {
        $pm = $this->stripe->paymentMethods->retrieve($paymentMethodId);

        return [
            'id' => $pm->id,
            'type' => $pm->type,
            'brand' => $pm->card->brand ?? null,
            'last_four' => $pm->card->last4 ?? null,
            'exp_month' => $pm->card->exp_month ?? null,
            'exp_year' => $pm->card->exp_year ?? null,
            'billing_address' => $pm->billing_details->address ?? null,
        ];
    }

    public function listPaymentMethods(User $user): array
    {
        $methods = $this->stripe->paymentMethods->all([
            'customer' => $user->stripe_id,
            'type' => 'card',
        ]);

        return collect($methods->data)->map(fn ($pm) => [
            'id' => $pm->id,
            'type' => $pm->type,
            'brand' => $pm->card->brand ?? null,
            'last_four' => $pm->card->last4 ?? null,
            'exp_month' => $pm->card->exp_month ?? null,
            'exp_year' => $pm->card->exp_year ?? null,
            'is_default' => false,
        ])->toArray();
    }

    public function setDefaultPaymentMethod(User $user, string $paymentMethodId): void
    {
        $this->stripe->customers->update($user->stripe_id, [
            'invoice_settings' => [
                'default_payment_method' => $paymentMethodId,
            ],
        ]);
    }
```

## Subscription Lifecycle Billing

### Plan Upgrade/Downgrade

```php
class PlanChangeService
{
    public function __construct(
        private BillingProvider $billing,
    ) {}

    public function upgradePlan(Subscription $subscription, Plan $newPlan): void
    {
        $this->validateUpgrade($subscription, $newPlan);

        // Calculate proration
        $proration = $this->calculateProration($subscription, $newPlan);

        // Update at provider
        $this->billing->updateSubscriptionPlan($subscription, $newPlan);

        // Update local subscription
        $subscription->update([
            'plan_id' => $newPlan->id,
            'quantity' => $subscription->quantity,
        ]);

        // Create proration invoice line item
        if ($proration > 0) {
            $this->createProrationInvoice($subscription, $proration, 'upgrade');
        }

        PlanUpgraded::dispatch($subscription, $newPlan, $proration);
    }

    public function downgradePlan(Subscription $subscription, Plan $newPlan): void
    {
        $this->validateDowngrade($subscription, $newPlan);

        $proration = $this->calculateProration($subscription, $newPlan);

        $this->billing->updateSubscriptionPlan($subscription, $newPlan);

        $subscription->update([
            'plan_id' => $newPlan->id,
            'pending_plan_id' => null,
            'ends_at' => null,
        ]);

        if ($proration < 0) {
            $this->createProrationInvoice($subscription, $proration, 'downgrade');
        }

        PlanDowngraded::dispatch($subscription, $newPlan, $proration);
    }

    public function schedulePlanChange(Subscription $subscription, Plan $newPlan): void
    {
        $subscription->update([
            'pending_plan_id' => $newPlan->id,
        ]);

        $this->billing->schedulePlanChange($subscription, $newPlan);

        PlanChangeScheduled::dispatch($subscription, $newPlan);
    }

    private function calculateProration(Subscription $subscription, Plan $newPlan): int
    {
        $currentPlan = $subscription->plan;
        $now = now();
        $periodStart = $subscription->current_period_start;
        $periodEnd = $subscription->current_period_end;

        $daysRemaining = $periodEnd->diffInDays($now);
        $totalDays = $periodEnd->diffInDays($periodStart);

        $currentDailyRate = $currentPlan->price / $totalDays;
        $newDailyRate = $newPlan->price / $totalDays;

        $creditForUnused = (int) ($currentDailyRate * $daysRemaining);
        $chargeForRemaining = (int) ($newDailyRate * $daysRemaining);

        return $chargeForRemaining - $creditForUnused;
    }

    private function createProrationInvoice(
        Subscription $subscription,
        int $amount,
        string $type
    ): void {
        $this->billing->createInvoiceItem(
            $subscription,
            [
                'description' => ucfirst($type) . ' proration',
                'amount' => abs($amount),
                'metadata' => [
                    'type' => $type,
                    'subscription_id' => $subscription->id,
                ],
            ]
        );
    }

    private function validateUpgrade(Subscription $subscription, Plan $newPlan): void
    {
        if ($newPlan->price <= $subscription->plan->price) {
            throw new NotAnUpgradeException('New plan must cost more than current plan.');
        }
    }

    private function validateDowngrade(Subscription $subscription, Plan $newPlan): void
    {
        if ($newPlan->price >= $subscription->plan->price) {
            throw new NotADowngradeException('New plan must cost less than current plan.');
        }
    }
}
```

### Trial Management

```php
class TrialBillingService
{
    public function __construct(
        private BillingProvider $billing,
    ) {}

    public function startTrial(Subscription $subscription): void
    {
        $plan = $subscription->plan;

        if (!$plan->has_trial) {
            throw new PlanDoesNotHaveTrialException();
        }

        $subscription->update([
            'trial_ends_at' => now()->addDays($plan->trial_days),
            'status' => 'trialing',
        ]);

        TrialStarted::dispatch($subscription);
    }

    public function endTrial(Subscription $subscription): void
    {
        $subscription->update([
            'trial_ends_at' => null,
            'status' => 'active',
        ]);

        // Charge the subscription
        $this->billing->charge($subscription);

        TrialEnded::dispatch($subscription);
    }

    public function convertTrialToSubscription(Subscription $subscription): void
    {
        if ($subscription->status !== 'trialing') {
            throw new SubscriptionNotInTrialException();
        }

        $this->endTrial($subscription);
    }

    public function cancelTrial(Subscription $subscription): void
    {
        $subscription->update([
            'trial_ends_at' => null,
            'status' => 'cancelled',
            'cancelled_at' => now(),
        ]);

        TrialCancelled::dispatch($subscription);
    }

    public function extendTrial(Subscription $subscription, int $days): void
    {
        $newEnd = $subscription->trial_ends_at
            ? $subscription->trial_ends_at->addDays($days)
            : now()->addDays($days);

        $subscription->update([
            'trial_ends_at' => $newEnd,
        ]);

        TrialExtended::dispatch($subscription, $days);
    }

    public function getTrialDaysRemaining(Subscription $subscription): int
    {
        if (!$subscription->trial_ends_at) {
            return 0;
        }

        return max(0, now()->diffInDays($subscription->trial_ends_at, false));
    }

    public function handleTrialExpiring(Subscription $subscription): void
    {
        $daysRemaining = $this->getTrialDaysRemaining($subscription);

        if ($daysRemaining <= config('billing.trial_expiry_reminder_days', 3)) {
            TrialExpiring::dispatch($subscription, $daysRemaining);
        }
    }
}
```

```php
class RefundService
{
    public function __construct(
        private BillingProvider $billing,
    ) {}

    public function processRefund(
        Receipt $receipt,
        ?int $amount = null,
        string $reason = 'requested_by_customer'
    ): Refund {
        $refundAmount = $amount ?? $receipt->amount;

        if ($refundAmount > ($receipt->amount - $receipt->refund_amount)) {
            throw new RefundExceedsPaymentException();
        }

        $providerRefund = $this->billing->createRefund(
            $receipt,
            $refundAmount,
            $reason
        );

        $refund = Refund::create([
            'receipt_id' => $receipt->id,
            'user_id' => $receipt->user_id,
            'provider' => config('billing.default'),
            'provider_refund_id' => $providerRefund['id'],
            'amount' => $refundAmount,
            'currency' => $receipt->currency,
            'reason' => $reason,
            'status' => 'completed',
            'refunded_at' => now(),
        ]);

        $receipt->increment('refund_amount', $refundAmount);

        if ($receipt->refund_amount >= $receipt->amount) {
            $receipt->update(['status' => 'refunded']);
        }

        RefundProcessed::dispatch($refund);

        return $refund;
    }

    public function getRefunds(User $user, int $perPage = 15): LengthAwarePaginator
    {
        return Refund::where('user_id', $user->id)
            ->with('receipt')
            ->orderByDesc('created_at')
            ->paginate($perPage);
    }

    public function calculateRefundAmount(Subscription $subscription): int
    {
        $now = now();
        $periodStart = $subscription->current_period_start;
        $periodEnd = $subscription->current_period_end;

        $daysUsed = $now->diffInDays($periodStart);
        $totalDays = $periodEnd->diffInDays($periodStart);
        $dailyRate = $subscription->plan->price / $totalDays;
        $daysRemaining = $totalDays - $daysUsed;

        return (int) ($dailyRate * $daysRemaining);
    }
}
```

```php
class CouponService
{
    public function applyCoupon(Subscription $subscription, string $code): void
    {
        $coupon = Coupon::where('code', $code)->firstOrFail();

        if (!$coupon->canBeApplied()) {
            throw new CouponNotApplicableException();
        }

        $subscription->coupons()->attach($coupon->id, [
            'applied_at' => now(),
        ]);

        $coupon->increment('redeem_count');

        $this->billing->applyCoupon($subscription, $coupon);

        CouponApplied::dispatch($subscription, $coupon);
    }

    public function removeCoupon(Subscription $subscription, Coupon $coupon): void
    {
        $subscription->coupons()->detach($coupon->id);

        $coupon->decrement('redeem_count');

        $this->billing->removeCoupon($subscription, $coupon);

        CouponRemoved::dispatch($subscription, $coupon);
    }

    public function validateCoupon(string $code, ?int $amount = null): CouponValidation
    {
        $coupon = Coupon::where('code', $code)->first();

        if (!$coupon) {
            return new CouponValidation(
                valid: false,
                message: 'Coupon not found.',
            );
        }

        if (!$coupon->canBeApplied()) {
            $reason = match (true) {
                $coupon->isExpired() => 'Coupon has expired.',
                $coupon->isFullyRedeemed() => 'Coupon has been fully redeemed.',
                !$coupon->valid => 'Coupon is no longer valid.',
                default => 'Coupon cannot be applied.',
            };

            return new CouponValidation(
                valid: false,
                message: $reason,
            );
        }

        $discount = $amount ? $coupon->calculateDiscount($amount) : null;

        return new CouponValidation(
            valid: true,
            coupon: $coupon,
            discount: $discount,
            message: 'Coupon is valid.',
        );
    }
}
```

## Error Handling & Recovery

### Payment Failure Handler

```php
class PaymentFailureHandler
{
    public function handle(Subscription $subscription, string $failureCode, string $failureMessage): void
    {
        // Update subscription status
        $subscription->update([
            'status' => 'past_due',
            'last_payment_failure' => now(),
            'failure_code' => $failureCode,
            'failure_message' => $failureMessage,
        ]);

        // Notify user
        Notification::send($subscription->user, new PaymentFailedNotification(
            subscription: $subscription,
            failureCode: $failureCode,
            failureMessage: $failureMessage,
        ));

        // Attempt to retry payment
        $retryAttempt = $subscription->retry_count + 1;
        $maxRetries = config('billing.max_retry_attempts', 3);

        if ($retryAttempt <= $maxRetries) {
            $subscription->update([
                'retry_count' => $retryAttempt,
                'next_retry_at' => now()->addHours(
                    config('billing.retry_interval_hours', 24) * $retryAttempt
                ),
            ]);

            ScheduleRetryPayment::dispatch($subscription)
                ->delay(now()->addHours(
                    config('billing.retry_interval_hours', 24) * $retryAttempt
                ));
        } else {
            // Max retries exceeded — cancel subscription
            $subscription->update([
                'status' => 'cancelled',
                'cancelled_at' => now(),
                'cancellation_reason' => 'payment_failed_after_max_retries',
            ]);

            Notification::send($subscription->user, new SubscriptionCancelledNotification(
                subscription: $subscription,
                reason: 'Maximum payment retry attempts exceeded.',
            ));

            PaymentRetryExhausted::dispatch($subscription);
        }

        PaymentFailed::dispatch($subscription, $failureCode, $failureMessage);
    }
}
```

### Dunning Management

```php
class DunningService
{
    public function processDunning(Subscription $subscription): void
    {
        $daysSinceFailure = now()->diffInDays($subscription->last_payment_failure);

        $dunningSteps = config('billing.dunning_steps', [
            1 => 'grace_period',
            3 => 'first_reminder',
            7 => 'second_reminder',
            14 => 'final_warning',
            30 => 'cancel',
        ]);

        foreach ($dunningSteps as $days => $action) {
            if ($daysSinceFailure >= $days) {
                match ($action) {
                    'grace_period' => $this->handleGracePeriod($subscription),
                    'first_reminder' => $this->sendReminder($subscription, 1),
                    'second_reminder' => $this->sendReminder($subscription, 2),
                    'final_warning' => $this->sendFinalWarning($subscription),
                    'cancel' => $this->cancelForNonPayment($subscription),
                };
            }
        }
    }

    private function handleGracePeriod(Subscription $subscription): void
    {
        // No action needed during grace period
    }

    private function sendReminder(Subscription $subscription, int $attempt): void
    {
        Notification::send($subscription->user, new PaymentReminderNotification(
            subscription: $subscription,
            attempt: $attempt,
        ));
    }

    private function sendFinalWarning(Subscription $subscription): void
    {
        Notification::send($subscription->user, new FinalPaymentWarningNotification(
            subscription: $subscription,
        ));
    }

    private function cancelForNonPayment(Subscription $subscription): void
    {
        app(BillingProvider::class)->cancelSubscription($subscription);

        $subscription->update([
            'status' => 'cancelled',
            'cancelled_at' => now(),
            'cancellation_reason' => 'dunning_cancelled',
        ]);

        SubscriptionCancelledForNonPayment::dispatch($subscription);
    }
}
```
