```php
            'daysRemaining' => now()->diffInDays($subscription->period()->end),
        ]);
    }
```

```php
class PlanOverview extends Component
{
    public User $user;

    public function mount(): void
    {
        $this->user = auth()->user();
    }

    public function getSubscriptionProperty(): ?Subscription
    {
        return $this->user->activeSubscription();
    }

    public function getPlanProperty(): ?Plan
    {
        return $this->subscription?->plan;
    }

    public function getEntitlementServiceProperty(): EntitlementService
    {
        return app(EntitlementService::class);
    }

    public function getUsageProperty(): array
    {
        if (!$this->subscription) {
            return [];
        }

        return [
            'downloads' => [
                'used' => $this->periodUsage('downloads'),
                'limit' => $this->entitlementService->value($this->user, 'limit.downloads_per_month') ?? 10,
                'unlimited' => $this->entitlementService->has($this->user, EntitlementKey::DownloadUnlimited),
            ],
            'ai_credits' => [
                'used' => $this->periodUsage('ai_chat_credits'),
                'limit' => $this->entitlementService->value($this->user, 'limit.ai_credits_per_month') ?? 50,
            ],
            'mcp_connections' => [
                'used' => $this->periodUsage('mcp_connections'),
                'limit' => $this->entitlementService->value($this->user, 'limit.mcp_connections') ?? 1,
            ],
        ];
    }

    public function getNextBillingDateProperty(): ?Carbon
    {
        return $this->subscription?->current_period_end;
    }

    public function getDaysRemainingProperty(): ?int
    {
        return $this->nextBillingDate
            ? now()->diffInDays($this->nextBillingDate)
            : null;
    }

    public function getTrialDaysRemainingProperty(): ?int
    {
        if (!$this->subscription || $this->subscription->status !== 'trialing') {
            return null;
        }

        return max(0, now()->diffInDays($this->subscription->trial_ends_at, false));
    }

    private function periodUsage(string $metric): int
    {
        if (!$this->subscription) {
            return 0;
        }

        return DB::table('usage_records')
            ->where('user_id', $this->user->id)
            ->where('metric', $metric)
            ->whereBetween('created_at', [
                $this->subscription->current_period_start,
                $this->subscription->current_period_end,
            ])
            ->sum('value');
    }

    public function render(): View
    {
        return view('user.dashboard.plan-overview', [
            'subscription' => $this->subscription,
            'plan' => $this->plan,
            'usage' => $this->usage,
            'nextBillingDate' => $this->nextBillingDate,
            'daysRemaining' => $this->daysRemaining,
            'trialDaysRemaining' => $this->trialDaysRemaining,
        ]);
    }
}
```

### Change Plan Component

```php
class ChangePlan extends Component
{
    public Subscription $subscription;
    public ?string $selectedPlanId = null;
    public bool $showConfirmation = false;

    public function mount(): void
    {
        $this->subscription = auth()->user()->activeSubscription();
    }

    public function getPlansProperty(): Collection
    {
        return Plan::where('is_active', true)
            ->where('is_visible', true)
            ->orderBy('sort_order')
            ->get();
    }

    public function getSelectedPlanProperty(): ?Plan
    {
        return $this->selectedPlanId ? Plan::find($this->selectedPlanId) : null;
    }

    public function getIsUpgradeProperty(): bool
    {
        if (!$this->selectedPlan) {
            return false;
        }

        return $this->selectedPlan->price > $this->subscription->plan->price;
    }

    public function getProrationAmountProperty(): int
    {
        if (!$this->selectedPlan) {
            return 0;
        }

        return app(PlanChangeService::class)
            ->calculateProration($this->subscription, $this->selectedPlan);
    }

    public function getNewFeaturesProperty(): array
    {
        if (!$this->selectedPlan) {
            return [];
        }

        $currentFeatures = $this->subscription->plan->features ?? [];
        $newFeatures = $this->selectedPlan->features ?? [];

        return array_diff($newFeatures, $currentFeatures);
    }

    public function selectPlan(string $planId): void
    {
        $this->selectedPlanId = $planId;
        $this->showConfirmation = true;
    }

    public function confirmChange(): void
    {
        if (!$this->selectedPlan) {
            return;
        }

        if ($this->isUpgrade) {
            app(PlanChangeService::class)->upgradePlan(
                $this->subscription,
                $this->selectedPlan
            );

            $this->dispatch('notify', type: 'success', message: 'Plan upgraded successfully!');
        } else {
            app(PlanChangeService::class)->downgradePlan(
                $this->subscription,
                $this->selectedPlan
            );

            $this->dispatch('notify', type: 'success', message: 'Plan downgraded. Changes take effect at the end of your billing period.');
        }

        $this->showConfirmation = false;
        $this->selectedPlanId = null;

        $this->dispatch('plan-changed');
    }

    public function render(): View
    {
        return view('user.dashboard.change-plan', [
            'plans' => $this->plans,
            'selectedPlan' => $this->selectedPlan,
            'isUpgrade' => $this->isUpgrade,
            'prorationAmount' => $this->prorationAmount,
            'newFeatures' => $this->newFeatures,
        ]);
    }
}
```

### Cancel Subscription Component

```php
class CancelSubscription extends Component
{
    public Subscription $subscription;
    public string $reason = '';
    public bool $showSaveOffer = false;
    public ?SaveOffer $saveOffer = null;

    public function mount(): void
    {
        $this->subscription = auth()->user()->activeSubscription();

        if (!$this->subscription || $this->subscription->status !== 'active') {
            redirect()->route('billing.index');
            return;
        }

        $this->saveOffer = app(SaveOffer::class)->eligible($this->subscription);
        $this->showSaveOffer = $this->saveOffer !== null;
    }

    public function getReasonsProperty(): array
    {
        return [
            'too_expensive' => __('Too expensive'),
            'missing_features' => __('Missing features'),
            'not_using_enough' => __('Not using it enough'),
            'switching_tools' => __('Switching to another tool'),
            'other' => __('Other'),
        ];
    }

    public function cancel(): void
    {
        $this->validate([
            'reason' => 'required|string',
        ]);

        app(CancellationService::class)->cancel(
            $this->subscription,
            reason: $this->reason,
            immediate: false,
        );

        $this->dispatch('notify', type: 'success', message: 'Subscription canceled. Access continues until the end of your billing period.');
        $this->dispatch('subscription-canceled');

        redirect()->route('billing.index');
    }

    public function acceptSaveOffer(): void
    {
        if (!$this->saveOffer) {
            return;
        }

        // Apply the save offer (e.g., 1 month free)
        $this->subscription->extendTrial($this->saveOffer->trialDays);

        $this->showSaveOffer = false;
        $this->dispatch('notify', type: 'success', message: "Great! We've added {$this->saveOffer->trialDays} free days to your subscription.");

        redirect()->route('billing.index');
    }

    public function render(): View
    {
        return view('user.dashboard.cancel', [
            'subscription' => $this->subscription,
            'reasons' => $this->reasons,
            'showSaveOffer' => $this->showSaveOffer,
            'saveOffer' => $this->saveOffer,
        ]);
    }
}
```

### Usage Tracking Service

```php
class UsageTrackingService
{
    public function record(string $metric, int $value = 1, ?User $user = null): void
    {
        $user = $user ?? auth()->user();
        $subscription = $user->activeSubscription();

        if (!$subscription) {
            return;
        }

        // Check limit before recording
        $limit = app(EntitlementService::class)
            ->value($user, "limit.{$metric}_per_month");

        if ($limit !== null) {
            $currentUsage = $this->getPeriodUsage($user, $metric, $subscription);

            if (($currentUsage + $value) > $limit) {
                throw new UsageLimitExceededException(
                    "Usage limit exceeded for metric: {$metric}"
                );
            }
        }

        UsageRecord::create([
            'user_id' => $user->id,
            'subscription_id' => $subscription->id,
            'metric' => $metric,
            'value' => $value,
        ]);
    }

    public function getPeriodUsage(User $user, string $metric, Subscription $subscription): int
    {
        return UsageRecord::where('user_id', $user->id)
            ->where('metric', $metric)
            ->whereBetween('created_at', [
                $subscription->current_period_start,
                $subscription->current_period_end,
            ])
            ->sum('value');
    }

    public function getUsageSummary(User $user): array
    {
        $subscription = $user->activeSubscription();

        if (!$subscription) {
            return [];
        }

        $entitlementService = app(EntitlementService::class);

        return [
            'downloads' => [
                'used' => $this->getPeriodUsage($user, 'downloads', $subscription),
                'limit' => $entitlementService->value($user, 'limit.downloads_per_month') ?? 10,
                'unlimited' => $entitlementService->has($user, EntitlementKey::DownloadUnlimited),
            ],
            'ai_credits' => [
                'used' => $this->getPeriodUsage($user, 'ai_chat_credits', $subscription),
                'limit' => $entitlementService->value($user, 'limit.ai_credits_per_month') ?? 50,
            ],
            'mcp_connections' => [
                'used' => $this->getPeriodUsage($user, 'mcp_connections', $subscription),
                'limit' => $entitlementService->value($user, 'limit.mcp_connections') ?? 1,
            ],
        ];
    }

    public function resetPeriodUsage(User $user): void
    {
        $subscription = $user->activeSubscription();

        if (!$subscription) {
            return;
        }

        UsageRecord::where('user_id', $user->id)
            ->where('created_at', '<', $subscription->current_period_start)
            ->delete();
    }
}
```

### Usage Limit Middleware

```php
class CheckUsageLimitMiddleware
{
    public function handle(Request $request, Closure $next, string $metric): Response
    {
        $user = auth()->user();

        if (!$user) {
            return $next($request);
        }

        $subscription = $user->activeSubscription();

        if (!$subscription) {
            return response()->json([
                'error' => 'no_active_subscription',
                'message' => 'No active subscription.',
            ], 403);
        }

        $entitlementService = app(EntitlementService::class);
        $limit = $entitlementService->value($user, "limit.{$metric}_per_month");

        if ($limit === null) {
            return $next($request); // No limit set
        }

        $currentUsage = app(UsageTrackingService::class)
            ->getPeriodUsage($user, $metric, $subscription);

        if ($currentUsage >= $limit) {
            return response()->json([
                'error' => 'usage_limit_reached',
                'message' => "You have reached your {$metric} limit.",
                'used' => $currentUsage,
                'limit' => $limit,
                'reset_date' => $subscription->current_period_end->toISOString(),
                'upgrade_url' => route('billing.change'),
            ], 403);
        }

        $request->attributes->set('usage_remaining', $limit - $currentUsage);

        return $next($request);
    }
}
```

### Dashboard Cache Service

```php
class DashboardCacheService
{
    private const CACHE_PREFIX = 'user:dashboard:';
    private const CACHE_TTL = 300; // 5 minutes

    public function getUserDashboardData(User $user): array
    {
        $cacheKey = self::CACHE_PREFIX . $user->id;

        return Cache::remember($cacheKey, self::CACHE_TTL, function () use ($user) {
            $subscription = $user->activeSubscription();

            if (!$subscription) {
                return [
                    'subscription' => null,
                    'plan' => null,
                    'usage' => [],
                    'invoices' => [],
                    'payment_methods' => [],
                ];
            }

            return [
                'subscription' => $subscription->toArray(),
                'plan' => $subscription->plan->toArray(),
                'usage' => app(UsageTrackingService::class)->getUsageSummary($user),
                'invoices' => $user->invoices()
                    ->orderByDesc('created_at')
                    ->limit(5)
                    ->get()
                    ->toArray(),
                'payment_methods' => $user->paymentMethods->toArray(),
            ];
        });
    }

    public function invalidateDashboardCache(User $user): void
    {
        Cache::forget(self::CACHE_PREFIX . $user->id);
    }

    public function invalidateAllDashboards(): void
    {
        Cache::tags([self::CACHE_PREFIX])->flush();
    }
}
```

```php
    public function via(object $notifiable): array
    {
        return ['mail', 'database'];
    }

    public function toMail(object $notifiable): MailMessage
    {
        return (new MailMessage)
            ->subject(__('Your subscription has been canceled'))
            ->greeting(__('Hello :name!', ['name' => $notifiable->name]))
            ->line(__('Your :plan subscription has been canceled.', [
                'plan' => $this->subscription->plan->name,
            ]))
            ->line(__('You will continue to have access until :date.', [
                'date' => $this->subscription->current_period_end->format('M j, Y'),
            ]))
            ->line(__('After that date, you will be downgraded to the Free plan.'))
            ->action(__('Resubscribe'), route('pricing'));
    }

    public function toArray(object $notifiable): array
    {
        return [
            'type' => 'subscription_canceled',
            'subscription_id' => $this->subscription->id,
            'ends_at' => $this->subscription->current_period_end->toISOString(),
        ];
    }
}
```

### Error States & Fallbacks

```php
class DashboardErrorHandler
{
    public function handleSubscriptionLoadError(User $user): array
    {
        Log::warning("Failed to load subscription for user {$user->id}");

        return [
            'subscription' => null,
            'plan' => null,
            'usage' => [],
            'error' => 'Unable to load subscription data. Please try again later.',
        ];
    }

    public function handleUsageLoadError(User $user): array
    {
        Log::warning("Failed to load usage data for user {$user->id}");

        return [
            'downloads' => ['used' => 0, 'limit' => 10, 'unlimited' => false],
            'ai_credits' => ['used' => 0, 'limit' => 50],
            'mcp_connections' => ['used' => 0, 'limit' => 1],
            'error' => 'Unable to load usage data.',
        ];
    }

    public function handleInvoiceLoadError(User $user): array
    {
        Log::warning("Failed to load invoices for user {$user->id}");

        return [
            'invoices' => [],
            'error' => 'Unable to load invoice history.',
        ];
    }
}
```
