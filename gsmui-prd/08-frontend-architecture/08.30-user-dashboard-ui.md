# Purpose

This document defines the end-user dashboard: an authenticated shell with the user's projects,
usage stats, recent activity, and profile/settings entry points. It is distinct from the
admin/platform dashboard (`08.21`) — it scopes to a single tenant user, not platform operations.

---

## 1. Page Composition

| Region | GsmUi Components | Livewire |
| --- | --- | --- |
| Shell | `gsmui:layouts.app` + `gsmui:sidebar` (user nav) | host `UserDashboard` |
| Projects | `gsmui:card` grid | `UserDashboard::projects()` |
| Usage | `gsmui:stat`, `gsmui:chart` (adapter) | `wire:model="period"` |
| Activity | `gsmui:table` + `gsmui:pagination` | `WithPagination` |
| Profile | `gsmui:avatar`, `gsmui:button` (settings) | `Profile` |

## 2. Component Tree

```text
gsmui:layouts.app
├── gsmui:sidebar
│   ├── gsmui:avatar (user)
│   └── gsmui:navlist (projects/usage/settings)
└── <content>
    ├── gsmui:card × N (project tiles)
    │   ├── gsmui:heading, gsmui:badge (status)
    │   └── gsmui:button (open)
    ├── gsmui:stat (usage, tabular-nums)  × 3
    ├── gsmui:chart (:series="$usageSeries")
    └── gsmui:table (:rows="$activity") + gsmui:pagination
```

## 3. Data Flow — User → Server

```text
gsmui:stat / gsmui:chart wire:model="period"
        │  input + change events
        ▼
UserDashboard::$period  ──▶  recompute usage (05.25)
        │
        ▼
$usageSeries  ──▶  <g-gsmui-chart> MutationObserver re-render

gsmui:button "Open project"
        │  wire:navigate / wire:click
        ▼
route to project (Livewire host swap)
```

- The chart web component reacts to `value` attribute changes via `MutationObserver` (AGENTS.md).
- `gsmui:avatar` shows initials fallback (no external image service); self-hosted avatar only.
- Disabled project tiles use `[&[disabled]]:shadow-none!`.

## 4. CSP & Styling

- Usage numbers use `tabular-nums` (rapid updates during polling).
- Dark mode via `gsmui:theme-switcher`; inline SVG icons only.

## 5. Livewire Component Specifications

### 5.1 `UserDashboard` Livewire Component

```php
<?php

namespace App\Livewire\User;

use Livewire\Component;
use Livewire\Attributes\Layout;
use Livewire\WithPagination;
use Illuminate\Support\Facades\Auth;

#[Layout('gsmui::layouts.app')]
class UserDashboard extends Component
{
    use WithPagination;

    public string $period = '30d';
    public string $search = '';
    public string $sortBy = 'updated_at';
    public string $sortDirection = 'desc';
    public int $perPage = 12;
    public bool $showStats = true;

    protected array $listeners = [
        'refreshDashboard' => '$refresh',
    ];

    protected function rules(): array
    {
        return [
            'period' => 'sometimes|string|in:7d,30d,90d,12m',
            'search' => 'sometimes|string|max:255',
            'sortBy' => 'sometimes|string|in:name,created_at,updated_at,status',
            'sortDirection' => 'sometimes|string|in:asc,desc',
        ];
    }

    public function updatedPeriod(): void
    {
        $this->showStats = true;
    }

    public function getProjectsProperty()
    {
        $user = Auth::user();

        return $user->projects()
            ->when($this->search, fn($q) => $q->where('name', 'like', "%{$this->search}%"))
            ->orderBy($this->sortBy, $this->sortDirection)
            ->paginate($this->perPage);
    }

    public function getUsageProperty(): array
    {
        $user = Auth::user();
        $days = match($this->period) {
            '7d' => 7,
            '30d' => 30,
            '90d' => 90,
            '12m' => 365,
            default => 30,
        };

        $usage = $user->usage()
            ->where('created_at', '>=', now()->subDays($days))
            ->get();

        return [
            'totalRequests' => $usage->sum('requests'),
            'totalTokens' => $usage->sum('tokens'),
            'totalCost' => $usage->sum('cost'),
            'activeDays' => $usage->unique('created_at')->count(),
            'dailyAverage' => round($usage->sum('requests') / max($days, 1)),
        ];
    }

    public function getUsageSeriesProperty(): array
    {
        $user = Auth::user();
        $days = match($this->period) {
            '7d' => 7,
            '30d' => 30,
            '90d' => 90,
            '12m' => 365,
            default => 30,
        };

        $daily = $user->usage()
            ->where('created_at', '>=', now()->subDays($days))
            ->selectRaw('DATE(created_at) as date, SUM(requests) as requests, SUM(tokens) as tokens')
            ->groupBy('date')
            ->orderBy('date')
            ->get();

        return [
            'labels' => $daily->pluck('date')->map(fn($d) => \Carbon\Carbon::parse($d)->format('M j'))->toArray(),
            'requests' => $daily->pluck('requests')->toArray(),
            'tokens' => $daily->pluck('tokens')->toArray(),
        ];
    }

    public function getActivityProperty()
    {
        $user = Auth::user();

        return $user->activities()
            ->latest()
            ->paginate(10);
    }

    public function render()
    {
        return view('user.dashboard', [
            'projects' => $this->projects,
            'usage' => $this->usage,
            'usageSeries' => $this->usageSeries,
            'activity' => $this->activity,
        ]);
    }
}
```

### 5.2 `UserProfile` Livewire Component

```php
<?php

namespace App\Livewire\User;

use Livewire\Component;
use Livewire\WithFileUploads;
use Livewire\Attributes\Layout;
use Illuminate\Support\Facades\Auth;
use Illuminate\Validation\Rule;

#[Layout('gsmui::layouts.app')]
class UserProfile extends Component
{
    use WithFileUploads;

    public string $name = '';
    public string $email = '';
    public ?string $avatar = null;
    public bool $showAvatarUpload = false;

    public function mount(): void
    {
        $user = Auth::user();
        $this->name = $user->name;
        $this->email = $user->email;
        $this->avatar = $user->avatar;
    }

    public function rules(): array
    {
        return [
            'name' => 'required|string|max:255',
            'email' => [
                'required',
                'email',
                'max:255',
                Rule::unique('users')->ignore(Auth::id()),
            ],
            'avatar' => 'nullable|image|max:1024|mimes:jpg,jpeg,png,webp',
        ];
    }

    public function save(): void
    {
        $this->validate();

        $user = Auth::user();

        $data = [
            'name' => $this->name,
            'email' => $this->email,
        ];

        if ($this->avatar && is_string($this->avatar)) {
            $data['avatar'] = $this->avatar;
        }

        $user->update($data);

        $this->dispatch('toast', message: __('Profile updated successfully.'));
    }

    public function updatedAvatar(): void
    {
        $this->validate(['avatar' => 'nullable|image|max:1024|mimes:jpg,jpeg,png,webp']);

        if ($this->avatar) {
            $path = $this->avatar->store('avatars', 'public');
            Auth::user()->update(['avatar' => $path]);
            $this->avatar = $path;
        }
    }

    public function deleteAvatar(): void
    {
        $user = Auth::user();
        if ($user->avatar) {
            \Storage::disk('public')->delete($user->avatar);
            $user->update(['avatar' => null]);
            $this->avatar = null;
        }
    }

    public function render()
    {
        return view('user.profile');
    }
}
```

### 5.3 `UserSettings` Livewire Component

```php
<?php

namespace App\Livewire\User;

use Livewire\Component;
use Livewire\Attributes\Layout;
use Illuminate\Support\Facades\Auth;

#[Layout('gsmui::layouts.app')]
class UserSettings extends Component
{
    public string $timezone = 'UTC';
    public string $locale = 'en';
    public bool $emailNotifications = true;
    public bool $pushNotifications = true;
    public bool $marketingEmails = false;
    public string $theme = 'system';

    public function mount(): void
    {
        $user = Auth::user();
        $settings = $user->settings ?? [];
        $this->timezone = $settings['timezone'] ?? 'UTC';
        $this->locale = $settings['locale'] ?? 'en';
        $this->emailNotifications = $settings['email_notifications'] ?? true;
        $this->pushNotifications = $settings['push_notifications'] ?? true;
        $this->marketingEmails = $settings['marketing_emails'] ?? false;
        $this->theme = $settings['theme'] ?? 'system';
    }

    public function save(): void
    {
        $this->validate([
            'timezone' => 'required|string|timezone',
            'locale' => 'required|string|in:en,ar,fr,de,es',
            'theme' => 'required|string|in:light,dark,system',
        ]);

        Auth::user()->update([
            'settings' => [
                'timezone' => $this->timezone,
                'locale' => $this->locale,
                'email_notifications' => $this->emailNotifications,
                'push_notifications' => $this->pushNotifications,
                'marketing_emails' => $this->marketingEmails,
                'theme' => $this->theme,
            ],
        ]);

        session(['theme' => $this->theme === 'system' ? (config('app.dark_mode') ? 'dark' : 'light') : $this->theme]);

        $this->dispatch('toast', message: __('Settings saved successfully.'));
    }

    public function updatedTheme(string $value): void
    {
        $theme = $value === 'system' ? (config('app.dark_mode') ? 'dark' : 'light') : $value;
        session(['theme' => $theme]);
        $this->dispatch('themeChanged', theme: $theme);
    }

    public function render()
    {
        return view('user.settings');
    }
}
```

## 21. API Key Management

### 21.1 API Keys Component

```php
<?php

namespace App\Livewire;

use Livewire\Component;
use App\Models\ApiKey;
use Illuminate\Support\Str;

class ApiKeyManager extends Component
{
    public array $keys = [];
    public bool $showCreateModal = false;
    public string $newKeyName = '';
    public string $newKeyScope = 'read';
    public ?string $newKeySecret = null;

    protected array $rules = [
        'newKeyName' => 'required|string|min:2|max:100',
        'newKeyScope' => 'required|in:read,write,admin',
    ];

    public function mount(): void
    {
        $this->loadKeys();
    }

    private function loadKeys(): void
    {
        $this->keys = ApiKey::where('user_id', auth()->id())
            ->orderBy('created_at', 'desc')
            ->get()
            ->toArray();
    }

    public function createKey(): void
    {
        $this->validate();

        $secret = 'gsmui_' . bin2hex(random_bytes(32));

        $key = ApiKey::create([
            'user_id' => auth()->id(),
            'name' => $this->newKeyName,
            'scope' => $this->newKeyScope,
            'key_hash' => hash('sha256', $secret),
            'key_prefix' => substr($secret, 0, 12),
            'last_used_at' => null,
        ]);

        $this->newKeySecret = $secret;
        $this->showCreateModal = false;
        $this->reset(['newKeyName', 'newKeyScope']);
        $this->loadKeys();
    }

    public function revokeKey(int $keyId): void
    {
        ApiKey::where('id', $keyId)
            ->where('user_id', auth()->id())
            ->delete();

        $this->loadKeys();
        $this->dispatch('toast', message: __('API key revoked.'));
    }

    public function copySecret(): void
    {
        $this->dispatch('copy-to-clipboard', text: $this->newKeySecret);
    }

    public function render()
    {
        return view('dashboard.api-keys', [
            'keys' => $this->keys,
        ]);
    }
}
```

```php
                'layout' => $this->layout,
            ],
        ]);

        $this->dispatch('toast', message: __('Dashboard preferences saved.'));
        $this->dispatch('dashboard-refresh');
    }

    {
        return view('dashboard.customize');
```

```php
<?php

namespace App\Livewire;

use Livewire\Component;
use Livewire\Attributes\Url;

class ProjectFilters extends Component
{
    #[Url(as: 'status')]
    public ?string $statusFilter = null;

    #[Url(as: 'sort')]
    public string $sortBy = 'updated_at';

    #[Url(as: 'q')]
    public string $search = '';

    public array $statusOptions = [
        'all' => 'All',
        'active' => 'Active',
        'paused' => 'Paused',
        'error' => 'Error',
    ];

    public array $sortOptions = [
        'updated_at' => 'Last updated',
        'name' => 'Name',
        'created_at' => 'Created',
        'status' => 'Status',
    ];

    public function clearFilters(): void
    {
        $this->statusFilter = null;
        $this->sortBy = 'updated_at';
        $this->search = '';
    }

    public function getHasFiltersProperty(): bool
    {
        return filled($this->statusFilter) || filled($this->search) || $this->sortBy !== 'updated_at';
    }

    public function render()
    {
        return view('dashboard.project-filters');
    }
}
```
