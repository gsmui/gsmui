# Git Hooks Configuration

## Pre-Commit Hook

```bash
#!/bin/bash
# .git/hooks/pre-commit

echo "Running pre-commit checks..."

# Run PHP CS Fixer
echo "Checking PHP code style..."
vendor/bin/pint --test
if [ $? -ne 0 ]; then
    echo "PHP CS Fixer failed. Run 'vendor/bin/pint --fix' to fix."
    exit 1
fi

# Run PHPStan
echo "Running static analysis..."
vendor/bin/phpstan analyse --no-progress
if [ $? -ne 0 ]; then
    echo "PHPStan found errors. Please fix them."
    exit 1
fi

# Run ESLint on changed JS files
CHANGED_JS=$(git diff --cached --name-only --diff-filter=ACM | grep -E '\.(js|ts|tsx)$')
if [ -n "$CHANGED_JS" ]; then
    echo "Checking JavaScript code style..."
    npx eslint $CHANGED_JS
    if [ $? -ne 0 ]; then
        echo "ESLint found errors. Please fix them."
        exit 1
    fi
fi

# Check for secrets
echo "Scanning for secrets..."
if git diff --cached --name-only | xargs grep -l -E '(password|secret|api_key|token)\s*=\s*["\x27]' 2>/dev/null; then
    echo "WARNING: Possible secrets detected in staged files!"
    echo "Please review your changes before committing."
    exit 1
fi

echo "Pre-commit checks passed!"
```

## Commit Message Hook

```bash
#!/bin/bash
# .git/hooks/commit-msg

commit_msg=$(cat "$1")

# Check commit message format
if ! echo "$commit_msg" | grep -qE '^(feat|fix|docs|style|refactor|test|chore|perf|ci|build|revert)(\(.+\))?: .{1,72}'; then
    echo "ERROR: Commit message does not follow conventional format."
    echo ""
    echo "Expected format: <type>(<scope>): <subject>"
    echo ""
    echo "Types: feat, fix, docs, style, refactor, test, chore, perf, ci, build, revert"
    echo "Example: feat(components): add date picker component"
    exit 1
fi

# Check subject line length
subject=$(echo "$commit_msg" | head -1)
if [ ${#subject} -gt 72 ]; then
    echo "ERROR: Subject line exceeds 72 characters."
    echo "Current length: ${#subject}"
    exit 1
fi

# Check for period at end of subject
if echo "$subject" | grep -q '\.$'; then
    echo "WARNING: Subject line should not end with a period."
fi

echo "Commit message validation passed!"
```

## Backup Types

| Type | Frequency | Retention | Storage | Method |
|------|-----------|-----------|---------|--------|
| Full backup | Daily (2am EST) | 30 days | S3 Standard | mysqldump |
| Incremental | Hourly | 7 days | S3 Standard | Binlog |
| Transaction logs | Real-time | 3 days | S3 Standard | Binlog streaming |
| Schema only | On each migration | Permanent | S3 Standard | mysqldump --no-data |
| Cross-region | Daily | 30 days | S3 (us-west-2) | S3 replication |
