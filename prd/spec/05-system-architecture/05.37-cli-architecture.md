# Purpose

The CLI provides command-line tools for developers to interact with and extend the GsmUi platform. It offers commands for component generation, theme management, debugging, and deployment automation.

---

# Architecture Overview

The CLI is built on Laravel's Artisan command system with custom namespaces and organization. Commands are grouped by domain (components, themes, debugging, deployment) and follow consistent patterns for input, output, and error handling.

The architecture separates command definition from business logic. Commands handle user interaction, argument parsing, and output formatting, while service classes encapsulate the actual work. This separation enables testing commands in isolation and reusing business logic across different interfaces.

The CLI supports both interactive and non-interactive modes. Interactive mode provides prompts and confirmations for manual use, while non-interactive mode uses flags and options for automation in scripts and CI pipelines.

# API Design

## CLI Commands

All CLI commands follow the `gsmui:` namespace prefix. Commands accept arguments and options following Artisan conventions:

- Arguments: Required, positional
- Options: Optional, prefixed with `--`
- Flags: Boolean options without values

## Exit Codes

- `0`: Success
- `1`: General failure
- `2`: Invalid arguments
- `3`: Permission denied
- `4`: Resource not found
- `5`: Timeout

# Command Registry

## Dynamic Command Registration

The CLI supports dynamic command registration from plugins and extensions. Commands are discovered at boot time and registered with the Artisan kernel.

```php
<?php

namespace App\Console;

use Illuminate\Console\Scheduling\Schedule;
use Illuminate\Foundation\Console\Kernel as ConsoleKernel;

class Kernel extends ConsoleKernel
{
    protected function commands(): void
    {
        $this->load(__DIR__.'/Commands');

        $this->loadPluginCommands();
        $this->loadExtensionCommands();
    }

    private function loadPluginCommands(): void
    {
        $pluginCommands = \Illuminate\Support\Facades\Cache::remember(
            'gsmui.plugin.commands',
            3600,
            function () {
                $commands = [];
                $pluginPath = base_path('plugins');

                if (!is_dir($pluginPath)) {
                    return [];
                }

                foreach (glob("{$pluginPath}/*/gsmui-plugin.json") as $manifestPath) {
                    $manifest = json_decode(file_get_contents($manifestPath), true);
                    $commands = array_merge($commands, $manifest['commands'] ?? []);
                }

                return $commands;
            }
        );

        foreach ($pluginCommands as $commandClass) {
            if (class_exists($commandClass)) {
                $this->app[$commandClass];
            }
        }
    }

    private function loadExtensionCommands(): void
    {
        $extensionCommands = \Illuminate\Support\Facades\Cache::remember(
            'gsmui.extension.commands',
            3600,
            function () {
                $commands = [];
                $extensionPath = base_path('extensions');

                if (!is_dir($extensionPath)) {
                    return [];
                }

                foreach (glob("{$extensionPath}/*/src/Commands/*.php") as $commandFile) {
                    require_once $commandFile;
                    $className = $this->resolveCommandClass($commandFile);
                    if ($className) {
                        $commands[] = $className;
                    }
                }

                return $commands;
            }
        );

        foreach ($extensionCommands as $commandClass) {
            $this->app[$commandClass];
        }
    }

    protected function schedule(Schedule $schedule): void
    {
        $schedule->command('gsmui:analytics:compute-daily')->daily();
        $schedule->command('gsmui:backup:create')->daily();
        $schedule->command('gsmui:health:check')->everyFiveMinutes();
        $schedule->command('gsmui:cache:warm')->hourly();
        $schedule->command('gsmui:logs:rotate')->daily();
        $schedule->command('gsmui:plugins:check-updates')->daily();
    }
}
```

# Command Output Formatter

## Rich Terminal Output

```php
<?php

namespace App\Console\Output;

use Symfony\Component\Console\Helper\Table;
use Symfony\Component\Console\Helper\TableCell;
use Symfony\Component\Console\Helper\TableSeparator;
use Symfony\Component\Console\Output\OutputInterface;

class CommandOutput
{
    private OutputInterface $output;

    public function __construct(OutputInterface $output)
    {
        $this->output = $output;
    }

    public function success(string $message): void
    {
        $this->output->writeln("<info>✓</info> {$message}");
    }

    public function error(string $message): void
    {
        $this->output->writeln("<error>✗</error> {$message}");
    }

    public function warning(string $message): void
    {
        $this->output->writeln("<comment>⚠</comment> {$message}");
    }

    public function info(string $message): void
    {
        $this->output->writeln("<info>i</info> {$message}");
    }

    public function table(array $headers, array $rows, array $options = []): void
    {
        $table = new Table($this->output);
        $table->setHeaders($headers);
        $table->setRows($rows);

        if (isset($options['style'])) {
            $table->setStyle($options['style']);
        }

        if (isset($options['column widths'])) {
            $table->setColumnWidths($options['column widths']);
        }

        $table->render();
    }

    public function keyValueTable(array $data, string $title = ''): void
    {
        if ($title) {
            $this->output->writeln("<info>{$title}</info>");
            $this->output->writeln('');
        }

        $table = new Table($this->output);
        $table->setHeaders(['Key', 'Value']);
        $table->setRows(array_map(
            fn($key, $value) => [$key, is_bool($value) ? ($value ? 'Yes' : 'No') : (string) $value],
            array_keys($data),
            array_values($data)
        ));

        $table->setStyle('compact');
        $table->render();
    }

    public function progressBar(int $max): \Symfony\Component\Console\Helper\ProgressBarHelper
    {
        return $this->output->createProgressBar($max);
    }

    public function section(string $title): void
    {
        $this->output->writeln('');
        $this->output->writeln("<info>━━━ {$title} ━━━</info>");
        $this->output->writeln('');
    }

    public function listing(array $items, string $prefix = '  '): void
    {
        foreach ($items as $item) {
            $this->output->writeln("{$prefix}• {$item}");
        }
    }

    public function json(mixed $data): void
    {
        $this->output->writeln(json_encode($data, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES));
    }

    public function csv(array $headers, array $rows): void
    {
        $this->output->writeln(implode(',', $headers));
        foreach ($rows as $row) {
            $this->output->writeln(implode(',', array_map(function ($cell) {
                return str_contains($cell, ',') || str_contains($cell, '"')
                    ? '"' . str_replace('"', '""', $cell) . '"'
                    : $cell;
            }, $row)));
        }
    }

    public function statusTable(array $items): void
    {
        $headers = ['Item', 'Status', 'Details'];
        $rows = array_map(function ($item) {
            $status = $item['status'] ?? 'unknown';
            $statusColor = match ($status) {
                'ok', 'active', 'healthy' => "<info>{$status}</info>",
                'warning', 'degraded' => "<comment>{$status}</comment>",
                'error', 'failed', 'critical' => "<error>{$status}</error>",
                default => $status,
            };

            return [
                $item['name'] ?? 'Unknown',
                $statusColor,
                $item['details'] ?? '',
            ];
        }, $items);

        $this->table($headers, $rows);
    }
}
```

```php
                            {--radius= : Border radius (none, sm, md, lg, full)}
                            {--shadow= : Shadow intensity (none, sm, md, lg)}
                            {--force : Overwrite existing theme}';

    protected $description = 'Generate a new GsmUi theme';

    public function handle(): int
    {
        $name = $this->argument('name');
        $slug = Str::kebab($name);
        $description = $this->option('desc') ?? "A custom GsmUi theme";
        $base = $this->option('base');
        $palette = $this->option('palette') ? json_decode($this->option('palette'), true) : null;
        $fontBody = $this->option('font-body');
        $fontHeading = $this->option('font-heading');
        $radius = $this->option('radius') ?? 'md';
        $shadow = $this->option('shadow') ?? 'md';
        $force = $this->option('force');

        $themePath = resource_path("css/gsmui/themes/{$slug}");

        if (is_dir($themePath) && !$force) {
            $this->error("Theme '{$slug}' already exists. Use --force to overwrite.");
            return self::FAILURE;
        }

        $this->info("Generating theme: {$name}");

        $tokens = $this->generateTokens($slug, $palette, $fontBody, $fontHeading, $radius, $shadow);
        $this->createThemeFile($themePath, $slug, $tokens);
        $this->createThemeConfig($themePath, $slug, $description, $tokens);
        $this->createThemePreview($themePath, $slug, $tokens);

        $this->info("Theme '{$name}' generated successfully!");
        $this->info("Activate with: php artisan gsmui:theme:activate {$slug}");

        return self::SUCCESS;

    private function generateTokens(string $slug, ?array $palette, ?string $fontBody, ?string $fontHeading, string $radius, string $shadow): array
    {

                'body' => $fontBody ?? 'Inter, system-ui, sans-serif',

    private function createThemeFile(string $path, string $slug, array $tokens): void

        }

        $css .= "}\n";
```

```php
    protected $description = 'Run platform health checks';

    public function handle(): int
    {
        $this->info('Running GsmUi health checks...');
        $this->newLine();

        $checks = [
            $this->checkDatabase(),
            $this->checkCache(),
            $this->checkQueue(),
            $this->checkDiskSpace(),
            $this->checkPhpVersion(),
            $this->checkExtensions(),

        if ($this->option('json')) {
            $this->output->writeln(json_encode($checks, JSON_PRETTY_PRINT));
            return self::SUCCESS;
        }

        $this->displayResults($checks);

        $failed = array_filter($checks, fn($check) => $check['status'] === 'critical');

        }

    private function checkDatabase(): array
    {
        try {
            $start = microtime(true);
            DB::select('SELECT 1');
            $duration = (microtime(true) - $start) * 1000;

            return [
                'name' => 'Database',
                'status' => $duration < 100 ? 'ok' : ($duration < 500 ? 'warning' : 'critical'),
                'details' => "Connected ({$duration}ms)",
                'metrics' => ['response_time_ms' => $duration],
            ];
        } catch (\Throwable $e) {
            return [
                'name' => 'Database',
                'status' => 'critical',
                'details' => "Connection failed: {$e->getMessage()}",
            ];
        }
    }

    private function checkCache(): array
    {
        try {
            $start = microtime(true);
            Cache::put('health_check', 'ok', 60);
            $value = Cache::get('health_check');
            $duration = (microtime(true) - $start) * 1000;

            return [
                'name' => 'Cache',
                'status' => $value === 'ok' ? 'ok' : 'critical',
                'details' => $value === 'ok' ? "Operational ({$duration}ms)" : 'Read/write mismatch',
                'metrics' => ['response_time_ms' => $duration],
            ];
        } catch (\Throwable $e) {
            return [
                'name' => 'Cache',
                'status' => 'critical',
                'details' => "Cache unavailable: {$e->getMessage()}",

    {
        try {
            $connection = config('queue.default');

            return [
                'name' => 'Queue',

                'status' => 'ok',
                'details' => 'No plugins directory',

        foreach ($plugins as $manifestPath) {

        return [
            'name' => 'Plugins',
            'status' => 'ok',
            'details' => count($plugins) . ' installed, ' . $activeCount . ' active',

    {
        $headers = ['Check', 'Status', 'Details'];
        $rows = array_map(function ($check) {
            $statusIcon = match ($check['status']) {
                'ok' => '<info>✓ OK</info>',
                'warning' => '<comment>⚠ WARNING</comment>',
                'critical' => '<error>✗ CRITICAL</error>',
                default => $check['status'],
            };

            return [
                $check['name'],
                $statusIcon,
                $check['details'],
            ];
        }, $checks);

        $this->table($headers, $rows);

        $critical = count(array_filter($checks, fn($c) => $c['status'] === 'critical'));
        $warnings = count(array_filter($checks, fn($c) => $c['status'] === 'warning'));

        $this->newLine();

        if ($critical > 0) {
            $this->error("{$critical} critical issue(s) found");
        } elseif ($warnings > 0) {
            $this->warning("{$warnings} warning(s) found");
        } else {
            $this->success('All checks passed');
        }
    }

    private function attemptFixes(array $failedChecks): void
    {
        foreach ($failedChecks as $check) {
            $this->info("Attempting fix for: {$check['name']}");

            switch ($check['name']) {
                case 'Cache':
                    $this->call('cache:clear');
                    break;
                case 'Queue':
                    $this->call('queue:restart');
                    break;
                case 'Migrations':
                    $this->call('migrate');
                    break;
            }
        }
    }

    private function formatBytes(int $bytes, int $precision = 2): string
    {
        $units = ['B', 'KB', 'MB', 'GB', 'TB'];
        $bytes = max($bytes, 0);
        $pow = floor(($bytes ? log($bytes) : 0) / log(1024));
        $pow = min($pow, count($units) - 1);
        $bytes /= pow(1024, $pow);

        return round($bytes, $precision) . ' ' . $units[$pow];
    }
```

```php
<?php

namespace App\Console\Commands\Gsmui;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\Process;

class DeployCommand extends Command
{
    protected $signature = 'gsmui:deploy
                            {environment=production : Target environment}
                            {--dry-run : Show what would be done without executing}
                            {--skip-tests : Skip test execution}
                            {--skip-backup : Skip pre-deploy backup}
                            {--force : Skip confirmation prompt}';

    protected $description = 'Deploy the GsmUi application';

    public function handle(): int
    {
        $environment = $this->argument('environment');
        $dryRun = $this->option('dry-run');
        $skipTests = $this->option('skip-tests');
        $skipBackup = $this->option('skip-backup');
        $force = $this->option('force');

        $this->info("Deploying to {$environment}...");
        $this->newLine();

        $steps = [
            'pre-flight' => $this->preFlightChecks($environment),
            'backup' => $skipBackup ? 'skipped' : $this->createBackup(),
            'tests' => $skipTests ? 'skipped' => $this->runTests(),
            'build' => $this->buildAssets(),
            'migrate' => $this->runMigrations(),
            'cache' => $this->clearCaches(),
            'reload' => $this->reloadServices(),
            'verify' => $this->verifyDeployment(),
        ];

        if ($dryRun) {
            $this->displayDryRun($steps);
            return self::SUCCESS;
        }

        if (!$force) {
            if (!$this->confirm('Ready to deploy?')) {
                return self::SUCCESS;
            }
        }

        $results = [];
        foreach ($steps as $step => $action) {
            $this->info("Running: {$step}");

            if ($action === 'skipped') {
                $this->comment("  Skipped");
                $results[$step] = 'skipped';
                continue;
            }

            $start = microtime(true);
            $result = $action();
            $duration = round(microtime(true) - $start, 2);

            $results[$step] = $result;
            $status = $result ? '✓' : '✗';
            $this->line("  {$status} Completed in {$duration}s");

            if (!$result && !$this->option('force')) {
                $this->error("Deployment failed at step: {$step}");
                return self::FAILURE;
            }
        }

        $this->newLine();
        $this->displayDeploymentSummary($results);

        return self::SUCCESS;
    }

    private function preFlightChecks(string $environment): callable
    {
        return function () use ($environment) {
            $checks = [
                $this->checkGitStatus(),
                $this->checkEnvironment($environment),
                $this->checkDependencies(),
            ];

            return !in_array(false, $checks);
        };
    }

    private function checkGitStatus(): bool
    {
        $status = Process::run('git status --porcelain');

        if (!empty(trim($status->output()))) {
            $this->warning('  Working directory is not clean');
            return false;
        }

        return true;
    }

    private function checkEnvironment(string $environment): bool
    {
        $envFile = base_path(".env.{$environment}");

        if (!file_exists($envFile)) {
            $this->warning("  Environment file .env.{$environment} not found");
            return false;
        }

        return true;
    }

    private function checkDependencies(): bool
    {
        $result = Process::run('composer install --no-dev --optimize-autoloader --no-interaction');

        return $result->successful();
    }

    private function createBackup(): callable
    {
        return function () {
            return $this->call('gsmui:backup', ['--quiet']) === self::SUCCESS;
        };
    }

    private function runTests(): callable
    {
        return function () {
            $result = Process::run('php artisan test --parallel');
            return $result->successful();
        };
    }

    private function buildAssets(): callable
    {
        return function () {
            $result = Process::run('npm ci && npm run build');
            return $result->successful();
        };
    }

    private function runMigrations(): callable
    {
        return function () {
            return $this->call('gsmui:migrate', ['--force']) === self::SUCCESS;
        };
    }

    private function clearCaches(): callable
    {
        return function () {
            $this->call('gsmui:cache:clear', ['--quiet']);
            return true;
        };
    }

    private function reloadServices(): callable
    {
        return function () {
            $this->call('queue:restart');
            return true;
        };
    }

    private function verifyDeployment(): callable
    {
        return function () {
            return $this->call('gsmui:health', ['--json']) === self::SUCCESS;
        };
    }

    private function displayDryRun(array $steps): void
    {
        $this->info('Dry run - the following steps would be executed:');
        $this->newLine();

        foreach ($steps as $step => $action) {
            $status = $action === 'skipped' ? '(skipped)' : '(would execute)';
            $this->line("  {$step} {$status}");
        }
    }

    private function displayDeploymentSummary(array $results): void
    {
        $headers = ['Step', 'Status'];
        $rows = array_map(fn($step, $result) => [
            $step,
            $result === 'skipped' ? '<comment>Skipped</comment>' : ($result ? '<info>Success</info>' : '<error>Failed</error>'),
        ], array_keys($results), array_values($results));

        $this->table($headers, $rows);

        $failed = array_filter($results, fn($r) => $r === false);

        if (empty($failed)) {
            $this->success('Deployment completed successfully!');
        } else {
            $this->error('Deployment completed with failures');
        }
    }
}
```

# Backup Command

## Automated Backup

```php
<?php

namespace App\Console\Commands\Gsmui;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Facades\Storage;

class BackupCommand extends Command
{
    protected $signature = 'gsmui:backup
                            {--type=full : Backup type (full, database, files, config)}
                            {--disk=local : Storage disk for backup}
                            {--compression=gzip : Compression method}
                            {--retention=30 : Days to keep backups}
                            {--quiet : Suppress output}';

    protected $description = 'Create a platform backup';

    public function handle(): int
    {
        $type = $this->option('type');
        $disk = $this->option('disk');
        $compression = $this->option('compression');
        $retention = (int) $this->option('retention');
        $quiet = $this->option('quiet');

        $backupId = date('Y-m-d_His');
        $backupPath = storage_path("app/backups/{$backupId}");

        if (!is_dir($backupPath)) {
            mkdir($backupPath, 0755, true);
        }

        $this->info("Creating {$type} backup...");

        $startTime = microtime(true);

        try {
            match ($type) {
                'full' => $this->fullBackup($backupPath),
                'database' => $this->databaseBackup($backupPath),
                'files' => $this->filesBackup($backupPath),
                'config' => $this->configBackup($backupPath),
            };

            $archivePath = $this->compress($backupPath, $compression);
            $this->uploadToDisk($archivePath, $disk, $backupId);
            $this->cleanupOldBackups($disk, $retention);

            $duration = round(microtime(true) - $startTime, 2);
            $size = $this->formatBytes(filesize($archivePath));

            if (!$quiet) {
                $this->success("Backup completed in {$duration}s ({$size})");
            }

            return self::SUCCESS;
        } catch (\Throwable $e) {
            if (!$quiet) {
                $this->error("Backup failed: {$e->getMessage()}");
            }

            return self::FAILURE;
        }
    }

    private function fullBackup(string $path): void
    {
        $this->databaseBackup($path);
        $this->filesBackup($path);
        $this->configBackup($path);
    }

    private function databaseBackup(string $path): void
    {
        $this->info('  Backing up database...');

        $tables = DB::select('SHOW TABLES');
        $dbName = DB::getDatabaseName();

        $dump = "-- GsmUi Database Backup\n";
        $dump .= "-- Date: " . date('Y-m-d H:i:s') . "\n";
        $dump .= "-- Database: {$dbName}\n\n";
        $dump .= "SET FOREIGN_KEY_CHECKS=0;\n\n";

        foreach ($tables as $table) {
            $tableName = $table->{"Tables_in_{$dbName}"} ?? reset($table);

            $dump .= "-- Table: {$tableName}\n";
            $dump .= "DROP TABLE IF EXISTS `{$tableName}`;\n";

            $createTable = DB::select("SHOW CREATE TABLE `{$tableName}`");
            $dump .= $createTable[0]->{'Create Table'} ?? '';
            $dump .= ";\n\n";

            $rows = DB::table($tableName)->get();

            foreach ($rows as $row) {
                $values = array_map(function ($value) {
                    if (is_null($value)) return 'NULL';
                    if (is_bool($value)) return $value ? '1' : '0';
                    return "'" . addslashes($value) . "'";
                }, (array) $row);

                $dump .= "INSERT INTO `{$tableName}` VALUES (" . implode(', ', $values) . ");\n";
            }

            $dump .= "\n";
        }

        $dump .= "SET FOREIGN_KEY_CHECKS=1;\n";

        file_put_contents("{$path}/database.sql", $dump);
    }

    private function filesBackup(string $path): void
    {
        $this->info('  Backing up files...');

        $directories = [
            'public/uploads' => 'uploads',
            'storage/app/public' => 'storage-public',
            'resources/views' => 'views-custom',
        ];

        foreach ($directories as $source => $dest) {
            $sourcePath = base_path($source);

            if (is_dir($sourcePath)) {
                $this->copyDirectory($sourcePath, "{$path}/files/{$dest}");
            }
        }
    }

    private function configBackup(string $path): void
    {
        $this->info('  Backing up configuration...');

        $configFiles = [
            '.env',
            'config/gsmui.php',
            'config/gsmui-pro.php',
            'composer.json',
            'package.json',
        ];

        foreach ($configFiles as $file) {
            $sourcePath = base_path($file);

            if (file_exists($sourcePath)) {
                $destDir = dirname("{$path}/config/{$file}");

                if (!is_dir($destDir)) {
                    mkdir($destDir, 0755, true);
                }

                copy($sourcePath, "{$path}/config/{$file}");
            }
        }
    }

    private function compress(string $path, string $method): string
    {
        $archivePath = "{$path}.tar.gz";

        $command = "tar -czf \"{$archivePath}\" -C \"" . dirname($path) . "\" \"" . basename($path) . "\"";
        exec($command, $output, $returnCode);

        if ($returnCode !== 0) {
            throw new \RuntimeException("Compression failed");
        }

        return $archivePath;
    }

    private function uploadToDisk(string $archivePath, string $disk, string $backupId): void
    {
        $filename = basename($archivePath);
        $contents = file_get_contents($archivePath);

        Storage::disk($disk)->put("backups/{$filename}", $contents);
    }

    private function cleanupOldBackups(string $disk, int $retentionDays): void
    {
        $files = Storage::disk($disk)->files('backups');
        $cutoff = now()->subDays($retentionDays);

        foreach ($files as $file) {
            $lastModified = Storage::disk($disk)->lastModified($file);

            if ($lastModified < $cutoff->timestamp) {
                Storage::disk($disk)->delete($file);
            }
        }
    }

    private function copyDirectory(string $source, string $destination): void
    {
        if (!is_dir($destination)) {
            mkdir($destination, 0755, true);
        }

        $files = scandir($source);

        foreach ($files as $file) {
            if ($file === '.' || $file === '..') continue;

            $sourcePath = "{$source}/{$file}";
            $destPath = "{$destination}/{$file}";

            if (is_dir($sourcePath)) {
                $this->copyDirectory($sourcePath, $destPath);
            } else {
                copy($sourcePath, $destPath);
            }
        }
    }

    private function formatBytes(int $bytes, int $precision = 2): string
    {
        $units = ['B', 'KB', 'MB', 'GB'];
        $bytes = max($bytes, 0);
        $pow = floor(($bytes ? log($bytes) : 0) / log(1024));
        $pow = min($pow, count($units) - 1);
        $bytes /= pow(1024, $pow);
        return round($bytes, $precision) . ' ' . $units[$pow];
    }
}
```
