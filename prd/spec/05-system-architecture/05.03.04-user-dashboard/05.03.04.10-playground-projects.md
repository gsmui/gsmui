## PlaygroundProjects Component

```php
// app/Livewire/Playground/PlaygroundProjects.php
class PlaygroundProjects extends Component
{
    public string $viewMode = 'grid';
    public string $sortBy = 'last_saved_at';
    public string $sortDirection = 'desc';
    public string $search = '';
    public array $projects = [];
    public int $totalProjects = 0;
    public bool $showCreateModal = false;
    public string $newProjectName = '';
    public string $newProjectDescription = '';

    public function mount(): void
    {
        $this->loadProjects();
    }

    public function loadProjects(): void
    {
        $query = PlaygroundProject::where('user_id', auth()->id());

        if ($this->search) {
            $query->where(function ($q) {
                $q->where('name', 'like', "%{$this->search}%")
                  ->orWhere('description', 'like', "%{$this->search}%");
            });
        }

        $this->projects = $query->orderBy($this->sortBy, $this->sortDirection)
            ->get()
            ->toArray();

        $this->totalProjects = count($this->projects);
    }

    public function createProject(): void
    {
        $this->validate([
            'newProjectName' => 'required|string|max:100',
            'newProjectDescription' => 'nullable|string|max:500',
        ]);

        $project = PlaygroundProject::create([
            'user_id' => auth()->id(),
            'name' => $this->newProjectName,
            'description' => $this->newProjectDescription,
            'components' => [],
            'custom_css' => '',
            'custom_js' => '',
        ]);

        event(new ProjectCreated($project));

        $this->showCreateModal = false;
        $this->newProjectName = '';
        $this->newProjectDescription = '';

        return redirect()->to("/playground/{$project->id}");
    }

    public function duplicateProject(string $projectId): void
    {
        $original = PlaygroundProject::where('user_id', auth()->id())
            ->findOrFail($projectId);

        $duplicate = $original->replicate();
        $duplicate->name = $original->name . ' (Copy)';
        $duplicate->save();

        event(new ProjectDuplicated($original, $duplicate));

        $this->loadProjects();

        $this->dispatch('profileSaved', message: 'Project duplicated');
    }

    public function deleteProject(string $projectId): void
    {
        $project = PlaygroundProject::where('user_id', auth()->id())
            ->findOrFail($projectId);

        $project->delete();

        event(new ProjectDeleted($project));

        $this->loadProjects();

        $this->dispatch('profileSaved', message: 'Project deleted');
    }

    public function toggleViewMode(): void
    {
        $this->viewMode = $this->viewMode === 'grid' ? 'list' : 'grid';
    }

    public function render()
    {
        return view('livewire.playground.projects')->layout('layouts.dashboard');
    }
}
```

## ProjectEditor Component

```php
// app/Livewire/Playground/ProjectEditor.php
class ProjectEditor extends Component
{
    public string $projectId;
    public array $project = [];
    public array $components = [];
    public string $customCss = '';
    public string $customJs = '';
    public ?string $selectedComponentId = null;
    public bool $saving = false;
    public string $saveStatus = 'saved';
    public bool $showPreview = true;
    public bool $showCode = false;
    public string $activeCodeTab = 'html';
    public int $version = 1;

    private ?\Timer $autoSaveTimer = null;

    protected $listeners = [
        'componentAdded' => 'addComponent',
        'componentRemoved' => 'removeComponent',
        'componentUpdated' => 'updateComponent',
        'componentReordered' => 'reorderComponents',
        'cssChanged' => 'updateCss',
        'jsChanged' => 'updateJs',
    ];

    public function mount(string $projectId): void
    {
        $this->projectId = $projectId;
        $this->loadProject();
    }

    private function loadProject(): void
    {
        $project = PlaygroundProject::where('user_id', auth()->id())
            ->with('components')
            ->findOrFail($this->projectId);

        $this->project = $project->toArray();
        $this->components = $project->components->toArray();
        $this->customCss = $project->custom_css;
        $this->customJs = $project->custom_js;
        $this->version = $project->version;
    }

    public function addComponent(array $componentData): void
    {
        $component = [
            'id' => Str::uuid(),
            'type' => $componentData['type'],
            'config' => $componentData['config'] ?? [],
            'position' => count($this->components),
            'custom_css' => '',
            'custom_js' => '',
        ];

        $this->components[] = $component;
        $this->triggerAutoSave();
    }

    public function removeComponent(string $componentId): void
    {
        $this->components = array_filter(
            $this->components,
            fn($c) => $c['id'] !== $componentId
        );

        $this->components = array_values($this->components);

        if ($this->selectedComponentId === $componentId) {
            $this->selectedComponentId = null;
        }

        $this->triggerAutoSave();
    }

    public function updateComponent(array $data): void
    {
        $componentId = $data['id'];
        unset($data['id']);

        $this->components = array_map(function ($component) use ($componentId, $data) {
            if ($component['id'] === $componentId) {
                return array_merge($component, $data);
            }
            return $component;
        }, $this->components);

        $this->triggerAutoSave();
    }

    public function reorderComponents(array $newOrder): void
    {
        $this->components = collect($this->components)
            ->sortBy(fn($c) => array_search($c['id'], $newOrder) ?? 999)
            ->values()
            ->toArray();

        $this->triggerAutoSave();
    }

    public function selectComponent(?string $componentId): void
    {
        $this->selectedComponentId = $componentId;
    }

    public function updateCss(string $css): void
    {
        $this->customCss = $css;
        $this->triggerAutoSave();
    }

    public function updateJs(string $js): void
    {
        $this->customJs = $js;
        $this->triggerAutoSave();
    }

    private function triggerAutoSave(): void
    {
        $this->saveStatus = 'saving';

        // Debounce save - cancel previous timer
        if ($this->autoSaveTimer) {
            $this->autoSaveTimer->cancel();
        }

        $this->autoSaveTimer = timer(300)->then(function () {
            $this->saveProject();
        });
    }

    public function saveProject(): void
    {
        $this->saving = true;

        $project = PlaygroundProject::where('user_id', auth()->id())
            ->findOrFail($this->projectId);

        $project->update([
            'components' => $this->components,
            'custom_css' => $this->customCss,
            'custom_js' => $this->customJs,
            'component_count' => count($this->components),
            'last_saved_at' => now(),
            'version' => $this->version + 1,
        ]);

        // Create version snapshot
        $this->createSnapshot($project);

        $this->version = $project->version;
        $this->saving = false;
        $this->saveStatus = 'saved';

        event(new ProjectSaved($project));
    }

    private function createSnapshot(PlaygroundProject $project): void
    {
        $previousSnapshot = PlaygroundSnapshot::where('project_id', $project->id)
            ->latest()
            ->first();

        $currentData = [
            'components' => $this->components,
            'custom_css' => $this->customCss,
            'custom_js' => $this->customJs,
        ];

        $delta = $previousSnapshot
            ? $this->calculateDelta($previousSnapshot->components_snapshot, $currentData)
            : null;

        PlaygroundSnapshot::create([
            'project_id' => $project->id,
            'version' => $project->version,
            'components_snapshot' => $currentData,
            'delta_from_previous' => $delta,
        ]);
    }

    private function calculateDelta(array $old, array $new): ?string
    {
        $diff = $this->computeDiff($old, $new);
        return empty($diff) ? null : json_encode($diff);
    }

    public function getPreviewHtml(): string
    {
        $html = '<!DOCTYPE html><html><head>';
        $html .= '<script src="https://cdn.tailwindcss.com"></script>';
        $html .= '<style>' . $this->customCss . '</style>';
        $html .= '</head><body class="p-4">';
        $html .= '<div id="app">';

        foreach ($this->components as $component) {
            $html .= $this->renderComponent($component);
        }

        $html .= '</div>';
        $html .= '<script>' . $this->customJs . '</script>';
        $html .= '</body></html>';

        return $html;
    }

    private function renderComponent(array $component): string
    {
        $type = $component['type'];
        $config = $component['config'];

        return match($type) {
            'button' => "<button class=\"px-4 py-2 bg-blue-600 text-white rounded-lg\">" .
                        ($config['text'] ?? 'Button') . "</button>",
            'input' => "<input type=\"text\" placeholder=\"" . ($config['placeholder'] ?? '') .
                       "\" class=\"px-4 py-2 border rounded-lg\" />",
            'card' => "<div class=\"p-6 border rounded-lg\"><h3 class=\"font-bold\">" .
                      ($config['title'] ?? 'Card') . "</h3></div>",
            default => "<div class=\"p-4 border rounded\">Unknown component</div>",
        };
    }

    public function render()
    {
        return view('livewire.playground.project-editor');
    }
}
```

```php
class ProjectShareModal extends Component
{
    public string $projectId;
    public bool $isPublic = false;
    public ?string $shareToken = null;
    public ?string $shareUrl = null;
    public ?string $embedCode = null;
    public string $privacy = 'private';
    public bool $showShareLink = false;
    public bool $showEmbedCode = false;

    public function mount(string $projectId): void
    {
        $this->projectId = $projectId;
        $this->loadShareSettings();
    }

    private function loadShareSettings(): void
    {
        $project = PlaygroundProject::where('user_id', auth()->id())
            ->findOrFail($this->projectId);

        $this->isPublic = $project->is_public;
        $this->shareToken = $project->share_token;
        $this->privacy = $project->is_public ? 'public' : 'private';

        if ($this->shareToken) {
            $this->shareUrl = url("/playground/shared/{$this->shareToken}");
            $this->embedCode = '<iframe src="' . $this->shareUrl . '" width="100%" height="500" frameborder="0"></iframe>';
        }
    }

    public function updatedPrivacy(string $value): void
    {
        $this->isPublic = $value === 'public';
    }

    public function generateShareLink(): void
    {
        $project = PlaygroundProject::where('user_id', auth()->id())
            ->findOrFail($this->projectId);

        if (!$project->share_token) {
            $project->update([
                'share_token' => Str::random(32),
                'is_public' => $this->isPublic,
            ]);
        }

        $this->loadShareSettings();
        $this->showShareLink = true;

        event(new ProjectShared($project));
    }

    public function copyShareLink(): void
    {
        $this->dispatch('copyToClipboard', text: $this->shareUrl);
    }

    public function copyEmbedCode(): void
    {
        $this->dispatch('copyToClipboard', text: $this->embedCode);
    }

    public function revokeShareLink(): void
    {
        $project = PlaygroundProject::where('user_id', auth()->id())
            ->findOrFail($this->projectId);

        $project->update([
            'share_token' => null,
            'is_public' => false,
        ]);

        $this->shareToken = null;
        $this->shareUrl = null;
        $this->embedCode = null;
        $this->showShareLink = false;

        $this->dispatch('profileSaved', message: 'Share link revoked');
    }

    public function render()
    {
        return view('livewire.playground.project-share-modal');
    }
}
```

```php
class ProjectExportModal extends Component
{
    public string $projectId;
    public string $exportFormat = 'html';
    public string $exportType = 'code';
    public ?string $exportedCode = null;
    public ?string $exportUrl = null;

        'vue' => 'Vue SFC',

    public array $types = [

    ];

    public function mount(string $projectId): void
    {
        $this->projectId = $projectId;
    }

    public function exportProject(): void
    {
        $this->exporting = true;

        $project = PlaygroundProject::where('user_id', auth()->id())
            ->with('components')
            ->findOrFail($this->projectId);

        $this->exportedCode = match($this->exportFormat) {
            'html' => $this->exportAsHtml($project),
            'blade' => $this->exportAsBlade($project),
            'react' => $this->exportAsReact($project),
            'vue' => $this->exportAsVue($project),
            default => '',
        };

        if ($this->exportType === 'zip') {
            $this->exportUrl = $this->generateZipExport($project);
        }

        $this->exporting = false;

    }

    private function exportAsHtml(PlaygroundProject $project): string
    {
        $html = '<!DOCTYPE html>' . PHP_EOL;

        $filename = match($this->exportFormat) {
            'html' => 'index.html',
            'blade' => 'welcome.blade.php',
            'react' => 'Component.jsx',
            'vue' => 'Component.vue',
        };

        $zip->addFromString($filename, $this->exportedCode);

        // Add additional files based on format
        if ($this->exportFormat === 'blade') {

        }

        $exportPath = 'playground/exports/' . Str::uuid() . '.zip';
        Storage::disk('s3')->put($exportPath, file_get_contents($zipPath));

        return Storage::disk('s3')->temporaryUrl($exportPath, now()->addHour());
```

## VersionHistoryModal Component

```php
// app/Livewire/Playground/VersionHistoryModal.php
class VersionHistoryModal extends Component
{
    public string $projectId;
    public array $versions = [];
    public ?string $selectedVersionId = null;
    public ?array $selectedVersionData = null;
    public bool $loading = false;

    public function mount(string $projectId): void
    {
        $this->projectId = $projectId;
        $this->loadVersions();
    }

    private function loadVersions(): void
    {
        $this->versions = PlaygroundSnapshot::where('project_id', $this->projectId)
            ->orderBy('version', 'desc')
            ->get()
            ->map(fn($snapshot) => [
                'id' => $snapshot->id,
                'version' => $snapshot->version,
                'created_at' => $snapshot->created_at->diffForHumans(),
                'component_count' => count($snapshot->components_snapshot['components'] ?? []),
            ])
            ->toArray();
    }

    public function selectVersion(string $versionId): void
    {
        $this->selectedVersionId = $versionId;
        $this->loading = true;

        $snapshot = PlaygroundSnapshot::findOrFail($versionId);
        $this->selectedVersionData = $snapshot->components_snapshot;

        $this->loading = false;
    }

    public function restoreVersion(): void
    {
        if (!$this->selectedVersionId) {
            return;
        }

        $snapshot = PlaygroundSnapshot::findOrFail($this->selectedVersionId);

        $project = PlaygroundProject::where('user_id', auth()->id())
            ->findOrFail($this->projectId);

        $project->update([
            'components' => $snapshot->components_snapshot['components'] ?? [],
            'custom_css' => $snapshot->components_snapshot['custom_css'] ?? '',
            'custom_js' => $snapshot->components_snapshot['custom_js'] ?? '',
            'version' => $project->version + 1,
        ]);

        event(new VersionRestored($project, $snapshot));

        $this->dispatch('profileSaved', message: 'Version restored successfully');
        $this->dispatch('versionRestored');
    }

    public function render()
    {
        return view('livewire.playground.version-history-modal');
    }
}
```

# Edge Cases

| Scenario | Handling |
|----------|----------|
| Auto-save fails | Show error toast, retry automatically |
| Project with 50+ components | Virtual rendering, lazy-load off-screen |
| Concurrent edits from 2 tabs | Last-write-wins with notification |
| Public link accessed without auth | View-only mode, no edit capabilities |
| Export with invalid code | Validate before export, show errors |
| Version restore with unsaved changes | Prompt to save or discard |
| Large project (>1MB) | Compress storage, lazy-load components |
| Component type removed | Show warning, allow migration |
| Share link expired | Regenerate on demand |
| Project with custom JS errors | Sandboxed iframe, errors contained |

## Component Resize

```javascript
// Alpine.js component resize handles
Alpine.data('componentResizer', () => ({
    resizing: false,
    startX: 0,
    startY: 0,
    startWidth: 0,
    startHeight: 0,

    initResizer(e, componentId) {
        this.resizing = true;
        this.startX = e.clientX;
        this.startY = e.clientY;

        const el = e.target.closest('[data-component]');
        this.startWidth = el.offsetWidth;
        this.startHeight = el.offsetHeight;

        const onMouseMove = (e) => {
            if (!this.resizing) return;

            const dx = e.clientX - this.startX;
            const dy = e.clientY - this.startY;

            const newWidth = Math.max(100, this.startWidth + dx);
            const newHeight = Math.max(50, this.startHeight + dy);

            el.style.width = `${newWidth}px`;
            el.style.height = `${newHeight}px`;
        };

        const onMouseUp = () => {
            this.resizing = false;
            document.removeEventListener('mousemove', onMouseMove);
            document.removeEventListener('mouseup', onMouseUp);

            const el = document.querySelector(`[data-component="${componentId}"]`);
            this.$wire.updateComponent({
                id: componentId,
                config: {
                    width: el.offsetWidth,
                    height: el.offsetHeight
                }
            });
        };

        document.addEventListener('mousemove', onMouseMove);
        document.addEventListener('mouseup', onMouseUp);
    }
}));
```

```javascript
export function initCodeEditor(element, language, content, onChange) {
    const langMap = {
        html: html(),
        css: css(),
        js: javascript(),
    };

    const state = EditorState.create({
        doc: content,
        extensions: [
            basicSetup,
            langMap[language] || html(),
            oneDark,

        ],
    });

    return new EditorView({

        parent: element,
    });
```

# Auto-Save System

## Debounced Save

```javascript
// resources/js/playground/auto-save.js
export class AutoSaveManager {
    constructor(wireComponent, delay = 300) {
        this.wireComponent = wireComponent;
        this.delay = delay;
        this.timer = null;
        this.pendingChanges = false;
        this.lastSavedContent = null;
    }

    scheduleSave() {
        this.pendingChanges = true;

        if (this.timer) {
            clearTimeout(this.timer);
        }

        this.timer = setTimeout(() => {
            this.save();
        }, this.delay);
    }

    async save() {
        if (!this.pendingChanges) return;

        this.wireComponent.saveStatus = 'saving';

        try {
            await this.wireComponent.call('saveProject');
            this.lastSavedContent = JSON.stringify(this.wireComponent.components);
            this.pendingChanges = false;
            this.wireComponent.saveStatus = 'saved';
        } catch (error) {
            console.error('Auto-save failed:', error);
            this.wireComponent.saveStatus = 'error';

            // Retry after 5 seconds
            setTimeout(() => this.save(), 5000);
        }
    }

    hasUnsavedChanges() {
        return this.pendingChanges ||
               JSON.stringify(this.wireComponent.components) !== this.lastSavedContent;
    }

    destroy() {
        if (this.timer) {
            clearTimeout(this.timer);
        }
    }
}
```

# Version Diff Engine

## Delta Calculation

```php
// app/Services/VersionDiffCalculator.php
class VersionDiffCalculator
{
    public function calculateDiff(array $oldSnapshot, array $newSnapshot): array
    {
        $diff = [
            'added' => [],
            'removed' => [],
            'modified' => [],
            'css_changed' => false,
            'js_changed' => false,
        ];

        $oldComponents = collect($oldSnapshot['components'] ?? []);
        $newComponents = collect($newSnapshot['components'] ?? []);

        // Find added components
        $oldIds = $oldComponents->pluck('id');
        $newIds = $newComponents->pluck('id');

        $diff['added'] = $newComponents
            ->filter(fn($c) => !$oldIds->contains($c['id']))
            ->toArray();

        // Find removed components
        $diff['removed'] = $oldComponents
            ->filter(fn($c) => !$newIds->contains($c['id']))
            ->toArray();

        // Find modified components
        $commonIds = $oldIds->intersect($newIds);
        foreach ($commonIds as $id) {
            $oldComponent = $oldComponents->firstWhere('id', $id);
            $newComponent = $newComponents->firstWhere('id', $id);

            if ($oldComponent !== $newComponent) {
                $diff['modified'][] = [
                    'id' => $id,
                    'type' => $newComponent['type'],
                    'changes' => $this->getComponentChanges($oldComponent, $newComponent),
                ];
            }
        }

        // Check CSS/JS changes
        $diff['css_changed'] = ($oldSnapshot['custom_css'] ?? '') !== ($newSnapshot['custom_css'] ?? '');
        $diff['js_changed'] = ($oldSnapshot['custom_js'] ?? '') !== ($newSnapshot['custom_js'] ?? '');

        return $diff;
    }

    private function getComponentChanges(array $old, array $new): array
    {
        $changes = [];

        // Config changes
        $oldConfig = $old['config'] ?? [];
        $newConfig = $new['config'] ?? [];

        foreach ($newConfig as $key => $value) {
            if (!isset($oldConfig[$key]) || $oldConfig[$key] !== $value) {
                $changes['config'][$key] = [
                    'old' => $oldConfig[$key] ?? null,
                    'new' => $value,
                ];
            }
        }

        foreach ($oldConfig as $key => $value) {
            if (!isset($newConfig[$key])) {
                $changes['config'][$key] = [
                    'old' => $value,
                    'new' => null,
                ];
            }
        }

        // Position changes
        if (($old['position'] ?? 0) !== ($new['position'] ?? 0)) {
            $changes['position'] = [
                'old' => $old['position'] ?? 0,
                'new' => $new['position'] ?? 0,
            ];
        }

        // Custom CSS/JS changes
        if (($old['custom_css'] ?? '') !== ($new['custom_css'] ?? '')) {
            $changes['custom_css'] = true;
        }

        if (($old['custom_js'] ?? '') !== ($new['custom_js'] ?? '')) {
            $changes['custom_js'] = true;
        }

        return $changes;
    }

    public function formatDiff(array $diff): string
    {
        $output = [];

        if (!empty($diff['added'])) {
            $output[] = count($diff['added']) . ' component(s) added';
        }

        if (!empty($diff['removed'])) {
            $output[] = count($diff['removed']) . ' component(s) removed';
        }

        if (!empty($diff['modified'])) {
            $output[] = count($diff['modified']) . ' component(s) modified';
        }

        if ($diff['css_changed']) {
            $output[] = 'Custom CSS updated';
        }

        if ($diff['js_changed']) {
            $output[] = 'Custom JavaScript updated';
        }

        return implode(', ', $output) ?: 'No changes';
    }
}
```
