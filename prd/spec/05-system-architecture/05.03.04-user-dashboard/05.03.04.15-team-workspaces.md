```php
// app/Livewire/Team/TeamWorkspaces.php
class TeamWorkspaces extends Component
{
    public ?array $currentTeam = null;
    public array $teams = [];
    public string $activeTab = 'overview';
    public bool $showCreateModal = false;
    public string $newTeamName = '';

    public function mount(): void
    {
        $this->loadTeams();
        $this->selectDefaultTeam();
    }

    private function loadTeams(): void
    {
        $this->teams = auth()->user()->teams()
            ->withCount('members')
            ->get()
            ->map(fn($team) => [
                'id' => $team->id,
                'name' => $team->name,
                'member_count' => $team->members_count,
                'role' => $team->pivot->role,
                'subscription_tier' => $team->subscription_tier,
                'created_at' => $team->created_at->diffForHumans(),
            ])
            ->toArray();
    }

    private function selectDefaultTeam(): void
    {
        if (!empty($this->teams)) {
            $defaultTeamId = auth()->user()->default_team_id ?? $this->teams[0]['id'];
            $this->selectTeam($defaultTeamId);
        }
    }

    public function selectTeam(string $teamId): void
    {
        $this->currentTeam = Team::withCount(['members', 'sharedComponents'])
            ->with('owner')
            ->findOrFail($teamId)
            ->toArray();

        event(new TeamSelected(auth()->user(), $this->currentTeam['id']));
    }

    public function createTeam(): void
    {
        $this->validate([
            'newTeamName' => 'required|string|max:100',
        ]);

        $team = Team::create([
            'name' => $this->newTeamName,
            'owner_id' => auth()->id(),
        ]);

        $team->members()->attach(auth()->id(), ['role' => 'owner']);

        event(new TeamCreated($team));

        $this->showCreateModal = false;
        $this->newTeamName = '';

        $this->loadTeams();
        $this->selectTeam($team->id);

        $this->dispatch('profileSaved', message: 'Team created successfully');
    }

    public function setTab(string $tab): void
    {
        $this->activeTab = $tab;
    }

    public function render()
    {
        return view('livewire.team.workspaces')->layout('layouts.dashboard');
    }
}
```

```php
// app/Livewire/Team/MemberManagement.php
class MemberManagement extends Component
{
    public string $teamId;
    public array $members = [];
    public array $pendingInvitations = [];
    public string $inviteEmail = '';
    public string $inviteRole = 'member';
    public bool $showRemoveModal = false;
    public string $removeMemberId = '';
    public string $removeMemberName = '';

    protected $rules = [
        'inviteEmail' => 'required|email',
        'inviteRole' => 'required|in:admin,member',
    ];

    public function mount(string $teamId): void
    {
        $this->teamId = $teamId;
        $this->loadMembers();
        $this->loadPendingInvitations();
    }

    private function loadMembers(): void
    {
        $this->members = TeamMember::where('team_id', $this->teamId)
            ->with('user')
            ->get()
            ->map(fn($member) => [
                'id' => $member->user_id,
                'name' => $member->user->name,
                'email' => $member->user->email,
                'avatar' => $member->user->avatar_url,
                'role' => $member->role,
                'joined_at' => $member->joined_at->diffForHumans(),
                'is_online' => $this->checkOnlineStatus($member->user_id),
            ])
            ->toArray();
    }

    private function loadPendingInvitations(): void
    {
        $this->pendingInvitations = TeamInvitation::where('team_id', $this->teamId)
            ->whereNull('accepted_at')
            ->where('expires_at', '>', now())
            ->get()
            ->map(fn($inv) => [
                'id' => $inv->id,
                'email' => $inv->email,
                'role' => $inv->role,
                'invited_by' => $inv->inviter->name,
                'expires_at' => $inv->expires_at->diffForHumans(),
            ])
            ->toArray();
    }

    private function checkOnlineStatus(int $userId): bool
    {
        return Cache::get("user_online:{$userId}", false);
    }

    public function inviteMember(): void
    {
        $this->validate();

        $team = Team::findOrFail($this->teamId);

        // Check seat limit
        $currentMembers = $team->members()->count();
        $seatLimit = $team->seat_limit;

        if ($seatLimit > 0 && $currentMembers >= $seatLimit) {
            $this->addError('inviteEmail', 'Team seat limit reached. Please upgrade your plan.');
            return;
        }

        // Check if already a member
        $existingMember = TeamMember::where('team_id', $this->teamId)
            ->whereHas('user', fn($q) => $q->where('email', $this->inviteEmail))
            ->exists();

        if ($existingMember) {
            $this->addError('inviteEmail', 'User is already a team member.');
            return;
        }

        // Create invitation
        $invitation = TeamInvitation::create([
            'team_id' => $this->teamId,
            'email' => $this->inviteEmail,
            'role' => $this->inviteRole,
            'invited_by' => auth()->id(),
            'token' => Str::random(64),
            'expires_at' => now()->addDays(7),
        ]);

        // Send invitation email
        TeamInvitationEmail::dispatch($invitation);

        event(new MemberInvited($team, $invitation));

        $this->inviteEmail = '';
        $this->inviteRole = 'member';

        $this->loadPendingInvitations();

        $this->dispatch('profileSaved', message: 'Invitation sent successfully');
    }

    public function changeRole(string $memberId, string $newRole): void
    {
        $team = Team::findOrFail($this->teamId);

        // Only owner can change roles
        if (auth()->id() !== $team->owner_id) {
            abort(403, 'Only the team owner can change member roles.');
        }

        // Cannot change own role
        if ($memberId == auth()->id()) {
            $this->addError('role', 'You cannot change your own role.');
            return;
        }

        // Cannot remove last admin
        if ($newRole !== 'admin') {
            $adminCount = TeamMember::where('team_id', $this->teamId)
                ->where('role', 'admin')
                ->count();

            if ($adminCount <= 1) {
                $this->addError('role', 'Team must have at least one admin.');
                return;
            }
        }

        TeamMember::where('team_id', $this->teamId)
            ->where('user_id', $memberId)
            ->update(['role' => $newRole]);

        event(new RoleChanged($team, $memberId, $newRole));

        $this->loadMembers();

        $this->dispatch('profileSaved', message: 'Member role updated');
    }

    public function confirmRemove(string $memberId, string $memberName): void
    {
        $this->removeMemberId = $memberId;
        $this->removeMemberName = $memberName;
        $this->showRemoveModal = true;
    }

    public function removeMember(): void
    {
        $team = Team::findOrFail($this->teamId);

        // Only owner or admin can remove members
        $currentUserRole = $team->members()
            ->where('user_id', auth()->id())
            ->first()?->role;

        if (!in_array($currentUserRole, ['owner', 'admin'])) {
            abort(403, 'Insufficient permissions.');
        }

        // Cannot remove owner
        if ($this->removeMemberId == $team->owner_id) {
            $this->addError('remove', 'Cannot remove the team owner.');
            return;
        }

        TeamMember::where('team_id', $this->teamId)
            ->where('user_id', $this->removeMemberId)
            ->delete();

        event(new MemberRemoved($team, $this->removeMemberId));

        $this->showRemoveModal = false;
        $this->removeMemberId = '';
        $this->removeMemberName = '';

        $this->loadMembers();

        $this->dispatch('profileSaved', message: 'Member removed from team');
    }

    public function cancelInvitation(string $invitationId): void
    {
        TeamInvitation::where('id', $invitationId)
            ->where('team_id', $this->teamId)
            ->delete();

        $this->loadPendingInvitations();

        $this->dispatch('profileSaved', message: 'Invitation cancelled');
    }

    public function render()
    {
        return view('livewire.team.member-management');
    }
}
```

```php
class SharedResources extends Component

                'shared_by' => $shared->sharer->name,
                'shared_at' => $shared->shared_at->diffForHumans(),

    {
        $this->teamTemplates = TeamTemplate::where('team_id', $this->teamId)
            ->with('template')
            ->get()
            ->map(fn($teamTemplate) => [
                'id' => $teamTemplate->id,
                'template_id' => $teamTemplate->template_id,
                'name' => $teamTemplate->template->name ?? 'Unknown',
                'category' => $teamTemplate->template->category ?? 'General',
                'added_by' => $teamTemplate->adder->name,
                'added_at' => $teamTemplate->created_at->diffForHumans(),
                'install_count' => $teamTemplate->install_count,
            ])
            ->toArray();
    }

    private function loadWorkspaceThemes(): void
    {
        $this->workspaceThemes = WorkspaceTheme::where('team_id', $this->teamId)
            ->get()
            ->map(fn($theme) => [
                'id' => $theme->id,
                'name' => $theme->name,
                'colors' => $theme->colors,
                'fonts' => $theme->fonts,
                'created_by' => $theme->creator->name,
                'created_at' => $theme->created_at->diffForHumans(),
                'is_active' => $theme->is_active,
            ])
            ->toArray();
    }

    public function shareComponent(string $componentId): void
    {
        // Check if already shared
        $exists = TeamSharedComponent::where('team_id', $this->teamId)
            ->where('component_id', $componentId)
            ->exists();

        if ($exists) {
            $this->dispatch('profileSaved', message: 'Component already shared');
            return;
        }

        TeamSharedComponent::create([
            'team_id' => $this->teamId,
            'component_id' => $componentId,
            'shared_by' => auth()->id(),
            'shared_at' => now(),
        ]);

        event(new ComponentSharedToTeam($this->teamId, $componentId));

        $this->loadSharedComponents();

        $this->dispatch('profileSaved', message: 'Component shared with team');
    }

    public function removeSharedComponent(string $sharedId): void
    {
        TeamSharedComponent::where('id', $sharedId)
            ->where('team_id', $this->teamId)
            ->delete();

        $this->loadSharedComponents();

        $this->dispatch('profileSaved', message: 'Component removed from shared library');
    }

    public function setActiveTheme(string $themeId): void
    {
        // Deactivate all themes
        WorkspaceTheme::where('team_id', $this->teamId)
            ->update(['is_active' => false]);

        // Activate selected theme
        WorkspaceTheme::where('id', $themeId)
            ->update(['is_active' => true]);

        $this->loadWorkspaceThemes();

        $this->dispatch('profileSaved', message: 'Theme activated');
    }

    public function render()
    {
        return view('livewire.team.shared-resources');
    }
}
```

```php
// app/Livewire/Team/TeamActivity.php
class TeamActivity extends Component
{
    public string $teamId;
    public string $activityType = '';
    public string $dateFrom = '';
    public string $dateTo = '';
    public array $activities = [];
    public int $currentPage = 1;
    public int $perPage = 50;
    public int $totalActivities = 0;

    public function mount(string $teamId): void
    {
        $this->teamId = $teamId;
        $this->loadActivities();
    }

    private function loadActivities(): void
    {
        $query = TeamActivity::where('team_id', $this->teamId)
            ->with('actor');

        if ($this->activityType) {
            $query->where('action', $this->activityType);
        }

        if ($this->dateFrom) {
            $query->where('created_at', '>=', $this->dateFrom);
        }

        if ($this->dateTo) {
            $query->where('created_at', '<=', $this->dateTo . ' 23:59:59');
        }

        $this->totalActivities = $query->count();

        $this->activities = $query->orderByDesc('created_at')
            ->skip(($this->currentPage - 1) * $this->perPage)
            ->take($this->perPage)
            ->get()
            ->map(fn($activity) => [
                'id' => $activity->id,
                'actor' => [
                    'name' => $activity->actor->name,
                    'avatar' => $activity->actor->avatar_url,
                ],
                'action' => $activity->action,
                'description' => $activity->description,
```

```php
class OwnershipTransferModal extends Component
{
    public string $teamId;
    public string $newOwnerId = '';
    public string $password = '';
    public bool $processing = false;
    public bool $completed = false;

    public function mount(string $teamId): void
    {
        $this->teamId = $teamId;
    }

    public function transfer(): void
    {
        $this->validate([
            'newOwnerId' => 'required|exists:users,id',
            'password' => 'required',
        ]);

        $user = auth()->user();

        // Verify password
        if (!Hash::check($this->password, $user->password)) {
            $this->addError('password', 'Password is incorrect.');
            return;
        }

        $this->processing = true;

        $team = Team::findOrFail($this->teamId);

        // Verify current user is owner
        if ($team->owner_id !== $user->id) {
            abort(403, 'Only the team owner can transfer ownership.');
        }

        // Verify new owner is a team member
        $newOwnerMember = TeamMember::where('team_id', $this->teamId)
            ->where('user_id', $this->newOwnerId)
            ->first();

        if (!$newOwnerMember) {
            $this->addError('newOwnerId', 'New owner must be a team member.');
            return;
        }

        // Transfer ownership
        $team->update(['owner_id' => $this->newOwnerId]);

        // Update roles
        $newOwnerMember->update(['role' => 'owner']);

        // Demote current owner to admin
        TeamMember::where('team_id', $this->teamId)
            ->where('user_id', $user->id)
            ->update(['role' => 'admin']);

        // Send notifications
        OwnershipTransferred::dispatch($team, $user, $this->newOwnerId);

        $this->processing = false;
        $this->completed = true;

        $this->dispatch('profileSaved', message: 'Team ownership transferred successfully');
    }

    public function render()
    {
        return view('livewire.team.ownership-transfer-modal');
    }
}
```

```php
class CheckTeamRole
{
    public function handle(Request $request, Closure $next, string ...$roles)
    {
        $teamId = $request->route('teamId') ?? $request->input('teamId');

        if (!$teamId) {
            abort(400, 'Team ID required');
        }

        $teamMember = TeamMember::where('team_id', $teamId)
            ->where('user_id', auth()->id())
            ->first();

        if (!$teamMember) {
            abort(403, 'You are not a member of this team');
        }

        if (!in_array($teamMember->role, $roles)) {
            abort(403, 'Insufficient permissions for this action');
        }

        $request->attributes->set('teamMember', $teamMember);
        $request->attributes->set('teamRole', $teamMember->role);

        return $next($request);
    }
}
```

## Permission Cache

```php
// app/Services/TeamPermissionCache.php
class TeamPermissionCache
{
    private int $ttl = 300; // 5 minutes

    public function getPermissions(User $user, Team $team): array
    {
        $cacheKey = "team_permissions:{$user->id}:{$team->id}";

        return Cache::remember($cacheKey, $this->ttl, function () use ($user, $team) {
            $member = $team->members()->where('user_id', $user->id)->first();

            if (!$member) {
                return [];
            }

            return $this->buildPermissions($member->role, $team);
        });
    }

    private function buildPermissions(string $role, Team $team): array
    {
        $basePermissions = [
            'view_team' => true,
            'use_shared_components' => true,
            'share_components' => true,
            'view_activity' => true,
        ];

        $adminPermissions = array_merge($basePermissions, [
            'invite_members' => true,
            'remove_members' => true,
            'manage_settings' => true,
        ]);

        $ownerPermissions = array_merge($adminPermissions, [
            'change_roles' => true,
            'transfer_ownership' => true,
            'manage_billing' => true,
            'delete_team' => true,
            'export_data' => true,
        ]);

        return match($role) {
            'owner' => $ownerPermissions,
            'admin' => $adminPermissions,
            'member' => $basePermissions,
            default => [],
        };
    }

    public function invalidate(User $user, Team $team): void
    {
        $cacheKey = "team_permissions:{$user->id}:{$team->id}";
        Cache::forget($cacheKey);
    }

    public function invalidateAll(Team $team): void
    {
        $members = $team->members()->pluck('user_id');

        foreach ($members as $memberId) {
```

```php
        return match($action) {
            'team.created' => "{$actor} created the team",
            'member.invited' => "{$actor} invited {$metadata['email']} as {$metadata['role']}",
            'member.joined' => "{$actor} joined the team",
            'member.removed' => "{$actor} removed {$metadata['member_name']} from the team",
            'role.changed' => "{$actor} changed {$metadata['member_name']}'s role to {$metadata['new_role']}",
            'ownership.transferred' => "{$actor} transferred ownership to {$metadata['new_owner_name']}",
            'component.shared' => "{$actor} shared component {$metadata['component_name']}",
            'template.added' => "{$actor} added template {$metadata['template_name']}",
            default => "{$actor} performed {$action}",
        };
    }

    public function getTeamActivities(string $teamId, array $filters = []): array
    {
        $query = TeamActivity::where('team_id', $teamId)
            ->with('actor');

        if (isset($filters['action'])) {
            $query->where('action', $filters['action']);
        }

        if (isset($filters['from'])) {
            $query->where('created_at', '>=', $filters['from']);
        }

        if (isset($filters['to'])) {
            $query->where('created_at', '<=', $filters['to']);
        }

        return $query->orderByDesc('created_at')
            ->limit($filters['limit'] ?? 100)
            ->get()
            ->toArray();
```

```php
// app/Services/TeamSeatManager.php
class TeamSeatManager
{
    public function canAddMember(Team $team): bool
    {
        $currentSeats = $team->members()->count();
        $seatLimit = $team->seat_limit;

        // -1 means unlimited
        if ($seatLimit === -1) {
            return true;
        }

        return $currentSeats < $seatLimit;
    }

    public function getSeatUsage(Team $team): array
    {
        $currentSeats = $team->members()->count();
        $seatLimit = $team->seat_limit;

        return [
            'used' => $currentSeats,
            'limit' => $seatLimit,
            'remaining' => $seatLimit === -1 ? -1 : max(0, $seatLimit - $currentSeats),
            'percentage' => $seatLimit === -1 ? 0 : min(100, round(($currentSeats / $seatLimit) * 100)),
        ];
    }

    public function updateSeatLimit(Team $team, int $newLimit): void
    {
        $currentMembers = $team->members()->count();

        if ($newLimit < $currentMembers && $newLimit !== -1) {
            throw new \InvalidArgumentException(
                "Cannot set seat limit below current member count ({$currentMembers})"
            );
        }

        $team->update(['seat_limit' => $newLimit]);

        // Update Stripe subscription if needed
        $this->syncStripeSeats($team, $newLimit);
    }

    private function syncStripeSeats(Team $team, int $seats): void
    {
        $subscription = $team->subscription;

        if (!$subscription) {
            return;
        }

        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        $stripe->subscriptions->update($subscription->stripe_id, [
            'items' => [
                [
                    'id' => $subscription->asStripeSubscription()->items->data[0]->id,
                    'quantity' => $seats === -1 ? 1 : $seats,
                ],
            ],
        ]);
    }
}
```

```php
class TeamInvoiceService
{
    public function getTeamInvoices(Team $team): array
    {
        return Invoice::where('team_id', $team->id)
            ->orderByDesc('created_at')
            ->get()
            ->map(fn($invoice) => [

                'number' => $invoice->number,
                'amount' => $invoice->amount,
                'currency' => $invoice->currency,
                'status' => $invoice->status,
                'pdf_url' => $invoice->pdf_path
                    ? Storage::disk('s3')->temporaryUrl($invoice->pdf_path, now()->addHour())
                    : null,
                'created_at' => $invoice->created_at->format('M j, Y'),
            ])
            ->toArray();
    }

    public function generateTeamInvoice(Team $team, array $items): Invoice
    {
        $total = collect($items)->sum(fn($item) => $item['amount'] * $item['quantity']);

        $invoice = Invoice::create([
            'team_id' => $team->id,
            'user_id' => $team->owner_id,
            'number' => $this->generateInvoiceNumber(),
            'amount' => $total,
            'currency' => 'USD',
            'status' => 'open',
            'items' => $items,
        ]);

        // Generate PDF
        $pdfPath = app(InvoicePdfGenerator::class)->generate($invoice);
        $invoice->update(['pdf_path' => $pdfPath]);

        return $invoice;
    }

    private function generateInvoiceNumber(): string
    {
        $lastInvoice = Invoice::orderByDesc('id')->first();
        $number = $lastInvoice ? intval(substr($lastInvoice->number, 4)) + 1 : 1;

        return 'TEAM' . str_pad($number, 6, '0', STR_PAD_LEFT);
    }
}
```
