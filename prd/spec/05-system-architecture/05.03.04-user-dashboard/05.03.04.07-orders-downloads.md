```php
// app/Livewire/Orders/OrdersList.php
class OrdersList extends Component
{
    public string $search = '';
    public string $statusFilter = '';
    public string $dateFrom = '';
    public string $dateTo = '';
    public string $sortField = 'created_at';
    public string $sortDirection = 'desc';
    public int $perPage = 25;
    public int $currentPage = 1;
    public array $orders = [];
    public int $totalOrders = 0;
    public bool $loading = false;

    protected $listeners = [
        'orderStatusChanged' => 'refreshOrders',
    ];

    public function mount(): void
    {
        $this->loadOrders();
    }

    public function updatedSearch(): void
    {
        $this->currentPage = 1;
        $this->loadOrders();
    }

    public function updatedStatusFilter(): void
    {
        $this->currentPage = 1;
        $this->loadOrders();
    }

    public function updatedDateFrom(): void
    {
        $this->currentPage = 1;
        $this->loadOrders();
    }

    public function updatedDateTo(): void
    {
        $this->currentPage = 1;
        $this->loadOrders();
    }

    public function sortBy(string $field): void
    {
        if ($this->sortField === $field) {
            $this->sortDirection = $this->sortDirection === 'asc' ? 'desc' : 'asc';
        } else {
            $this->sortField = $field;
            $this->sortDirection = 'desc';
        }

        $this->loadOrders();
    }

    public function loadOrders(): void
    {
        $this->loading = true;

        $query = Order::where('user_id', auth()->id())
            ->with(['items', 'invoice']);

        // Search
        if ($this->search) {
            $query->where(function ($q) {
                $q->where('number', 'like', "%{$this->search}%")
                  ->orWhereHas('items', function ($q) {
                      $q->where('name', 'like', "%{$this->search}%");
                  });
            });
        }

        // Status filter
        if ($this->statusFilter) {
            $query->where('status', $this->statusFilter);
        }

        // Date range
        if ($this->dateFrom) {
            $query->where('created_at', '>=', $this->dateFrom);
        }
        if ($this->dateTo) {
            $query->where('created_at', '<=', $this->dateTo . ' 23:59:59');
        }

        // Sort
        $query->orderBy($this->sortField, $this->sortDirection);

        // Paginate
        $this->totalOrders = $query->count();
        $this->orders = $query->skip(($this->currentPage - 1) * $this->perPage)
            ->take($this->perPage)
            ->get()
            ->toArray();

        $this->loading = false;
    }

    public function goToPage(int $page): void
    {
        $this->currentPage = $page;
        $this->loadOrders();
    }

    public function refreshOrders(): void
    {
        $this->loadOrders();
    }

    public function render()
    {
        return view('livewire.orders.orders-list');
    }
}
```

## OrderDetail Component

```php
// app/Livewire/Orders/OrderDetail.php
class OrderDetail extends Component
{
    public ?array $order = null;
    public bool $loading = false;
    public bool $showInvoice = false;

    public function mount(string $orderId): void
    {
        $this->loadOrder($orderId);
    }

    private function loadOrder(string $orderId): void
    {
        $this->loading = true;

        $this->order = Order::where('user_id', auth()->id())
            ->where('id', $orderId)
            ->with(['items.product', 'invoice', 'downloads.file'])
            ->first()
            ?->toArray();

        if (!$this->order) {
            abort(404, 'Order not found');
        }

        $this->loading = false;
    }

    public function downloadFile(string $fileId): string
    {
        $download = $this->order['downloads'] ?? [];
        $file = collect($download)->firstWhere('file.id', $fileId);

        if (!$file) {
            abort(404, 'File not found');
        }

        // Check rate limiting
        if (RateLimiter::tooManyAttempts('download:' . auth()->id(), 10)) {
            abort(429, 'Too many download attempts. Please try again later.');
        }

        RateLimiter::hit('download:' . auth()->id(), 60);

        // Generate signed URL
        $signedUrl = Storage::disk('s3')->temporaryUrl(
            $file['file']['path'],
            now()->addHour(),
            ['ResponseContentDisposition' => 'attachment; filename="' . $file['file']['name'] . '"']
        );

        // Log download
        DownloadLog::create([
            'user_id' => auth()->id(),
            'order_id' => $this->order['id'],
            'file_id' => $fileId,
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
        ]);

        // Increment download count
        OrderDownload::where('order_id', $this->order['id'])
            ->where('file_id', $fileId)
            ->increment('download_count');

        event(new DownloadInitiated(auth()->id(), $this->order['id'], $fileId));

        return $signedUrl;
    }

    public function downloadInvoice(): string
    {
        if (!$this->order['invoice']) {
            abort(404, 'Invoice not found');
        }

        return Storage::disk('s3')->temporaryUrl(
            $this->order['invoice']['pdf_path'],
            now()->addHour()
        );
    }

    public function render()
    {
        return view('livewire.orders.order-detail');
    }
}
```

## DownloadsList Component

```php
// app/Livewire/Orders/DownloadsList.php
class DownloadsList extends Component
{
    public string $search = '';
    public string $sortBy = 'last_downloaded_at';
    public string $sortDirection = 'desc';
    public int $currentPage = 1;
    public int $perPage = 25;
    public array $downloads = [];
    public int $totalDownloads = 0;
    public array $selectedDownloads = [];
    public bool $loading = false;

    public function mount(): void
    {
        $this->loadDownloads();
    }

    public function loadDownloads(): void
    {
        $this->loading = true;

        $query = OrderDownload::where('user_id', auth()->id())
            ->with(['file', 'order']);

        if ($this->search) {
            $query->whereHas('file', function ($q) {
                $q->where('name', 'like', "%{$this->search}%");
            });
        }

        $this->totalDownloads = $query->count();
        $this->downloads = $query
            ->orderBy($this->sortBy, $this->sortDirection)
            ->skip(($this->currentPage - 1) * $this->perPage)
            ->take($this->perPage)
            ->get()
            ->toArray();

        $this->loading = false;
    }

    public function toggleSelect(string $downloadId): void
    {
        if (in_array($downloadId, $this->selectedDownloads)) {
            $this->selectedDownloads = array_filter(
                $this->selectedDownloads,
                fn($id) => $id !== $downloadId
            );
        } else {
            $this->selectedDownloads[] = $downloadId;
        }
    }

    public function selectAll(): void
    {
        $this->selectedDownloads = array_column($this->downloads, 'id');
    }

    public function deselectAll(): void
    {
        $this->selectedDownloads = [];
    }

    public function downloadSelected(): void
    {
        if (empty($this->selectedDownloads)) {
            return;
        }

        // Generate ZIP archive of selected files
        $zip = new \ZipArchive();
        $zipPath = tempnam(sys_get_temp_dir(), 'gsmui_downloads_');
        $zip->open($zipPath);

        foreach ($this->selectedDownloads as $downloadId) {
            $download = collect($this->downloads)->firstWhere('id', $downloadId);
            if ($download) {
                $fileContent = Storage::disk('s3')->get($download['file']['path']);
                $zip->addFromString($download['file']['name'], $fileContent);
            }
        }

        $zip->close();

        // Store zip and return signed URL
        $zipName = 'downloads_' . Str::uuid() . '.zip';
        Storage::disk('s3')->put('temp/' . $zipName, file_get_contents($zipPath));

        return Storage::disk('s3')->temporaryUrl(
            'temp/' . $zipName,
            now()->addHour()
        );
    }

    public function render()
    {
        return view('livewire.orders.downloads-list');
    }
}
```

```php
class LicensesList extends Component

        $this->loadLicenses();

    {
        $query = License::where('user_id', auth()->id())
            ->with(['orderItem.order']);

        if ($this->search) {
            $query->where('key', 'like', "%{$this->search}%")
```

```
User navigates to Orders page
  ↓
Order list loads with filters
  ↓
User searches for specific order
  ↓
Results filtered in real-time
  ↓
User clicks "View details"
  ↓
Order detail page loads
  ↓
User views items, payment, downloads
  ↓
User downloads invoice PDF
```

## Flow 2: Download Files

```
User navigates to Downloads page
  ↓
Download list loads with all files
  ↓
User selects files via checkboxes
  ↓
User clicks "Download Selected"
  ↓
ZIP archive created server-side
  ↓
Signed download URL generated
  ↓
Browser downloads ZIP file
  ↓
Download count incremented
  ↓
Activity logged
```

## Flow 3: License Management

```
User navigates to Licenses page
  ↓
License list loads with status
  ↓
User clicks on license to expand
  ↓
Full license key displayed
  ↓
User copies license key
  ↓
Clipboard notification shown
  ↓
User uses license in their project
```

## Flow 4: Export Order Data

```
User clicks "Export CSV"
  ↓
Export job queued
  ↓
Processing indicator shown
  ↓
Export file generated
  ↓
Download link sent via notification
  ↓
User downloads CSV file
  ↓
CSV contains all order data
```

# Edge Cases

| Scenario | Handling |
|----------|----------|
| Download link expires during transfer | Resumable download with Range headers |
| User tries to download refunded order | Block download, show "Refunded" status |
| Large file (>100MB) download | Chunked transfer with progress indicator |
| License key already activated | Show activation details, no re-activation |
| Order with mixed items (paid + free) | Show individual item statuses |
| Invoice PDF generation fails | Fallback to HTML invoice |
| Concurrent download attempts | Rate limiting with queue |
| User downloads same file 100 times | Track count, no limit (DRM-free) |
| Currency conversion display | Show original + converted amount |
| Partial refund on order | Show refund status per item |

```php
// app/Services/InvoicePdfGenerator.php
class InvoicePdfGenerator
{
    public function generate(Order $order): string
    {
        $invoice = $order->invoice;
        $user = $order->user;
        $items = $order->items;

        $pdf = Pdf::loadView('invoices.pdf', [
            'invoice' => $invoice,
            'user' => $user,
            'order' => $order,
            'items' => $items,
            'company' => [
                'name' => 'GsmUi',
                'address' => '123 Developer Street',
                'city' => 'San Francisco, CA 94105',
                'country' => 'United States',
                'email' => 'billing@gsmui.com',
                'website' => 'https://gsmui.com',
            ],
        ]);

        $pdf->setPaper('a4');
```

```php
// app/Services/DownloadUrlGenerator.php
class DownloadUrlGenerator
{
    private int $expiryMinutes = 60;
    private int $maxDownloads = 100;

    public function generateUrl(OrderDownload $download): string
    {
        $user = $download->order->user;
        $file = $download->file;

        // Rate limiting check
        if ($this->isRateLimited($user->id)) {
            throw new DownloadRateLimitException('Download rate limit exceeded');
        }

        // Generate signed URL
        $signedUrl = Storage::disk('s3')->temporaryUrl(
            $file->path,
            now()->addMinutes($this->expiryMinutes),
            [
                'ResponseContentDisposition' => 'attachment; filename="' . $file->name . '"',
                'ResponseContentType' => $file->mime_type ?? 'application/octet-stream',
            ]
        );

        // Log download attempt
        DownloadLog::create([
            'user_id' => $user->id,
            'order_id' => $download->order_id,
            'file_id' => $file->id,
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
            'signed_url_expires_at' => now()->addMinutes($this->expiryMinutes),
        ]);

        // Increment download count
        $download->increment('download_count');
        $download->update(['last_downloaded_at' => now()]);

        return $signedUrl;
    }

    private function isRateLimited(int $userId): bool
    {
        $key = "download_rate:{$userId}";
        $count = Cache::get($key, 0);

        if ($count >= $this->maxDownloads) {
            return true;
        }

        Cache::put($key, $count + 1, now()->addHour());
        return false;
    }

    public function generateBulkUrls(array $downloadIds): array
    {
        return collect($downloadIds)
            ->map(fn($id) => OrderDownload::find($id))
            ->filter()
            ->mapWithKeys(fn($download) => [
                $download->id => $this->generateUrl($download),
            ])
            ->toArray();
    }
}
```

```php
// app/Services/ZipArchiveGenerator.php
class ZipArchiveGenerator
{
    public function generate(array $downloadIds): string
    {
        $zipPath = tempnam(sys_get_temp_dir(), 'gsmui_');
        $zip = new \ZipArchive();

        if ($zip->open($zipPath, \ZipArchive::CREATE | \ZipArchive::OVERWRITE) !== true) {
            throw new \RuntimeException('Failed to create ZIP archive');
        }

        foreach ($downloadIds as $downloadId) {
            $download = OrderDownload::with('file')->find($downloadId);

            if (!$download || !$download->file) {
                continue;
            }

            $fileContent = Storage::disk('s3')->get($download->file->path);
            $zip->addFromString($download->file->name, $fileContent);
        }

        $zip->close();

        // Store to S3
        $archiveName = 'downloads/zip/' . Str::uuid() . '.zip';
        Storage::disk('s3')->put($archiveName, file_get_contents($zipPath));

        // Clean up temp file
        @unlink($zipPath);

        return $archiveName;
    }

    public function generateTemporaryUrl(string $archivePath): string
    {
        return Storage::disk('s3')->temporaryUrl(
            $archivePath,
            now()->addHour()
        );
    }
}
```

## Retry Logic

```php
// app/Services/DownloadRetryHandler.php
class DownloadRetryHandler
{
    private int $maxRetries = 3;
    private int $retryDelaySeconds = 5;

    public function downloadWithRetry(callable $downloadFunction): string
    {
        $attempts = 0;

        while ($attempts < $this->maxRetries) {
            try {
                return $downloadFunction();
            } catch (DownloadFailedException $e) {
                $attempts++;

                if ($attempts >= $this->maxRetries) {
                    throw $e;
                }

                // Exponential backoff
                sleep($this->retryDelaySeconds * pow(2, $attempts - 1));

                Log::warning('Download retry', [
                    'attempt' => $attempts,
                    'error' => $e->getMessage(),
                ]);
            }
        }

        throw new \RuntimeException('Download failed after retries');
    }
}
```

# Order Status Transitions

## Status State Machine

```
pending → paid (payment confirmed)
pending → cancelled (user cancels or timeout)
paid → refunded (full refund)
paid → partially_refunded (partial refund)
paid → download_access_revoked (refund after download window)
refunded → (terminal state)
cancelled → (terminal state)
```

## Status Transition Handler

```php
// app/Services/OrderStatusHandler.php
class OrderStatusHandler
{
    private array $validTransitions = [
        'pending' => ['paid', 'cancelled'],
        'paid' => ['refunded', 'partially_refunded'],
        'partially_refunded' => ['refunded'],
        'refunded' => [],
        'cancelled' => [],
    ];

    public function transition(Order $order, string $newStatus, ?string $reason = null): bool
    {
        $currentStatus = $order->status;

        if (!isset($this->validTransitions[$currentStatus]) ||
            !in_array($newStatus, $this->validTransitions[$currentStatus])) {
            throw new InvalidStatusTransitionException(
                "Cannot transition from {$currentStatus} to {$newStatus}"
            );
        }

        $order->update([
            'status' => $newStatus,
            'status_changed_at' => now(),
            'status_reason' => $reason,
        ]);

        // Handle side effects
        $this->handleTransitionEffects($order, $currentStatus, $newStatus);

        // Log status change
        OrderStatusLog::create([
            'order_id' => $order->id,
            'from_status' => $currentStatus,
            'to_status' => $newStatus,
            'reason' => $reason,
            'changed_by' => auth()->id() ?? 'system',
        ]);

        event(new OrderStatusChanged($order, $currentStatus, $newStatus));

        return true;
    }

    private function handleTransitionEffects(Order $order, string $from, string $to): void
    {
        match($to) {
            'paid' => $this->handlePaymentConfirmed($order),
            'refunded' => $this->handleFullRefund($order),
            'partially_refunded' => $this->handlePartialRefund($order),
            'cancelled' => $this->handleCancellation($order),
            default => null,
        };
    }

    private function handlePaymentConfirmed(Order $order): void
    {
        // Generate invoice
        $this->generateInvoice($order);

        // Send confirmation email
        OrderConfirmationEmail::dispatch($order->user, $order);

        // Activate licenses
        foreach ($order->items as $item) {
            if ($item->type === 'template' || $item->type === 'component') {
                License::create([
                    'order_item_id' => $item->id,
                    'user_id' => $order->user_id,
                    'key' => Str::random(32),
                    'status' => 'active',
                ]);
            }
        }
    }

    private function handleFullRefund(Order $order): void
    {
        // Revoke all licenses
        $order->licenses()->update(['status' => 'revoked']);

        // Send refund confirmation
        RefundConfirmationEmail::dispatch($order->user, $order);

        // Log refund
        RefundLog::create([
            'order_id' => $order->id,
            'amount' => $order->total,
            'reason' => $order->status_reason,
            'processed_at' => now(),
        ]);
    }

    private function handleCancellation(Order $order): void
    {
        // Send cancellation confirmation
        CancellationEmail::dispatch($order->user, $order);
    }
}
```
