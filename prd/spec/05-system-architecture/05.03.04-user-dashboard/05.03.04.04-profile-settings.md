# Conclusion

Profile Settings is the user's control center for identity, preferences, and connected accounts in GsmUi. It must handle sensitive data (passwords, email) with strict security while remaining simple and intuitive. The profile page is also the gateway for OAuth integrations and notification preferences, making it critical for both account security and user engagement. Every change here must be validated, confirmed, and reflected across the platform instantly.

```php
// app/Livewire/Profile/ProfileInformationSection.php
class ProfileInformationSection extends Component
{
    public string $name = '';
    public string $email = '';
    public string $timezone = '';
    public string $locale = '';
    public ?string $avatarUrl = null;
    public bool $hasUnsavedChanges = false;
    public bool $saving = false;

    protected $listeners = [
        'avatarUploaded' => 'updateAvatarUrl',
    ];

    public function mount(): void
    {
        $user = auth()->user();
        $this->name = $user->name;
        $this->email = $user->email;
        $this->timezone = $user->timezone ?? 'UTC';
        $this->locale = $user->locale ?? 'en';
        $this->avatarUrl = $user->avatar_url;
    }

    public function updated($field): void
    {
        $this->hasUnsavedChanges = true;
    }

    public function save(): void
    {
        $this->validate([
            'name' => 'required|string|max:255',
            'email' => 'required|email|max:255',
            'timezone' => 'required|timezone',
            'locale' => 'required|string|in:en,es,fr,de,ar,zh,ja',
        ]);

        $this->saving = true;

        $user = auth()->user();

        // Check if email changed
        $emailChanged = $user->email !== $this->email;

        $user->update([
            'name' => $this->name,
            'email' => $this->email,
            'timezone' => $this->timezone,
            'locale' => $this->locale,
        ]);

        if ($emailChanged) {
            // Send verification to both addresses
            $user->sendEmailVerificationNotification();
            event(new EmailChangeRequested($user, $user->email));
        }

        event(new ProfileUpdated($user));

        $this->hasUnsavedChanges = false;
        $this->saving = false;

        $this->dispatch('profileSaved', message: 'Profile updated successfully');
    }

    public function updateAvatarUrl(string $url): void
    {
        $this->avatarUrl = $url;
    }

    public function render()
    {
        return view('livewire.profile.profile-information-section');
    }
}
```

```php
// app/Livewire/Profile/AvatarUploader.php
class AvatarUploader extends Component
{
    public ?string $avatarUrl = null;
    public bool $uploading = false;
    public bool $showCropper = false;
    public $temporaryImage = null;

    protected $listeners = [
        'cropComplete' => 'processCroppedImage',
    ];

    public function mount(?string $avatarUrl): void
    {
        $this->avatarUrl = $avatarUrl;
    }

    public function updatedTemporaryImage(): void
    {
        $this->validate([
            'temporaryImage' => 'required|file|image|max:2048|mimes:jpeg,png,gif,webp',
        ]);

        $this->showCropper = true;
    }

    public function processCroppedImage(string $croppedData): void
    {
        $this->uploading = true;

        $user = auth()->user();

        // Decode base64 cropped image
        $imageData = base64_decode(preg_replace('#^data:image/\w+;base64,#i', '', $croppedData));

        // Generate unique filename
        $filename = 'avatars/' . $user->id . '/' . Str::uuid() . '.webp';

        // Store to S3
        Storage::disk('s3')->put($filename, $imageData, 'public');

        // Resize and generate thumbnails
        $image = Image::make($imageData);
        $image->fit(256, 256, function ($constraint) {
            $constraint->upsize();
            $constraint->aspectRatio();
        });

        $thumbnailPath = 'avatars/' . $user->id . '/thumb.webp';
        Storage::disk('s3')->put($thumbnailPath, (string) $image->encode('webp', 90), 'public');

        // Update user avatar
        $user->update([
            'avatar_path' => $filename,
            'avatar_thumb_path' => $thumbnailPath,
        ]);

        // Delete old avatar if exists
        if ($user->avatar_path && $user->avatar_path !== $filename) {
            Storage::disk('s3')->delete($user->avatar_path);
            Storage::disk('s3')->delete($user->avatar_thumb_path);
        }

        $this->avatarUrl = Storage::disk('s3')->url($filename);
        $this->uploading = false;
        $this->showCropper = false;

        event(new AvatarUploaded($user));

        $this->dispatch('avatarUploaded', url: $this->avatarUrl);
        $this->dispatch('profileSaved', message: 'Avatar updated successfully');
    }

    public function removeAvatar(): void
    {
        $user = auth()->user();

        if ($user->avatar_path) {
            Storage::disk('s3')->delete($user->avatar_path);
            Storage::disk('s3')->delete($user->avatar_thumb_path);

            $user->update([
                'avatar_path' => null,
                'avatar_thumb_path' => null,
            ]);

            $this->avatarUrl = null;

            $this->dispatch('avatarUploaded', url: null);
            $this->dispatch('profileSaved', message: 'Avatar removed');
        }
    }

    public function render()
    {
        return view('livewire.profile.avatar-uploader');
    }
}
```

```php
// app/Livewire/Profile/PasswordSection.php
class PasswordSection extends Component
{
    public string $currentPassword = '';
    public string $newPassword = '';
    public string $confirmPassword = '';
    public int $passwordStrength = 0;
    public string $passwordStrengthLabel = '';
    public bool $showCurrentPassword = false;
    public bool $showNewPassword = false;
    public bool $changing = false;

    protected $rules = [
        'currentPassword' => 'required',
        'newPassword' => [
            'required',
            'string',
            'min:8',
            'max:128',
            'regex:/[A-Z]/',      // At least one uppercase
            'regex:/[a-z]/',      // At least one lowercase
            'regex:/[0-9]/',      // At least one number
            'regex:/[@$!%*?&]/',  // At least one special character
        ],
        'confirmPassword' => 'required|same:newPassword',
    ];

    protected $messages = [
        'newPassword.regex' => 'Password must include uppercase, lowercase, number, and special character.',
        'newPassword.min' => 'Password must be at least 8 characters.',
        'confirmPassword.same' => 'Passwords do not match.',
    ];

    public function updatedNewPassword(): void
    {
        $this->calculatePasswordStrength();
    }

    private function calculatePasswordStrength(): void
    {
        $password = $this->newPassword;
        $score = 0;

        // Length
        if (strlen($password) >= 8) $score++;
        if (strlen($password) >= 12) $score++;
        if (strlen($password) >= 16) $score++;

        // Character types
        if (preg_match('/[a-z]/', $password)) $score++;
        if (preg_match('/[A-Z]/', $password)) $score++;
        if (preg_match('/[0-9]/', $password)) $score++;
        if (preg_match('/[^a-zA-Z0-9]/', $password)) $score++;

        // Common patterns (negative)
        if (preg_match('/(123|abc|password|qwerty)/i', $password)) $score -= 2;

        $this->passwordStrength = max(0, min(4, $score));

        $this->passwordStrengthLabel = match($this->passwordStrength) {
            0, 1 => 'Very weak',
            2 => 'Weak',
            3 => 'Good',
            4 => 'Strong',
            default => 'Very weak',
        };
    }

    public function changePassword(): void
    {
        $this->validate();

        $user = auth()->user();

        // Verify current password
        if (!Hash::check($this->currentPassword, $user->password)) {
            $this->addError('currentPassword', 'Current password is incorrect.');
            return;
        }

        $this->changing = true;

        $user->update([
            'password' => Hash::make($this->newPassword),
        ]);

        // Invalidate all other sessions
        DB::table('sessions')
            ->where('user_id', $user->id)
            ->where('id', '!=', request()->session()->getId())
            ->delete();

        event(new PasswordChanged($user));

        // Clear form
        $this->currentPassword = '';
        $this->newPassword = '';
        $this->confirmPassword = '';
        $this->passwordStrength = 0;
        $this->changing = false;

        $this->dispatch('profileSaved', message: 'Password changed successfully. Other sessions have been logged out.');
    }

    public function render()
    {
        return view('livewire.profile.password-section');
    }
}
```

```php
// app/Livewire/Profile/ConnectedAccountsSection.php
class ConnectedAccountsSection extends Component
{
    public array $connectedAccounts = [];
    public array $availableProviders = ['github', 'google'];
    public bool $showDisconnectModal = false;
    public string $disconnectProvider = '';

    public function mount(): void
    {
        $this->loadConnectedAccounts();
    }

    private function loadConnectedAccounts(): void
    {
        $this->connectedAccounts = auth()->user()
            ->oauthConnections()
            ->get()
            ->map(fn($connection) => [
                'provider' => $connection->provider,
                'provider_id' => $connection->provider_user_id,
                'email' => $connection->provider_email,
                'connected_at' => $connection->created_at->diffForHumans(),
                'avatar' => $connection->provider_avatar,
            ])
            ->toArray();
    }

    public function connect(string $provider): RedirectResponse
    {
        if (!in_array($provider, $this->availableProviders)) {
            return redirect()->back()->withErrors(['provider' => 'Unsupported provider']);
        }

        return Socialite::driver($provider)
            ->scopes($this->getProviderScopes($provider))
            ->redirect();
    }

    public function confirmDisconnect(string $provider): void
    {
        $this->disconnectProvider = $provider;
        $this->showDisconnectModal = true;
    }

    public function disconnect(): void
    {
        $user = auth()->user();

        $connection = $user->oauthConnections()
            ->where('provider', $this->disconnectProvider)
            ->first();

        if ($connection) {
            // Revoke tokens if possible
            try {
                Socialite::driver($this->disconnectProvider)
                    ->stateless()
                    ->tokensFromResponse(
                        // Attempt to revoke
                    );
            } catch (\Exception $e) {
                // Token revocation failed, continue with deletion
            }

            $connection->delete();

            event(new OAuthDisconnected($user, $this->disconnectProvider));
        }

        $this->showDisconnectModal = false;
        $this->loadConnectedAccounts();

        $this->dispatch('profileSaved', message: ucfirst($this->disconnectProvider) . ' account disconnected');
    }

    private function getProviderScopes(string $provider): array
    {
        return match($provider) {
            'github' => ['user:email', 'read:user'],
            'google' => ['email', 'profile'],
            default => [],
        };
    }

    public function render()
    {
        return view('livewire.profile.connected-accounts-section');
    }
}
```

```php
// app/Livewire/Profile/NotificationPreferencesSection.php
class NotificationPreferencesSection extends Component
{
    public bool $emailNotifications = true;
    public bool $inAppNotifications = true;
    public string $digestFrequency = 'daily';

    // Granular notification preferences
    public array $preferences = [
        'order_confirmation' => ['email' => true, 'in_app' => true],
        'download_complete' => ['email' => false, 'in_app' => true],
        'subscription_expiry' => ['email' => true, 'in_app' => true],
        'team_invite' => ['email' => true, 'in_app' => true],
        'team_activity' => ['email' => false, 'in_app' => true],
        'security_alert' => ['email' => true, 'in_app' => true],
        'marketing' => ['email' => false, 'in_app' => false],
        'product_updates' => ['email' => true, 'in_app' => true],
        'newsletter' => ['email' => false, 'in_app' => false],
    ];

    public function mount(): void
    {
        $user = auth()->user();
        $savedPrefs = $user->preferences()->pluck('value', 'key')->toArray();

        foreach ($this->preferences as $key => $default) {
            if (isset($savedPrefs[$key])) {
                $this->preferences[$key] = json_decode($savedPrefs[$key], true);
            }
        }

        $this->digestFrequency = $savedPrefs['digest_frequency'] ?? 'daily';
    }

    public function togglePreference(string $type, string $channel): void
    {
        if (isset($this->preferences[$type][$channel])) {
            $this->preferences[$type][$channel] = !$this->preferences[$type][$channel];
        }
    }

    public function save(): void
    {
        $user = auth()->user();

        foreach ($this->preferences as $key => $value) {
            $user->preferences()->updateOrCreate(
                ['key' => $key],
                ['value' => json_encode($value)]
            );
        }

        $user->preferences()->updateOrCreate(
            ['key' => 'digest_frequency'],
            ['value' => $this->digestFrequency]
        );

        event(new PreferencesUpdated($user));

        $this->dispatch('profileSaved', message: 'Notification preferences saved');
    }

    public function render()
    {
        return view('livewire.profile.notification-preferences-section');
    }
}
```

```php
        // Queue export job
        ExportUserData::dispatch($user);

        $this->showExportModal = false;
        $this->exporting = false;

        $this->dispatch('profileSaved', message: 'Data export initiated. You will receive an email when ready.');
    }

    public function requestDeletion(): void
    {
        $this->showDeletionModal = true;
    }

    public function confirmDeletion(): void
    {
        $user = auth()->user();

        // Verify password
        if (!Hash::check($this->deletionConfirmation, $user->password)) {
            $this->addError('deletionConfirmation', 'Password is incorrect.');
            return;
        }

        $this->deleting = true;

        // Soft delete
        $user->update([
            'deleted_at' => now(),
            'email' => $user->id . '-deleted@deleted.invalid',
        ]);

        // Schedule hard delete after 30 days
        HardDeleteUser::dispatch($user)->delay(now()->addDays(30));

        // Invalidate all sessions
        DB::table('sessions')
            ->where('user_id', $user->id)
            ->delete();

        event(new AccountDeleted($user));

        auth()->logout();

        $this->dispatch('profileSaved', message: 'Account scheduled for deletion. You have 30 days to reactivate.');

        return redirect()->route('home');
    }

    public function render()
    {
        return view('livewire.profile.danger-zone-section');
    }
}
```

```
User navigates to Profile Settings
  ↓
Profile Information tab active
  ↓
User edits name, email, timezone, or locale
  ↓
"Unsaved changes" indicator appears
  ↓
User clicks "Save"

Form validation (client + server)

  ↓

  ↓

  ↓

  ↓
Toast: "Profile updated successfully"
```

## Flow 2: Avatar Upload

```
User clicks "Upload photo"
  ↓
File picker opens (accepts images only)
  ↓
User selects image
  ↓
Client-side validation:
  - File size < 2MB
  - File type: jpeg, png, gif, webp
  ↓
If valid → Cropper modal opens
  ↓
User adjusts crop area (1:1 aspect ratio)
  ↓
User clicks "Save"
  ↓
Cropped image sent to server
  ↓
Server processes:
  - Generate 256x256 thumbnail
  - Convert to WebP format
  - Store to S3
  - Delete old avatar
  ↓
Avatar URL updated across UI
  ↓
Toast: "Avatar updated successfully"
```

## Flow 3: Password Change

```
User navigates to Security tab
  ↓
Password change form displayed
  ↓
User enters current password
  ↓
User enters new password
  ↓
Password strength indicator updates in real-time
  ↓
User confirms new password
  ↓

  - Current password correct
  - New password meets requirements
  - Passwords match
  ↓

  ↓

```

| Scenario | Handling |
|----------|----------|

## Completeness Calculation

```php
// app/Services/ProfileCompleteness.php
class ProfileCompleteness
{
    private array $fields = [
        'name' => ['weight' => 15, 'label' => 'Full name'],
        'email' => ['weight' => 15, 'label' => 'Email address'],
        'avatar' => ['weight' => 20, 'label' => 'Profile photo'],
        'timezone' => ['weight' => 10, 'label' => 'Timezone'],
        'locale' => ['weight' => 10, 'label' => 'Language'],
        'password' => ['weight' => 15, 'label' => 'Password set'],
        'oauth' => ['weight' => 15, 'label' => 'Connected account'],
    ];

    public function calculate(User $user): array
    {
        $completed = 0;
        $total = array_sum(array_column($this->fields, 'weight'));
        $missing = [];

        foreach ($this->fields as $key => $field) {
            if ($this->isFieldComplete($user, $key)) {
                $completed += $field['weight'];
            } else {
                $missing[] = [
                    'key' => $key,
                    'label' => $field['label'],
                    'weight' => $field['weight'],
                ];
            }
        }

        $percent = round(($completed / $total) * 100);

        return [
            'percent' => $percent,
            'completed' => $completed,
            'total' => $total,
            'missing' => $missing,
            'is_complete' => $percent === 100,
        ];
    }

    private function isFieldComplete(User $user, string $field): bool
    {
        return match($field) {
            'name' => filled($user->name),
            'email' => filled($user->email) && $user->hasVerifiedEmail(),
            'avatar' => filled($user->avatar_path),
            'timezone' => filled($user->timezone),
            'locale' => filled($user->locale),
            'password' => filled($user->password),
            'oauth' => $user->oauthConnections()->exists(),
            default => false,
        };
    }
}
```

```php
    public ?string $qrCode = null;

    }

    {
```

## Profile Analytics Dashboard

```php
// app/Services/ProfileAnalytics.php
class ProfileAnalytics
{
    public function getProfileHealthScore(User $user): float
    {
        $completeness = app(ProfileCompleteness::class)->calculate($user);
        $securityScore = $this->calculateSecurityScore($user);
        $engagementScore = $this->calculateEngagementScore($user);

        // Weighted average
        return ($completeness['percent'] * 0.4) +
               ($securityScore * 0.35) +
               ($engagementScore * 0.25);
    }

    private function calculateSecurityScore(User $user): float
    {
        $score = 0;

        // Password strength
        if ($user->password) $score += 30;

        // 2FA enabled
        if ($user->hasTwoFactorEnabled()) $score += 40;

        // OAuth connected
        if ($user->oauthConnections()->count() > 0) $score += 15;

        // Recent password change
        if ($user->password_changed_at?->diffInDays(now()) < 90) $score += 15;

        return min(100, $score);
    }

    private function calculateEngagementScore(User $user): float
    {
        $score = 0;

        // Avatar uploaded
        if ($user->avatar_path) $score += 25;

        // Custom preferences
        if ($user->preferences()->count() > 3) $score += 25;

        // Recent profile update
        if ($user->updated_at->diffInDays(now()) < 30) $score += 25;

        // Activity in last 7 days
        if ($user->activities()->where('created_at', '>=', now()->subDays(7))->exists()) {
            $score += 25;
        }

        return min(100, $score);
    }
```
