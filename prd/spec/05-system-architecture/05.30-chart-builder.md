# Purpose

The Chart Builder provides an interactive interface for creating and customizing data visualizations. It allows users to transform raw data into meaningful charts through a visual editor, with support for multiple chart types, extensive customization options, and various export formats.

---

# Architecture Overview

The Chart Builder is a client-heavy component that renders charts in the browser using Canvas and SVG technologies. The backend handles data storage, sharing, and template management while the frontend handles all rendering and interaction logic.

The architecture separates data management from visualization rendering. Data flows through a pipeline: ingestion, transformation, mapping, and rendering. Each stage is independently testable and configurable.

The component integrates with the Gsmui component library for the builder interface (forms, buttons, panels) and uses a charting library (Chart.js or D3.js) for the actual visualization rendering. The builder UI is built with Livewire for state management and Alpine.js for client-side interactivity.

```php
            'datasets' => $datasets,

            'interaction' => [
                'mode' => 'nearest',
                'axis' => 'x',
                'intersect' => false,
            ],
```

### Statistical Functions

```php
class StatisticalProcessor
{
    public function calculateMovingAverage(array $data, int $window): array
    {
        $result = [];
        for ($i = 0; $i < count($data); $i++) {
            if ($i < $window - 1) {
                $result[] = null;
            } else {
                $windowData = array_slice($data, $i - $window + 1, $window);
                $result[] = array_sum($windowData) / count($windowData);
            }
        }
        return $result;
    }

    public function calculateTrendLine(array $data): array
    {
        $n = count($data);
        $x = range(0, $n - 1);
        $y = $data;

        $sumX = array_sum($x);
        $sumY = array_sum($y);
        $sumXY = array_sum(array_map(fn($xi, $yi) => $xi * $yi, $x, $y));
        $sumX2 = array_sum(array_map(fn($xi) => $xi * $xi, $x));

        $slope = ($n * $sumXY - $sumX * $sumY) / ($n * $sumX2 - $sumX * $sumX);
        $intercept = ($sumY - $slope * $sumX) / $n;

        return array_map(fn($xi) => $slope * $xi + $intercept, $x);
    }

    public function calculateRegression(array $xData, array $yData): RegressionResult
    {
        $n = count($xData);
        $sumX = array_sum($xData);
        $sumY = array_sum($yData);
        $sumXY = array_sum(array_map(fn($x, $y) => $x * $y, $xData, $yData));
        $sumX2 = array_sum(array_map(fn($x) => $x * $x, $xData));
        $sumY2 = array_sum(array_map(fn($y) => $y * $y, $yData));

        $slope = ($n * $sumXY - $sumX * $sumY) / ($n * $sumX2 - $sumX * $sumX);
        $intercept = ($sumY - $slope * $sumX) / $n;

        // R-squared
        $ssRes = 0;
        $ssTot = 0;
        $meanY = $sumY / $n;

        for ($i = 0; $i < $n; $i++) {
            $predicted = $slope * $xData[$i] + $intercept;
            $ssRes += ($yData[$i] - $predicted) ** 2;
            $ssTot += ($yData[$i] - $meanY) ** 2;
        }

        $rSquared = $ssTot > 0 ? 1 - ($ssRes / $ssTot) : 0;

        return new RegressionResult(
            slope: $slope,
            intercept: $intercept,
            r_squared: $rSquared,
            equation: "y = {$slope}x + {$intercept}",
        );
    }

    public function calculateCorrelation(array $xData, array $yData): float
    {
        $n = count($xData);
        $sumX = array_sum($xData);
        $sumY = array_sum($yData);
        $sumXY = array_sum(array_map(fn($x, $y) => $x * $y, $xData, $yData));
        $sumX2 = array_sum(array_map(fn($x) => $x * $x, $xData));
        $sumY2 = array_sum(array_map(fn($y) => $y * $y, $yData));

        $numerator = $n * $sumXY - $sumX * $sumY;
        $denominator = sqrt(($n * $sumX2 - $sumX ** 2) * ($n * $sumY2 - $sumY ** 2));

        return $denominator > 0 ? $numerator / $denominator : 0;
    }
}
```

### Data Transformation

```php
class DataTransformer
{
    public function pivot(array $data, string $indexColumn, string $pivotColumn, string $valueColumn): array
    {
        $result = [];
        $pivotValues = array_unique(array_column($data, $pivotColumn));

        foreach ($data as $row) {
            $index = $row[$indexColumn];
            if (!isset($result[$index])) {
                $result[$index] = array_fill_keys($pivotValues, null);
            }
            $result[$index][$row[$pivotColumn]] = $row[$valueColumn];
        }

        return $result;
    }

    public function aggregate(array $data, string $groupBy, string $aggregateColumn, string $function): array
    {
        $groups = [];
        foreach ($data as $row) {
            $key = $row[$groupBy];
            if (!isset($groups[$key])) {
                $groups[$key] = [];
            }
            $groups[$key][] = $row[$aggregateColumn];
        }

        return array_map(function ($values) use ($function) {
            return match($function) {
                'sum' => array_sum($values),
                'avg' => array_sum($values) / count($values),
                'count' => count($values),
                'min' => min($values),
                'max' => max($values),
                default => array_sum($values),
            };
        }, $groups);
    }

    public function normalize(array $data, float $min = 0, float $max = 1): array
    {
        $dataMin = min($data);
        $dataMax = max($data);
        $range = $dataMax - $dataMin;

        if ($range === 0) {
            return array_fill(0, count($data), ($min + $max) / 2);
        }

        return array_map(function ($value) use ($dataMin, $range, $min, $max) {
            return (($value - $dataMin) / $range) * ($max - $min) + $min;
        }, $data);
    }
}
```

```php
class PngExporter
{
    public function export(Chart $chart, int $scale = 2): string
    {
        $width = 800 * $scale;
        $height = 600 * $scale;

        $image = imagecreatetruecolor($width, $height);
        $bgColor = imagecolorallocate($image, 255, 255, 255);
        imagefilledrectangle($image, 0, 0, $width, $height, $bgColor);

        // Render chart on canvas
        $renderer = new CanvasChartRenderer($image, $width, $height);
        $renderer->render($chart);

        // Capture as PNG
        ob_start();
        imagepng($image);
        $pngData = ob_get_clean();
        imagedestroy($image);

        return $pngData;
    }
}
```

```php
class PdfExporter
{
    public function export(Chart $chart): string
    {
        $pdf = new FPDF('P', 'mm', 'A4');
        $pdf->AddPage();

        // Title
        $pdf->SetFont('Arial', 'B', 16);
        $pdf->Cell(0, 10, $chart->name, 0, 1, 'C');

        // Description
        if ($chart->description) {
            $pdf->SetFont('Arial', '', 10);
            $pdf->MultiCell(0, 5, $chart->description);
            $pdf->Ln(5);
        }

        // Chart image
        $pngExporter = new PngExporter();
        $pngData = $pngExporter->export($chart, 2);
        $tempPath = storage_path('app/temp/chart.png');
        file_put_contents($tempPath, $pngData);
        $pdf->Image($tempPath, 10, $pdf->GetY(), 190);
        unlink($tempPath);

        // Data table
        $pdf->AddPage();
        $pdf->SetFont('Arial', 'B', 12);
        $pdf->Cell(0, 10, 'Data Table', 0, 1);
        $this->renderDataTable($pdf, $chart);

        return $pdf->Output('S');
    }

    protected function renderDataTable(FPDF $pdf, Chart $chart): void
    {
        $pdf->SetFont('Arial', '', 8);
        $headers = array_merge(['Label'], array_column($chart->getSeries(), 'name'));

        foreach ($headers as $header) {
            $pdf->Cell(40, 7, $header, 1);
        }
        $pdf->Ln();

        foreach ($chart->getLabels() as $index => $label) {
            $pdf->Cell(40, 7, $label, 1);
            foreach ($chart->getSeries() as $series) {
                $pdf->Cell(40, 7, $series['data'][$index] ?? '', 1);
            }
            $pdf->Ln();
        }
    }
}
```

```php
class ApiConnector
{
    public function fetchData(array $config): array
    {
        $client = new \GuzzleHttp\Client([
            'timeout' => $config['timeout'] ?? 30,
        ]);

        $headers = $config['headers'] ?? [];
        if (isset($config['api_key'])) {
            $headers['Authorization'] = "Bearer {$config['api_key']}";
        }

        $response = $client->request(
            $config['method'] ?? 'GET',
            $config['url'],
            [
                'headers' => $headers,
                'query' => $config['params'] ?? [],
                'json' => $config['body'] ?? null,
            ]
        );

        $data = json_decode($response->getBody(), true);

        // Transform data if needed
        if (isset($config['transform'])) {
            $data = $this->transformData($data, $config['transform']);
        }

        return $data;
    }

    protected function transformData(array $data, array $transform): array
    {
        $result = $data;

        // Flatten nested structures
        if (isset($transform['flatten'])) {
            $result = $this->flatten($result, $transform['flatten']);
        }

        // Map fields
        if (isset($transform['map'])) {
            $result = array_map(function ($item) use ($transform['map']) {
                $mapped = [];
                foreach ($transform['map'] as $from => $to) {
                    $mapped[$to] = data_get($item, $from);
                }
                return $mapped;
            }, $result);
        }

        // Filter
        if (isset($transform['filter'])) {
            $result = array_filter($result, function ($item) use ($transform['filter']) {
                foreach ($transform['filter'] as $field => $condition) {
                    $value = data_get($item, $field);
                    if (!$this->matchCondition($value, $condition)) {
                        return false;
                    }
                }
                return true;
            });
        }

        return array_values($result);
    }
}
```

```php
        }

        if (isset($config['order_by'])) {

                $query->orderBy($order[0], $order[1] ?? 'asc');
            }
        }

        // Apply limit
```

```php
class CsvImporter
{
    public function import(UploadedFile $file, array $options = []): array
    {
        $handle = fopen($file->getPathname(), 'r');
        $headers = fgetcsv($handle);

        // Normalize headers
        $headers = array_map(function ($header) {
            return Str::slug($header, '_');
        }, $headers);

        $rows = [];
        $rowNumber = 0;

        while (($row = fgetcsv($handle)) !== false) {
            $rowNumber++;

            if (count($row) !== count($headers)) {
                continue; // Skip malformed rows
            }

            $rowData = array_combine($headers, $row);

            // Type casting
            if ($options['auto_detect_types'] ?? true) {
                $rowData = $this->castTypes($rowData);
            }

            $rows[] = $rowData;
        }

        fclose($handle);

        return [
            'headers' => $headers,
            'rows' => $rows,
            'row_count' => count($rows),
            'detected_types' => $this->detectColumnTypes($rows),
        ];
    }

    protected function castTypes(array $row): array
    {
        return array_map(function ($value) {
            if (is_numeric($value)) {
                return $value + 0; // Cast to number
            }
            if (filter_var($value, FILTER_VALIDATE_BOOLEAN, FILTER_NULL_ON_FAILURE) !== null) {
                return filter_var($value, FILTER_VALIDATE_BOOLEAN);
            }
            return $value;
        }, $row);
    }

    protected function detectColumnTypes(array $rows): array
    {
        if (empty($rows)) return [];

        $types = [];
        $firstRow = $rows[0];

        foreach ($firstRow as $column => $value) {
            $types[$column] = gettype($value);
        }

        return $types;
    }
}
```

```php
class JsonImporter
{
    public function import(UploadedFile $file, array $options = []): array
    {
        $content = file_get_contents($file->getPathname());
        $data = json_decode($content, true);

        if (json_last_error() !== JSON_ERROR_NONE) {
            throw new InvalidJsonException(json_last_error_msg());
        }

        // Handle nested structures
        if ($options['flatten'] ?? false) {
            $data = $this->flattenData($data);
        }

        // Handle different JSON structures
        if (isset($data['data']) && is_array($data['data'])) {
            $rows = $data['data'];
        } elseif (is_array($data) && !isset($data[0])) {
            $rows = [$data];
        } else {
            $rows = $data;
        }

        $headers = !empty($rows) ? array_keys($rows[0]) : [];

        return [
            'headers' => $headers,
            'rows' => $rows,
            'row_count' => count($rows),
        ];
    }

    protected function flattenData(array $data, string $prefix = ''): array
    {
        $result = [];

        foreach ($data as $key => $value) {
            $newKey = $prefix ? "{$prefix}.{$key}" : $key;

            if (is_array($value) && !array_is_list($value)) {
                $result = array_merge($result, $this->flattenData($value, $newKey));
            } else {
                $result[$newKey] = $value;
            }
        }

        return $result;
    }
}
```

```php
class ScheduledExportManager
{
    public function schedule(Chart $chart, array $config): ScheduledExport
    {
        return ScheduledExport::create([
            'chart_id' => $chart->id,
            'format' => $config['format'],
            'frequency' => $config['frequency'],
            'recipients' => $config['recipients'],
            'is_active' => true,

        ]);
    }

    public function processScheduledExports(): void
    {
        $dueExports = ScheduledExport::where('is_active', true)
            ->where('next_run_at', '<=', now())
            ->get();

        foreach ($dueExports as $export) {
            $this->processExport($export);
            $export->update(['next_run_at' => $this->calculateNextRun($export->frequency)]);
        }
    }

    protected function processExport(ScheduledExport $export): void
    {
        $chart = $export->chart;
        $exporter = match($export->format) {
            'pdf' => new ChartPdfReportGenerator(),
            'png' => new PngExporter(),
            'svg' => new SvgExporter(),
            'csv' => new CsvExporter(),
        };

        $output = $exporter->export($chart);

        foreach ($export->recipients as $recipient) {
            Notification::route('mail', $recipient)
                ->notify(new ChartExportNotification($chart, $output, $export->format));
        }
    }
}
```

## Chart Performance Optimization

### Data Aggregation Engine

```php
class DataAggregationEngine
{
    public function aggregate(array $data, string $method, int $interval): array
    {
        $aggregated = [];
        $chunks = array_chunk($data, $interval);

        foreach ($chunks as $chunk) {
            $aggregated[] = match($method) {
                'sum' => array_sum($chunk),
                'avg' => array_sum($chunk) / count($chunk),
                'min' => min($chunk),
                'max' => max($chunk),
                'count' => count($chunk),
                default => array_sum($chunk),
            };
        }

        return $aggregated;
    }

    public function downsample(array $data, int $targetPoints): array
    {
        if (count($data) <= $targetPoints) {
            return $data;
        }

        $step = count($data) / $targetPoints;
        $downsampled = [];

        for ($i = 0; $i < $targetPoints; $i++) {
            $index = (int) ($i * $step);
            $downsampled[] = $data[$index];
        }

        return $downsampled;
    }

    public function movingAverage(array $data, int $window): array
    {
        $result = [];
        for ($i = 0; $i < count($data); $i++) {
            $start = max(0, $i - $window + 1);
            $windowData = array_slice($data, $start, $i - $start + 1);
            $result[] = array_sum($windowData) / count($windowData);
        }
        return $result;
    }
}
```

### Rendering Optimization

```php
class ChartRenderingOptimizer
{
    public function optimizeForLargeDatasets(array $data, int $threshold = 10000): array
    {
        $totalPoints = count($data['labels'] ?? []);

        if ($totalPoints > $threshold) {
            $data = $this->downsampleData($data, $threshold);
            $data['_meta'] = ['downsampled' => true, 'original_count' => $totalPoints];
        }

        return $data;
    }

    protected function downsampleData(array $data, int $targetPoints): array
    {
        $engine = new DataAggregationEngine();

        if (isset($data['labels'])) {
            $data['labels'] = $engine->downsample($data['labels'], $targetPoints);
        }

        if (isset($data['series'])) {
            foreach ($data['series'] as &$series) {
                $series['data'] = $engine->downsample($series['data'], $targetPoints);
            }
        }

        return $data;
    }

    public function getCachingConfig(): array
    {
        return [
            'driver' => 'redis',
            'ttl' => 3600,
            'prefix' => 'chart:render:',
            'tags' => true,
        ];
    }
}
```
