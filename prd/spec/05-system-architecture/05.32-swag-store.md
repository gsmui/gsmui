# Purpose

The Swag Store provides a marketplace for GsmUi-branded merchandise and related products. It supports both digital goods (wallpapers, icons, fonts, design assets) and physical merchandise (t-shirts, hoodies, mugs, stickers, notebooks) with a streamlined shopping and fulfillment experience.

---

# Architecture Overview

The Swag Store is built as a standard e-commerce flow integrated into the GsmUi platform. It uses the existing authentication, billing, and notification systems while maintaining its own product catalog, cart, and order management modules.

The architecture separates concerns into product management, cart operations, checkout processing, and fulfillment. Each concern is handled by a dedicated service that communicates through events and contracts, enabling independent evolution and testing.

Physical and digital products follow different fulfillment paths. Digital products are delivered immediately through the download system, while physical products integrate with shipping providers for tracking and delivery confirmation. The store shares the payment processing infrastructure with the subscription system, reducing integration complexity.

```php
class ProductCatalogService
{
    public function getProducts(array $filters = [], int $page = 1, int $perPage = 20): LengthAwarePaginator
    {
        $query = SwagProduct::where('is_active', true);

        if (isset($filters['category'])) {
            $query->where('category_id', $filters['category']);
        }

        if (isset($filters['type'])) {
            $query->where('type', $filters['type']);
        }

        if (isset($filters['min_price'])) {
            $query->where('price', '>=', $filters['min_price']);
        }

        if (isset($filters['max_price'])) {
            $query->where('price', '<=', $filters['max_price']);
        }

        if (isset($filters['search'])) {
            $query->where(function ($q) use ($filters) {
                $q->where('name', 'LIKE', "%{$filters['search']}%")
                  ->orWhere('description', 'LIKE', "%{$filters['search']}%");
            });
        }

        $sort = $filters['sort'] ?? 'newest';
        $query = match($sort) {
            'price_asc' => $query->orderBy('price'),
            'price_desc' => $query->orderByDesc('price'),
            'popular' => $query->orderByDesc('download_count'),
            'name' => $query->orderBy('name'),
            default => $query->orderByDesc('created_at'),
        };

        return $query->with('category')->paginate($perPage, ['*'], 'page', $page);
    }

    public function getFeaturedProducts(int $limit = 6): Collection
    {
        return Cache::remember('store:featured', now()->addHours(2), function () use ($limit) {
            return SwagProduct::where('is_active', true)
                ->where('is_featured', true)
                ->limit($limit)
                ->with('category')
                ->get();
        });
    }

    public function getProductDetail(string $slug): SwagProduct
    {
        return Cache::remember("store:product:{$slug}", now()->addHour(), function () use ($slug) {
            return SwagProduct::where('slug', $slug)
                ->where('is_active', true)
                ->with('category', 'variants', 'reviews')
                ->firstOrFail();
        });
    }
}
```

### Inventory Management

```php
class InventoryManager
{
    public function checkAvailability(SwagProduct $product, array $variant = null): bool
    {
        if ($product->type === 'digital') {
            return true; // Digital products are always available
        }

        if ($variant) {
            $stock = $this->getVariantStock($product, $variant);
            return $stock > 0;
        }

        return $product->inventory_count > 0;
    }

    public function decrementStock(SwagProduct $product, int $quantity = 1, array $variant = null): void
    {
        if ($product->type === 'digital') {
            return;
        }

        if ($variant) {
            $this->decrementVariantStock($product, $variant, $quantity);
            return;
        }

        if ($product->inventory_count < $quantity) {
            throw new InsufficientStockException($product->id);
        }

        $product->decrement('inventory_count', $quantity);
    }

    public function getLowStockProducts(int $threshold = 10): Collection
    {
        return SwagProduct::where('type', 'physical')
            ->where('is_active', true)
            ->where('inventory_count', '<=', $threshold)
            ->get();
    }

    public function syncInventory(SwagProduct $product): void
    {
        if ($product->type !== 'physical') return;

        $provider = $this->getFulfillmentProvider($product);
        $providerStock = $provider->getStock($product->sku);

        $product->update(['inventory_count' => $providerStock]);
    }
}
```

## Cart Management

### Advanced Cart Operations

```php
class AdvancedCartManager
{
    public function mergeCarts(User $user, string $sessionId): void
    {
        $guestCart = SwagCart::where('session_id', $sessionId)->first();
        $userCart = $this->getOrCreateCart($user);

        if (!$guestCart) return;

        foreach ($guestCart->items as $guestItem) {
            $existingItem = $userCart->items()
                ->where('product_id', $guestItem->product_id)
                ->where('variant', $guestItem->variant)
                ->first();

            if ($existingItem) {
                $existingItem->increment('quantity', $guestItem->quantity);
            } else {
                $userCart->items()->create([
                    'product_id' => $guestItem->product_id,
                    'quantity' => $guestItem->quantity,
                    'variant' => $guestItem->variant,
                ]);
            }
        }

        $guestCart->delete();
    }

    public function calculateTotals(SwagCart $cart): CartTotals
    {
        $subtotal = $cart->items->sum(function ($item) {
            return $item->product->price * $item->quantity;
        });

        $discount = $this->calculateDiscount($cart, $subtotal);
        $tax = $this->calculateTax($cart, $subtotal - $discount);
        $shipping = $this->calculateShipping($cart);

        return new CartTotals(
            subtotal: $subtotal,
            discount: $discount,
            tax: $tax,
            shipping: $shipping,
            total: $subtotal - $discount + $tax + $shipping,
        );
    }

    protected function calculateDiscount(SwagCart $cart, float $subtotal): float
    {
        if (!$cart->coupon_code) return 0;

        $coupon = Coupon::where('code', $cart->coupon_code)
            ->where('is_active', true)
            ->first();

        if (!$coupon || !$coupon->isValid()) return 0;

        if ($coupon->type === 'percentage') {
            return $subtotal * ($coupon->value / 100);
        }

        return min($coupon->value, $subtotal);
    }

    protected function calculateTax(SwagCart $cart, float $amount): float
    {
        if (!$cart->shipping_address) return 0;

        $taxRate = $this->getTaxRate($cart->shipping_address['country'], $cart->shipping_address['state']);

        return round($amount * $taxRate, 2);
    }

    protected function calculateShipping(SwagCart $cart): float
    {
        $hasPhysicalItems = $cart->items->contains(fn($item) => $item->product->type === 'physical');

        if (!$hasPhysicalItems) return 0;

        $weight = $cart->items->sum(function ($item) {
            return ($item->product->weight ?? 0) * $item->quantity;
        });

        return $this->getShippingRate($cart->shipping_address, $weight);
    }
}
```

## Checkout Flow

### Multi-Step Checkout

```php
class MultiStepCheckout
{
    public function initiate(SwagCart $cart, array $data): CheckoutSession
    {
        $session = CheckoutSession::create([
            'cart_id' => $cart->id,
            'user_id' => auth()->id(),
            'step' => 'shipping',
            'shipping_address' => $data['shipping_address'] ?? null,
            'billing_address' => $data['billing_address'] ?? null,
            'payment_method' => null,
            'totals' => app(AdvancedCartManager::class)->calculateTotals($cart)->toArray(),
        ]);

        return $session;
    }

    public function processShipping(CheckoutSession $session, array $address): CheckoutSession
    {
        $validated = Validator::make($address, [
            'name' => 'required|string|max:255',
            'line1' => 'required|string|max:255',
            'line2' => 'nullable|string|max:255',
            'city' => 'required|string|max:255',
            'state' => 'required|string|max:255',
            'postal_code' => 'required|string|max:20',
            'country' => 'required|string|size:2',
            'phone' => 'required|string|max:20',
        ])->validate();

        $session->update([
            'shipping_address' => $validated,
            'step' => 'payment',
        ]);

        // Recalculate shipping costs
        $totals = app(AdvancedCartManager::class)->calculateTotals($session->cart);
        $session->update(['totals' => $totals->toArray()]);

        return $session;
    }

    public function processPayment(CheckoutSession $session, array $paymentData): CheckoutSession
    {
        $paymentIntent = Stripe::paymentIntents()->create([
            'amount' => (int) ($session->totals['total'] * 100),
            'currency' => 'usd',
            'payment_method' => $paymentData['payment_method_id'],
            'confirm' => true,
            'metadata' => [
                'checkout_session_id' => $session->id,
                'user_id' => $session->user_id,
            ],
        ]);

        $session->update([
            'payment_method' => $paymentData['payment_method_id'],
            'payment_intent_id' => $paymentIntent->id,
            'step' => 'review',
        ]);

        return $session;
    }

    public function confirm(CheckoutSession $session): Order
    {
        // Create order
        $order = Order::create([
            'user_id' => $session->user_id,
            'status' => 'confirmed',
            'subtotal' => $session->totals['subtotal'],
            'discount' => $session->totals['discount'],
            'tax' => $session->totals['tax'],
            'shipping_cost' => $session->totals['shipping'],
            'total' => $session->totals['total'],
            'shipping_address' => $session->shipping_address,
            'payment_intent_id' => $session->payment_intent_id,
        ]);

        // Create order items
        foreach ($session->cart->items as $item) {
            $orderItem = OrderItem::create([
                'order_id' => $order->id,
                'product_id' => $item->product_id,
                'quantity' => $item->quantity,
                'unit_price' => $item->product->price,
                'total' => $item->product->price * $item->quantity,
                'variant' => $item->variant,
            ]);

            // Fulfill digital products
            if ($item->product->type === 'digital') {
                $this->fulfillDigital($orderItem);
            }
        }

        // Decrement inventory for physical products
        foreach ($session->cart->items as $item) {
            if ($item->product->type === 'physical') {
                app(InventoryManager::class)->decrementStock($item->product, $item->quantity, $item->variant);
            }
        }

        // Clear cart
        $session->cart->items()->delete();
        $session->cart->delete();

        // Send confirmation
        event(new OrderConfirmed($order));

        return $order;
    }

    protected function fulfillDigital(OrderItem $item): void
    {
        $downloadUrl = $this->generateDownloadUrl($item);
        $licenseKey = $this->generateLicenseKey($item);

        $item->update([
            'download_url' => $downloadUrl,
            'license_key' => $licenseKey,
            'fulfillment_status' => 'fulfilled',
        ]);
    }
}
```

```php
class DownloadService
{
    public function generateDownloadUrl(OrderItem $item, int $expiresInHours = 24): string
    {
        $token = Str::random(64);

        DownloadToken::create([
            'order_item_id' => $item->id,
            'token' => hash('sha256', $token),
            'expires_at' => now()->addHours($expiresInHours),
            'max_downloads' => 5,
            'download_count' => 0,
        ]);

        return route('store.download', ['token' => $token]);
    }

    public function processDownload(string $token): StreamedResponse
    {
        $downloadToken = DownloadToken::where('token', hash('sha256', $token))
            ->where('expires_at', '>', now())
            ->where('download_count', '<', DB::raw('max_downloads'))
            ->firstOrFail();

        $item = $downloadToken->orderItem;
        $product = $item->product;

        // Increment download count
        $downloadToken->increment('download_count');

        // Log download
        DownloadLog::create([
            'order_item_id' => $item->id,
            'user_id' => auth()->id(),
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
        ]);

        // Stream file
        $filePath = storage_path("app/products/{$product->file_path}");

        return response()->stream(function () use ($filePath) {
            readfile($filePath);
        }, 200, [
            'Content-Type' => mime_content_type($filePath),
            'Content-Disposition' => "attachment; filename=\"" . basename($filePath) . "\"",
            'Content-Length' => filesize($filePath),
        ]);
    }

    public function generateLicenseKey(OrderItem $item): string
    {
        $key = strtoupper(implode('-', array_map(function () {
            return substr(strtoupper(Str::random(4)), 0, 4);
        }, range(1, 4))));

        LicenseKey::create([
            'order_item_id' => $item->id,
            'key' => $key,
            'is_active' => true,
        ]);

        return $key;
    }
}
```

```php
class PrintProviderIntegration

    public function createOrder(Order $order): PrintProviderOrder
    {

            'inprogress' => 'processing',

            'canceled' => 'cancelled',

        if ($newStatus === 'shipped') {
            $printfulOrder->order->update([
                'tracking_number' => $payload['tracking_number'] ?? null,
```

```php
{
    public function validate(string $code, SwagCart $cart): ?Coupon
    {
        $coupon = Coupon::where('code', $code)
            ->where('is_active', true)
            ->first();

        if (!$coupon) {
            throw new InvalidCouponException('Invalid coupon code');
        }

        if ($coupon->expires_at && $coupon->expires_at->isPast()) {
            throw new InvalidCouponException('Coupon has expired');
        }

        if ($coupon->usage_limit && $coupon->usage_count >= $coupon->usage_limit) {
            throw new InvalidCouponException('Coupon usage limit reached');
        }

        if ($coupon->min_cart_amount && $cart->subtotal() < $coupon->min_cart_amount) {
            throw new InvalidCouponException("Minimum cart amount is {$coupon->min_cart_amount}");
        }

        if ($coupon->product_ids && !$this->appliesToCart($coupon, $cart)) {
            throw new InvalidCouponException('Coupon does not apply to cart items');
        }

        return $coupon;
    }

    public function apply(string $code, SwagCart $cart): void
    {
        $coupon = $this->validate($code, $cart);

        $cart->update(['coupon_code' => $code]);
        $coupon->increment('usage_count');
    }
```

```php
class ShippingCalculator
{
    public function calculateRates(array $address, array $items): array
    {
        $totalWeight = collect($items)->sum(function ($item) {
            return ($item['weight'] ?? 0) * $item['quantity'];
        });

        $totalValue = collect($items)->sum(function ($item) {
            return ($item['price'] ?? 0) * $item['quantity'];
        });

        $rates = [];

        // Standard shipping
        $rates[] = [
            'method' => 'standard',
            'name' => 'Standard Shipping',
            'price' => $this->calculateStandardRate($address, $totalWeight, $totalValue),
            'estimated_days' => '5-7 business days',
        ];

        // Express shipping
        $rates[] = [
            'method' => 'express',
            'name' => 'Express Shipping',
            'price' => $this->calculateExpressRate($address, $totalWeight, $totalValue),
            'estimated_days' => '2-3 business days',
        ];

        // Free shipping threshold
        if ($totalValue >= config('store.free_shipping_threshold', 50)) {
            $rates[] = [
                'method' => 'free',
                'name' => 'Free Standard Shipping',
                'price' => 0,
                'estimated_days' => '5-7 business days',
                'applied' => true,
            ];
        }

        return $rates;
    }

    protected function calculateStandardRate(array $address, float $weight, float $value): float
    {
        $baseRate = 5.99;
        $weightRate = $weight * 0.5;
        $zoneRate = $this->getZoneRate($address['country']);

        return round($baseRate + $weightRate + $zoneRate, 2);
    }

    protected function calculateExpressRate(array $address, float $weight, float $value): float
    {
        $standardRate = $this->calculateStandardRate($address, $weight, $value);

        return round($standardRate * 2, 2);
    }

    protected function getZoneRate(string $country): float
    {
        $zones = [
            'US' => 0,
            'CA' => 2.00,
            'GB' => 3.50,
            'EU' => 3.50,
            'AU' => 4.50,
            'default' => 5.00,
        ];

        return $zones[$country] ?? $zones['default'];
    }
}
```

```php
    public function redeem(string $code, User $user): bool
    {
        $giftCard = GiftCard::where('code', $code)
            ->where('is_redeemed', false)
            ->where('expires_at', '>', now())
            ->first();

        if (!$giftCard) {
            return false;
        }

        $giftCard->update([
            'is_redeemed' => true,
            'redeemed_by' => $user->id,
            'redeemed_at' => now(),
        ]);

        // Add balance to user's account
        $user->increment('gift_card_balance', $giftCard->balance);

        event(new GiftCardRedeemed($giftCard, $user));

        return true;
    }

    public function applyBalance(Order $order, float $amount): void
    {
        $user = $order->user;
        $availableBalance = $user->gift_card_balance;

        if ($availableBalance <= 0) {
            throw new InsufficientGiftCardBalanceException();
        }

        $applyAmount = min($amount, $availableBalance);

        $user->decrement('gift_card_balance', $applyAmount);

        OrderPayment::create([
            'order_id' => $order->id,
            'method' => 'gift_card',
            'amount' => $applyAmount,
        ]);
    }

    {
        do {
            $code = 'GSM-' . strtoupper(Str::random(4) . '-' . Str::random(4));
        } while (GiftCard::where('code', $code)->exists());

        return $code;
```

```php
class LoyaltyProgram
{
    protected array $pointsConfig = [
        'purchase' => 1, // 1 point per dollar spent
        'review' => 10,
        'referral' => 50,
        'birthday' => 25,
    ];

    protected array $rewardsConfig = [
        100 => ['type' => 'discount', 'value' => 5],
        250 => ['type' => 'discount', 'value' => 15],
        500 => ['type' => 'discount', 'value' => 30],
        1000 => ['type' => 'free_product', 'value' => null],
    ];

    public function awardPoints(User $user, string $reason, array $context = []): int
    {
        $points = $this->pointsConfig[$reason] ?? 0;

        if ($reason === 'purchase') {
            $points = ($context['amount'] ?? 0) * $this->pointsConfig['purchase'];
        }

        if ($points <= 0) return 0;

        LoyaltyPoint::create([
            'user_id' => $user->id,
            'points' => $points,
            'reason' => $reason,
            'context' => $context,
        ]);

        $user->increment('loyalty_points', $points);

        $this->checkForRewards($user);

        return $points;
    }

    public function redeemPoints(User $user, int $points): ?LoyaltyReward
    {
        if ($user->loyalty_points < $points) {
            return null;
        }

        $reward = $this->findReward($points);
        if (!$reward) return null;

        $user->decrement('loyalty_points', $points);

        LoyaltyRedemption::create([
            'user_id' => $user->id,
            'points_redeemed' => $points,
            'reward_type' => $reward['type'],
            'reward_value' => $reward['value'],
        ]);

        return new LoyaltyReward($reward);
    }

    protected function checkForRewards(User $user): void
    {
        foreach ($this->rewardsConfig as $threshold => $reward) {
            if ($user->loyalty_points >= $threshold) {
                $existing = LoyaltyReward::where('user_id', $user->id)
                    ->where('threshold', $threshold)
                    ->exists();

                if (!$existing) {
                    LoyaltyReward::create([
                        'user_id' => $user->id,
                        'threshold' => $threshold,
                        'type' => $reward['type'],
                        'value' => $reward['value'],
                        'is_claimed' => false,
                    ]);

                    Notification::send($user, new LoyaltyRewardNotification($reward, $threshold));
                }
            }
        }
    }
}
```

```php
class ProductBundleManager
{
    public function createBundle(array $data): ProductBundle
    {
        $bundle = ProductBundle::create([
            'name' => $data['name'],
            'description' => $data['description'],
            'discount_type' => $data['discount_type'], // percentage, fixed
            'discount_value' => $data['discount_value'],
            'is_active' => true,
        ]);

        foreach ($data['products'] as $productData) {
            BundleItem::create([
                'bundle_id' => $bundle->id,
                'product_id' => $productData['product_id'],
                'quantity' => $productData['quantity'] ?? 1,
            ]);
        }

        return $bundle;
    }

    public function calculateBundlePrice(ProductBundle $bundle): BundlePricing
    {
        $items = $bundle->items->with('product')->get();
        $originalPrice = $items->sum(fn($item) => $item->product->price * $item->quantity);

        $discount = match($bundle->discount_type) {
            'percentage' => $originalPrice * ($bundle->discount_value / 100),
            'fixed' => $bundle->discount_value,
            default => 0,
        };

        $bundlePrice = $originalPrice - $discount;

        return new BundlePricing(
            original_price: $originalPrice,
            discount: $discount,
            bundle_price: $bundlePrice,
            savings: $discount,
            savings_percentage: $originalPrice > 0 ? round(($discount / $originalPrice) * 100) : 0,
        );
    }
}
```

```php
            'start_at' => $data['start_at'],
            'end_at' => $data['end_at'],
            'product_ids' => $data['product_ids'] ?? [],
            'category_ids' => $data['category_ids'] ?? [],
            'max_uses' => $data['max_uses'] ?? null,
            'current_uses' => 0,
            'is_active' => true,
        ]);
    }

    public function getActiveSales(): Collection
    {
        return FlashSale::where('is_active', true)
            ->where('start_at', '<=', now())
            ->where('end_at', '>=', now())
            ->get();
    }

    public function applySale(SwagProduct $product): ?array
    {
        $activeSales = $this->getActiveSales()
            ->filter(fn($sale) => $this->saleAppliesTo($sale, $product));

        if ($activeSales->isEmpty()) return null;

        $bestSale = $activeSale->sortByDesc('discount_value')->first();

        return [
            'sale_id' => $bestSale->id,
            'discount_type' => $bestSale->discount_type,
            'discount_value' => $bestSale->discount_value,
            'sale_price' => $this->calculateSalePrice($product, $bestSale),
            'ends_at' => $bestSale->end_at,
        ];
    }

    protected function saleAppliesTo(FlashSale $sale, SwagProduct $product): bool
    {
        if (!empty($sale->product_ids) && !in_array($product->id, $sale->product_ids)) {
            return false;
        }

        if (!empty($sale->category_ids) && !in_array($product->category_id, $sale->category_ids)) {
            return false;
        }

        if ($sale->max_uses && $sale->current_uses >= $sale->max_uses) {
            return false;
        }

        return true;
    }
}
```

```php
class TaxConfiguration

    public function calculateTax(float $amount, array $address): float
    {
        $taxRate = $this->getTaxRate($address['country'], $address['state'] ?? null);

        return round($amount * $taxRate, 2);

    {
        $rates = config('store.tax_rates', []);

        if (isset($rates[$country][$state])) {
            return $rates[$country][$state];

        if (isset($rates[$country]['*'])) {
            return $rates[$country]['*'];
        }

        return config('store.default_tax_rate', 0);
    }
```
