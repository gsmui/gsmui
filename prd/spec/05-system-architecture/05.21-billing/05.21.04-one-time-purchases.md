```php
// app/Services/OneTimePurchaseService.php
class OneTimePurchaseService
{
    public function __construct(
        private StripeClient $stripe,
        private LicenseService $licenseService,
        private DownloadService $downloadService,
    ) {}

    public function createOrder(User $user, Product $product, array $paymentData): Order
    {
        // Create Stripe payment intent
        $paymentIntent = $this->stripe->paymentIntents->create([
            'amount' => $product->price,
            'currency' => $product->currency ?? 'usd',
            'customer' => $user->stripe_id,
            'metadata' => [
                'user_id' => $user->id,
                'product_id' => $product->id,
                'product_name' => $product->name,
            ],
        ]);

        // Create local order
        $order = Order::create([
            'user_id' => $user->id,
            'status' => 'pending',
            'subtotal' => $product->price,
            'total' => $product->price,
            'currency' => $product->currency ?? 'usd',
            'payment_method' => 'stripe',
            'payment_id' => $paymentIntent->id,
        ]);

        // Create order item
        $order->items()->create([
            'product_id' => $product->id,
            'description' => $product->name,
            'amount' => $product->price,
            'quantity' => 1,
            'type' => 'product',
        ]);

        return $order;
    }

    public function completeOrder(Order $order, \Stripe\PaymentIntent $paymentIntent): void
    {
        $order->update([
            'status' => 'completed',
            'paid_at' => now(),
        ]);

        // Generate license for each product
        foreach ($order->items as $item) {
            $license = $this->licenseService->generate(
                order: $order,
                product: $item->product,
                user: $order->user,
            );

            // Create download links
            $this->downloadService->createLinks(
                order: $order,
                product: $item->product,
                license: $license,
            );
        }

        // Generate invoice
        $this->generateInvoice($order);

        // Send confirmation email
        OrderConfirmationMail::to($order->user)->queue($order);

        event(new OrderCompleted($order));
    }

    public function refundOrder(Order $order, string $reason): void
    {
        // Process refund in Stripe
        $refund = $this->stripe->refunds->create([
            'payment_intent' => $order->payment_id,
            'reason' => 'requested_by_customer',
        ]);

        $order->update([
            'status' => 'refunded',
            'refunded_at' => now(),
            'refund_reason' => $reason,
        ]);

        // Revoke licenses
        $this->licenseService->revokeAllForOrder($order);

        // Revoke download access
        $this->downloadService->revokeAllForOrder($order);

        event(new OrderRefunded($order, $reason));
    }
}
```

```php
// app/Services/OneTimeCheckoutService.php
class OneTimeCheckoutService
{
    public function createCheckoutSession(User $user, Product $product, ?string $couponCode = null): \Stripe\Checkout\Session
    {
        $lineItem = [
            'price_data' => [
                'currency' => $product->currency ?? 'usd',
                'product_data' => [
                    'name' => $product->name,
                    'description' => $product->short_description,
                    'images' => $product->preview_images ? [$product->preview_images[0]] : [],
                ],
                'unit_amount' => $product->price,
            ],
            'quantity' => 1,
        ];

        $params = [
            'payment_method_types' => ['card'],
            'mode' => 'payment',
            'customer' => $user->stripe_id,
            'line_items' => [$lineItem],
            'success_url' => route('checkout.success', ['session_id' => '{CHECKOUT_SESSION_ID}']),
            'cancel_url' => route('checkout.cancel'),
            'metadata' => [
                'user_id' => $user->id,
                'product_id' => $product->id,
                'type' => 'one_time',
            ],
        ];

        // Apply coupon if provided
        if ($couponCode) {
            $coupon = Coupon::where('code', $couponCode)
                ->where('is_active', true)
                ->first();

            if ($coupon && $coupon->isValidForUser($user, $product)) {
                $params['discounts'] = [
                    ['coupon' => $coupon->stripe_coupon_id],
                ];
            }
        }

        return $this->stripe->checkout->sessions->create($params);
    }

    public function handleSuccess(string $sessionId): Order
    {
        $session = $this->stripe->checkout->sessions->retrieve($sessionId);

        $order = Order::where('stripe_session_id', $sessionId)->first();

        if ($order && $order->status === 'pending') {
            app(OneTimePurchaseService::class)->completeOrder(
                $order,
                $this->stripe->paymentIntents->retrieve($session->payment_intent)
            );
        }

        return $order;
    }
}
```

```php
// app/Services/DownloadService.php
class DownloadService
{
    private const MAX_DOWNLOADS_PER_PRODUCT = 50;
    private const DOWNLOAD_LINK_TTL = 3600; // 1 hour

    public function createLinks(Order $order, Product $product, License $license): Collection
    {
        $files = $product->files()->orderByDesc('version')->get();

        $links = collect();

        foreach ($files as $file) {
            $download = Download::create([
                'order_id' => $order->id,
                'user_id' => $order->user_id,
                'product_id' => $product->id,
                'file_id' => $file->id,
                'license_id' => $license->id,
                'token' => Str::random(64),
                'expires_at' => now()->addDays(30),
            ]);

            $links->push([
                'version' => $file->version,
                'filename' => $file->filename,
                'size' => $file->size,
                'url' => $this->generateSignedUrl($download),
            ]);
        }

        return $links;
    }

    public function getDownloadUrl(Download $download): string
    {
        if ($download->download_count >= self::MAX_DOWNLOADS_PER_PRODUCT) {
            throw new DownloadLimitExceededException();
        }

        if ($download->expires_at->isPast()) {
            throw new DownloadExpiredException();
        }

        return Storage::disk('s3')->temporaryUrl(
            $download->file->path,
            now()->addSeconds(self::DOWNLOAD_LINK_TTL),
            [
                'ResponseContentDisposition' => "attachment; filename=\"{$download->file->filename}\"",
            ]
        );
    }

    public function recordDownload(Download $download): void
    {
        $download->increment('download_count');
        $download->update(['last_downloaded_at' => now()]);

        // Log for analytics
        DownloadEvent::create([
            'download_id' => $download->id,
            'user_id' => $download->user_id,
            'product_id' => $download->product_id,
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
        ]);
    }

    public function revokeAllForOrder(Order $order): void
    {
        Download::where('order_id', $order->id)->update(['revoked_at' => now()]);
    }

    public function getUserDownloads(User $user): Collection
    {
        return Download::where('user_id', $user->id)
            ->whereNull('revoked_at')
            ->with(['product', 'file'])
            ->get()
            ->groupBy('product_id')
            ->map(function ($downloads, $productId) {
                $product = $downloads->first()->product;
                $latestFile = $downloads->sortByDesc('file.version')->first()->file;

                return [
                    'product' => $product,
                    'latest_version' => $latestFile->version,
                    'downloads' => $downloads->map(fn($d) => [
                        'version' => $d->file->version,
                        'filename' => $d->file->filename,
                        'size' => $d->file->size,
                        'downloaded_at' => $d->last_downloaded_at,
                        'expires_at' => $d->expires_at,
                        'is_expired' => $d->expires_at->isPast(),
                        'download_count' => $d->download_count,
                    ]),
                ];
            });
    }
}
```

```php
// app/Services/RefundService.php
class RefundService
{
    private const REFUND_WINDOW_DAYS = 14;
    private const AUTO_APPROVE_THRESHOLD = 5000; // $50

    public function requestRefund(Order $order, User $user, string $reason, ?string $details = null): RefundRequest
    {
        // Validate refund eligibility
        $this->validateEligibility($order, $user);

        $refundRequest = RefundRequest::create([
            'order_id' => $order->id,
            'user_id' => $user->id,
            'reason' => $reason,
            'details' => $details,
            'status' => 'pending',
            'amount' => $order->total,
        ]);

        // Auto-approve small refunds
        if ($order->total <= self::AUTO_APPROVE_THRESHOLD) {
            $this->approveRefund($refundRequest);
        } else {
            // Notify admin for manual review
            NewRefundRequestNotification::toAdmins()->queue($refundRequest);
        }

        event(new RefundRequested($refundRequest));

        return $refundRequest;
    }

    public function approveRefund(RefundRequest $refundRequest): void
    {
        $order = $refundRequest->order;

        // Process refund in Stripe
        $refund = \Stripe\Refund::create([
            'payment_intent' => $order->payment_id,
            'amount' => $refundRequest->amount,
            'reason' => 'requested_by_customer',
            'metadata' => [
                'refund_request_id' => $refundRequest->id,
                'user_id' => $refundRequest->user_id,
            ],
        ]);

        $refundRequest->update([
            'status' => 'approved',
            'stripe_refund_id' => $refund->id,
            'processed_at' => now(),
        ]);

        $order->update([
            'status' => $order->total === $refundRequest->amount ? 'refunded' : 'partially_refunded',
            'refunded_at' => now(),
            'refund_reason' => $refundRequest->reason,
        ]);

        // Revoke licenses
        app(LicenseService::class)->revokeAllForOrder($order);

        // Revoke download access
        app(DownloadService::class)->revokeAllForOrder($order);

        // Send notification
        RefundApprovedNotification::to($refundRequest->user)->queue($refundRequest);

        event(new RefundProcessed($refundRequest));
    }

    public function denyRefund(RefundRequest $refundRequest, string $denialReason): void
    {
        $refundRequest->update([
            'status' => 'denied',
            'denial_reason' => $denialReason,
            'processed_at' => now(),
        ]);

        RefundDeniedNotification::to($refundRequest->user)->queue($refundRequest);
    }

    private function validateEligibility(Order $order, User $user): void
    {
        if ($order->user_id !== $user->id) {
            throw new UnauthorizedException();
        }

        if (!in_array($order->status, ['completed'])) {
            throw new InvalidOrderStatusException('Order is not eligible for refund');
        }

        if ($order->paid_at->diffInDays(now()) > self::REFUND_WINDOW_DAYS) {
            throw new RefundWindowExpiredException('Refund window has expired (14 days)');
        }

        if ($order->refund_requests()->where('status', 'pending')->exists()) {
            throw new PendingRefundException('A refund request is already pending');
        }
    }
}
```

```php
    public function submitReview(User $user, Product $product, int $rating, string $title, string $body): ProductReview
    {
        // Verify purchase
        $order = Order::where('user_id', $user->id)
            ->where('status', 'completed')
            ->whereHas('items', fn($q) => $q->where('product_id', $product->id))
            ->first();

        $review = ProductReview::create([
            'product_id' => $product->id,
            'user_id' => $user->id,
            'order_id' => $order?->id,
            'rating' => $rating,
            'title' => $title,
            'body' => $body,
            'is_verified' => $order !== null,
            'is_approved' => false,
        ]);

        // Auto-approve if verified buyer
        if ($order) {
            $review->update(['is_approved' => true]);
        }

        // Update product rating
        $this->updateProductRating($product);

        event(new ReviewSubmitted($review));

        return $review;
    }

    private function updateProductRating(Product $product): void
    {
        $stats = ProductReview::where('product_id', $product->id)
            ->where('is_approved', true)
            ->selectRaw('AVG(rating) as avg_rating, COUNT(*) as review_count')
            ->first();

        $product->update([
            'rating' => round($stats->avg_rating ?? 0, 1),
            'rating_count' => $stats->review_count ?? 0,
        ]);
    }
}
```

```php
class ProductComparisonService

    public function compare(array $productIds): array
    {
        $products = Product::whereIn('id', $productIds)->get();

        return [
            'products' => $products,
            'features' => $this->extractFeatures($products),
            'pricing' => $this->extractPricing($products),
            'ratings' => $this->extractRatings($products),
        ];
    }

    private function extractFeatures(Collection $products): array
    {
        $allFeatures = collect();

        foreach ($products as $product) {
            $features = $product->metadata['features'] ?? [];
            foreach ($features as $feature) {
                $allFeatures->push($feature);
            }
        }

        $uniqueFeatures = $allFeatures->unique()->values();

        return $uniqueFeatures->map(fn($feature) => [
            'name' => $feature,
            'values' => $products->map(fn($p) => in_array($feature, $p->metadata['features'] ?? [])),
        ])->toArray();
    }

    private function extractPricing(Collection $products): array
    {
        return $products->map(fn($p) => [
            'id' => $p->id,
            'name' => $p->name,
            'price' => $p->price,
            'price_formatted' => $p->price_formatted,
        ])->toArray();
    }
```

```php
              ->whereHas('items', fn($iq) => $iq->where('product_id', $product->id));
        })->get();

        foreach ($users as $user) {
            VersionAvailableNotification::to($user)->queue($product, $version);
        }
```

```php
<?php
// app/Services/Analytics/PurchaseAnalyticsService.php
namespace App\Services\Analytics;

use App\Models\Order;
use App\Models\Product;
use Illuminate\Support\Facades\DB;

class PurchaseAnalyticsService
{
    public function trackPurchase(Order $order): void
    {
        foreach ($order->items as $item) {
            DB::table('analytics_events')->insert([
                'event_type' => 'purchase',
                'user_id' => $order->user_id,
                'product_id' => $item->product_id,
                'order_id' => $order->id,
                'amount' => $item->price,
                'currency' => $order->currency,
                'properties' => json_encode([
                    'quantity' => $item->quantity,
                    'discount' => $item->discount,
                    'payment_method' => $order->payment_method,
                ]),
                'created_at' => now(),
            ]);
        }
    }

    public function getConversionRate(Product $product, string $period = '30d'): float
    {
        $views = DB::table('analytics_events')
            ->where('event_type', 'product_view')
            ->where('product_id', $product->id)
            ->where('created_at', '>=', now()->subDays((int) $period))
            ->count();

        $purchases = DB::table('analytics_events')
            ->where('event_type', 'purchase')
            ->where('product_id', $product->id)
            ->where('created_at', '>=', now()->subDays((int) $period))
            ->count();

        return $views > 0 ? round($purchases / $views * 100, 2) : 0;
    }

    public function getRevenueByProduct(string $period = '30d'): array
    {
        return DB::table('analytics_events')
            ->where('event_type', 'purchase')
            ->where('created_at', '>=', now()->subDays((int) $period))
            ->select('product_id', DB::raw('SUM(amount) as revenue'), DB::raw('COUNT(*) as orders'))
            ->groupBy('product_id')
            ->orderByDesc('revenue')
            ->get()
            ->toArray();
    }

    public function getAverageOrderValue(string $period = '30d'): float
    {
        $result = DB::table('analytics_events')
            ->where('event_type', 'purchase')
            ->where('created_at', '>=', now()->subDays((int) $period))
            ->avg('amount');

        return round($result, 2);
    }
}
```

```php
<?php
// app/Livewire/Admin/RefundAnalyticsDashboard.php
namespace App\Livewire\Admin;

use Livewire\Component;
use Illuminate\Support\Facades\DB;

class RefundAnalyticsDashboard extends Component
{
    public $period = '30d';
    public $refundRate;
    public $totalRefunds;
    public $totalRevenue;
    public $topRefundedProducts;

    public function mount(): void
    {
        $this->loadData();
    }

    public function loadData(): void
    {
        $days = (int) str_replace('d', '', $this->period);

        $this->totalRevenue = DB::table('analytics_events')
            ->where('event_type', 'purchase')
            ->where('created_at', '>=', now()->subDays($days))
            ->sum('amount');

        $this->totalRefunds = DB::table('analytics_events')
            ->where('event_type', 'refund')
            ->where('created_at', '>=', now()->subDays($days))
            ->sum('amount');

        $this->refundRate = $this->totalRevenue > 0
            ? round($this->totalRefunds / $this->totalRevenue * 100, 2)
            : 0;

        $this->topRefundedProducts = DB::table('analytics_events')
            ->where('event_type', 'refund')
            ->where('created_at', '>=', now()->subDays($days))
            ->select('product_id', DB::raw('COUNT(*) as count'), DB::raw('SUM(amount) as total'))
            ->groupBy('product_id')
            ->orderByDesc('total')
            ->limit(10)
            ->get();
    }

    public function render()
    {
        return view('livewire.admin.refund-analytics-dashboard');
    }
}
```
