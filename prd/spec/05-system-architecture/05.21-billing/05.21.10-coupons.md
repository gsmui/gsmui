## Discount Calculation Engine

```php
class DiscountCalculator
{
    public function calculate(Coupon $coupon, Cart $cart): DiscountResult
    {
        $subtotal = $cart->getSubtotal();

        return match ($coupon->type) {
            'percentage' => $this->calculatePercentage($coupon, $subtotal),
            'fixed_amount' => $this->calculateFixed($coupon, $subtotal),
            'free_trial' => $this->calculateFreeTrial($coupon, $cart),
            default => throw new InvalidCouponTypeException(),
        };
    }

    private function calculatePercentage(Coupon $coupon, int $subtotal): DiscountResult
    {
        $discount = (int) ($subtotal * ($coupon->value / 100));

        if ($coupon->max_discount && $discount > $coupon->max_discount) {
            $discount = $coupon->max_discount;
        }

        return new DiscountResult(
            amount: $discount,
            type: 'percentage',
            description: "{$coupon->value}% off",
        );
    }

    private function calculateFixed(Coupon $coupon, int $subtotal): DiscountResult
    {
        $discount = min($coupon->value, $subtotal);

        return new DiscountResult(
            amount: $discount,
            type: 'fixed',
            description: "$" . number_format($coupon->value / 100, 2) . " off",
        );
    }
}
```

## Code Generation Service

```php
class CouponCodeGenerator
{
    private const CHARSET = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
    private const EXCLUDED = ['0', 'O', 'I', '1', 'L'];

    public function generate(int $length = 8, ?string $prefix = null): string
    {
        do {
            $code = $this->generateRandom($length);
            if ($prefix) {
                $code = $prefix . $code;
            }
        } while ($this->codeExists($code));

        return $code;
    }

    private function generateRandom(int $length): string
    {
        $code = '';
        for ($i = 0; $i < $length; $i++) {
            $code .= self::Charset[random_int(0, strlen(self::Charset) - 1)];
        }
        return $code;
    }
}
```

## Validation Service

```php
class CouponValidator
{
    public function validate(string $code, Cart $cart, User $user): ValidationResult
    {
        $coupon = Coupon::where('code', $code)->first();

        if (!$coupon) {
            return ValidationResult::invalid('invalid_code', 'Invalid coupon code');
        }

        if (!$coupon->is_active) {
            return ValidationResult::invalid('inactive', 'Coupon is inactive');
        }

        if ($coupon->expires_at && $coupon->expires_at->isPast()) {
            return ValidationResult::invalid('expired', 'Coupon has expired');
        }

        if ($coupon->starts_at && $coupon->starts_at->isFuture()) {
            return ValidationResult::invalid('not_started', 'Coupon is not active yet');
        }

        if ($coupon->max_uses && $coupon->current_uses >= $coupon->max_uses) {
            return ValidationResult::invalid('usage_limit', 'Usage limit reached');
        }

        if ($coupon->max_uses_per_user) {
            $userUses = $coupon->usages()->where('user_id', $user->id)->count();
            if ($userUses >= $coupon->max_uses_per_user) {
                return ValidationResult::invalid('already_used', 'Already used this coupon');
            }
        }

        if ($coupon->min_purchase && $cart->getSubtotal() < $coupon->min_purchase) {
            return ValidationResult::invalid('min_not_met', 'Minimum purchase not met');
        }

        if ($coupon->product_ids && !$this->isProductEligible($coupon, $cart)) {
            return ValidationResult::invalid('product_restricted', 'Not valid for this product');
        }

        if ($coupon->plan_ids && !$this->isPlanEligible($coupon, $cart)) {
            return ValidationResult::invalid('plan_restricted', 'Not valid for this plan');
        }

        return ValidationResult::valid($coupon);
    }
}
```

## Duration Implementation

```php
class CouponDurationHandler
{
    public function shouldApply(Coupon $coupon, Subscription $subscription): bool
    {
        return match ($coupon->duration) {
            'once' => $this->isFirstPayment($subscription),
            'forever' => true,
            'repeating' => $this->isWithinCycleLimit($coupon, $subscription),
            default => false,
        };
    }

    private function isFirstPayment(Subscription $subscription): bool
    {
        return $subscription->payments()->count() === 0;
    }

    private function isWithinCycleLimit(Coupon $coupon, Subscription $subscription): bool
    {
        $appliedCount = $coupon->usages()
            ->where('subscription_id', $subscription->id)
            ->count();

        return $appliedCount < $coupon->duration_cycles;
    }
}
```

```php
class CouponROI
{
    public function calculate(Coupon $coupon, Carbon $start, Carbon $end): ROIResult
    {
        $usages = $coupon->usages()
            ->whereBetween('created_at', [$start, $end])
            ->get();

        $totalDiscount = $usages->sum('discount_amount');
        $totalRevenue = $usages->sum('order.total');
        $newUsers = $usages->where('user.is_new', true)->count();

        return new ROIResult(
            total_discount: $totalDiscount,
            total_revenue: $totalRevenue,
            roi: ($totalRevenue - $totalDiscount) / $totalDiscount * 100,
            new_users_acquired: $newUsers,
            cost_per_acquisition: $totalDiscount / max($newUsers, 1),
        );
    }
}
```

```php
class CouponStackingValidator
{
    public function canStack(Coupon $newCoupon, array $appliedCoupons): bool
    {
        // Rule 1: Single coupon per order (current)
        if (count($appliedCoupons) > 0) {
            return false;
        }

        // Rule 2: Check category conflicts
        foreach ($appliedCoupons as $applied) {
            if ($newCoupon->category === $applied->category) {
                return false;
            }
        }

        // Rule 3: Check maximum discount cap
        $currentDiscount = array_sum(array_column($appliedCoupons, 'discount_amount'));
        $newDiscount = $this->calculateDiscount($newCoupon);
        if ($currentDiscount + $newDiscount > $this->getMaxDiscountCap()) {
            return false;
        }

        return true;
    }
}
```
