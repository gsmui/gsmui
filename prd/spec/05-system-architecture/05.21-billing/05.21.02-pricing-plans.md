```php
// app/Services/PricingAnalytics.php
class PricingAnalytics
{
    public function trackPageView(User $user, string $variant): void
    {
        analytics()->track('pricing_page_view', [
            'user_id' => $user?->id,
            'variant' => $variant,
            'timestamp' => now()->toIso8601String(),
        ]);
    }

    public function trackPlanSelected(User $user, string $plan, string $variant): void
    {
        analytics()->track('pricing_plan_selected', [
            'user_id' => $user?->id,
            'plan' => $plan,
            'variant' => $variant,
        ]);
    }

    public function trackCheckoutStarted(User $user, string $plan, string $variant): void
    {
        analytics()->track('pricing_checkout_started', [
            'user_id' => $user?->id,
            'plan' => $plan,
            'variant' => $variant,
        ]);
    }

    public function trackConversion(User $user, string $plan, string $variant, int $amount): void
    {
        analytics()->track('pricing_converted', [
            'user_id' => $user->id,
            'plan' => $plan,
            'variant' => $variant,
            'amount' => $amount,
        ]);
    }

    public function getConversionRates(string $variant, Carbon $start, Carbon $end): array
    {
        $views = analytics()->count('pricing_page_view', $start, $end, ['variant' => $variant]);
        $checkouts = analytics()->count('pricing_checkout_started', $start, $end, ['variant' => $variant]);
        $conversions = analytics()->count('pricing_converted', $start, $end, ['variant' => $variant]);

        return [
            'views' => $views,
            'checkouts' => $checkouts,
            'conversions' => $conversions,
            'checkout_rate' => $views > 0 ? round(($checkouts / $views) * 100, 2) : 0,
            'conversion_rate' => $checkouts > 0 ? round(($conversions / $checkouts) * 100, 2) : 0,
        ];
    }
}
```

# Plan Limits Enforcement

## Usage Tracking Service

```php
// app/Services/UsageLimitService.php
class UsageLimitService
{
    public function canPerformAction(User $user, string $action, int $quantity = 1): UsageCheck
    {
        $plan = $user->subscription?->plan;
        $limit = $plan?->getLimit($action);

        if ($limit === null) {
            return new UsageCheck(allowed: true, remaining: null);
        }

        if ($limit === -1) {
            return new UsageCheck(allowed: true, remaining: -1); // Unlimited
        }

        $current = $this->getCurrentUsage($user, $action);
        $remaining = $limit - $current;

        return new UsageCheck(
            allowed: $remaining >= $quantity,
            remaining: $remaining,
            limit: $limit,
            current: $current,
            resets_at: $this->getResetTime($action),
        );
    }

    private function getCurrentUsage(User $user, string $action): int
    {
        return match ($action) {
            'projects' => $user->projects()->count(),
            'downloads_per_month' => $user->downloads()
                ->where('created_at', '>=', now()->startOfMonth())
                ->count(),
            'api_calls_per_day' => $user->apiTokens()->sum('calls_today'),
            'storage_mb' => (int) ($user->files()->sum('size') / 1024 / 1024),
            default => 0,
        };
    }

    private function getResetTime(string $action): ?Carbon
    {
        return match ($action) {
            'downloads_per_month' => now()->startOfMonth()->addMonth(),
            'api_calls_per_day' => now()->startOfDay()->addDay(),
            default => null,
        };
    }
}

class UsageCheck
{
    public function __construct(
        public readonly bool $allowed,
        public readonly ?int $remaining,
        public readonly ?int $limit = null,
        public readonly ?int $current = null,
        public readonly ?Carbon $resets_at = null,
    ) {}
}
```

## Limit Enforcement Middleware

```php
// app/Http/Middleware/EnforceUsageLimit.php
class EnforceUsageLimit
{
    public function handle(Request $request, Closure $next, string $action): Response
    {
        $user = $request->user();

        if (!$user) {
            return redirect()->route('login');
        }

        $check = app(UsageLimitService::class)->canPerformAction($user, $action);

        if (!$check->allowed) {
            if ($request->expectsJson()) {
                return response()->json([
                    'error' => 'Usage limit reached',
                    'limit' => $check->limit,
                    'current' => $check->current,
                    'resets_at' => $check->resets_at?->toIso8601String(),
                    'upgrade_url' => route('pricing'),
                ], 403);
            }

            return redirect()->route('pricing')
                ->with('error', "You've reached your {$action} limit. Please upgrade your plan.");
        }

        return $next($request);
    }
}
```

# Plan Feature Gating

## Feature Gate Service

```php
// app/Services/FeatureGateService.php
class FeatureGateService
{
    public function hasFeature(User $user, string $feature): bool
    {
        $plan = $user->subscription?->plan;

        if (!$plan) {
            return $this->getDefaultPlan()->includesFeature($feature);
        }

        return $plan->includesFeature($feature);
    }

    public function getFeatureValue(User $user, string $feature): mixed
    {
        $plan = $user->subscription?->plan;

        if (!$plan) {
            return $this->getDefaultPlan()->features[$feature] ?? null;
        }

        return $plan->features[$feature] ?? null;
    }

    public function getUpgradePlan(User $user, string $feature): ?Plan
    {
        return Plan::active()
            ->where('price', '>', $user->subscription?->plan?->price ?? 0)
            ->whereJsonContains('features', $feature)
            ->orderBy('price')
            ->first();
    }

    private function getDefaultPlan(): Plan
    {
        return Plan::where('slug', 'free')->firstOrFail();
    }
}
```

```php
// app/Services/PricingExperimentService.php
class PricingExperimentService
{
    public function getVariant(string $experimentKey, ?User $user = null): ?string
    {
        $experiment = config("pricing-experiments.experiments.{$experimentKey}");

        if (!$experiment) return null;

        // Check if user is already assigned
        if ($user) {
            $existing = $user->experiment_assignments
                ->where('experiment_key', $experimentKey)
                ->first();

            if ($existing) {
                return $existing->variant;
            }
        }

        // Assign new variant
        $variant = $this->assignVariant($experiment);

        if ($user) {
            $user->experiment_assignments()->create([
                'experiment_key' => $experimentKey,
                'variant' => $variant,
            ]);
        }

        return $variant;
    }

    private function assignVariant(array $experiment): string
    {
        $rand = mt_rand(1, 100);
        $cumulative = 0;

        foreach ($experiment['variants'] as $key => $config) {
            $cumulative += $config['weight'];
            if ($rand <= $cumulative) {
                return $key;
            }
        }

        return array_key_first($experiment['variants']);
    }

    public function trackConversion(string $experimentKey, string $event, User $user): void
    {
        $assignment = $user->experiment_assignments
            ->where('experiment_key', $experimentKey)
            ->first();

        if (!$assignment) return;

        ExperimentEvent::create([
            'experiment_key' => $experimentKey,
            'variant' => $assignment->variant,
            'event' => $event,
            'user_id' => $user->id,
            'metadata' => ['revenue' => $user->orders()->sum('total')],
        ]);
    }
}
```

```php
// app/Services/PlanMigrationService.php
class PlanMigrationService
{
    public function calculateUpgrade(Subscription $current, Plan $newPlan): UpgradePreview
    {
        $currentPlan = $current->plan;

        // Proration calculation
        $daysRemaining = now()->diffInDays($current->current_period_end);
        $totalDays = Carbon::parse($current->current_period_start)
            ->diffInDays($current->current_period_end);

        $currentDailyRate = $currentPlan->price / $totalDays;
        $newDailyRate = $newPlan->price / $totalDays;

        $creditForRemaining = (int) round($currentDailyRate * $daysRemaining);
        $chargeForNew = (int) round($newDailyRate * $daysRemaining);
        $prorationAmount = $chargeForNew - $creditForRemaining;

        return new UpgradePreview(
            currentPlan: $currentPlan,
            newPlan: $newPlan,
            credit: $creditForRemaining,
            charge: $chargeForNew,
            proration: $prorationAmount,
            effectiveDate: $current->current_period_end,
            immediateAccess: $prorationAmount > 0,
        );
    }

    public function executeUpgrade(Subscription $subscription, Plan $newPlan, int $proration): void
    {
        $stripe = new \Stripe\StripeClient(config('services.stripe.secret'));

        // Update Stripe subscription
        $stripe->subscriptions->update($subscription->stripe_id, [
            'items' => [
                [
                    'id' => $subscription->stripe_item_id,
                    'deleted' => true,
                ],
                [
                    'price' => $newPlan->stripe_price_id,
                ],
            ],
            'proration_behavior' => 'create_prorations',
            'metadata' => [
                'upgrade_from' => $subscription->plan->slug,
                'upgrade_to' => $newPlan->slug,
            ],
        ]);

        // Update local record
        $subscription->update([
            'plan_id' => $newPlan->id,
            'stripe_price_id' => $newPlan->stripe_price_id,
        ]);

        // Log the change
        SubscriptionChange::create([
            'subscription_id' => $subscription->id,
            'old_plan_id' => $subscription->plan_id,
            'new_plan_id' => $newPlan->id,
            'proration_amount' => $proration,
            'type' => 'upgrade',
        ]);

        event(new SubscriptionUpgraded($subscription, $newPlan, $proration));
    }
}

    {
        $amount = abs($this->proration);
        $prefix = $this->proration > 0 ? 'Charge' : 'Credit';

    }
```

## Recommendation Service

```php
// app/Services/PlanRecommendationService.php
class PlanRecommendationService
{
    public function recommend(User $user): PlanRecommendation
    {
        $usage = $this->analyzeUsage($user);
        $profile = $this->buildProfile($user);

        $plans = Plan::active()->ordered()->get();

        $scores = $plans->map(function ($plan) use ($usage, $profile) {
            $score = 0;

            // Feature match score (0-40)
            $score += $this->featureMatchScore($plan, $usage);

            // Usage fit score (0-30)
            $score += $this->usageFitScore($plan, $usage);

            // Budget fit score (0-20)
            $score += $this->budgetFitScore($plan, $profile);

            // Upgrade momentum score (0-10)
            $score += $this->momentumScore($plan, $user);

            return ['plan' => $plan, 'score' => $score];
        });

        $recommended = $scores->sortByDesc('score')->first();

        return new PlanRecommendation(
            plan: $recommended['plan'],
            score: $recommended['score'],
            reasons: $this->generateReasons($recommended['plan'], $usage, $profile),
            alternatives: $scores->slice(1, 2)->pluck('plan'),
        );
    }

    private function analyzeUsage(User $user): array
    {
        return [
            'components_used' => $user->projects()->with('components')->get()
                ->pluck('components')->flatten()->unique()->count(),
            'projects_count' => $user->projects()->count(),
            'downloads_this_month' => $user->downloads()
                ->where('created_at', '>=', now()->startOfMonth())
                ->count(),
            'api_calls_today' => $user->apiTokens()->sum('calls_today'),
            'storage_used_mb' => $user->files()->sum('size') / 1024 / 1024,
            'team_members' => $user->workspace?->members()->count() ?? 1,
            'build_minutes_this_month' => $user->builds()
                ->where('created_at', '>=', now()->startOfMonth())
                ->sum('duration_minutes'),
        ];
    }

    private function featureMatchScore(Plan $plan, array $usage): int
    {
        $score = 0;

        if ($usage['components_used'] > 20 && $plan->includesFeature('full_components')) {
            $score += 20;
        }

        if ($usage['projects_count'] > 5 && $plan->getLimit('projects') >= $usage['projects_count']) {
            $score += 10;
        }

        if ($usage['team_members'] > 1 && $plan->getLimit('team_members') >= $usage['team_members']) {
            $score += 10;
        }

        return min($score, 40);
    }

    private function generateReasons(Plan $plan, array $usage, array $profile): array
    {
        $reasons = [];

        if ($usage['components_used'] > 15) {
            $reasons[] = "You're using {$usage['components_used']} components — {$plan->name} gives you access to all 70+";
        }

        if ($usage['projects_count'] > 3) {
            $reasons[] = "With {$usage['projects_count']} projects, {$plan->name}'s {$plan->getLimit('projects')} project limit is ideal";
        }

        if ($usage['team_members'] > 1) {
            $reasons[] = "Your team of {$usage['team_members']} can collaborate with {$plan->name}'s team features";
        }

        return $reasons;
    }
}

class PlanRecommendation
{
    public function __construct(
        public Plan $plan,
        public int $score,
        public array $reasons,
        public Collection $alternatives,
    ) {}
}
```

```php
            $churn = $this->getChurnRate($plan);
            $ltv = $this->calculateLTV($plan, $churn);

            $analysis[$plan->slug] = [
                'subscribers' => $subscribers,
                'mrr' => $mrr,
                'churn_rate' => $churn,
                'ltv' => $ltv,
                'revenue_per_subscriber' => $subscribers > 0 ? round($mrr / $subscribers, 2) : 0,
                'recommended_price' => $this->getRecommendedPrice($plan, $ltv, $churn),
            ];
        }

        return $analysis;
    }

    private function getChurnRate(Plan $plan): float
    {
        $total = $plan->subscriptions()->count();
        $canceled = $plan->subscriptions()->where('status', 'canceled')->count();

        return $total > 0 ? round(($canceled / $total) * 100, 2) : 0;
    }

    private function calculateLTV(Plan $plan, float $churnRate): float
    {
        $monthlyRevenue = $plan->price / 100;
        $avgLifetimeMonths = $churnRate > 0 ? 100 / $churnRate : 12;

        return round($monthlyRevenue * $avgLifetimeMonths, 2);
    }

    private function getRecommendedPrice(Plan $plan, float $ltv, float $churn): int
    {
        $currentPrice = $plan->price;

        if ($churn > 10) {
            // High churn - consider lowering price
            return (int) round($currentPrice * 0.9);
        } elseif ($churn < 3) {
            // Low churn - room to increase price
            return (int) round($currentPrice * 1.1);
        }

        return $currentPrice;
    }
}
```

```php
    {
```

```php
            return response()->json([
                'error' => 'feature_not_in_plan',
                'message' => "The {$feature} feature is not available on the {$planSlug} plan.",
                'upgrade_url' => route('pricing.index'),
                'required_plan' => self::findLowestPlanWith($feature),
            ], 403);
        }

        // Check usage limit
        $limit = FeatureMatrix::limit($planSlug, $feature);
        if ($limit !== -1 && $limit > 0) {
            $usage = $this->getUsage($user, $feature);
            if ($usage >= $limit) {
                return response()->json([
                    'error' => 'feature_limit_reached',
                    'message' => "You have reached the {$limit} {$feature} limit.",
                    'usage' => $usage,
                    'limit' => $limit,
                    'upgrade_url' => route('pricing.index'),
                ], 429);
            }
        }

        return $next($request);
    }

    private function getUsage($user, string $feature): int
    {
        $modelMap = [
            'projects.count' => \App\Models\Project::class,
            'webhooks.count' => \App\Models\Webhook::class,
            'domains.count' => \App\Models\Domain::class,
            'team.members' => \App\Models\TeamMember::class,
        ];

        $model = $modelMap[$feature] ?? null;
        if ($model) {
            return $model::where('user_id', $user->id)->count();
        }

        // For API calls, check Redis counter
        if ($feature === 'api.calls_per_minute') {
            return (int) \Redis::get("api_calls:{$user->id}:" . date('Y-m-d-H-i'));
```

### Algorithm and Data Model

```php
<?php
// app/Services/PlanRecommendationEngine.php
namespace App\Services;

use App\Models\User;
use App\Models\UsageSnapshot;

class PlanRecommendationEngine
{
    /**
     * Analyze user usage patterns and recommend the optimal plan.
     */
    public function recommend(User $user): array
    {
        $usage = $this->analyzeUsage($user);
        $currentPlan = $user->currentSubscription?->plan?->slug ?? 'free';
        $projected = $this->projectUsage($usage, $user);
        $candidates = $this->findCandidatePlans($projected);

        $bestFit = $this->scoreCandidates($candidates, $projected, $user);

        return [
            'current_plan' => $currentPlan,
            'usage_analysis' => $usage,
            'projected_usage' => $projected,
            'recommended_plan' => $bestFit['plan'],
            'confidence_score' => $bestFit['confidence'],
            'reasons' => $bestFit['reasons'],
            'estimated_monthly_savings' => $this->calculateSavings($currentPlan, $bestFit['plan'], $projected),
            'upgrade_benefits' => FeatureMatrix::diff($currentPlan, $bestFit['plan']),
        ];
    }

    private function analyzeUsage(User $user): array
    {
        $snapshots = UsageSnapshot::where('user_id', $user->id)
            ->where('recorded_at', '>=', now()->subDays(90))
            ->orderBy('recorded_at')
            ->get();

        return [
            'ai_chat_messages_total' => $snapshots->sum('ai_chat_messages'),
            'ai_images_generated' => $snapshots->sum('ai_images_generated'),
            'storage_bytes_peak' => $snapshots->max('storage_bytes_used'),
            'api_calls_total' => $snapshots->sum('api_calls'),
            'projects_count' => $snapshots->max('projects_count'),
            'team_members_count' => $snapshots->max('team_members_count'),
            'active_days' => $snapshots->pluck('recorded_at->date')->unique()->count(),
            'avg_daily_api_calls' => $snapshots->sum('api_calls') / max(1, $snapshots->count()),
            'growth_rate' => $this->calculateGrowthRate($snapshots),
        ];
    }

    private function projectUsage(array $usage, User $user): array
    {
        $daysInPeriod = max(1, $usage['active_days']);
        $monthlyProjection = 30 / $daysInPeriod;

        return [
            'monthly_ai_chat' => (int) ($usage['ai_chat_messages_total'] * $monthlyProjection),
            'monthly_ai_images' => (int) ($usage['ai_images_generated'] * $monthlyProjection),
            'monthly_api_calls' => (int) ($usage['api_calls_total'] * $monthlyProjection),
            'storage_bytes_projected' => (int) ($usage['storage_bytes_peak'] * (1 + $usage['growth_rate'])),
            'projects_projected' => (int) ($usage['projects_count'] * (1 + $usage['growth_rate'])),
        ];
    }

    private function findCandidatePlans(array $projected): array
    {
        $candidates = [];
        foreach (FeatureMatrix::PLAN_FEATURES as $plan => $features) {
            $fits = true;

            if ($projected['monthly_api_calls'] > $features['api.calls_per_minute'] * 60 * 24 * 30) {
                $fits = false;
            }
            if ($features['storage.bytes'] > 0 && $projected['storage_bytes_projected'] > $features['storage.bytes']) {
                $fits = false;
            }
            if ($features['projects.count'] > 0 && $projected['projects_projected'] > $features['projects.count']) {
                $fits = false;
            }

            if ($fits) {
                $candidates[] = $plan;
            }
        }

        return $candidates;
    }

    private function scoreCandidates(array $candidates, array $projected, User $user): array
    {
        $scores = [];
        foreach ($candidates as $plan) {
            $score = 0;
            $reasons = [];
            $features = FeatureMatrix::get($plan);

            // Cost efficiency score
            $price = Plan::where('slug', $plan)->first()->price ?? 0;
            if ($price > 0) {
                $costPerApiCall = $price / max(1, $projected['monthly_api_calls']);
                $score += max(0, 100 - $costPerApiCall * 10000);
            }

            // Growth headroom score
            if ($features['projects.count'] > 0) {
                $headroom = ($features['projects.count'] - $projected['projects_projected']) / $features['projects.count'];
                $score += $headroom * 30;
                $reasons[] = "Project capacity: {$headroom}% headroom";
            }

            // Feature completeness score
            if ($features['ai.chat'] && $projected['monthly_ai_chat'] > 0) {
                $score += 20;
                $reasons[] = 'AI chat included';
            }
            if ($features['export.data']) {
                $score += 10;
                $reasons[] = 'Data export available';
            }

            $scores[$plan] = [
                'plan' => $plan,
                'score' => $score,
                'confidence' => min(100, (int) $score),
                'reasons' => $reasons,
            ];
        }

        usort($scores, fn($a, $b) => $b['score'] <=> $a['score']);

        return $scores[0] ?? ['plan' => 'free', 'score' => 0, 'confidence' => 0, 'reasons' => []];
    }

    private function calculateGrowthRate($snapshots): float
    {
        if ($snapshots->count() < 2) {
            return 0.0;
        }

        $mid = (int) ($snapshots->count() / 2);
        $firstHalf = $snapshots->take($mid)->avg('api_calls');
        $secondHalf = $snapshots->skip($mid)->avg('api_calls');

        if ($firstHalf == 0) {
            return 1.0;
        }

        return max(0, ($secondHalf - $firstHalf) / $firstHalf);
    }

    private function calculateSavings(string $currentPlan, string $recommendedPlan, array $projected): float
    {
        $currentPrice = Plan::where('slug', $currentPlan)->first()->price ?? 0;
        $recommendedPrice = Plan::where('slug', $recommendedPlan)->first()->price ?? 0;

        return $currentPrice - $recommendedPrice;
    }
}
```

```php
            if ($bucket < $cumulative) {
                return $variant;
            }
        }

        return $config['variants'][0];
```

```php
<?php
// app/Services/PlanMigrationService.php
namespace App\Services;

use App\Models\Plan;
use App\Models\Subscription;
use App\Models\User;
use App\Events\PlanUpgraded;
use App\Events\PlanDowngraded;
use Illuminate\Support\Facades\DB;

class PlanMigrationService
{
    public function calculateUpgrade(Subscription $subscription, Plan $newPlan): array
    {
        $currentPlan = $subscription->plan;
        $daysRemaining = $subscription->current_period_end->diffInDays(now());
        $dailyProration = $currentPlan->price / 30;

        $credit = $dailyProration * $daysRemaining;
        $newPlanDaily = $newPlan->price / 30;
        $charge = max(0, $newPlan->price - $credit);

        return [
            'currentPlan' => $currentPlan,
            'newPlan' => $newPlan,
            'days_remaining' => $daysRemaining,
            'credit' => round($credit, 2),
            'charge' => round($charge, 2),
            'effective_date' => now(),
            'next_billing_date' => $subscription->current_period_end,
        ];
    }

    public function executeUpgrade(User $user, Plan $newPlan): Subscription
    {
        return DB::transaction(function () use ($user, $newPlan) {
            $subscription = $user->currentSubscription;

            if (! $subscription) {
                return $this->createSubscription($user, $newPlan);
            }

            $preview = $this->calculateUpgrade($subscription, $newPlan);

            $subscription->update([
                'plan_id' => $newPlan->id,
                'status' => 'active',
                'upgraded_at' => now(),
            ]);

            if ($preview['charge'] > 0) {
                $this->processPayment($user, $preview['charge'], "Upgrade to {$newPlan->name}");
            }

            event(new PlanUpgraded($user, $subscription->plan, $newPlan, $preview));

            return $subscription;
        });
    }

    public function executeDowngrade(User $user, Plan $newPlan): Subscription
    {
        return DB::transaction(function () use ($user, $newPlan) {
            $subscription = $user->currentSubscription;
            $oldPlan = $subscription->plan;

            $subscription->update([
                'plan_id' => $newPlan->id,
                'downgrade_scheduled_at' => $subscription->current_period_end,
                'status' => 'active',
            ]);

            event(new PlanDowngraded($user, $oldPlan, $newPlan));

            return $subscription;
        });
    }

    {
        return Subscription::create([
            'user_id' => $user->id,
            'plan_id' => $plan->id,
            'status' => 'active',
            'current_period_start' => now(),
            'current_period_end' => now()->addMonth(),
        ]);
    }
```

### Health Check Endpoint

```php
<?php
// app/Http/Controllers/PricingHealthController.php
namespace App\Http\Controllers;

use App\Models\Plan;
use App\Models\Subscription;
use Illuminate\Http\JsonResponse;
use Illuminate\Support\Facades\DB;

class PricingHealthController extends Controller
{
    public function check(): JsonResponse
    {
        $checks = [
            'database' => $this->checkDatabase(),
            'stripe_sync' => $this->checkStripeSync(),
            'plan_integrity' => $this->checkPlanIntegrity(),
            'subscription_consistency' => $this->checkSubscriptionConsistency(),
        ];

        $healthy = ! in_array(false, array_column($checks, 'healthy'));

        return response()->json([
            'status' => $healthy ? 'healthy' : 'degraded',
            'checks' => $checks,
            'timestamp' => now()->toISOString(),
        ], $healthy ? 200 : 503);
    }

    private function checkDatabase(): array
    {
        try {
            DB::select('SELECT 1');
            return ['healthy' => true, 'latency_ms' => 0];
        } catch (\Exception $e) {
            return ['healthy' => false, 'error' => $e->getMessage()];
        }
    }

    private function checkStripeSync(): array
    {
        $plansWithoutStripe = Plan::where('is_active', true)
            ->whereNull('stripe_price_id_monthly')
            ->count();

        return [
            'healthy' => $plansWithoutStripe === 0,
            'unsynced_plans' => $plansWithoutStripe,
        ];
    }

    private function checkPlanIntegrity(): array
    {
        $orphaned = DB::table('subscriptions')
            ->leftJoin('plans', 'subscriptions.plan_id', '=', 'plans.id')
            ->whereNull('plans.id')
            ->count();

        return [
            'healthy' => $orphaned === 0,
            'orphaned_subscriptions' => $orphaned,
        ];
    }

    private function checkSubscriptionConsistency(): array
    {
        $pastDueNoGrace = Subscription::where('status', 'past_due')
            ->where('current_period_end', '<', now()->subDays(3))
            ->count();

        return [
            'healthy' => $pastDueNoGrace === 0,
            'stale_past_due' => $pastDueNoGrace,
        ];
    }
}
```

```php
// app/Services/CurrencyService.php
namespace App\Services;

class CurrencyService
{
    const SUPPORTED_CURRENCIES = [
        'USD' => ['symbol' => '$', 'decimal_places' => 2],
        'EUR' => ['symbol' => '€', 'decimal_places' => 2],
        'GBP' => ['symbol' => '£', 'decimal_places' => 2],
        'CAD' => ['symbol' => 'C$', 'decimal_places' => 2],
        'AUD' => ['symbol' => 'A$', 'decimal_places' => 2],
        'JPY' => ['symbol' => '¥', 'decimal_places' => 0],
        'BRL' => ['symbol' => 'R$', 'decimal_places' => 2],
        'INR' => ['symbol' => '₹', 'decimal_places' => 2],
    ];

    public function convert(Plan $plan, string $targetCurrency): array
    {
        $basePrice = (int) ($plan->price_monthly * 100); // cents
        $baseCurrency = new Currency('USD');
        $target = new Currency($targetCurrency);

        $converter = new Converter($this->getExchangeRates());
        $money = new Money($basePrice, $baseCurrency);
        $converted = $converter->convert($money, $target, 1);

        return [
            'amount' => $converted->getAmount(),
            'formatted' => $this->formatMoney($converted, $targetCurrency),
            'currency' => $targetCurrency,
        ];
    }

    private function getExchangeRates(): array
    {
        // In production, fetch from an API
        return [
            'USD' => 1.0,
            'EUR' => 0.92,
            'GBP' => 0.79,
            'CAD' => 1.36,
            'AUD' => 1.53,
            'JPY' => 149.5,
            'BRL' => 4.97,
            'INR' => 83.12,
        ];
    }

    private function formatMoney(Money $money, string $currency): string
    {
        $info = self::SUPPORTED_CURRENCIES[$currency];
        $amount = $money->getAmount() / pow(10, $info['decimal_places']);

        return $info['symbol'] . number_format($amount, $info['decimal_places']);
    }
}
```

```sql
-- Essential indexes for billing performance
CREATE INDEX idx_subscriptions_user_status ON subscriptions(user_id, status);

CREATE INDEX idx_usage_snapshots_user_date ON usage_snapshots(user_id, recorded_at);

-- Materialized view for MRR calculation (refreshed hourly)
CREATE MATERIALIZED VIEW mrr_by_plan AS
SELECT
    p.slug as plan_slug,
    p.price_monthly,
    COUNT(s.id) as active_subscriptions,
    SUM(p.price_monthly) as mrr
FROM plans p
LEFT JOIN subscriptions s ON p.id = s.plan_id
    AND s.status = 'active'
    AND s.current_period_end > NOW()
WHERE p.is_active = true
GROUP BY p.id, p.slug, p.price_monthly;
```
