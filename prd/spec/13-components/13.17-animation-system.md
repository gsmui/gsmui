# 13.17 Animation System — GsmUi Animation Specification

## Overview

The GsmUi animation system provides consistent, performant, and accessible motion across all components. Animations are built on CSS transitions/animations with JavaScript orchestration for complex sequences.

## Design Principles

1. **Respect Reduced Motion** - All animations honor `prefers-reduced-motion`
2. **Performance First** - Use transform/opacity only; avoid layout-triggering properties
3. **Consistent Timing** - Standardized durations and easings
4. **Purposeful Motion** - Animations communicate state changes, not decoration
5. **Interruptible** - Animations can be cancelled/reversed mid-flight

## JavaScript Animation Controller

```javascript
// gsmui-animation.js
class AnimationController {
    constructor(element) {
        this.element = element;
        this.animations = new Map();
        this.reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    }
    
    // Play a keyframe animation
    animate(keyframes, options = {}) {
        if (this.reducedMotion) {
            options.duration = 0;
        }
        
        const animation = this.element.animate(keyframes, {
            duration: options.duration ?? 200,
            easing: options.easing ?? 'var(--gsmui-ease-out)',
            fill: options.fill ?? 'forwards',
            ...options,
        });
        
        const id = Symbol('animation');
        this.animations.set(id, animation);
        
        animation.finished.finally(() => {
            this.animations.delete(id);
        });
        
        return {
            id,
            cancel: () => animation.cancel(),
            pause: () => animation.pause(),
            play: () => animation.play(),
            reverse: () => animation.reverse(),
            finished: animation.finished,
        };
    }
    
    // Fade in
    fadeIn(duration = 200) {
        return this.animate([
            { opacity: 0 },
            { opacity: 1 }
        ], { duration, easing: 'var(--gsmui-ease-out)' });
    }
    
    // Fade out
    fadeOut(duration = 150) {
        return this.animate([
            { opacity: 1 },
            { opacity: 0 }
        ], { duration, easing: 'var(--gsmui-ease-in)' });
    }
    
    // Slide in from direction
    slideIn(direction = 'up', duration = 200) {
        const transforms = {
            up: 'translateY(-100%)',
            down: 'translateY(100%)',
            left: 'translateX(-100%)',
            right: 'translateX(100%)',
        };
        
        return this.animate([
            { opacity: 0, transform: transforms[direction] },
            { opacity: 1, transform: 'translate(0)' }
        ], { duration, easing: 'var(--gsmui-ease-spring)' });
    }
    
    // Scale in
    scaleIn(duration = 200) {
        return this.animate([
            { opacity: 0, transform: 'scale(0.95)' },
            { opacity: 1, transform: 'scale(1)' }
        ], { duration, easing: 'var(--gsmui-ease-spring)' });
    }
    
    // Shake (for errors)
    shake(duration = 300) {
        return this.animate([
            { transform: 'translateX(0)' },
            { transform: 'translateX(-8px)' },
            { transform: 'translateX(8px)' },
            { transform: 'translateX(-8px)' },
            { transform: 'translateX(0)' }
        ], { duration, easing: 'var(--gsmui-ease-out)' });
    }
    
    // Cancel all
    cancelAll() {
        for (const animation of this.animations.values()) {
            animation.cancel();
        }
        this.animations.clear();
    }
}

// Usage
const controller = new AnimationController(element);
await controller.fadeIn().finished;
await controller.slideIn('up').finished;
```

```javascript
async function openModal(modal) {
    const backdrop = modal.querySelector('.gsmui-modal-backdrop');
```

### Staggered Animations

```javascript
function staggerChildren(container, selector, animationFn, delay = 50) {
    const children = container.querySelectorAll(selector);
    const controllers = Array.from(children).map((child, i) => {
        const controller = new AnimationController(child);
        return new Promise(resolve => {
            setTimeout(() => {
                animationFn(controller).finished.then(resolve);
            }, i * delay);
        });
    });
    
    return Promise.all(controllers);
}
```

## Advanced Animation Patterns

### Scroll-Triggered Animations

```javascript
class ScrollAnimator {
    constructor() {
        this.observers = new Map();
        this.setupIntersectionObserver();
    }
    
    setupIntersectionObserver() {
        this.observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    this.animateElement(entry.target);
                }
            });
        }, {
            threshold: 0.1,
            rootMargin: '0px 0px -50px 0px',
        });
    }
    
    observe(element, animation = 'fade-in-up') {
        element.dataset.animation = animation;
        element.style.opacity = '0';
        this.observer.observe(element);
    }
    
    animateElement(element) {
        const animation = element.dataset.animation;
        const controller = new AnimationController(element);
        
        const animations = {
            'fade-in-up': () => controller.slideIn('up', 300),
            'fade-in-down': () => controller.slideIn('down', 300),
            'fade-in-left': () => controller.slideIn('left', 300),
            'fade-in-right': () => controller.slideIn('right', 300),
            'scale-in': () => controller.scaleIn(300),
            'fade-in': () => controller.fadeIn(300),
        };
        
        (animations[animation] || animations['fade-in'])();
        this.observer.unobserve(element);
    }
    
    destroy() {
        this.observer.disconnect();
    }
}
```

### Parallax Effects

```javascript
class ParallaxEffect {
    constructor(element, options = {}) {
        this.element = element;
        this.options = {
            speed: 0.5,
            direction: 'vertical',
            ...options,
        };
        
        this.boundHandleScroll = this.handleScroll.bind(this);
        this.setup();
    }
    
    setup() {
        window.addEventListener('scroll', this.boundHandleScroll, { passive: true });
        this.element.style.willChange = 'transform';
    }
    
    handleScroll() {
        const scrolled = window.pageYOffset;
        const elementTop = this.element.offsetTop;
        const elementHeight = this.element.offsetHeight;
        
        if (scrolled + window.innerHeight > elementTop && 
            scrolled < elementTop + elementHeight) {
            
            const translateY = (scrolled - elementTop) * this.options.speed;
            
            if (this.options.direction === 'vertical') {
                this.element.style.transform = `translateY(${translateY}px)`;
            } else {
                this.element.style.transform = `translateX(${translateY}px)`;
            }
        }
    }
    
    destroy() {
        window.removeEventListener('scroll', this.boundHandleScroll);
        this.element.style.willChange = '';
        this.element.style.transform = '';
    }
}
```

### Staggered List Animations

```javascript
class StaggeredAnimator {
    constructor(container, options = {}) {
        this.container = container;
        this.options = {
            itemSelector: ':scope > *',
            animation: 'slide-in',
            delay: 50,
            duration: 300,
            ...options,
        };
    }
    
    animate() {
        const items = this.container.querySelectorAll(this.options.itemSelector);
        
        items.forEach((item, index) => {
            item.style.opacity = '0';
            item.style.transform = 'translateY(20px)';
            
            setTimeout(() => {
                const controller = new AnimationController(item);
                
                const animations = {
                    'slide-in': () => controller.slideIn('up', this.options.duration),
                    'fade-in': () => controller.fadeIn(this.options.duration),
                    'scale-in': () => controller.scaleIn(this.options.duration),
                };
                
                (animations[this.options.animation] || animations['slide-in'])();
            }, index * this.options.delay);
        });
    }
    
    reset() {
        const items = this.container.querySelectorAll(this.options.itemSelector);
        
        items.forEach(item => {
            item.style.opacity = '0';
            item.style.transform = 'translateY(20px)';
        });
    }
}
```

## Animation Utilities

### Animation Queue

```javascript
class AnimationQueue {
    constructor() {
        this.queue = [];
        this.running = false;
        this.currentAnimation = null;
    }
    
    add(animation, options = {}) {
        return new Promise((resolve) => {
            this.queue.push({
                animation,
                options,
                resolve,
            });
            
            if (!this.running) {
                this.process();
            }
        });
    }
    
    async process() {
        if (this.queue.length === 0) {
            this.running = false;
            return;
        }
        
        this.running = true;
        const { animation, options, resolve } = this.queue.shift();
        
        this.currentAnimation = await animation();
        
        if (options.wait !== false && this.currentAnimation?.finished) {
            await this.currentAnimation.finished;
        }
        
        resolve();
        this.process();
    }
    
    clear() {
        this.queue = [];
        this.currentAnimation?.cancel();
    }
    
    pause() {
        this.currentAnimation?.pause();
    }
    
    resume() {
        this.currentAnimation?.play();
    }
}
```

```javascript
        this.lastTime = 0;
    }
    
    schedule(element, keyframes, options = {}) {
        const animation = {
            element,
            keyframes,
            options: {
                duration: 200,
                easing: 'ease-out',
                fill: 'forwards',
                ...options,
            },
            startTime: null,
            delay: options.delay || 0,
        };
        
        this.animations.push(animation);
        
        if (!this.frameId) {
            this.start();
        }
        
        return new Promise((resolve) => {
            animation.resolve = resolve;
        });
    }
    
    start() {
        this.lastTime = performance.now();
        this.tick();
    }
    
    tick() {
        const now = performance.now();
        const delta = now - this.lastTime;
        
        this.animations.forEach((animation, index) => {
            if (animation.startTime === null) {
                animation.startTime = now + animation.delay;
            }
            
            if (now >= animation.startTime && !animation.started) {
                animation.animation = animation.element.animate(
                    animation.keyframes,
                    animation.options
                );
                animation.started = true;
                
                animation.animation.finished.then(() => {
                    animation.resolve();
                    const idx = this.animations.indexOf(animation);
                    if (idx > -1) {
                        this.animations.splice(idx, 1);
                    }
                });
            }
        });
        
        if (this.animations.length > 0) {
            this.frameId = requestAnimationFrame(() => this.tick());
        } else {
            this.frameId = null;
        }
    }
    
    cancelAll() {
        this.animations.forEach(animation => {
            animation.animation?.cancel();
            animation.resolve?.();
        });
        this.animations = [];
        
        if (this.frameId) {
            cancelAnimationFrame(this.frameId);
            this.frameId = null;
        }
    }
}
```

### Performance Monitoring

```javascript
class AnimationPerformanceMonitor {
    constructor() {
        this.metrics = {
            frameDrops: 0,
            longFrames: 0,
            averageFPS: 0,
            animations: new Map(),
        };
        
        this.lastFrameTime = performance.now();
        this.frameCount = 0;
        this.fpsHistory = [];
        
        this.setupMonitoring();
    }
    
    setupMonitoring() {
        let lastTime = performance.now();
        let frameCount = 0;
        
        const measureFPS = (currentTime) => {
            frameCount++;
            
            if (currentTime - lastTime >= 1000) {
                const fps = Math.round((frameCount * 1000) / (currentTime - lastTime));
                this.fpsHistory.push(fps);
                
                if (this.fpsHistory.length > 60) {
                    this.fpsHistory.shift();
                }
                
                this.metrics.averageFPS = Math.round(
                    this.fpsHistory.reduce((a, b) => a + b, 0) / this.fpsHistory.length
                );
                
                frameCount = 0;
                lastTime = currentTime;
            }
            
            requestAnimationFrame(measureFPS);
        };
        
        requestAnimationFrame(measureFPS);
    }
    
    trackAnimation(name, element) {
        const start = performance.now();
        
        const track = () => {
            const now = performance.now();
            const frameTime = now - this.lastFrameTime;
            
            if (frameTime > 16.67) {
                this.metrics.frameDrops++;
            }
            
            if (frameTime > 32) {
                this.metrics.longFrames++;
            }
            
            this.lastFrameTime = now;
        };
        
        const frameId = requestAnimationFrame(track);
        
        this.metrics.animations.set(name, {
            element,
            startTime: start,
            frameId,
        });
        
        return {
            stop: () => {
                cancelAnimationFrame(frameId);
                const duration = performance.now() - start;
                this.metrics.animations.delete(name);
                return { duration, name };
            },
        };
    }
    
    getReport() {
        return {
            ...this.metrics,
            animationCount: this.metrics.animations.size,
            dropRate: this.metrics.frameDrops / (this.frameCount || 1),
        };
    }
    
    reset() {
        this.metrics.frameDrops = 0;
        this.metrics.longFrames = 0;
        this.fpsHistory = [];
    }
}
```

### Animation Performance Tips

```javascript
// Good: Use transform and opacity
element.animate([
    { transform: 'translateX(0)', opacity: 1 },
    { transform: 'translateX(100px)', opacity: 0 }
], { duration: 300 });

// Bad: Animating layout properties
element.animate([
    { width: '100px', marginLeft: '0' },
    { width: '200px', marginLeft: '100px' }
], { duration: 300 });

// Use will-change for animations
element.style.willChange = 'transform, opacity';
element.animate([...]);

// Clean up after animation
animation.finished.then(() => {
    element.style.willChange = '';
});

// Use requestAnimationFrame for custom animations
function customAnimation(element, from, to, duration) {
    const start = performance.now();
    
    function tick(now) {
        const elapsed = now - start;
        const progress = Math.min(elapsed / duration, 1);
        
        const eased = easeOutCubic(progress);
        const value = from + (to - from) * eased;
        
        element.style.transform = `translateX(${value}px)`;
        
        if (progress < 1) {
            requestAnimationFrame(tick);
        }
    }
    
    requestAnimationFrame(tick);
}

function easeOutCubic(t) {
    return 1 - Math.pow(1 - t, 3);
}
```

## Animation Best Practices

### Animation Guidelines

#### Do's and Don'ts

#### Do

1. **Use transform and opacity** — These properties are GPU-accelerated
2. **Keep animations short** — 100-300ms is usually sufficient
3. **Use appropriate easing** — Ease-out for entering, ease-in for exiting
4. **Respect reduced motion** — Always check `prefers-reduced-motion`
5. **Test performance** — Use DevTools to monitor frame rate

#### Don't

1. **Don't animate layout properties** — Avoid `width`, `height`, `margin`, `padding`
2. **Don't use infinite animations** — Except for spinners/loaders
3. **Don't animate too many elements** — Keep animation count low
4. **Don't ignore accessibility** — Animations can cause seizures
5. **Don't use JavaScript for simple animations** — CSS is sufficient

### Performance Guidelines

#### GPU-Accelerated Properties

```css
/* Good: GPU-accelerated */
.gsmui-component {
    transform: translateX(100px);
    transform: scale(1.1);
    transform: rotate(45deg);
    opacity: 0.5;
}

/* Bad: Layout-triggering */
.gsmui-component {
    width: 100px;
    height: 100px;
    margin-left: 100px;
    padding: 20px;
    top: 100px;
    left: 100px;
}
```

#### will-change Usage

```css
/* Add will-change before animation starts */
.gsmui-component {
    will-change: transform, opacity;
}

/* Remove after animation completes */
.gsmui-component.animated {
    will-change: auto;
}
```

### Timing Guidelines

| Duration | Use Case | Example |
|----------|----------|---------|
| 0ms | Instant feedback | Button press |
| 100ms | Micro-interactions | Hover, focus |
| 200ms | Standard transitions | State changes |
| 300ms | Complex animations | Popovers, modals |
| 500ms | Page transitions | Route changes |

### Easing Guidelines

| Easing | Use Case | Example |
|--------|----------|---------|
| Ease-in | Exiting elements | Dropdown close |
| Ease-out | Entering elements | Dropdown open |
| Ease-in-out | State changes | Toggle switch |
| Spring | Playful interactions | Popovers, tooltips |

### Animation Checklist

- [ ] Animations are short (100-300ms)
- [ ] Animations use GPU-accelerated properties
- [ ] Reduced motion is respected
- [ ] Animations don't cause layout shifts
- [ ] Animations are tested on slow devices
- [ ] Animations are documented
- [ ] Animations don't block user interaction
- [ ] Animations have appropriate easing

```javascript
        this.element = element;
        this.animations = new Map();
        this.reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
        
        this.setupReducedMotionListener();
    }
    
    setupReducedMotionListener() {
        window.matchMedia('(prefers-reduced-motion: reduce)').addEventListener('change', (e) => {
            this.reducedMotion = e.matches;
            
            if (this.reducedMotion) {
                this.cancelAll();
            }
        });
    }
    
    animate(keyframes, options = {}) {
        if (this.reducedMotion) {
            options.duration = 0;
        }
        
        const animation = this.element.animate(keyframes, {
            duration: options.duration ?? 200,
            easing: options.easing ?? 'ease-out',
            fill: options.fill ?? 'forwards',
            ...options,
        });
        
        const id = Symbol('animation');
        this.animations.set(id, animation);
        
        animation.finished.finally(() => {
            this.animations.delete(id);
        });
        
        return {
            id,
            cancel: () => animation.cancel(),
            pause: () => animation.pause(),
            play: () => animation.play(),
            reverse: () => animation.reverse(),
            finished: animation.finished,
        };
    }
    
    fadeIn(duration = 200) {
        return this.animate([
            { opacity: 0 },
            { opacity: 1 }
        ], { duration, easing: 'ease-out' });
    }
    
    fadeOut(duration = 150) {
        return this.animate([
            { opacity: 1 },
            { opacity: 0 }
        ], { duration, easing: 'ease-in' });
    }
    
    slideIn(direction = 'up', duration = 200) {
        const transforms = {
            up: 'translateY(-100%)',
            down: 'translateY(100%)',
            left: 'translateX(-100%)',
            right: 'translateX(100%)',
        };
        
        return this.animate([
            { opacity: 0, transform: transforms[direction] },
            { opacity: 1, transform: 'translate(0)' }
        ], { duration, easing: 'cubic-bezier(0.34, 1.56, 0.64, 1)' });
    }
    
    scaleIn(duration = 200) {
        return this.animate([
            { opacity: 0, transform: 'scale(0.95)' },
            { opacity: 1, transform: 'scale(1)' }
        ], { duration, easing: 'cubic-bezier(0.34, 1.56, 0.64, 1)' });
    }
    
    shake(duration = 300) {
        return this.animate([
            { transform: 'translateX(0)' },
            { transform: 'translateX(-8px)' },
            { transform: 'translateX(8px)' },
            { transform: 'translateX(-8px)' },
            { transform: 'translateX(0)' }
        ], { duration, easing: 'ease-out' });
    }
    
    cancelAll() {
        for (const animation of this.animations.values()) {
            animation.cancel();
        }
        this.animations.clear();
    }
    
    pauseAll() {
        for (const animation of this.animations.values()) {
            animation.pause();
        }
    }
    
    resumeAll() {
        for (const animation of this.animations.values()) {
            animation.play();
        }
    }
}
```

### Transition Manager

```javascript
class TransitionManager {
    constructor(element) {
        this.element = element;
        this.currentTransition = null;
    }
    
    async transition(options = {}) {
        const {
            enter = [],
            leave = [],
            duration = 200,
            onEnter = null,
            onLeave = null,
        } = options;
        
        // Cancel any current transition
        if (this.currentTransition) {
            this.currentTransition.cancel();
        }
        
        // Apply enter classes
        this.element.classList.add(...enter);
        
        if (onEnter) onEnter();
        
        // Wait for transition
        await new Promise(resolve => setTimeout(resolve, duration));
        
        // Remove enter classes
        this.element.classList.remove(...enter);
        
        return {
            close: async () => {
                // Apply leave classes
                this.element.classList.add(...leave);
                
                if (onLeave) onLeave();
                
                // Wait for transition
                await new Promise(resolve => setTimeout(resolve, duration));
                
                // Remove leave classes
                this.element.classList.remove(...leave);
            }
        };
    }
    
    cancel() {
        if (this.currentTransition) {
            this.currentTransition.cancel();
            this.currentTransition = null;
        }
    }
}
```

### Animation Scheduler

```javascript
class AnimationScheduler {
    constructor() {
        this.queue = [];
        this.running = false;
    }
    
    async add(animation, options = {}) {
        return new Promise((resolve) => {
            this.queue.push({
                animation,
                options,
                resolve,
            });
            
            if (!this.running) {
                this.process();
            }
        });
    }
    
    async process() {
        if (this.queue.length === 0) {
            this.running = false;
            return;
        }
        
        this.running = true;
        const { animation, options, resolve } = this.queue.shift();
        
        try {
            await animation();
        } catch (error) {
            console.error('Animation error:', error);
        }
        
        resolve();
        this.process();
    }
    
    clear() {
        this.queue = [];
    }
}
```

```javascript
class StaggerAnimator {
    constructor(container, options = {}) {
        this.container = container;
        this.options = {
            itemSelector: ':scope > *',
            animation: 'fadeIn',
            delay: 50,
            duration: 200,
            ...options,
        };
    }
    
    async animate() {
        const items = this.container.querySelectorAll(this.options.itemSelector);
        const controllers = [];
        
        items.forEach((item, index) => {
            setTimeout(() => {
                const controller = new AnimationController(item);
                controller[this.options.animation](this.options.duration);
                controllers.push(controller);
            }, index * this.options.delay);
        });
        
        return new Promise(resolve => {
            setTimeout(resolve, items.length * this.options.delay + this.options.duration);
        });
    }
}
```
