# 13.03 Component Anatomy — GsmUi Component Structure

## Component File Structure

Every GsmUi component follows this standardized structure:

```
packages/gsmui/src/Components/{Category}/{ComponentName}/
├── {ComponentName}.php              # Main PHP component class
├── {ComponentName}Contract.php      # Interface (if applicable)
├── Concerns/
│   └── {TraitName}.php              # Reusable traits
├── {ComponentName}Provider.php      # Service provider (if needed)
└── resources/
    └── views/
        └── components/
            └── {category}/
                └── {component-name}/
                    ├── index.blade.php           # Main template
                    ├── {sub-component}.blade.php # Sub-components
                    └── variants/
                        ├── {variant-name}.blade.php
                        └── ...
```

## PHP Component Class Anatomy

```php
<?php

namespace Gsmui\Components\{Category};

use Gsmui\Contracts\ComponentContract;
use Gsmui\Concerns\Controllable;
use Gsmui\Concerns\Disableable;
use Gsmui\Support\GsmuiClasses;
use Illuminate\Contracts\View\View;
use Livewire\Attributes\Locked;
use Livewire\Component as LivewireComponent;

class {ComponentName} extends LivewireComponent implements ComponentContract
{
    use Controllable, Disableable, /* other traits */;

    // ─── Props ──────────────────────────────────────────────
    
    /** @var string|null */
    #[Locked]
    public ?string $variant = 'default';
    
    /** @var string|null */
    #[Locked]
    public ?string $size = 'md';
    
    /** @var string|null */
    #[Locked]
    public ?string $color = 'primary';
    
    /** @var bool */
    public bool $disabled = false;
    
    /** @var bool */
    public bool $loading = false;
    
    // ... additional props with PHPDoc for IDE support
    
    // ─── Slots ──────────────────────────────────────────────
    
    /** @var string|null */
    public ?string $icon = null;
    
    /** @var string|null */
    public ?string $label = null;
    
    /** @var string|null */
    public ?string $description = null;
    
    // ─── Computed Properties ────────────────────────────────
    
    public function classes(): string
    {
        return GsmuiClasses::join([
            $this->baseClasses(),
            $this->variantClasses(),
            $this->sizeClasses(),
            $this->colorClasses(),
            $this->stateClasses(),
        ]);
    }
    
    protected function baseClasses(): string
    {
        return 'inline-flex items-center justify-center font-medium transition-colors focus-visible:outline-2 focus-visible:outline-[-webkit-focus-ring-color]';
    }
    
    protected function variantClasses(): string
    {
        return match ($this->variant) {
            'primary' => 'bg-(--gsmui-color-primary) text-white hover:bg-(--gsmui-color-primary)/90',
            'secondary' => 'bg-(--gsmui-color-zinc-100) text-zinc-900 dark:bg-(--gsmui-color-zinc-800) dark:text-white hover:bg-(--gsmui-color-zinc-200) dark:hover:bg-(--gsmui-color-zinc-700)',
            'outline' => 'border border-(--gsmui-color-zinc-300) dark:border-(--gsmui-color-zinc-600) bg-transparent hover:bg-(--gsmui-color-zinc-50) dark:hover:bg-(--gsmui-color-zinc-800/50)',
            'ghost' => 'bg-transparent hover:bg-(--gsmui-color-zinc-100) dark:hover:bg-(--gsmui-color-zinc-800)',
            'link' => 'bg-transparent text-(--gsmui-color-primary) underline-offset-2 hover:underline',
            default => '',
        };
    }
    
    protected function sizeClasses(): string
    {
        return match ($this->size) {
            'xs' => 'h-6 px-2 text-xs rounded-md',
            'sm' => 'h-8 px-3 text-sm rounded-md',
            'md' => 'h-10 px-4 text-sm rounded-lg',
            'lg' => 'h-12 px-6 text-base rounded-lg',
            'xl' => 'h-14 px-8 text-lg rounded-xl',
            default => '',
        };
    }
    
    protected function colorClasses(): string
    {
        if ($this->variant !== 'primary') return '';
        
        return match ($this->color) {
            'primary' => '',
            'secondary' => 'bg-(--gsmui-color-zinc-600) hover:bg-(--gsmui-color-zinc-700)',
            'success' => 'bg-(--gsmui-color-emerald-600) hover:bg-(--gsmui-color-emerald-700)',
            'warning' => 'bg-(--gsmui-color-amber-600) hover:bg-(--gsmui-color-amber-700)',
            'danger' => 'bg-(--gsmui-color-red-600) hover:bg-(--gsmui-color-red-700)',
            default => '',
        };
    }
    
    protected function stateClasses(): string
    {
        $classes = [];
        
        if ($this->disabled) {
            $classes[] = 'opacity-50 cursor-not-allowed pointer-events-none';
        }
        
        if ($this->loading) {
            $classes[] = 'relative pr-8';
        }
        
        return implode(' ', $classes);
    }
    
    // ─── Lifecycle ──────────────────────────────────────────
    
    public function mount(): void
    {
        // Initialize Controllable trait
        $this->initializeControllable();
    }
    
    // ─── Public Methods ─────────────────────────────────────
    
    public function render(): View
    {
        return view('gsmui::components.{category}.{component-name}.index');
    }
}
```

## Sub-Component Pattern

For complex components with multiple parts (e.g., Select, DatePicker):

```
{component-name}/
├── index.blade.php           # Root component
├── Trigger.blade.php         # Trigger button/input
├── Content.blade.php         # Popover/content area
├── Option.blade.php          # Individual option
├── Group.blade.php           # Option group
├── Input.blade.php           # Search input
├── Empty.blade.php           # Empty state
└── variants/
    ├── default.blade.php
    ├── inline.blade.php
    └── ...
```

Sub-components are registered as anonymous components:
```blade
{{-- In parent index.blade.php --}}
<x-gsmui:select.trigger {{ $triggerAttributes }} />
<x-gsmui:select.content {{ $contentAttributes }} />
```

### Core Component Class Pattern

```javascript
// resources/js/components/forms/select/Select.js

import { Controllable } from '../../mixins/Controllable.js';
import { Disableable } from '../../mixins/Disableable.js';
import { Focusable } from '../../mixins/Focusable.js';
import { detangle } from '../../utils/detangle.js';
import { setAttribute, removeAttribute } from '../../utils/dom.js';

export class Select extends HTMLElement {
    static get observedAttributes() {
        return ['value', 'disabled', 'multiple', 'searchable', 'placeholder'];
    }

    constructor() {
        super();
        this.attachShadow({ mode: 'open', delegatesFocus: true });
        
        // State
        this._value = null;
        this._internalValue = null;
        this._open = false;
        this._focusedIndex = -1;
        this._options = [];
        
        // Mixins
        this.controllable = new Controllable(this, {
            valueAttribute: 'value',
            eventName: 'change',
            detangle: true,
        });
        
        this.disableable = new Disableable(this);
        this.focusable = new Focusable(this);
    }

    connectedCallback() {
        this.render();
        this.bindEvents();
        this.controllable.connect();
    }

    disconnectedCallback() {
        this.controllable.disconnect();
    }

    attributeChangedCallback(name, oldValue, newValue) {
        if (oldValue === newValue) return;
        
        switch (name) {
            case 'value':
                this.controllable.setValue(newValue);
                break;
            case 'disabled':
                this.disableable.setDisabled(newValue !== null);
                break;
            // ... other attributes
        }
    }

    // ─── Public API ────────────────────────────────────────

    get value() { return this._value; }
    set value(val) { this.controllable.setValue(val); }

    get open() { return this._open; }
    set open(val) { this.setOpen(val); }

    // ─── Internal Methods ──────────────────────────────────

    setOpen(open) {
        if (this._open === open) return;
        this._open = open;
        setAttribute(this, 'open', open ? '' : null);
        this.dispatchEvent(new CustomEvent('open-change', { detail: { open } }));
        
        if (open) {
            this.focusFirstOption();
            document.addEventListener('click', this.handleOutsideClick);
            document.addEventListener('keydown', this.handleKeydown);
        } else {
            document.removeEventListener('click', this.handleOutsideClick);
            document.removeEventListener('keydown', this.handleKeydown);
        }
    }

    // ─── Event Handlers ────────────────────────────────────

    handleOutsideClick = (event) => {
        if (!this.contains(event.target) && !this.popover?.contains(event.target)) {
            this.setOpen(false);
        }
    };

    handleKeydown = (event) => {
        if (!this._open) return;
        
        switch (event.key) {
            case 'ArrowDown':
                event.preventDefault();
                this.focusNextOption();
                break;
            case 'ArrowUp':
                event.preventDefault();
                this.focusPreviousOption();
                break;
            case 'Enter':
            case ' ':
                event.preventDefault();
                this.selectFocusedOption();
                break;
            case 'Escape':
                this.setOpen(false);
                break;
            case 'Home':
                event.preventDefault();
                this.focusFirstOption();
                break;
            case 'End':
                event.preventDefault();
                this.focusLastOption();
                break;
        }
    };

    // ─── Rendering ─────────────────────────────────────────

    render() {
        this.shadowRoot.innerHTML = `
            <style>
                ${this.styles()}
            </style>
            <slot name="trigger"></slot>
            <slot name="content"></slot>
        `;
    }

    styles() {
        return `
            :host { display: inline-block; }
            :host([hidden]) { display: none; }
            :host([disabled]) { opacity: 0.5; pointer-events: none; }
        `;
    }
}

customElements.define('gsmui-select', Select);
```

## Trait/Concern Anatomy

```php
<?php

namespace Gsmui\Concerns;

use Livewire\Attributes\Locked;

trait Controllable
{
    /** @var mixed */
    #[Locked]
    public $value = null;

    /** @var mixed */
    #[Locked]
    public $model = null;

    /** @var string */
    public $wireModel = 'model';

    /** @var bool */
    public $detangle = true;

    public function initializeControllable(): void
    {
        $this->syncModel();
    }

    public function syncModel(): void
    {
        if ($this->model !== null) {
            $this->value = $this->model;
        }
    }

    public function updatedValue($value): void
    {
        if ($this->detangle) {
            $this->detangle();
        }
        $this->dispatch('input', $value);
        $this->dispatch('change', $value);
    }

    public function detangle(): void
    {
        // Prevent sync loops between Controllable and internal state
        $this->model = $this->value;
    }

    public function getValueAttribute(): mixed
    {
        return $this->value ?? $this->model;
    }
}
```

## Base Component Class

### Abstract Base

```php
<?php

namespace Gsmui\Components;

use Gsmui\Contracts\ComponentContract;
use Gsmui\Concerns\HasProps;
use Gsmui\Concerns\HasSlots;
use Gsmui\Concerns\HasEvents;
use Gsmui\Concerns\Disableable;
use Gsmui\Concerns\Accessible;
use Gsmui\Support\GsmuiClasses;
use Illuminate\Contracts\View\View;
use Livewire\Component;

abstract class BaseComponent extends Component implements ComponentContract
{
    use HasProps, HasSlots, HasEvents, Disableable, Accessible;
    
    // ─── Standard Props ─────────────────────────────────────
    
    /** @var string|null */
    public ?string $id = null;
    
    /** @var string */
    public string $class = '';
    
    /** @var array */
    public array $attributes = [];
    
    /** @var string */
    public string $variant = 'default';
    
    /** @var string */
    public string $size = 'md';
    
    /** @var string */
    public string $color = 'primary';
    
    /** @var bool */
    public bool $disabled = false;
    
    /** @var bool */
    public bool $loading = false;
    
    /** @var bool */
    public bool $hidden = false;
    
    // ─── Computed Properties ────────────────────────────────
    
    public function classes(): string
    {
        return GsmuiClasses::join([
            $this->baseClasses(),
            $this->variantClasses(),
            $this->sizeClasses(),
            $this->colorClasses(),
            $this->stateClasses(),
            $this->class,
            $this->attributes->get('class', ''),
        ]);
    }
    
    abstract protected function baseClasses(): string;
    
    protected function variantClasses(): string
    {
        return '';
    }
    
    protected function sizeClasses(): string
    {
        return '';
    }
    
    protected function colorClasses(): string
    {
        return '';
    }
    
    protected function stateClasses(): string
    {
        $classes = [];
        
        if ($this->disabled) {
            $classes[] = 'opacity-50 cursor-not-allowed pointer-events-none';
        }
        
        if ($this->loading) {
            $classes[] = 'relative';
        }
        
        if ($this->hidden) {
            $classes[] = 'hidden';
        }
        
        return implode(' ', $classes);
    }
    
    // ─── Lifecycle ──────────────────────────────────────────
    
    public function mount(): void
    {
        $this->initializeId();
        $this->initializeAccessible();
    }
    
    protected function initializeId(): void
    {
        if ($this->id === null) {
            $this->id = 'gsmui-' . $this->componentName . '-' . Str::random(8);
        }
    }
    
    // ─── Rendering ──────────────────────────────────────────
    
    public function render(): View
    {
        return view($this->bladeView());
    }
    
    abstract protected function bladeView(): string;
    
    // ─── Arrayable ──────────────────────────────────────────
    
    public function toArray(): array
    {
        return [
            'id' => $this->id,
            'class' => $this->class,
            'variant' => $this->variant,
            'size' => $this->size,
            'color' => $this->color,
            'disabled' => $this->disabled,
            'loading' => $this->loading,
            'hidden' => $this->hidden,
        ];
    }
}
```

## Trait Anatomy Deep Dive

### Controllable Trait

```php
<?php

namespace Gsmui\Concerns;

use Livewire\Attributes\Locked;
use Livewire\Attributes\On;

trait Controllable
{
    /** @var mixed */
    #[Locked]
    public $value = null;
    
    /** @var mixed */
    public $model = null;
    
    /** @var string */
    public string $wireModel = 'model';
    
    /** @var bool */
    public bool $detangle = true;
    
    /** @var bool */
    protected bool $syncing = false;
    
    public function initializeControllable(): void
    {
        $this->syncModel();
    }
    
    public function syncModel(): void
    {
        if ($this->model !== null && !$this->syncing) {
            $this->syncing = true;
            $this->value = $this->model;
            $this->syncing = false;
        }
    }
    
    public function updatedValue(mixed $value): void
    {
        if ($this->detangle) {
            $this->detangle();
        }
        
        $this->dispatch('input', value: $value);
        $this->dispatch('change', value: $value);
        
        if ($this->wireModel) {
            $this->syncing = true;
            $this->{$this->wireModel} = $value;
            $this->syncing = false;
        }
    }
    
    public function detangle(): void
    {
        $this->syncing = true;
        $this->model = $this->value;
        $this->syncing = false;
    }
    
    public function getValue(): mixed
    {
        return $this->value;
    }
    
    public function setValue(mixed $value): void
    {
        $this->value = $value;
    }
    
    #[On('set-value')]
    public function handleSetValue(mixed $value): void
    {
        $this->setValue($value);
    }
}
```

```javascript
// resources/js/components/actions/button/Button.js

export class GsmuiButton extends HTMLElement {
    static get observedAttributes() {
        return ['disabled', 'loading', 'variant', 'size'];
    }
    
    constructor() {
        super();
        this.attachShadow({ mode: 'open', delegatesFocus: true });
        
        // Internal state
        this._disabled = false;
        this._loading = false;
        
        // Bind event handlers
        this.handleClick = this.handleClick.bind(this);
        this.handleKeydown = this.handleKeydown.bind(this);
    }
    
    connectedCallback() {
        this.render();
        this.shadowRoot.addEventListener('click', this.handleClick);
        this.shadowRoot.addEventListener('keydown', this.handleKeydown);
        
        if (this.hasAttribute('autofocus')) {
            this.focus();
        }
    }
    
    disconnectedCallback() {
        this.shadowRoot.removeEventListener('click', this.handleClick);
        this.shadowRoot.removeEventListener('keydown', this.handleKeydown);
    }
    
    attributeChangedCallback(name, oldValue, newValue) {
        if (oldValue === newValue) return;
        
        switch (name) {
            case 'disabled':
                this._disabled = newValue !== null;
                this.updateDisabled();
                break;
            case 'loading':
                this._loading = newValue !== null;
                this.updateLoading();
                break;
        }
    }
    
    // ─── Public API ────────────────────────────────────────
    
    get disabled() { return this._disabled; }
    set disabled(val) {
        if (val) {
            this.setAttribute('disabled', '');
        } else {
            this.removeAttribute('disabled');
        }
    }
    
    get loading() { return this._loading; }
    set loading(val) {
        if (val) {
            this.setAttribute('loading', '');
        } else {
            this.removeAttribute('loading');
        }
    }
    
    // ─── Event Handlers ────────────────────────────────────
    
    handleClick(event) {
        if (this._disabled || this._loading) {
            event.preventDefault();
            event.stopPropagation();
            return;
        }
        
        this.dispatchEvent(new CustomEvent('gsmui-click', {
            bubbles: true,
            composed: true,
            detail: { originalEvent: event },
        }));
    }
    
    handleKeydown(event) {
        if (event.key === 'Enter' || event.key === ' ') {
            event.preventDefault();
            this.click();
        }
    }
    
    // ─── Rendering ─────────────────────────────────────────
    
    render() {
        this.shadowRoot.innerHTML = `
            <style>${this.styles()}</style>
            <slot></slot>
        `;
    }
    
    styles() {
        return `
            :host {
                display: inline-flex;
                align-items: center;
                justify-content: center;
                cursor: pointer;
                user-select: none;
            }
            
            :host([disabled]) {
                opacity: 0.5;
                cursor: not-allowed;
                pointer-events: none;
            }
            
            :host([loading]) {
                position: relative;
                pointer-events: none;
            }
        `;
    }
    
    updateDisabled() {
        this.shadowRoot.host.classList.toggle('disabled', this._disabled);
    }
    
    updateLoading() {
        this.shadowRoot.host.classList.toggle('loading', this._loading);
    }
}

customElements.define('gsmui-button', GsmuiButton);
```

```javascript
// resources/js/mixins/Controllable.js

export function Controllable(Base) {
    return class extends Base {
        constructor() {
            super();
            this._value = null;
            this._internalValue = null;
            this._detangle = false;
        }
        
        get value() { return this._value; }
        set value(val) {
            const oldValue = this._value;
            this._value = val;
            
            if (!this._detangle) {
                this._internalValue = val;
                this.dispatchEvent(new CustomEvent('value-change', {
                    detail: { value: val, oldValue },
                }));
            }
        }
        
        syncValue(val) {
            this._detangle = true;
            this._internalValue = val;
            this._value = val;
            this._detangle = false;
        }
    };
}
```

```blade
</form>
```

### JavaScript Anatomy

```
export class {ComponentName} extends HTMLElement {
    static get observedAttributes() {
        return ['value', 'disabled', 'variant', 'size'];
    }
    
    constructor() {
        super();
        this.attachShadow({ mode: 'open' });
        this._value = null;
        this._disabled = false;
    }
    
    connectedCallback() {
        this.render();
        this.bindEvents();
    }
    
    disconnectedCallback() {
        this.unbindEvents();
    }
    
    attributeChangedCallback(name, oldValue, newValue) {
        if (oldValue === newValue) return;
        this.updateAttribute(name, newValue);
    }
    
    get value() { return this._value; }
    set value(val) { this._value = val; }
    
    render() {
        this.shadowRoot.innerHTML = `<style>${this.styles()}</style><slot></slot>`;
    }
    
    styles() {
        return `:host { display: inline-block; }`;
    }
}
```
