# 13.21 Playground — GsmUi Playground Specification

## Overview

The GsmUi Playground is an interactive, in-browser component explorer that allows developers to:
- Browse all components and their variants
- Edit props in real-time
- Copy production-ready code snippets
- Test accessibility states
- Preview in light/dark mode
- Test responsive breakpoints

### PropEditor Component

```tsx
// components/PropEditor.tsx
interface PropEditorProps {
    metadata: ComponentMetadata;
    values: Record<string, any>;
    onChange: (name: string, value: any) => void;
    filter?: string;
}

export function PropEditor({ metadata, values, onChange, filter }: PropEditorProps) {
    const [expandedCategories, setExpandedCategories] = useState<Set<string>>(new Set(['content', 'appearance']));
    
    const categorizedProps = useMemo(() => {
        return metadata.props.reduce((acc, prop) => {
            if (!filter || prop.name.includes(filter) || prop.description?.includes(filter)) {
                if (!acc[prop.category]) acc[prop.category] = [];
                acc[prop.category].push(prop);
            }
            return acc;
        }, {} as Record<string, PropMetadata[]>);
    }, [metadata.props, filter]);
    
    return (
        <div className="prop-editor">
            <input 
                type="text" 
                placeholder="Filter props..." 
                className="prop-filter"
                onChange={(e) => setFilter(e.target.value)}
            />
            
            {Object.entries(categorizedProps).map(([category, props]) => (
                <details key={category} defaultOpen={expandedCategories.has(category)}>
                    <summary onClick={() => toggleCategory(category)}>
                        {category.charAt(0).toUpperCase() + category.slice(1)} ({props.length})
                    </summary>
                    <div className="prop-list">
                        {props.map(prop => (
                            <PropControl 
                                key={prop.name}
                                prop={prop}
                                value={values[prop.name] ?? prop.default}
                                onChange={(val) => onChange(prop.name, val)}
                            />
                        ))}
                    </div>
                </details>
            ))}
        </div>
    );
}
```

### PropControl Components

```tsx
// PropControl components by type
function PropControl({ prop, value, onChange }: { prop: PropMetadata; value: any; onChange: (val: any) => void }) {
    switch (prop.type) {
        case 'boolean':
            return <ToggleSwitch checked={value} onChange={onChange} label={prop.name} />;
        case 'string':
            if (prop.options) {
                return <Select options={prop.options} value={value} onChange={onChange} placeholder={prop.default} />;
            }
            return <TextInput value={value} onChange={onChange} placeholder={prop.default} />;
        case 'number':
            return <NumberInput value={value} onChange={onChange} />;
        case 'array':
            return <TagInput value={value} onChange={onChange} itemType={prop.itemType} />;
        case 'object':
            return <JsonEditor value={value} onChange={onChange} />;
        default:
            return <TextInput value={JSON.stringify(value)} onChange={onChange} />;
    }
}
```

## Live Preview

### Preview Modes

```tsx
// hooks/useLivePreview.ts
interface PreviewOptions {
    componentName: string;
    props: Record<string, any>;
    slots: Record<string, string>;
    mode: 'iframe' | 'shadow' | 'inline';
}

function useLivePreview(options: PreviewOptions) {
    const [previewHtml, setPreviewHtml] = useState('');
    const [error, setError] = useState<string | null>(null);
    
    useEffect(() => {
        renderPreview();
    }, [options.props, options.slots]);
    
    async function renderPreview() {
        try {
            if (options.mode === 'iframe') {
                // Render in isolated iframe
                const html = await fetch('/api/playground/render', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        component: options.componentName,
                        props: options.props,
                        slots: options.slots,
                    }),
                }).then(r => r.text());
                
                setPreviewHtml(html);
            } else {
                // Client-side rendering with Alpine/Livewire
                const html = await clientSideRender(options);
                setPreviewHtml(html);
            }
        } catch (e) {
            setError(e.message);
        }
    }
    
    return { previewHtml, error };
}
```

```php
// routes/playground.php
Route::post('/playground/render', function (Request $request) {
    $request->validate([
        'component' => 'required|string',
        'props' => 'array',
        'slots' => 'array',
    ]);
    
    $component = "GsmUi\\Components\\" . Str::studly($request->component);
    
    if (!class_exists($component)) {
        return response()->json(['error' => 'Component not found'], 404);
    }
    
    $instance = app($component, [
        'attributes' => new \Illuminate\View\ComponentAttributeBag($request->props),
    ]);
    
    foreach ($request->slots as $name => $content) {
        $instance->setSlot($name, $content);
    }
    
    $html = $instance->render()->render();
    
    return response()->json(['html' => $html]);
})->middleware('web');
```

## Code Generation

### Blade Code Generator

```typescript
// utils/codeGenerator.ts
export function generateBladeCode(
    componentName: string,
    props: Record<string, any>,
    slots: Record<string, string>
): string {
    const propString = Object.entries(props)
        .filter(([_, value]) => value !== undefined && value !== '')
        .map(([key, value]) => formatProp(key, value))
        .join('\n    ');
    
    const slotString = Object.entries(slots)
        .map(([name, content]) => {
            if (name === 'default') {
                return `    ${content}`;
            }
            return `    <x-slot name="${name}">\n${indent(content, 2)}\n    </x-slot>`;
        })
        .join('\n');
    
    if (slotString && !slots.default) {
        return `<x-gsmui:${componentName}
    ${propString}
>
${slotString}
</x-gsmui:${componentName}>`;
    }
    
    if (slots.default) {
        return `<x-gsmui:${componentName}
    ${propString}
>
    ${slots.default}
</x-gsmui:${componentName}>`;
    }
    
    return `<x-gsmui:${componentName}
    ${propString}
/>`;
}

function formatProp(key: string, value: any): string {
    if (typeof value === 'boolean') {
        return value ? key : '';
    }
    if (typeof value === 'string') {
        return `${key}="${value}"`;
    }
    if (typeof value === 'number') {
        return `:${key}="${value}"`;
    }
    if (Array.isArray(value) || typeof value === 'object') {
        return `:${key}="${JSON.stringify(value).replace(/"/g, '"')}"`;
    }
    return '';
}

function indent(str: string, spaces: number): string {
    return str.split('\n').map(line => ' '.repeat(spaces) + line).join('\n');
}
```

### Livewire Code Generator

```typescript
function generateLivewireCode(
    componentName: string,
    props: Record<string, any>,
    slots: Record<string, string>
): string {
    // Extract wire:model
    const wireModel = props.wireModel;
    const otherProps = { ...props };
    delete otherProps.wireModel;
    
    const propString = Object.entries(otherProps)
        .filter(([_, v]) => v !== undefined && v !== '')
        .map(([k, v]) => formatProp(k, v))
        .join('\n    ');
    
    return `<x-gsmui:${toKebabCase(componentName)}
    ${propString}
    ${wireModel ? `wire:model="${wireModel}"` : ''}
>
    ${slots.default || ''}
</x-gsmui:${toKebabCase(componentName)}>`;
}
```

## URL State Management

### Encoding State

```typescript
// utils/urlState.ts
interface PlaygroundState {
    component: string;
    props: Record<string, any>;
    slots: Record<string, string>;
    theme: 'light' | 'dark';
    breakpoint: 'mobile' | 'tablet' | 'desktop';
    mode: 'iframe' | 'shadow';
}

export function encodeState(state: PlaygroundState): string {
    // Remove default values to keep URL short
    const metadata = componentRegistry.get(state.component);
    const minimalProps = Object.entries(state.props)
        .filter(([key, value]) => {
            const propDef = metadata?.props.find(p => p.name === key);
            return value !== undefined && value !== '' && value !== propDef?.default;
        })
        .reduce((acc, [k, v]) => ({ ...acc, [k]: v }), {});
    
    const compressed = {
        c: state.component,
        p: minimalProps,
        s: state.slots,
        t: state.theme,
        b: state.breakpoint,
        m: state.mode,
    };
    
    return btoa(JSON.stringify(compressed));
}

export function decodeState(hash: string): Partial<PlaygroundState> | null {
    try {
        const decoded = JSON.parse(atob(hash));
        return {
            component: decoded.c,
            props: decoded.p || {},
            slots: decoded.s || {},
            theme: decoded.t || 'light',
            breakpoint: decoded.b || 'desktop',
            mode: decoded.m || 'iframe',
        };
    } catch {
        return null;
    }
}
```

### URL Sync Hook

```typescript
// hooks/useUrlState.ts
export function useUrlState(initialState: PlaygroundState) {
    const [state, setState] = useState<PlaygroundState>(() => {
        if (typeof window !== 'undefined') {
            return decodeState(window.location.hash.slice(1)) || initialState;
        }
        return initialState;
    });
    
    useEffect(() => {
        const hash = encodeState(state);
        window.history.replaceState(null, '', `#${hash}`);
    }, [state]);
    
    // Handle browser back/forward
    useEffect(() => {
        const handlePopState = () => {
            const decoded = decodeState(window.location.hash.slice(1));
            if (decoded) setState(decoded);
        };
        
        window.addEventListener('popstate', handlePopState);
        return () => window.removeEventListener('popstate', handlePopState);
    }, []);
    
    return [state, setState] as const;
}
```

## Shareable Links

```tsx
// components/ShareButton.tsx
export function ShareButton({ state }: { state: PlaygroundState }) {
    const [copied, setCopied] = useState(false);
    const url = `${window.location.origin}/playground#${encodeState(state)}`;
    
    const handleCopy = async () => {
        await navigator.clipboard.writeText(url);
        setCopied(true);
        setTimeout(() => setCopied(false), 2000);
    };
    
    return (
        <button 
            className="share-button"
            onClick={handleCopy}
            aria-label="Copy playground link"
        >
            {copied ? (
                <>
                    <CheckIcon /> Copied!
                </>
            ) : (
                <>
                    <LinkIcon /> Share
                </>
            )}
        </button>
    );
}
```

```tsx
// hooks/useKeyboardShortcuts.ts
export function useKeyboardShortcuts(actions: Record<string, () => void>) {
    useEffect(() => {
        const handleKeyDown = (e: KeyboardEvent) => {
            if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) {
                return;
            }
            
            const shortcut = [
                e.ctrlKey || e.metaKey ? 'ctrl' : null,
                e.shiftKey ? 'shift' : null,
                e.altKey ? 'alt' : null,
                e.key.toLowerCase(),
            ].filter(Boolean).join('+');
            
            if (actions[shortcut]) {
                e.preventDefault();
                actions[shortcut]();
            }
        };
        
        window.addEventListener('keydown', handleKeyDown);
        return () => window.removeEventListener('keydown', handleKeyDown);
    }, [actions]);
}
```

## Playground Architecture Deep Dive

### System Architecture Overview

The GsmUi Playground consists of three main subsystems:

**1. Frontend Application (React/TypeScript)**
- User interface for browsing components
- Prop editing controls
- Live preview rendering
- Code generation engine
- URL state management

**2. Backend API (Laravel)**
- Component metadata serving
- Server-side rendering endpoint
- User session management
- Analytics tracking
- Rate limiting

**3. Component Registry (YAML/JSON)**
- Component definitions
- Prop metadata
- Example configurations
- Accessibility information
- Version compatibility data

## Prop Editor Advanced Features

### Prop Type Detection and Rendering

```typescript
// Enhanced prop control detection
function getPropControl(prop: PropMetadata): React.ComponentType<PropControlProps> {
    // Check for enum type first
    if (prop.type === 'enum' || prop.options) {
        if (prop.options && prop.options.length <= 5) {
            return RadioGroupControl;
        }
        return SelectControl;
    }
    
    // Check for boolean
    if (prop.type === 'boolean') {
        return ToggleControl;
    }
    
    // Check for number with range
    if (prop.type === 'number' && prop.min !== undefined && prop.max !== undefined) {
        return SliderControl;
    }
    
    // Check for number
    if (prop.type === 'number') {
        return NumberInputControl;
    }
    
    // Check for array
    if (prop.type === 'array') {
        return TagInputControl;
    }
    
    // Check for object
    if (prop.type === 'object') {
        return JsonEditorControl;
    }
    
    // Check for function
    if (prop.type === 'function') {
        return CodeEditorControl;
    }
    
    // Default to text input
    return TextInputControl;
}
```

### Prop Validation

```typescript
interface PropValidator {
    validate(value: any, prop: PropMetadata): ValidationResult;
}

interface ValidationResult {
    valid: boolean;
    errors: string[];
    warnings: string[];
}

class CompositePropValidator implements PropValidator {
    private validators: PropValidator[] = [
        new RequiredValidator(),
        new TypeValidator(),
        new RangeValidator(),
        new PatternValidator(),
        new EnumValidator(),
        new CustomValidator(),
    ];
    
    validate(value: any, prop: PropMetadata): ValidationResult {
        const errors: string[] = [];
        const warnings: string[] = [];
        
        for (const validator of this.validators) {
            const result = validator.validate(value, prop);
            errors.push(...result.errors);
            warnings.push(...result.warnings);
        }
        
        return {
            valid: errors.length === 0,
            errors,
            warnings,
        };
    }
}

class TypeValidator implements PropValidator {
    validate(value: any, prop: PropMetadata): ValidationResult {
        const errors: string[] = [];
        const warnings: string[] = [];
        
        switch (prop.type) {
            case 'string':
                if (typeof value !== 'string') {
                    errors.push(`Expected string, got ${typeof value}`);
                }
                break;
            case 'number':
                if (typeof value !== 'number' || isNaN(value)) {
                    errors.push(`Expected number, got ${typeof value}`);
                }
                break;
            case 'boolean':
                if (typeof value !== 'boolean') {
                    errors.push(`Expected boolean, got ${typeof value}`);
                }
                break;
            case 'array':
                if (!Array.isArray(value)) {
                    errors.push(`Expected array, got ${typeof value}`);
                }
                break;
            case 'object':
                if (typeof value !== 'object' || value === null || Array.isArray(value)) {
                    errors.push(`Expected object, got ${typeof value}`);
                }
                break;
        }
        
        return { valid: errors.length === 0, errors, warnings };
    }
}
```

```typescript
class BladeCodeGenerator {
    generate(component: string, props: Record<string, any>, slots: Record<string, string>): string {
        const lines: string[] = [];
        const componentName = this.toKebabCase(component);
        
        // Start tag
        let tag = `<x-gsmui:${componentName}`;
        
        // Add props
        const propLines = this.formatProps(props);
        if (propLines.length > 0) {
            tag += '\n';
            tag += propLines.map(p => `    ${p}`).join('\n');
        }
        
        // Check for slots
        const hasSlots = Object.keys(slots).length > 0;
        
        if (hasSlots) {
            tag += '\n>';
            lines.push(tag);
            
            // Add slot content
            for (const [name, content] of Object.entries(slots)) {
                if (name === 'default' && content) {
                    lines.push(`    ${content}`);
                } else if (content) {
                    lines.push(`    <x-slot name="${name}">`);
                    lines.push(`        ${content}`);
                    lines.push(`    </x-slot>`);
                }
            }
            
            lines.push(`</x-gsmui:${componentName}>`);
        } else {
            tag += ' />';
            lines.push(tag);
        }
        
        return lines.join('\n');
    }
    
    private formatProps(props: Record<string, any>): string[] {
        const formatted: string[] = [];
        
        for (const [key, value] of Object.entries(props)) {
            if (value === undefined || value === null) continue;
            
            if (typeof value === 'boolean') {
                if (value) {
                    formatted.push(key);
                }
            } else if (typeof value === 'string') {
                formatted.push(`${key}="${this.escapeHtml(value)}"`);
            } else if (typeof value === 'number') {
                formatted.push(`:${key}="${value}"`);
            } else if (Array.isArray(value) || typeof value === 'object') {
                formatted.push(`:${key}="${this.escapeJson(value)}"`);
            }
        }
        
        return formatted;
    }
    
    private toKebabCase(str: string): string {
        return str.replace(/([a-z0-9])([A-Z])/g, '$1-$2').toLowerCase();
    }
    
    private escapeHtml(str: string): string {
        return str.replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
    }
    
    private escapeJson(obj: any): string {
        return JSON.stringify(obj).replace(/"/g, '&quot;');
    }
}
```

```typescript
class StateCompressor {
    private maxUrlLength: number = 2000;
    
    compress(state: PlaygroundState): string {
        const compressed = {
            c: state.component,
            p: this.compressProps(state.props),
            s: this.compressSlots(state.slots),
            t: state.theme[0], // 'l' or 'd'
            b: state.breakpoint[0], // 'm', 't', or 'd'
        };
        
        const json = JSON.stringify(compressed);
        const base64 = btoa(json);
        
        // If URL too long, use more aggressive compression
        if (base64.length > this.maxUrlLength) {
            return this.compressAggressive(compressed);
        }
        
        return base64;
    }
    
    private compressProps(props: Record<string, any>): Record<string, any> {
        const compressed: Record<string, any> = {};
        
        for (const [key, value] of Object.entries(props)) {
            // Skip default values
            if (this.isDefaultValue(key, value)) continue;
            
            // Shorten common keys
            const shortKey = this.shortenKey(key);
            compressed[shortKey] = value;
        }
        
        return compressed;
    }
    
    private shortenKey(key: string): string {
        const abbreviations: Record<string, string> = {
            'disabled': 'd',
            'variant': 'v',
            'size': 's',
            'color': 'c',
            'icon': 'i',
            'placeholder': 'ph',
            'label': 'lb',
        };
        return abbreviations[key] || key;
    }
    
    private compressAggressive(compressed: any): string {
        // Use LZ-string or similar compression
        return LZString.compressToBase64(JSON.stringify(compressed));
    }
    
    decompress(hash: string): Partial<PlaygroundState> | null {
        try {
            let json: string;
            
            // Try base64 first
            try {
                json = atob(hash);
            } catch {
                // Try LZ-string decompression
                json = LZString.decompressFromBase64(hash);
            }
            
            const compressed = JSON.parse(json);
            
            return {
                component: compressed.c,
                props: this.decompressProps(compressed.p || {}),
                slots: compressed.s || {},
                theme: compressed.t === 'd' ? 'dark' : 'light',
                breakpoint: this.expandBreakpoint(compressed.b),
            };
        } catch {
            return null;
        }
    }
}
```

```typescript
class StateSynchronizer {
    private state: PlaygroundState;
    private urlHash: string;
    private syncInterval: number = 100; // ms
    
    constructor(initialState: PlaygroundState) {
        this.state = initialState;
        this.urlHash = this.compressState(initialState);
        
        this.startSync();
    }
    
    private startSync(): void {
        // Sync to URL on state change
        setInterval(() => {
            const newHash = this.compressState(this.state);
            if (newHash !== this.urlHash) {
                this.urlHash = newHash;
                window.history.replaceState(null, '', `#${newHash}`);

                const newState = this.decompressState(hash);
                if (newState) {
                    this.state = { ...this.state, ...newState };
                    this.onStateChange?.(this.state);
                }
            }
        });
    }

        this.state = { ...this.state, ...partial };
        this.onStateChange?.(this.state);
    }
    
    getState(): PlaygroundState {
        return { ...this.state };
    }
```

```typescript
class LazyComponentLoader {
    private loadedComponents: Map<string, ComponentMetadata> = new Map();
    private loadingPromises: Map<string, Promise<ComponentMetadata>> = new Map();
    
    async load(componentName: string): Promise<ComponentMetadata> {
        // Return cached if available
        if (this.loadedComponents.has(componentName)) {
            return this.loadedComponents.get(componentName)!;
        }
        
        // Return existing promise if already loading
        if (this.loadingPromises.has(componentName)) {
            return this.loadingPromises.get(componentName)!;
        }
        
        // Start new load
        const promise = this.fetchComponent(componentName);
        this.loadingPromises.set(componentName, promise);
        
        try {
            const metadata = await promise;
            this.loadedComponents.set(componentName, metadata);
            this.loadingPromises.delete(componentName);
            return metadata;
        } catch (error) {
            this.loadingPromises.delete(componentName);
            throw error;
        }
    }
    
    private async fetchComponent(name: string): Promise<ComponentMetadata> {
        const response = await fetch(`/api/playground/components/${name}`);
        if (!response.ok) {
            throw new Error(`Failed to load component: ${name}`);
        }
        return response.json();
    }
    
    preload(componentNames: string[]): void {
        for (const name of componentNames) {
            this.load(name).catch(() => {}); // Fire and forget
        }
    }
}
```

```typescript
class VirtualComponentList {
    private container: HTMLElement;
    private items: ComponentMetadata[] = [];
    private visibleItems: ComponentMetadata[] = [];
    private scrollTop: number = 0;

    private containerHeight: number = 0;

        this.render();

        const endIndex = Math.min(
            startIndex + Math.ceil(this.containerHeight / this.itemHeight) + 1,

        );

                    <span class="component-name">${item.displayName}</span>
                    <span class="component-category">${item.category}</span>
                </div>
            `;
        }).join('');
        
        this.container.innerHTML = html;
    }
```

```typescript
    private currentIndex: number = 0;
    
    constructor() {

    }
    
    private setupKeyboardHandlers(): void {
        document.addEventListener('keydown', (e) => {
            switch (e.key) {
                case 'Tab':
                    this.handleTab(e);
                    break;
                case 'ArrowDown':
                    this.handleArrowDown();
                    break;
                case 'ArrowUp':
                    this.handleArrowUp();
                    break;
                case 'Enter':
                case ' ':
                    this.handleActivation();
                    break;
                case 'Escape':
                    this.handleEscape();
                    break;
            }
        });
    }
    
    private handleTab(e: KeyboardEvent): void {
        const focusable = this.getFocusableElements();
        const currentIndex = focusable.indexOf(document.activeElement as HTMLElement);
        
        if (e.shiftKey) {
            // Shift+Tab: Move backwards
            const prevIndex = currentIndex <= 0 ? focusable.length - 1 : currentIndex - 1;
            focusable[prevIndex]?.focus();
        } else {
            // Tab: Move forwards
            const nextIndex = currentIndex >= focusable.length - 1 ? 0 : currentIndex + 1;
            focusable[nextIndex]?.focus();
        }
        
        e.preventDefault();
    }
    
    private getFocusableElements(): HTMLElement[] {
        return Array.from(document.querySelectorAll(
            'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
        ));
    }
}
```
