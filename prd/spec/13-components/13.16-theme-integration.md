# 13.16 Theme Integration — GsmUi Theme Specification

## Overview

GsmUi components are fully themeable through a CSS custom property (design token) system. Themes can be switched at runtime, support light/dark modes, and allow per-component customization.

## Architecture

### Token Hierarchy

```
┌─────────────────────────────────────────────────────────────┐
│                    PRIMITIVE TOKENS                         │
│  --color-blue-500, --spacing-4, --radius-lg, --font-size-sm │
├─────────────────────────────────────────────────────────────┤
│                    SEMANTIC TOKENS                          │
│  --color-primary, --color-bg-primary, --color-text-primary  │
├─────────────────────────────────────────────────────────────┤
│                   COMPONENT TOKENS                          │
│  --btn-bg-primary, --input-border, --card-shadow           │
├─────────────────────────────────────────────────────────────┤
│                    COMPONENT STYLES                         │
│  .gsmui-button { background: var(--btn-bg-primary) }       │
└─────────────────────────────────────────────────────────────┘
```

### CSS Generation

```php
<?php

namespace GsmUi\Support;

class ThemeCompiler
{
    public function compile(array $theme): string
    {
        $css = ":root {\n";
        
        // Primitive tokens
        $css .= $this->compilePrimitives($theme);
        
        // Semantic tokens
        $css .= $this->compileSemantic($theme);
        
        // Component tokens
        $css .= $this->compileComponents($theme);
        
        $css .= "}\n\n";
        
        // Dark mode
        $css .= "@media (prefers-color-scheme: dark) {\n";
        $css .= "    :root {\n";
        $css .= $this->compileDarkMode($theme);
        $css .= "    }\n";
        $css .= "}\n";
        
        return $css;
    }
    
    protected function compilePrimitives(array $theme): string
    {
        // Returns all primitive tokens from Tailwind
        return file_get_contents(__DIR__ . '/../resources/css/primitives.css');
    }
    
    protected function compileSemantic(array $theme): string
    {
        $css = '';
        $colors = $theme['colors'] ?? [];
        
        foreach ($colors as $name => $shades) {
            if (is_array($shades) && isset($shades['light'], $shades['dark'])) {
                $css .= "    --gsmui-color-{$name}: {$shades['light']};\n";
            }
        }
        
        return $css;
    }
    
    protected function compileDarkMode(array $theme): string
    {
        $css = '';
        $colors = $theme['colors'] ?? [];
        
        foreach ($colors as $name => $shades) {
            if (is_array($shades) && isset($shades['light'], $shades['dark'])) {
                $css .= "        --gsmui-color-{$name}: {$shades['dark']};\n";
            }
        }
        
        return $css;
    }
}
```

### Publishing Theme CSS

```php
// ThemeServiceProvider
public function boot(): void
{
    $this->commands([
        \GsmUi\Commands\CompileTheme::class,
    ]);
}

// Command: php artisan gsmui:compile-theme
class CompileTheme extends Command
{
    protected $signature = 'gsmui:compile-theme {--theme=default} {--output=public/vendor/gsmui/css/theme.css}';
    
    public function handle(): int
    {
        $theme = config("gsmui.theme.themes.{$this->option('theme')}");
        $css = app(ThemeCompiler::class)->compile($theme);
        
        $output = $this->option('output');
        File::put($output, $css);
        
        $this->info("Theme compiled to {$output}");
        return self::SUCCESS;
    }
}
```

## Runtime Theme Switching

### Theme Service

```php
<?php

namespace GsmUi\Services;

class ThemeManager
{
    protected string $currentTheme = 'default';
    protected array $themes = [];
    
    public function setTheme(string $name): void
    {
        if (!isset($this->themes[$name])) {
            throw new \InvalidArgumentException("Theme {$name} not found");
        }
        
        $this->currentTheme = $name;
        session(['gsmui_theme' => $name]);
        
        // Update CSS variables on root
        $this->applyTheme($name);
    }
    
    public function getCurrentTheme(): string
    {
        return $this->currentTheme;
    }
    
    public function getThemeVariables(string $name): array
    {
        return $this->themes[$name]['cssVariables'] ?? [];
    }
    
    protected function applyTheme(string $name): void
    {
        $variables = $this->getThemeVariables($name);
        
        $script = "document.documentElement.style.cssText += '";
        foreach ($variables as $key => $value) {
            $script .= "--{$key}: {$value}; ";
        }
        $script .= "';";
        
        // Inject via Livewire or direct response
        if (app()->bound('livewire')) {
            \Livewire\Livewire::dispatch('gsmui:theme-changed', [
                'theme' => $name,
                'variables' => $variables,
            ]);
        }
    }
}
```

```javascript
        this.currentTheme = localStorage.getItem('gsmui-theme') || 'default';
        this.applyTheme(this.currentTheme);
    }
    
    setTheme(theme) {
        this.currentTheme = theme;
        localStorage.setItem('gsmui-theme', theme);
        this.applyTheme(theme);
        
        // Dispatch event for Livewire
        window.dispatchEvent(new CustomEvent('gsmui:theme-changed', {
            detail: { theme }
        }));
    }
    
    applyTheme(theme) {
        document.documentElement.setAttribute('data-gsmui-theme', theme);
        
        // If using class-based dark mode
        if (theme === 'dark') {
            document.documentElement.classList.add('dark');
        } else {
            document.documentElement.classList.remove('dark');
        }
    }
    
    toggleDarkMode() {
        const isDark = document.documentElement.classList.toggle('dark');
        localStorage.setItem('gsmui-dark-mode', isDark);
    }
}
```

## CSS Variable Injection

### Dynamic Token Injection at Runtime

GsmUi supports injecting CSS variables at runtime without requiring a page reload or CSS file regeneration. This enables per-user, per-tenant, and per-context theming.

### Runtime Injection Manager

```javascript
class GsmuiVariableInjector {
    constructor() {
        this.root = document.documentElement;
        this.injected = new Map();
        this.styleElement = null;
        this.init();
    }

    init() {
        this.styleElement = document.createElement('style');
        this.styleElement.id = 'gsmui-runtime-tokens';
        this.styleElement.setAttribute('data-gsmui', 'true');
        document.head.appendChild(this.styleElement);
    }

    setVariable(name, value, scope = ':root') {
        const key = `${scope}--${name}`;
        if (!this.injected.has(scope)) {
            this.injected.set(scope, new Map());
        }
        this.injected.get(scope).set(name, value);
        this.flush();
    }

    setVariables(variables, scope = ':root') {
        if (!this.injected.has(scope)) {
            this.injected.set(scope, new Map());
        }
        const map = this.injected.get(scope);
        for (const [name, value] of Object.entries(variables)) {
            map.set(name, value);
        }
        this.flush();
    }

    removeVariable(name, scope = ':root') {
        const map = this.injected.get(scope);
        if (map) {
            map.delete(name);
            this.flush();
        }
    }

    flush() {
        let css = '';
        for (const [scope, vars] of this.injected) {
            css += `${scope} {\n`;
            for (const [name, value] of vars) {
                css += `  --${name}: ${value};\n`;
            }
            css += '}\n';
        }
        this.styleElement.textContent = css;
    }

    getVariable(name, scope = ':root') {
        const map = this.injected.get(scope);
        return map?.get(name) || null;
    }

    getAllVariables() {
        const result = {};
        for (const [scope, vars] of this.injected) {
            result[scope] = Object.fromEntries(vars);
        }
        return result;
    }

    clear(scope = null) {
        if (scope) {
            this.injected.delete(scope);
        } else {
            this.injected.clear();
        }
        this.flush();
    }

    destroy() {
        this.styleElement?.remove();
        this.injected.clear();
    }
}

window.GsmuiVariables = new GsmuiVariableInjector();
```

```php
namespace GsmUi\Services;

class TokenInjector
{
    protected array $overrides = [];
    protected string $scope = ':root';

    public function scope(string $scope): self
    {
        $clone = clone $this;
        $clone->scope = $scope;
        return $clone;
    }

    {

        $clone->overrides[$name] = $value;
        return $clone;

        }
        $css .= "}\n";
        return $css;
    }

    public function render(): string
    {
        return "<style data-gsmui-injected>{$this->compile()}</style>";
    }
```

## Dark Mode Implementation

### Multi-Strategy Dark Mode

GsmUi supports three dark mode strategies, selectable per-application:

### Strategy 1: System Preference (Default)

```css
@media (prefers-color-scheme: dark) {
    :root {
        --gsmui-color-bg-primary: var(--gsmui-color-zinc-900);
        --gsmui-color-text-primary: var(--gsmui-color-white);
        /* ... all dark overrides ... */
    }
}
```

### Strategy 2: Class-Based

```javascript
class GsmuiDarkModeManager {
    constructor(options = {}) {
        this.strategy = options.strategy || 'class';
        this.darkClass = options.darkClass || 'dark';
        this.storageKey = options.storageKey || 'gsmui-dark-mode';
        this.root = document.documentElement;
        this.onChange = options.onChange || null;
        this.mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
        this.init();
    }

    init() {
        const stored = localStorage.getItem(this.storageKey);
        if (stored !== null) {
            this.setDarkMode(stored === 'true');
        } else if (this.strategy === 'system') {
            this.setDarkMode(this.mediaQuery.matches);
            this.mediaQuery.addEventListener('change', (e) => {
                this.setDarkMode(e.matches);
            });
        }
    }

    setDarkMode(isDark) {
        if (this.strategy === 'class' || this.strategy === 'both') {
            this.root.classList.toggle(this.darkClass, isDark);
        }
        if (this.strategy === 'attribute' || this.strategy === 'both') {
            this.root.setAttribute('data-gsmui-theme', isDark ? 'dark' : 'light');
        }
        this.root.setAttribute('color-scheme', isDark ? 'dark' : 'light');
        localStorage.setItem(this.storageKey, isDark);
        this.onChange?.(isDark);
    }

    toggle() {
        const isDark = this.isDark();
        this.setDarkMode(!isDark);
        return !isDark;
    }

    isDark() {
        return this.root.classList.contains(this.darkClass);
    }

    setSystem() {
        localStorage.removeItem(this.storageKey);
        this.setDarkMode(this.mediaQuery.matches);
    }
}

window.GsmuiDarkMode = new GsmuiDarkModeManager({ strategy: 'class' });
```

```javascript
class GsmuiRTLManager {
    constructor() {
        this.root = document.documentElement;
        this.isRTL = false;
        this.init();
    }

    init() {
        this.isRTL = this.root.getAttribute('dir') === 'rtl';
        const observer = new MutationObserver((mutations) => {
            for (const mutation of mutations) {
                if (mutation.attributeName === 'dir') {
                    this.isRTL = this.root.getAttribute('dir') === 'rtl';
                    this.onDirectionChange();
                }
            }
        });
        observer.observe(this.root, { attributes: true, attributeFilter: ['dir'] });
    }

    setRTL(enabled) {
        this.root.setAttribute('dir', enabled ? 'rtl' : 'ltr');
        this.isRTL = enabled;
        localStorage.setItem('gsmui-dir', enabled ? 'rtl' : 'ltr');
        this.onDirectionChange();
    }

    toggle() {
        this.setRTL(!this.isRTL);
    }

    onDirectionChange() {
        window.dispatchEvent(new CustomEvent('gsmui:direction-changed', {
            detail: { rtl: this.isRTL },
        }));
    }
}

window.GsmuiRTL = new GsmuiRTLManager();
```

### Token Generator

```php
<?php

namespace GsmUi\Support;

class TokenGenerator
{
    protected array $config;

    public function __construct(array $config)
    {
        $this->config = $config;
    }

    public function generate(): string
    {
        $css = ":root {\n";
        $css .= $this->generatePrimitives();
        $css .= $this->generateSemantic();
        $css .= $this->generateComponentTokens();
        $css .= "}\n\n";
        $css .= $this->generateDarkMode();
        return $css;
    }

    protected function generatePrimitives(): string
    {
        $css = '';
        $prefix = $this->config['prefix'] ?? 'gsmui';
        foreach ($this->config['categories'] as $category => $scales) {
            foreach ($scales as $scale => $value) {
                if (is_array($value)) {
                    foreach ($value as $shade => $hex) {
                        $css .= "  --{$prefix}-{$category}-{$scale}-{$shade}: {$hex};\n";
                    }
                } else {
                    $css .= "  --{$prefix}-{$category}-{$scale}: {$value};\n";
                }
            }
        }
        return $css;
    }

    protected function generateSemantic(): string
    {
        $css = '';
        $prefix = $this->config['prefix'] ?? 'gsmui';
        foreach ($this->config['semantic'] as $name => $shades) {
            $css .= "  --{$prefix}-color-{$name}: var(--{$prefix}-{$this->resolveReference($shades['light'])});\n";
        }
        return $css;
    }

    protected function resolveReference(string $ref): string
    {
        if (preg_match('/^{(.+)}$/', $ref, $matches)) {
            return str_replace('.', '-', $matches[1]);
        }
        return $ref;
    }

    protected function generateComponentTokens(): string
    {
        return "  /* Component tokens generated from semantic tokens */\n";
    }

    protected function generateDarkMode(): string
    {
        $css = "@media (prefers-color-scheme: dark) {\n  :root {\n";
        $prefix = $this->config['prefix'] ?? 'gsmui';
        foreach ($this->config['semantic'] as $name => $shades) {
            $css .= "    --{$prefix}-color-{$name}: var(--{$prefix}-{$this->resolveReference($shades['dark'])});\n";
        }
        $css .= "  }\n}\n";
        return $css;
    }
}
```

```php
        $this->themes = $themes;
    }

    public function compile(string $themeName): string
    {
        $resolved = $this->resolve($themeName);
        $css = $this->generateCSS($resolved);
        return $css;
    }

    public function resolve(string $themeName, array $visited = []): array
    {
        if (!isset($this->themes[$themeName])) {
            throw new \InvalidArgumentException("Theme '{$themeName}' not found");
        }

        $theme = $this->themes[$themeName];

        if (isset($theme['extends'])) {
            if (in_array($theme['extends'], $visited)) {
                throw new \RuntimeException("Circular theme inheritance detected: {$theme['extends']}");
            }
            $visited[] = $theme['extends'];
            $parent = $this->resolve($theme['extends'], $visited);
            return $this->merge($parent, $theme);
        }

        return $theme;
    }

    protected function merge(array $parent, array $child): array
    {
        $merged = $parent;
        foreach ($child as $key => $value) {
            if ($key === 'extends') continue;
            if (is_array($value) && isset($merged[$key]) && is_array($merged[$key])) {
                $merged[$key] = array_merge_recursive($merged[$key], $value);
            } else {
                $merged[$key] = $value;
            }
        }
        return $merged;
    }

    protected function generateCSS(array $theme): string
    {
        $css = ":root {\n";
        if (isset($theme['colors'])) {
            foreach ($theme['colors'] as $name => $shades) {
                $css .= "  --gsmui-color-{$name}: {$shades['light']};\n";
            }
        }
        if (isset($theme['typography']['fontFamily'])) {
            $css .= "  --gsmui-font-sans: {$theme['typography']['fontFamily']};\n";
        }
        $css .= "}\n\n";

        $css .= "@media (prefers-color-scheme: dark) {\n  :root {\n";
        if (isset($theme['colors'])) {
            foreach ($theme['colors'] as $name => $shades) {
                if (isset($shades['dark'])) {
                    $css .= "    --gsmui-color-{$name}: {$shades['dark']};\n";
                }
            }
        }
        $css .= "  }\n}\n";

        return $css;
    }
}
```

### Theme Switching Service

```php
<?php

namespace GsmUi\Services;

use Illuminate\Support\Facades\Session;
use Illuminate\Support\Facades\Cookie;

class RuntimeThemeSwitcher
{
    protected string $sessionKey = 'gsmui_active_theme';
    protected string $cookieKey = 'gsmui_theme';

    public function switchTo(string $theme): void
    {
        Session::put($this->sessionKey, $theme);
        Cookie::queue($this->cookieKey, $theme, 60 * 24 * 365);
    }

    public function current(): string
    {
        return Session::get($this->sessionKey, config('gsmui.theme.default', 'default'));
    }

    public function available(): array
    {
        return array_keys(config('gsmui.theme.themes', []));
    }

    public function isDark(): bool
    {
        return Session::get('gsmui_dark_mode', false);
    }

    public function toggleDarkMode(): bool
    {
        $isDark = !$this->isDark();
        Session::put('gsmui_dark_mode', $isDark);
        Cookie::queue('gsmui_dark_mode', $isDark, 60 * 24 * 365);
        return $isDark;
    }

    public function getCSSVariables(string $theme = null): array
    {
        $theme = $theme ?? $this->current();
        $config = config("gsmui.theme.themes.{$theme}", []);
        $variables = [];

        if (isset($config['colors'])) {
            foreach ($config['colors'] as $name => $shades) {
                $mode = $this->isDark() ? 'dark' : 'light';
                $variables["gsmui-color-{$name}"] = $shades[$mode] ?? $shades['light'] ?? '';
            }
        }

        return $variables;
    }

    public function renderInlineCSS(string $theme = null): string
    {
        $vars = $this->getCSSVariables($theme);
        $css = ':root { ';
        foreach ($vars as $name => $value) {
            $css .= "--{$name}: {$value}; ";
        }
        $css .= '}';
        return $css;
    }
}
```

## Theme Persistence

### Storage Strategies

```javascript
class GsmuiThemePersistence {
    constructor(options = {}) {
        this.strategy = options.strategy || 'localStorage';
        this.storageKey = options.storageKey || 'gsmui-theme';
        this.cookieName = options.cookieName || 'gsmui_theme';
        this.cookieDays = options.cookieDays || 365;
    }

    save(theme) {
        switch (this.strategy) {
            case 'localStorage':
                localStorage.setItem(this.storageKey, JSON.stringify(theme));
                break;
            case 'sessionStorage':
                sessionStorage.setItem(this.storageKey, JSON.stringify(theme));
                break;
            case 'cookie':
                this.setCookie(this.cookieName, JSON.stringify(theme), this.cookieDays);
                break;
            case 'server':
                this.saveToServer(theme);
                break;
        }
    }

    load() {
        switch (this.strategy) {
            case 'localStorage':
                const stored = localStorage.getItem(this.storageKey);
                return stored ? JSON.parse(stored) : null;
            case 'sessionStorage':
                const sessionStored = sessionStorage.getItem(this.storageKey);
                return sessionStored ? JSON.parse(sessionStored) : null;
            case 'cookie':
                const cookieValue = this.getCookie(this.cookieName);
                return cookieValue ? JSON.parse(cookieValue) : null;
            case 'server':
                return this.loadFromServer();
        }
    }

    clear() {
        switch (this.strategy) {
            case 'localStorage':
                localStorage.removeItem(this.storageKey);
                break;
            case 'sessionStorage':
                sessionStorage.removeItem(this.storageKey);
                break;
            case 'cookie':
                this.setCookie(this.cookieName, '', -1);
                break;
        }
    }

    setCookie(name, value, days) {
        const expires = new Date(Date.now() + days * 864e5).toUTCString();
        document.cookie = `${name}=${encodeURIComponent(value)}; expires=${expires}; path=/; SameSite=Lax`;
    }

    getCookie(name) {
        return document.cookie.split('; ').reduce((r, c) => {
            const [key, val] = c.split('=');
            return key === name ? decodeURIComponent(val) : r;
        }, null);
    }

    async saveToServer(theme) {
        await fetch('/api/gsmui/theme', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json', 'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]')?.content },
            body: JSON.stringify({ theme }),
        });
    }

    async loadFromServer() {
        try {
            const resp = await fetch('/api/gsmui/theme');
            const data = await resp.json();
            return data.theme || null;
        } catch {
            return null;
        }
    }
}

window.GsmuiThemePersistence = new GsmuiThemePersistence({ strategy: 'localStorage' });
```

### Efficient Theme Loading

```javascript
class GsmuiThemeOptimizer {
    constructor() {
        this.cache = new Map();
        this.preloaded = new Set();
    }

    async preloadTheme(name) {
        if (this.preloaded.has(name)) return;
        const link = document.createElement('link');
        link.rel = 'prefetch';
        link.href = `/css/themes/${name}.css`;
        document.head.appendChild(link);
        this.preloaded.add(name);
    }

    async loadThemeCSS(name) {
        if (this.cache.has(name)) return this.cache.get(name);
        const resp = await fetch(`/css/themes/${name}.css`);
        const css = await resp.text();
        this.cache.set(name, css);
        return css;
    }

    applyThemeImmediate(name, cssVars) {
        const style = document.createElement('style');
        style.id = `gsmui-theme-${name}`;
        let css = `:root { `;
        for (const [key, value] of Object.entries(cssVars)) {
            css += `--${key}: ${value}; `;
        }
        css += `}`;
        style.textContent = css;
        document.head.appendChild(style);
    }
}
```

### Theme CSS Caching Strategy

```php
<?php

namespace GsmUi\Services;

class ThemeCache
{
    protected string $cacheDir;

    public function __construct()
    {
        $this->cacheDir = storage_path('app/gsmui/themes');
        if (!is_dir($this->cacheDir)) {
            mkdir($this->cacheDir, 0755, true);
        }
    }

    public function get(string $theme): ?string
    {
        $path = "{$this->cacheDir}/{$theme}.css";
        if (!file_exists($path)) return null;
        return file_get_contents($path);
    }

    public function put(string $theme, string $css): void
    {
        file_put_contents("{$this->cacheDir}/{$theme}.css", $css);
    }

    public function clear(string $theme = null): void
    {
        if ($theme) {
            @unlink("{$this->cacheDir}/{$theme}.css");
        } else {
            $files = glob("{$this->cacheDir}/*.css");
            foreach ($files as $file) {
                @unlink($file);
            }
        }
    }

    public function version(string $theme): string
    {
        $path = "{$this->cacheDir}/{$theme}.css";
        if (!file_exists($path)) return 'none';
        return md5_file($path);
    }
}
```
