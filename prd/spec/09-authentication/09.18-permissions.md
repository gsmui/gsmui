# 09.18 — Permissions

## Purpose

Define the complete permission system for GsmUi, including permission naming, groups, assignment, and enforcement.

## Permission Naming Convention

Format: `resource.action`

```text
themes.create

themes.update
themes.delete
```

```php
public function run(): void

    $permissions = [

        'components' => ['create', 'read', 'update', 'delete', 'publish'],
        'templates' => ['create', 'read', 'update', 'delete', 'install'],
        'marketplace' => ['browse', 'purchase', 'manage'],
        'users' => ['create', 'read', 'update', 'delete', 'manage'],
        'roles' => ['create', 'read', 'update', 'delete', 'manage'],
        'billing' => ['read', 'manage'],
        'documentation' => ['create', 'read', 'update', 'delete'],
        'blog' => ['create', 'read', 'update', 'delete', 'publish'],
        'admin' => ['access', 'manage'],
    ];

    foreach ($permissions as $resource => $actions) {
        foreach ($actions as $action) {
            Permission::firstOrCreate([
                'name' => "{$resource}.{$action}",
                'guard_name' => 'web',
            ]);
        }
    }
}
```

## Custom Permissions

Admins can create custom permissions through the admin panel:

```text
Admin → Permissions → Create Permission

Name: [custom_action]
Group: [custom___]
Description: [Custom permission for specific feature]
```

## Permission Inheritance

Permissions do not inherit between roles. Each role must have its permissions explicitly assigned. This prevents accidental permission escalation.

## Security Notes

* Permission checks logged for audit
* Permission changes logged via Activity Log
* Cache cleared on permission changes
* Custom permissions require Super Admin approval
* Permission names must follow `resource.action` format

## Permission Service Layer

### Core Permission Service

```php
// app/Services/PermissionService.php

namespace App\Services;

use App\Models\User;
use App\Models\Permission;
use App\Models\Role;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class PermissionService
{
    /**
     * Get all permissions grouped by resource.
     */
    public function getGroupedPermissions(): array
    {
        return Cache::remember('permissions:grouped', 3600, function () {
            return Permission::all()
                ->groupBy('group_name')
                ->map(fn($group) => $group->keyBy('name'))
                ->toArray();
        });
    }

    /**
     * Check if a user has a specific permission.
     */
    public function userHasPermission(User $user, string $permission): bool
    {
        $cacheKey = "user:{$user->id}:permission:{$permission}";

        return Cache::remember($cacheKey, 3600, function () use ($user, $permission) {
            return $user->hasPermissionTo($permission);
        });
    }

    /**
     * Check if a user has any of the given permissions.
     */
    public function userHasAnyPermission(User $user, array $permissions): bool
    {
        foreach ($permissions as $permission) {
            if ($this->userHasPermission($user, $permission)) {
                return true;
            }
        }

        return false;
    }

    /**
     * Check if a user has all of the given permissions.
     */
    public function userHasAllPermissions(User $user, array $permissions): bool
    {
        foreach ($permissions as $permission) {
            if (! $this->userHasPermission($user, $permission)) {
                return false;
            }
        }

        return true;
    }

    /**
     * Get all permissions for a user across all roles.
     */
    public function getUserPermissions(User $user): Collection
    {
        return Cache::remember("user:{$user->id}:all-permissions", 3600, function () use ($user) {
            return $user->getAllPermissions();
        });
    }

    /**
     * Get permissions a user is missing for a given set.
     */
    public function getMissingPermissions(User $user, array $requiredPermissions): array
    {
        $userPermissions = $this->getUserPermissions($user)->pluck('name')->toArray();
        return array_diff($requiredPermissions, $userPermissions);
    }

    /**
     * Grant a direct permission to a user (bypassing roles).
     */
    public function grantPermission(User $user, string $permissionName): void
    {
        $permission = Permission::where('name', $permissionName)->firstOrFail();
        $user->givePermissionTo($permission);

        $this->clearUserPermissionCache($user);

        activity('permission-change')
            ->performedOn($user)
            ->withProperties(['permission' => $permissionName, 'action' => 'granted'])
            ->log("Permission '{$permissionName}' granted directly to user");
    }

    /**
     * Revoke a direct permission from a user.
     */
    public function revokePermission(User $user, string $permissionName): void
    {
        $permission = Permission::where('name', $permissionName)->first();
        if ($permission) {
            $user->revokePermissionTo($permission);
            $this->clearUserPermissionCache($user);

            activity('permission-change')
                ->performedOn($user)
                ->withProperties(['permission' => $permissionName, 'action' => 'revoked'])
                ->log("Permission '{$permissionName}' revoked from user");
        }
    }

    /**
     * Clear all permission caches for a user.
     */
    public function clearUserPermissionCache(User $user): void
    {
        $permissions = Permission::all();
        foreach ($permissions as $permission) {
            Cache::forget("user:{$user->id}:permission:{$permission->name}");
        }
        Cache::forget("user:{$user->id}:all-permissions");
    }

    /**
     * Clear all permission caches globally.
     */
    public function clearAllPermissionCaches(): void
    {
        Cache::forget('permissions:grouped');

        User::pluck('id')->each(function ($userId) {
            $this->clearUserPermissionCache(User::find($userId));
        });
    }

    /**
     * Validate a permission name format.
     */
    public function isValidPermissionName(string $name): bool
    {
        return (bool) preg_match('/^[a-z]+\.[a-z_]+$/', $name);
    }

    /**
     * Get permissions by group.
     */
    public function getPermissionsByGroup(string $groupName): Collection
    {
        return Cache::remember("permissions:group:{$groupName}", 3600, function () use ($groupName) {
            return Permission::where('group_name', $groupName)->get();
        });
    }

    /**
     * Get all available groups.
     */
    public function getAvailableGroups(): array
    {
        return Cache::remember('permissions:groups', 3600, function () {
            return Permission::distinct()
                ->pluck('group_name')
                ->filter()
                ->sort()
                ->toArray();
        });
    }

    /**
     * Get permission usage statistics.
     */
    public function getPermissionStats(): array
    {
        $permissions = Permission::all();
        $stats = [];

        foreach ($permissions as $permission) {
            $stats[$permission->name] = [
                'direct_users' => User::whereHas('permissions', function ($q) use ($permission) {
                    $q->where('permissions.id', $permission->id);
                })->count(),
                'role_users' => User::whereHas('roles.permissions', function ($q) use ($permission) {
                    $q->where('permissions.id', $permission->id);
                })->count(),
            ];
        }

        return $stats;
    }

    /**
     * Check if a permission is used by any role.
     */
    public function isPermissionInUse(int $permissionId): bool
    {
        return DB::table('role_has_permissions')
            ->where('permission_id', $permissionId)
            ->exists();
    }

    /**
     * Create a new permission.
     */
    public function createPermission(array $data): Permission
    {
        if (! $this->isValidPermissionName($data['name'])) {
            throw new \InvalidArgumentException(
                "Invalid permission name format: {$data['name']}. Must follow 'resource.action' format."
            );
        }

        $permission = Permission::create([
            'name' => $data['name'],
            'guard_name' => $data['guard_name'] ?? 'web',
            'display_name' => $data['display_name'] ?? null,
            'description' => $data['description'] ?? null,
            'group_name' => $data['group_name'] ?? null,
        ]);

        $this->clearAllPermissionCaches();

        activity('permission-management')
            ->performedOn($permission)
            ->log('Permission created');

        return $permission;
    }

    /**
     * Delete a permission that is not in use.
     */
    public function deletePermission(int $permissionId): void
    {
        $permission = Permission::findOrFail($permissionId);

        if ($this->isPermissionInUse($permissionId)) {
            throw new \RuntimeException('Cannot delete permission that is assigned to roles.');
        }

        $permission->delete();
        $this->clearAllPermissionCaches();

        activity('permission-management')
            ->withProperties(['permission' => $permission->name])
            ->log('Permission deleted');
    }
}
```

## Permission Models

### Permission Model Extension

```php
// app/Models/Permission.php (extensions to Spatie model)

namespace App\Models;

use Spatie\Permission\Models\Permission as SpatiePermission;

class Permission extends SpatiePermission
{
    protected $fillable = [
        'name',
        'guard_name',
        'display_name',
        'description',
        'group_name',
    ];

    protected $casts = [
        'created_at' => 'datetime',
        'updated_at' => 'datetime',
    ];

    /**
     * Get roles that have this permission.
     */
    public function roles()
    {
        return $this->belongsToMany(Role::class, 'role_has_permissions');
    }

    /**
     * Get users with this permission directly (not via roles).
     */
    public function directUsers()
    {
        return $this->morphedByMany(User::class, 'model', 'model_has_permissions');
    }

    /**
     * Get users with this permission via any role.
     */
    public function users()
    {
        return User::whereHas('roles.permissions', function ($q) {
            $q->where('permissions.id', $this->id);
        });
    }

    /**
     * Get group display name.
     */
    public function getGroupDisplayNameAttribute(): string
    {
        return ucfirst(str_replace('_', ' ', $this->group_name ?? 'general'));
    }

    /**
     * Get action from permission name.
     */
    public function getActionAttribute(): string
    {
        return substr(strrchr($this->name, '.'), 1);
    }

    /**
     * Get resource from permission name.
     */
    public function getResourceAttribute(): string
    {
        return explode('.', $this->name)[0];
    }

    /**
     * Scope: permissions in a specific group.
     */
    public function scopeInGroup($query, string $groupName)
    {
        return $query->where('group_name', $groupName);
    }

    /**
     * Scope: search permissions by name.
     */
    public function scopeSearch($query, string $search)
    {
        return $query->where('name', 'like', "%{$search}%")
            ->orWhere('display_name', 'like', "%{$search}%");
    }
}
```

## Permission Middleware

### Permission Middleware Implementation

```php
// app/Http/Middleware/CheckPermission.php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class CheckPermission
{
    public function handle(Request $request, Closure $next, string ...$permissions): Response
    {
        if (! $request->user()) {
            return redirect()->route('login');
        }

        foreach ($permissions as $permission) {
            if (! $request->user()->can($permission)) {
                if ($request->expectsJson()) {
                    return response()->json([
                        'message' => 'Unauthorized.',
                        'error' => "Missing permission: {$permission}",
                    ], 403);
                }

                abort(403, "Unauthorized. Missing permission: {$permission}");
            }
        }

        return $next($request);
    }
}

// app/Http/Middleware/CheckAnyPermission.php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class CheckAnyPermission
{
    public function handle(Request $request, Closure $next, string ...$permissions): Response
    {
        if (! $request->user()) {
            return redirect()->route('login');
        }

        $hasAny = false;
        foreach ($permissions as $permission) {
            if ($request->user()->can($permission)) {
                $hasAny = true;
                break;
            }
        }

        if (! $hasAny) {
            if ($request->expectsJson()) {
                return response()->json([
                    'message' => 'Unauthorized.',
                    'error' => 'Missing any of permissions: ' . implode(', ', $permissions),
                ], 403);
            }

            abort(403, 'Unauthorized. You do not have the required permissions.');
        }

        return $next($request);
    }
}
```

```php
// app/Livewire/Admin/Permissions/PermissionList.php

namespace App\Livewire\Admin\Permissions;

use Livewire\Component;
use App\Models\Permission;
use App\Services\PermissionService;

class PermissionList extends Component
{
    public string $search = '';
    public ?string $filterGroup = null;
    public string $view = 'grouped';

    protected PermissionService $permissionService;

    public function boot(PermissionService $permissionService): void
    {
        $this->permissionService = $permissionService;
    }

    public function getPermissionsProperty()
    {
        return Permission::query()
            ->when($this->search, function ($query) {
                $query->search($this->search);
            })
            ->when($this->filterGroup, function ($query) {
                $query->inGroup($this->filterGroup);
            })
            ->withCount('roles')
            ->orderBy('group_name')
            ->orderBy('name')
            ->get();
    }

    public function getGroupedPermissionsProperty(): array
    {
        return $this->permissions
            ->groupBy('group_name')
            ->toArray();
    }

    public function getAvailableGroupsProperty(): array
    {
        return $this->permissionService->getAvailableGroups();
    }

    public function deletePermission(int $permissionId): void
    {
        try {
            $this->permissionService->deletePermission($permissionId);
            session()->flash('success', 'Permission deleted successfully.');
        } catch (\Exception $e) {
            session()->flash('error', $e->getMessage());
        }
    }

    public function render()
    {
        return view('livewire.admin.permissions.permission-list', [
            'permissions' => $this->permissions,
            'groupedPermissions' => $this->groupedPermissions,
            'availableGroups' => $this->availableGroups,
        ]);
    }
}
```

### Permission Form Component

```php
// app/Livewire/Admin/Permissions/PermissionForm.php

namespace App\Livewire\Admin\Permissions;

use Livewire\Component;
use App\Models\Permission;
use App\Services\PermissionService;

class PermissionForm extends Component
{
    public ?int $permissionId = null;
    public string $name = '';
    public string $displayName = '';
    public string $description = '';
    public string $groupName = '';

    protected PermissionService $permissionService;

    protected array $rules = [
        'name' => ['required', 'string', 'max:125', 'regex:/^[a-z]+\.[a-z_]+$/'],
        'displayName' => ['nullable', 'string', 'max:255'],
        'description' => ['nullable', 'string', 'max:1000'],
        'groupName' => ['required', 'string', 'max:125'],
    ];

    protected array $messages = [
        'name.regex' => 'Permission name must follow "resource.action" format (e.g., themes.create).',
    ];

    public function boot(PermissionService $permissionService): void
    {
        $this->permissionService = $permissionService;
    }

    public function mount(?int $permissionId = null): void
    {
        if ($permissionId) {
            $permission = Permission::findOrFail($permissionId);
            $this->permissionId = $permission->id;
            $this->name = $permission->name;
            $this->displayName = $permission->display_name ?? '';
            $this->description = $permission->description ?? '';
            $this->groupName = $permission->group_name ?? '';
        }
    }

    public function save(): void
    {
        $this->validate();

        try {
            if ($this->permissionId) {
                $permission = Permission::findOrFail($this->permissionId);
                $permission->update([
                    'name' => $this->name,
                    'display_name' => $this->displayName,
                    'description' => $this->description,
                    'group_name' => $this->groupName,
                ]);

                $this->permissionService->clearAllPermissionCaches();

                activity('permission-management')
                    ->performedOn($permission)
                    ->log('Permission updated');

                session()->flash('success', 'Permission updated successfully.');
            } else {
                $permission = $this->permissionService->createPermission([
                    'name' => $this->name,
                    'display_name' => $this->displayName,
                    'description' => $this->description,
                    'group_name' => $this->groupName,
                ]);

                session()->flash('success', 'Permission created successfully.');
            }

            redirect()->route('admin.permissions.index');
        } catch (\Exception $e) {
            session()->flash('error', $e->getMessage());
        }
    }

    public function render()
    {
        return view('livewire.admin.permissions.permission-form');
    }
}
```

```php
// app/Http/Controllers/Api/PermissionController.php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Permission;
use App\Services\PermissionService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class PermissionController extends Controller
{
    public function __construct(
        protected PermissionService $permissionService
    ) {}

    public function index(): JsonResponse
    {
        $permissions = Permission::withCount('roles')
            ->orderBy('group_name')
            ->orderBy('name')
            ->get();

        return response()->json(['data' => $permissions]);
    }

    public function store(Request $request): JsonResponse
    {
        $validated = $request->validate([
            'name' => ['required', 'string', 'max:125', 'unique:permissions,name'],
            'display_name' => ['nullable', 'string', 'max:255'],
            'description' => ['nullable', 'string', 'max:1000'],
            'group_name' => ['required', 'string', 'max:125'],
        ]);

        $permission = $this->permissionService->createPermission($validated);

        return response()->json([
            'data' => $permission,
            'message' => 'Permission created successfully.',
        ], 201);
    }

    public function show(Permission $permission): JsonResponse
    {
        return response()->json([
            'data' => $permission->load('roles'),
        ]);
    }

    public function update(Request $request, Permission $permission): JsonResponse
    {
        $validated = $request->validate([
            'name' => ['sometimes', 'string', 'max:125', 'unique:permissions,name,' . $permission->id],
            'display_name' => ['nullable', 'string', 'max:255'],
            'description' => ['nullable', 'string', 'max:1000'],
            'group_name' => ['sometimes', 'string', 'max:125'],
        ]);

        $permission->update($validated);
        $this->permissionService->clearAllPermissionCaches();

        activity('permission-management')
            ->performedOn($permission)
            ->log('Permission updated via API');

        return response()->json([
            'data' => $permission,
            'message' => 'Permission updated successfully.',
        ]);
    }

    public function destroy(Permission $permission): JsonResponse
    {
        try {
            $this->permissionService->deletePermission($permission->id);
            return response()->json(['message' => 'Permission deleted successfully.']);
        } catch (\RuntimeException $e) {
            return response()->json(['message' => $e->getMessage()], 422);
        }
    }

    public function groups(): JsonResponse
    {
        $groups = $this->permissionService->getAvailableGroups();

        return response()->json(['data' => $groups]);
    }

    public function byGroup(string $group): JsonResponse
    {
        $permissions = $this->permissionService->getPermissionsByGroup($group);

        return response()->json(['data' => $permissions]);
    }
}
```

## Permission Cache Strategy

### Cache Architecture

```text
Permission Cache Layers:
    Layer 1: In-memory (per-request) — avoids repeated DB queries
    Layer 2: Redis/Memcached — shared across requests
    Layer 3: Database — source of truth

Cache Key Structure:
    user:{id}:permission:{name}       — single permission check
    user:{id}:all-permissions          — all user permissions
    permissions:grouped                — all permissions grouped
    permissions:group:{name}           — permissions in a group
    permissions:groups                 — list of all groups
```

### Cache Warming

```php
// app/Services/PermissionCacheWarmer.php

namespace App\Services;

use App\Models\User;
use App\Models\Permission;
use Illuminate\Support\Facades\Cache;

class PermissionCacheWarmer
{
    /**
     * Warm cache for a specific user.
     */
    public function warmUserCache(User $user): void
    {
        $permissions = $user->getAllPermissions();

        foreach ($permissions as $permission) {
            Cache::remember(
                "user:{$user->id}:permission:{$permission->name}",
                config('permissions.cache.ttl', 3600),
                fn() => true
            );
        }

        Cache::remember(
            "user:{$user->id}:all-permissions",
            config('permissions.cache.ttl', 3600),
            fn() => $permissions
        );
    }

    /**
     * Warm cache for all active users.
     */
    public function warmAllUserCaches(): void
    {
        User::where('last_active_at', '>=', now()->subDays(30))
            ->chunkById(100, function ($users) {
                foreach ($users as $user) {
                    $this->warmUserCache($user);
                }
            });
    }

    /**
     * Warm grouped permissions cache.
     */
    public function warmGroupedPermissionsCache(): void
    {
        Cache::remember(
            'permissions:grouped',
            config('permissions.cache.ttl', 3600),
            fn() => Permission::all()->groupBy('group_name')->toArray()
        );
    }

    /**
     * Warm all caches.
     */
    public function warmAll(): void
    {
        $this->warmGroupedPermissionsCache();
        $this->warmAllUserCaches();
    }
}
```

```php
namespace App\Services;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class PermissionCacheMetrics
{
    /**
     * Get cache hit/miss ratio for permissions.
     */
    public function getCacheStats(): array
    {
        $totalChecks = (int) Cache::get('metrics:permission:total_checks', 0);
        $cacheHits = (int) Cache::get('metrics:permission:cache_hits', 0);
        $cacheMisses = (int) Cache::get('metrics:permission:cache_misses', 0);

        return [
            'total_checks' => $totalChecks,
            'cache_hits' => $cacheHits,
            'cache_misses' => $cacheMisses,
            'hit_rate' => $totalChecks > 0
                ? round(($cacheHits / $totalChecks) * 100, 2)
                : 0,
        ];
    }

    /**
     * Record a cache hit.
     */
    public function recordHit(): void
    {
        Cache::increment('metrics:permission:total_checks');
        Cache::increment('metrics:permission:cache_hits');
    }

    /**
     * Record a cache miss.
     */
    public function recordMiss(): void
    {
        Cache::increment('metrics:permission:total_checks');
        Cache::increment('metrics:permission:cache_misses');
    }

    /**
     * Reset metrics.
     */
    public function reset(): void
    {
        Cache::forget('metrics:permission:total_checks');
        Cache::forget('metrics:permission:cache_hits');
        Cache::forget('metrics:permission:cache_misses');
    }
```

```php
class PermissionReportService

            'by_group' => $this->getByGroup(),

            'most_granted' => $this->getMostGrantedPermissions(),

     * Get summary statistics.

    {

            'total_permissions' => Permission::count(),
            'total_roles' => Role::count(),
            'total_users' => User::count(),
            'direct_assignments' => DB::table('model_has_permissions')->count(),
            'role_assignments' => DB::table('role_has_permissions')->count(),
        ];
    }

    /**
     * Get permissions grouped by resource.
     */
    public function getByGroup(): array
    {
        return Permission::select('group_name', DB::raw('count(*) as count'))
            ->groupBy('group_name')
            ->get()
            ->pluck('count', 'group_name')
            ->toArray();
    }

    /**
     * Get permissions not assigned to any role.
     */
    public function getOrphanedPermissions(): Collection
    {
        return Permission::whereDoesntHave('roles')
            ->whereDoesntHave('directUsers')
            ->get();
    }

    /**
     * Get most frequently granted permissions.
     */
    public function getMostGrantedPermissions(int $limit = 10): Collection
    {
        return Permission::withCount('roles')
            ->orderBy('roles_count', 'desc')
            ->limit($limit)
            ->get();
    }

    /**
     * Get least frequently used permissions.
     */
    public function getLeastUsedPermissions(int $limit = 10): Collection
    {
        return Permission::withCount('roles')
            ->orderBy('roles_count', 'asc')
            ->limit($limit)
            ->get();
    }

    /**
     * Get user coverage for each permission.
     */
    public function getUserCoverage(): array
    {
        $totalUsers = User::count();
        $results = [];

        $permissions = Permission::withCount('roles')->get();

        foreach ($permissions as $permission) {
            $usersWithPermission = User::whereHas('roles.permissions', function ($q) use ($permission) {
                $q->where('permissions.id', $permission->id);
            })->count();

            $results[$permission->name] = [
                'total_users' => $usersWithPermission,
                'percentage' => $totalUsers > 0
                    ? round(($usersWithPermission / $totalUsers) * 100, 2)
                    : 0,
            ];
        }

        return $results;
    }
```

```php
// app/Services/PermissionImporter.php

namespace App\Services;

use App\Models\Permission;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class PermissionImporter
{
    /**
     * Import permissions from a JSON structure.
     */
    public function importFromJson(array $data, bool $updateExisting = false): array
    {
        $results = [
            'created' => 0,
            'updated' => 0,
            'skipped' => 0,
            'errors' => [],
        ];

        DB::beginTransaction();

        try {
            foreach ($data as $group => $permissions) {
                foreach ($permissions as $permissionData) {
                    $name = $permissionData['name'] ?? "{$group}.{$permissionData['action']}";

                    $existing = Permission::where('name', $name)->first();

                    if ($existing) {
                        if ($updateExisting) {
                            $existing->update([
                                'display_name' => $permissionData['display_name'] ?? $existing->display_name,
                                'description' => $permissionData['description'] ?? $existing->description,
                                'group_name' => $group,
                            ]);
                            $results['updated']++;
                        } else {
                            $results['skipped']++;
                        }
                    } else {
                        Permission::create([
                            'name' => $name,
                            'guard_name' => $permissionData['guard'] ?? 'web',
                            'display_name' => $permissionData['display_name'] ?? null,
                            'description' => $permissionData['description'] ?? null,
                            'group_name' => $group,
                        ]);
                        $results['created']++;
                    }
                }
            }

            DB::commit();
            Log::info('Permission import completed', $results);

            return $results;
        } catch (\Exception $e) {
            DB::rollBack();
            $results['errors'][] = $e->getMessage();
            Log::error('Permission import failed', ['error' => $e->getMessage()]);
            return $results;
        }
    }

    /**
     * Import permissions from CSV.
     */
    public function importFromCsv(string $filePath, bool $updateExisting = false): array
    {
        $results = [
            'created' => 0,
            'updated' => 0,
            'skipped' => 0,
            'errors' => [],
        ];

        if (! file_exists($filePath)) {
            $results['errors'][] = "File not found: {$filePath}";
            return $results;
        }

        $handle = fopen($filePath, 'r');
        $headers = fgetcsv($handle);

        while (($row = fgetcsv($handle)) !== false) {
            $data = array_combine($headers, $row);

            try {
                $name = $data['name'] ?? '';
                $group = $data['group'] ?? null;

                if (! $name) {
                    $results['errors'][] = 'Row missing name: ' . json_encode($data);
                    continue;
                }

                $existing = Permission::where('name', $name)->first();

                if ($existing) {
                    if ($updateExisting) {
                        $existing->update([
                            'display_name' => $data['display_name'] ?? $existing->display_name,
                            'description' => $data['description'] ?? $existing->description,
                            'group_name' => $group,
                        ]);
                        $results['updated']++;
                    } else {
                        $results['skipped']++;
                    }
                } else {
                    Permission::create([
                        'name' => $name,
                        'guard_name' => $data['guard'] ?? 'web',
                        'display_name' => $data['display_name'] ?? null,
                        'description' => $data['description'] ?? null,
                        'group_name' => $group,
                    ]);
                    $results['created']++;
                }
            } catch (\Exception $e) {
                $results['errors'][] = "Error processing row: " . $e->getMessage();
            }
        }

        fclose($handle);
        return $results;
    }
}
```

```php
// app/Livewire/Admin/Permissions/PermissionMatrix.php

namespace App\Livewire\Admin\Permissions;

use Livewire\Component;
use App\Models\Role;
use App\Models\Permission;

class PermissionMatrix extends Component
{
    public array $matrix = [];
    public bool $isSaving = false;

    public function mount(): void
    {
        $roles = Role::with('permissions')->orderBy('hierarchy_level', 'desc')->get();
        $permissions = Permission::orderBy('group_name')->orderBy('name')->get();

        foreach ($roles as $role) {
            $this->matrix[$role->id] = $permissions->pluck('id')->mapWithKeys(function ($permId) use ($role) {
                return [$permId => $role->permissions->contains('id', $permId)];
            })->toArray();
        }

        $this->roles = $roles;
        $this->permissions = $permissions;
    }

    public function togglePermission(int $roleId, int $permissionId): void
    {
        $role = Role::findOrFail($roleId);
        $permission = Permission::findOrFail($permissionId);

        if ($this->matrix[$roleId][$permissionId]) {
            $role->revokePermissionTo($permission);
            $this->matrix[$roleId][$permissionId] = false;
        } else {
            $role->givePermissionTo($permission);
            $this->matrix[$roleId][$permissionId] = true;
        }
    }

    public function grantAllToRole(int $roleId): void
    {
        $role = Role::findOrFail($roleId);
        $permissionIds = Permission::pluck('id')->toArray();

        $role->syncPermissions($permissionIds);

        foreach ($permissionIds as $permId) {
            $this->matrix[$roleId][$permId] = true;
        }
    }

    public function revokeAllFromRole(int $roleId): void
    {
        $role = Role::findOrFail($roleId);
        $permissionIds = Permission::pluck('id')->toArray();

        $role->syncPermissions([]);

        foreach ($permissionIds as $permId) {
            $this->matrix[$roleId][$permId] = false;
        }
    }

    public function render()
    {
        return view('livewire.admin.permissions.permission-matrix', [
            'roles' => $this->roles,
            'permissions' => $this->permissions,
        ]);
    }
}
```
