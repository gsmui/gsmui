# 09.08 — Passkeys / WebAuthn

## Purpose

Define the future Passkey (WebAuthn) authentication system for GsmUi, enabling passwordless, phishing-resistant authentication.

## Status: Phase 3 (Future)

Passkeys are a future feature. This document defines the architecture for implementation.

## What Are Passkeys

Passkeys are FIDO2/WebAuthn credentials that replace passwords entirely. They use public-key cryptography and are bound to the user's device.

## Detailed WebAuthn Architecture

### What Are Passkeys

Passkeys are FIDO2/WebAuthn credentials that replace passwords entirely. They use public-key cryptography and are bound to the user's device. Passkeys provide:

1. **Phishing resistance** — Credentials are bound to specific origins
2. **No shared secrets** — Public-key cryptography, no passwords transmitted
3. **Device-bound** — Credentials cannot be copied or transferred
4. **User verification** — Biometric or PIN required on device
5. **Replay protection** — Challenge-response mechanism
6. **Clone detection** — Counter prevents credential duplication

### WebAuthn Terminology

| Term | Definition |
|---|---|
| Relying Party (RP) | The server (GsmUi) that relies on the authenticator |
| Authenticator | The device that creates and stores the credential (phone, security key) |
| Credential | The public-key/private-key pair |
| Attestation | Proof that the credential was created by a genuine authenticator |
| Assertion | Proof that the user controls the private key |
| User Verification | Biometric or PIN check on the authenticator |
| User Presence | Simple touch/click on the authenticator |

### WebAuthn Registration Implementation

```php
// app/Services/WebAuthnService.php

class WebAuthnService
{
    public function generateRegistrationChallenge(User $user): array
    {
        $rp = $this->getRelyingParty();
        $userEntity = $this->getUserEntity($user);

        // Get existing credentials for exclusion list
        $excludedCredentials = $user->passkeys->map(function ($passkey) {
            return new PublicKeyCredentialDescriptor(
                PublicKeyCredentialDescriptor::TYPE_PUBLIC_KEY,
                base64_decode($passkey->credential_id),
                $passkey->transports
            );
        })->toArray();

        $authenticatorSelection = new AuthenticatorSelectionCriteria(
            residentKey: 'preferred',
            userVerification: 'preferred',
        );

        $challenge = $this->server->generatePublicKeyCredentialCreationOptions(
            user: $userEntity,
            publicKeyCredentialRp: $rp,
            authenticatorSelection: $authenticatorSelection,
            excludedCredentials: $excludedCredentials,
        );

        // Store challenge in session
        session(['webauthn_registration_challenge' => $challenge]);

        return $challenge;
    }

    public function verifyRegistration(User $user, array $data): Passkey
    {
        $challenge = session('webauthn_registration_challenge');

        $publicKeyCredential = $this->server->decodeAndParseAttestation(
            $data,
            $challenge,
            new AuthenticatorAssertionResponseValidator()
        );

        // Store the credential
        $passkey = Passkey::create([
            'user_id' => $user->id,
            'name' => $data['name'] ?? 'Passkey',
            'credential_id' => base64_encode($publicKeyCredential->getCredentialId()),
            'public_key' => base64_encode($publicKeyCredential->getPublicKey()),
            'counter' => $publicKeyCredential->getCounter(),
            'authenticator_type' => $this->getAuthenticatorType($publicKeyCredential),
            'transports' => $publicKeyCredential->getTransports(),
        ]);

        event(new PasskeyRegistered($user, $passkey));

        return $passkey;
    }

    protected function getRelyingParty(): PublicKeyCredentialRpEntity
    {
        return new PublicKeyCredentialRpEntity(
            name: config('webauthn.rp_name'),
            id: config('webauthn.rp_id'),
        );
    }

    protected function getUserEntity(User $user): PublicKeyCredentialUserEntity
    {
        return new PublicKeyCredentialUserEntity(
            name: $user->email,
            id: $user->id . '',
            displayName: $user->name,
        );
    }
}
```

### WebAuthn Authentication Implementation

```php
// app/Livewire/Auth/PasskeyLogin.php

class PasskeyLogin extends Component
{
    public string $email = '';
    public bool $showPasskeyPrompt = false;
    public ?string $challenge = null;

    public function mount(): void
    {
        $this->email = session('login_email', '');
    }

    public function startPasskeyLogin(): void
    {
        $this->validate(['email' => ['required', 'email']]);

        $user = User::where('email', $this->email)->first();

        if (! $user || ! $user->passkeys()->exists()) {
            $this->addError('email', 'No passkeys found for this email.');
            return;
        }

        $passkeys = $user->passkeys->map(function ($passkey) {
            return new PublicKeyCredentialDescriptor(
                PublicKeyCredentialDescriptor::TYPE_PUBLIC_KEY,
                base64_decode($passkey->credential_id),
                $passkey->transports
            );
        })->toArray();

        $challenge = $this->server->generatePublicKeyCredentialRequestOptions(
            userVerification: 'preferred',
            allowedCredentials: $passkeys,
        );

        $this->challenge = json_encode($challenge);
        $this->showPasskeyPrompt = true;
    }

    public function completePasskeyLogin(array $data): void
    {
        $user = User::where('email', $this->email)->first();

        $passkey = $user->passkeys()
            ->where('credential_id', base64_encode($data['id']))
            ->first();

        if (! $passkey) {
            $this->addError('email', 'Invalid passkey.');
            return;
        }

        $publicKeyCredentialSource = $this->server->decodeAndParseAssertion(
            $data,
            $this->challenge,
            new PublicKeyCredentialSource(
                id: base64_decode($passkey->credential_id),
                type: PublicKeyCredentialSource::TYPE_PUBLIC_KEY,
                transports: $passkey->transports,
                PublicKeyCredentialSource::USER_HANDLE => $user->id . '',
            )
        );

        // Verify counter
        if ($publicKeyCredentialSource->getCounter() <= $passkey->counter) {
            $this->addError('email', 'Passkey verification failed.');
            return;
        }

        // Update counter
        $passkey->update([
            'counter' => $publicKeyCredentialSource->getCounter(),
        ]);

        // Login user
        Auth::login($user);

        activity()
            ->performedOn($user)
            ->withProperties([
                'passkey_id' => $passkey->id,
                'ip' => request()->ip(),
            ])
            ->log('auth.passkey_login');

        redirect()->intended('/dashboard');
    }
}
```

### Passkey Management

```php
// app/Livewire/Settings/PasskeyManager.php

class PasskeyManager extends Component
{
    public Collection $passkeys;
    public string $newPasskeyName = '';
    public bool $showAddModal = false;

    public function mount(): void
    {
        $this->passkeys = Auth::user()->passkeys()
            ->orderBy('last_used_at', 'desc')
            ->get();
    }

    public function addPasskey(): void
    {
        $this->validate([
            'newPasskeyName' => ['required', 'string', 'max:255'],
        ]);

        $challenge = app(WebAuthnService::class)
            ->generateRegistrationChallenge(Auth::user());

        $this->showAddModal = true;
    }

    public function confirmAddPasskey(array $data): void
    {
        $passkey = app(WebAuthnService::class)
            ->verifyRegistration(Auth::user(), $data);

        $passkey->update(['name' => $this->newPasskeyName]);

        $this->passkeys = Auth::user()->passkeys()->get();
        $this->showAddModal = false;
        $this->newPasskeyName = '';
    }

    public function deletePasskey(Passkey $passkey): void
    {
        $this->authorize('delete', $passkey);

        $passkey->delete();

        $this->passkeys = Auth::user()->passkeys()->get();
    }

    public function renamePasskey(Passkey $passkey, string $name): void
    {
        $this->authorize('update', $passkey);

        $passkey->update(['name' => $name]);

        $this->passkeys = Auth::user()->passkeys()->get();
    }
}
```

### Passkey Security Checklist

- [ ] Challenge-response prevents replay attacks
- [ ] Counter prevents clone attacks
- [ ] User verification required on device
- [ ] Credentials bound to origin (phishing resistant)
- [ ] Passkey registration requires password confirmation
- [ ] Passkey deletion logged
- [ ] Passkey last_used_at updated on each use
- [ ] Transports stored for WebAuthn API
- [ ] Attestation verification (optional)
- [ ] Challenge timeout enforced (60 seconds)

### Passkey Troubleshooting

| Issue | Cause | Solution |
|---|---|---|
| Passkey not available | Browser doesn't support WebAuthn | Use a supported browser |
| Registration fails | User cancelled or timeout | Retry registration |
| Authentication fails | Counter mismatch or invalid signature | Re-register passkey |
| Passkey not syncing | Platform sync not enabled | Enable platform sync |
| Can't delete last passkey | Would lock user out | Keep at least one passkey |
| Challenge timeout | User took too 60 seconds | Retry with new challenge |

```php
class PasskeyController extends Controller
{

        $passkeys = Auth::user()->passkeys()
            ->orderBy('last_used_at', 'desc')
            ->get();

        return view('settings.passkeys', compact('passkeys'));

    public function store(Request $request): JsonResponse

        ]);

        $user = $request->user();

        $service = app(WebAuthnService::class);

        $service = app(WebAuthnService::class);

                'passkey' => [

                    'name' => $passkey->name,

                'message' => 'Passkey registration failed.',
            ], 422);
        }
    }

        $passkey->delete();

        event(new PasskeyDeleted(Auth::user(), $passkey));

        return response()->json([
            'message' => 'Passkey deleted successfully.',
        ]);
    }

    public function rename(Request $request, Passkey $passkey): JsonResponse
    {
        $this->authorize('update', $passkey);

        $request->validate([
            'name' => ['required', 'string', 'max:255'],
        ]);

        $passkey->update(['name' => $request->name]);
```

### WebAuthn Service

```php
// app/Services/WebAuthnService.php

class WebAuthnService
{
    protected Server $server;

    public function __construct()
    {
        $this->server = new Server(
            new TruePathResolver(),
            new ArrayTrustRepository()
        );

        $this->server->setRpId(config('webauthn.rp_id'));
        $this->server->setOrigin(config('webauthn.origin'));
    }

    public function generateRegistrationChallenge(User $user): array
    {
        $rp = new PublicKeyCredentialRpEntity(
            config('webauthn.rp_name'),
            config('webauthn.rp_id')
        );

        $userEntity = new PublicKeyCredentialUserEntity(
            $user->email,
            $user->id . '',
            $user->name
        );

        $excludedCredentials = $user->passkeys->map(function ($passkey) {
            return new PublicKeyCredentialDescriptor(
                PublicKeyCredentialDescriptor::TYPE_PUBLIC_KEY,
                base64_decode($passkey->credential_id),
                $passkey->transports
            );
        })->toArray();

        $authenticatorSelection = new AuthenticatorSelectionCriteria(
            residentKey: 'preferred',
            userVerification: 'preferred'
        );

        return $this->server->generatePublicKeyCredentialCreationOptions(
            user: $userEntity,
            publicKeyCredentialRp: $rp,
            authenticatorSelection: $authenticatorSelection,
            excludedCredentials: $excludedCredentials
        );
    }

    public function verifyRegistration(User $user, array $data): Passkey
    {
        $challenge = session('webauthn_registration_challenge');

        $publicKeyCredential = $this->server->decodeAndParseAttestation(
            $data,
            $challenge
        );

        $passkey = Passkey::create([
            'user_id' => $user->id,
            'name' => $data['name'] ?? 'Passkey',
            'credential_id' => base64_encode($publicKeyCredential->getCredentialId()),
            'public_key' => base64_encode($publicKeyCredential->getPublicKey()),
            'counter' => $publicKeyCredential->getCounter(),
            'authenticator_type' => $this->getAuthenticatorType($publicKeyCredential),
            'transports' => $publicKeyCredential->getTransports(),
        ]);

        event(new PasskeyRegistered($user, $passkey));

        return $passkey;
    }

    public function generateAuthenticationChallenge(User $user): array
    {
        $passkeys = $user->passkeys->map(function ($passkey) {
            return new PublicKeyCredentialDescriptor(
                PublicKeyCredentialDescriptor::TYPE_PUBLIC_KEY,
                base64_decode($passkey->credential_id),
                $passkey->transports
            );
        })->toArray();

        return $this->server->generatePublicKeyCredentialRequestOptions(
            userVerification: 'preferred',
            allowedCredentials: $passkeys
        );
    }

    public function verifyAuthentication(User $user, array $data): void
    {
        $challenge = session('webauthn_authentication_challenge');

        $credential = $user->passkeys()
            ->where('credential_id', base64_encode($data['id']))
            ->first();

        if (! $credential) {
            throw new PasskeyNotFoundException();
        }

        $publicKeyCredentialSource = $this->server->decodeAndParseAssertion(
            $data,
            $challenge
        );

        if ($publicKeyCredentialSource->getCounter() <= $credential->counter) {
            throw new PasskeyCloneDetectedException();
        }

        $credential->update([
            'counter' => $publicKeyCredentialSource->getCounter(),
            'last_used_at' => now(),
        ]);
    }

    protected function getAuthenticatorType($credential): string
    {
        return match(true) {
            $credential->isUserPresent() && $credential->isUserVerified() => 'biometric',
            $credential->isUserPresent() => 'security_key',
            default => 'platform',
        };
    }
}
```

### Passkey Security Checklist

- [ ] Challenge-response prevents replay attacks
- [ ] Counter prevents clone attacks
- [ ] User verification required on device
- [ ] Credentials bound to origin (phishing resistant)
- [ ] Passkey registration requires password confirmation
- [ ] Passkey deletion logged
- [ ] Passkey last_used_at updated on each use
- [ ] Transports stored for WebAuthn API
- [ ] Attestation verification (optional)
- [ ] Challenge timeout enforced (60 seconds)
- [ ] Only one passkey deletion allowed (keep at least one)
- [ ] Activity logging for all passkey events
- [ ] IP logging for passkey authentication
- [ ] User agent logging for passkey authentication
- [ ] Passkey names are user-defined

### Passkey Service Class

```php
// app/Services/PasskeyService.php

namespace App\Services;

use App\Models\Passkey;
use App\Models\User;
use Cose\Algorithm\Manager;
use Cose\Algorithm\Sign\EC2\ES256;
use Cose\Algorithm\Sign\RSA\PS256;
use Cose\Key;
use Illuminate\Support\Str;
use LBernardoni\SimpleWebAuthn\AuthenticationResponse;
use LBernardoni\SimpleWebAuthn\RegistrationResponse;
use LBernardoni\SimpleWebAuthn\Services\WebAuthn;

class PasskeyService
{
    protected WebAuthn $webAuthn;

    public function __construct()
    {
        $this->webAuthn = new WebAuthn(
            config('app.name'),
            config('app.url'),
            60 // Challenge timeout in seconds
        );
    }

    public function generateRegistrationChallenge(User $user): array
    {
        $excludeCredentials = $user->passkeys->map(fn ($passkey) => [
            'type' => 'public-key',
            'id' => $passkey->credential_id,
            'transports' => $passkey->transports ?? ['internal'],
        ])->toArray();

        $challenge = $this->webAuthn->generateRegistrationChallenge(
            userId: $user->id,
            userName: $user->email,
            userDisplayName: $user->name,
            excludeCredentials: $excludeCredentials
        );

        return [
            'challenge' => $challenge['challenge'],
            'rp' => $challenge['rp'],
            'user' => $challenge['user'],
            'pubKeyCredParams' => $challenge['pubKeyCredParams'],
            'timeout' => $challenge['timeout'],
            'attestation' => $challenge['attestation'],
            'excludeCredentials' => $excludeCredentials,
            'authenticatorSelection' => [
                'authenticatorAttachment' => 'cross-platform',
                'userVerification' => 'preferred',
                'residentKey' => 'preferred',
            ],
        ];
    }

    public function verifyRegistration(User $user, array $response, string $challenge): Passkey
    {
        $registrationResponse = RegistrationResponse::create($response);

        $credential = $this->webAuthn->validateRegistration(
            $registrationResponse,
            $challenge
        );

        return $user->passkeys()->create([
            'name' => $response['name'] ?? 'Unnamed Passkey',
            'credential_id' => $credential->getCredentialId(),
            'public_key' => base64_encode($credential->getPublicKey()),
            'counter' => $credential->getSignCount(),
            'authenticator_type' => $credential->getAuthenticatorType(),
            'transports' => $response['transports'] ?? ['internal'],
            'last_used_at' => now(),
        ]);
    }

    public function generateAuthenticationChallenge(string $email): ?array
    {
        $user = User::where('email', $email)->first();

        if (! $user || $user->passkeys->isEmpty()) {
            return null;
        }

        $allowedCredentials = $user->passkeys->map(fn ($passkey) => [
            'type' => 'public-key',
            'id' => $passkey->credential_id,
            'transports' => $passkey->transports ?? ['internal'],
        ])->toArray();

        $challenge = $this->webAuthn->generateAuthenticationChallenge(
            allowedCredentials: $allowedCredentials
        );

        return [
            'challenge' => $challenge['challenge'],
            'timeout' => $challenge['timeout'],
            'rpId' => $challenge['rpId'],
            'userVerification' => $challenge['userVerification'],
            'allowCredentials' => $allowedCredentials,
        ];
    }

    public function verifyAuthentication(array $response, string $challenge): ?Passkey
    {
        $authResponse = AuthenticationResponse::create($response);

        $passkey = Passkey::where('credential_id', $authResponse->getId())->first();

        if (! $passkey) {
            return null;
        }

        $credential = $this->webAuthn->validateAuthentication(
            $authResponse,
            $challenge,
            $passkey->public_key,
            $passkey->counter
        );

        if (! $credential) {
            return null;
        }

        // Check for clone attack (counter should have incremented)
        if ($credential->getSignCount() <= $passkey->counter && $passkey->counter > 0) {
            event(new PasskeyCloneDetected($passkey->user, $passkey));
            return null;
        }

        $passkey->update([
            'counter' => $credential->getSignCount(),
            'last_used_at' => now(),
        ]);

        return $passkey;
    }
}
```

### Passkey JavaScript Integration

```javascript
// resources/js/passkeys.js

export class PasskeyManager {
    constructor() {
        this.webAuthnSupported = window.PublicKeyCredential !== undefined;
    }

    async isPlatformAuthenticatorAvailable() {
        if (!this.webAuthnSupported) return false;

        return await PublicKeyCredential.isUserVerifyingPlatformAuthenticatorAvailable();
    }

    async register(options) {
        if (!this.webAuthnSupported) {
            throw new Error('WebAuthn is not supported in this browser');
        }

        const credential = await navigator.credentials.create({
            publicKey: {
                challenge: this.base64ToBuffer(options.challenge),
                rp: {
                    name: options.rp.name,
                    id: options.rp.id,
                },
                user: {
                    id: this.base64ToBuffer(options.user.id),
                    name: options.user.name,
                    displayName: options.user.displayName,
                },
                pubKeyCredParams: options.pubKeyCredParams.map(param => ({
                    ...param,
                    alg: param.alg,
                })),
                authenticatorSelection: {
                    authenticatorAttachment: 'cross-platform',
                    userVerification: 'preferred',
                    residentKey: 'preferred',
                },
                timeout: options.timeout || 60000,
                attestation: 'none',
                excludeCredentials: (options.excludeCredentials || []).map(cred => ({
                    type: 'public-key',
                    id: this.base64ToBuffer(cred.id),
                    transports: cred.transports || ['internal'],
                })),
            },
        });

        return {
            id: credential.rawId,
            rawId: this.bufferToBase64(credential.rawId),
            response: {
                attestationObject: this.bufferToBase64(credential.response.attestationObject),
                clientDataJSON: this.bufferToBase64(credential.response.clientDataJSON),
            },
            type: credential.type,
            transports: credential.response.getTransports(),
        };
    }

    async authenticate(options) {
        if (!this.webAuthnSupported) {
            throw new Error('WebAuthn is not supported in this browser');
        }

        const assertion = await navigator.credentials.get({
            publicKey: {
                challenge: this.base64ToBuffer(options.challenge),
                rpId: options.rpId,
                allowCredentials: (options.allowCredentials || []).map(cred => ({
                    type: 'public-key',
                    id: this.base64ToBuffer(cred.id),
                    transports: cred.transports || ['internal'],
                })),
                userVerification: options.userVerification || 'preferred',
                timeout: options.timeout || 60000,
            },
        });

        return {
            id: assertion.rawId,
            rawId: this.bufferToBase64(assertion.rawId),
            response: {
                authenticatorData: this.bufferToBase64(assertion.response.authenticatorData),
                clientDataJSON: this.bufferToBase64(assertion.response.clientDataJSON),
                signature: this.bufferToBase64(assertion.response.signature),
                userHandle: assertion.response.userHandle
                    ? this.bufferToBase64(assertion.response.userHandle)
                    : null,
            },
            type: assertion.type,
        };
    }

    base64ToBuffer(base64) {
        const binary = atob(base64.replace(/-/g, '+').replace(/_/g, '/'));
        const bytes = new Uint8Array(binary.length);
        for (let i = 0; i < binary.length; i++) {
            bytes[i] = binary.charCodeAt(i);
        }
        return bytes.buffer;
    }

    bufferToBase64(buffer) {
        const bytes = new Uint8Array(buffer);
        let binary = '';
        for (let i = 0; i < bytes.length; i++) {
            binary += String.fromCharCode(bytes[i]);
        }
        return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    }
}

document.addEventListener('alpine:init', () => {

                this.error = err.message || 'An error occurred.';
            } finally {
                this.isProcessing = false;
            }
        },

        async authenticate(email) {
            this.isProcessing = true;
            this.error = null;

            try {
                const challengeResponse = await fetch('/auth/passkey/challenge', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content,
                    },
                    body: JSON.stringify({ email }),
                });

                const challenge = await challengeResponse.json();

                if (!challenge) {
                    this.error = 'No passkeys found for this email.';
                    return;
                }

                const assertion = await this.manager.authenticate(challenge);

                const verifyResponse = await fetch('/auth/passkey/verify', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content,
                    },
                    body: JSON.stringify(assertion),
                });

                if (verifyResponse.ok) {
                    window.location.href = '/dashboard';
                } else {
                    this.error = 'Passkey authentication failed.';
                }
            } catch (err) {
                this.error = err.message || 'An error occurred.';
            } finally {
                this.isProcessing = false;
            }
        },
    }));
});
```

```php
// app/Services/PasskeyMigrationService.php

class PasskeyMigrationService
{
    public function migrateFromLegacyWebAuthn(User $user): void
    {
        // If user has legacy WebAuthn credentials
        $legacyCredentials = $user->webauthn_credentials ?? [];
        
        foreach ($legacyCredentials as $credential) {
            Passkey::create([
                'user_id' => $user->id,
                'name' => $credential['name'] ?? 'Migrated Passkey',
                'credential_id' => $credential['credential_id'],
                'public_key' => $credential['public_key'],
                'counter' => $credential['counter'] ?? 0,
                'authenticator_type' => $credential['type'] ?? 'unknown',
                'transports' => $credential['transports'] ?? ['internal'],
                'last_used_at' => $credential['last_used_at'] ?? null,
            ]);
        }
        
        // Clear legacy field
        $user->update(['webauthn_credentials' => null]);
    }

    public function importFromArray(User $user, array $passkeys): int
    {
        $imported = 0;
        
        foreach ($passkeys as $passkeyData) {
            if (! Passkey::where('credential_id', $passkeyData['credential_id'])->exists()) {
                Passkey::create([
                    'user_id' => $user->id,
                    'name' => $passkeyData['name'] ?? 'Imported Passkey',
                    'credential_id' => $passkeyData['credential_id'],
                    'public_key' => $passkeyData['public_key'],
                    'counter' => $passkeyData['counter'] ?? 0,
                    'authenticator_type' => $passkeyData['authenticator_type'] ?? 'unknown',
                    'transports' => $passkeyData['transports'] ?? ['internal'],
                    'last_used_at' => $passkeyData['last_used_at'] ?? null,
                ]);
                $imported++;
            }
        }
        
        return $imported;
    }
```

### Passkey Disaster Recovery

```php
// app/Console/Commands/PasskeyDisasterRecovery.php

class PasskeyDisasterRecovery extends Command
{
    protected $signature = 'passkey:disaster-recovery 
                            {--user-id= : Specific user ID}
                            {--rebuild-counters : Rebuild counters from auth logs}
                            {--remove-duplicates : Remove duplicate credential IDs}
                            {--dry-run : Show what would be done}';

    protected $description = 'Disaster recovery for passkey data';

    public function handle(): int
    {
        $query = Passkey::query();
        
        if ($this->option('user-id')) {
            $query->where('user_id', $this->option('user-id'));
        }
        
        $passkeys = $query->get();
        $fixed = 0;
        
        foreach ($passkeys as $passkey) {
            // Check for duplicate credential IDs
            if ($this->option('remove-duplicates')) {
                $duplicates = Passkey::where('credential_id', $passkey->credential_id)
                    ->where('id', '!=', $passkey->id)
                    ->get();
                
                foreach ($duplicates as $dup) {
                    if (! $this->option('dry-run')) {
                        $dup->delete();
                    }
                    $fixed++;
                }
            }
            
            // Rebuild counters from authentication logs
            if ($this->option('rebuild-counters')) {
                $maxCounter = ActivityLog::where('log_name', 'passkey_auth')
                    ->where('properties->passkey_id', $passkey->id)
                    ->max('properties->counter');
                
                if ($maxCounter && $maxCounter > $passkey->counter) {
                    if (! $this->option('dry-run')) {
                        $passkey->update(['counter' => $maxCounter]);
                    }
                    $fixed++;
                }
            }
            
            // Validate public key format
            if (! $this->isValidPublicKey($passkey->public_key)) {
                $this->warn("Invalid public key for passkey {$passkey->id}");
            }
        }
        
        $this->info("Fixed {$fixed} issues.");
        return 0;
    }
    
    protected function isValidPublicKey(string $key): bool
    {
        try {
            $decoded = base64_decode($key, true);
            return $decoded !== false && strlen($decoded) > 0;
        } catch (\Exception $e) {
            return false;
        }
    }
}
```
