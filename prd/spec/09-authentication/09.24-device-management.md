## Device Data

| Field | Source | Purpose |
|---|---|---|
| Browser | User-Agent parsing | Device identification |
| OS | User-Agent parsing | Device identification |
| IP Address | Request | Location + security |
| Location | IP geolocation | Security monitoring |
| Last Active | Session activity | Inactivity detection |
| First Seen | Session creation | Trust building |
| Trusted | User action | Skip 2FA on trusted devices |

## Device Tracking

```php
// app/Services/DeviceService.php

class DeviceService
{
    public function trackDevice(User $user): UserDevice
    {
        $userAgent = request()->userAgent();

        return UserDevice::updateOrCreate(
            [
                'user_id' => $user->id,
                'session_id' => session()->getId(),
            ],
            [
                'browser' => $this->parseBrowser($userAgent),
                'os' => $this->parseOS($userAgent),
                'ip_address' => request()->ip(),
                'location' => $this->getLocation(request()->ip()),
                'last_active_at' => now(),
            ]
        );
    }

    public function revokeDevice(User $user, int $deviceId): void
    {
        $device = UserDevice::where('user_id', $user->id)
            ->findOrFail($deviceId);

        // Delete the session
        Session::where('id', $device->session_id)->delete();

        $device->delete();
    }

    public function revokeAllDevices(User $user): void
    {
        // Keep current session
        $currentSessionId = session()->getId();

        Session::where('user_id', $user->id)
            ->where('id', '!=', $currentSessionId)
            ->delete();

        UserDevice::where('user_id', $user->id)
            ->where('session_id', '!=', $currentSessionId)
            ->delete();
    }
}
```

## Trusted Devices

Trusted devices can skip 2FA verification:

## Security Notes

* Device data stored with session lifecycle
* Trusted device status persists across sessions
* IP changes trigger re-verification
* Suspicious device additions alert user via email
* Device revocation revokes the associated session
* All device management actions logged

### Device Fingerprint Service

```php
// app/Services/DeviceFingerprintService.php

namespace App\Services;

use Illuminate\Http\Request;

class DeviceFingerprintService
{
    public function generateFingerprint(Request $request): string
    {
        $components = [
            $request->userAgent(),
            $request->header('Accept-Language', ''),
            $request->header('Accept-Encoding', ''),
            $this->getScreenResolution($request),
            $this->getTimezone($request),
        ];

        return hash('sha256', implode('|', $components));
    }

    public function getDeviceType(string $userAgent): string
    {
        $userAgent = strtolower($userAgent);

        if (preg_match('/tablet|ipad/i', $userAgent)) {
            return 'tablet';
        }

        if (preg_match('/mobile|android|iphone|windows phone/i', $userAgent)) {
            return 'mobile';
        }

        if (preg_match('/desktop|windows|mac|linux/i', $userAgent)) {
            return 'desktop';
        }

        return 'unknown';
    }

    public function parseBrowser(string $userAgent): array
    {
        $browsers = [
            'Edge' => '/Edg[e\/]([\d.]+)/i',
            'Chrome' => '/Chrome\/([\d.]+)/i',
            'Firefox' => '/Firefox\/([\d.]+)/i',
            'Safari' => '/Version\/([\d.]+).*Safari/i',
            'Opera' => '/(?:OPR|Opera)\/([\d.]+)/i',
            'IE' => '/(?:MSIE |Trident.*rv:)([\d.]+)/i',
        ];

        foreach ($browsers as $name => $pattern) {
            if (preg_match($pattern, $userAgent, $matches)) {
                return [
                    'name' => $name,
                    'version' => $matches[1] ?? null,
                ];
            }
        }

        return ['name' => 'Unknown', 'version' => null];
    }

    public function parseOS(string $userAgent): array
    {
        $operatingSystems = [
            'Windows 11' => '/Windows NT 10.0.*Build\/(\d{5,})/i',
            'Windows 10' => '/Windows NT 10\.0/i',
            'Windows 8.1' => '/Windows NT 6\.3/i',
            'Windows 7' => '/Windows NT 6\.1/i',
            'macOS' => '/Mac OS X ([\d_]+)/i',
            'Linux' => '/Linux/i',
            'Android' => '/Android ([\d.]+)/i',
            'iOS' => '/(?:iPhone|iPad).*OS ([\d_]+)/i',
            'ChromeOS' => '/CrOS/i',
        ];

        foreach ($operatingSystems as $name => $pattern) {
            if (preg_match($pattern, $userAgent, $matches)) {
                $version = $matches[1] ?? null;
                if ($version) {
                    $version = str_replace('_', '.', $version);
                }
                return ['name' => $name, 'version' => $version];
            }
        }

        return ['name' => 'Unknown', 'version' => null];
    }

    private function getScreenResolution(Request $request): string
    {
        return $request->input('screen_resolution', 'unknown');
    }

    private function getTimezone(Request $request): string
    {
        return $request->input('timezone', 'unknown');
    }
}
```

## Enhanced Device Tracking Service

```php
// app/Services/DeviceTrackingService.php

namespace App\Services;

use App\Models\IpGeolocationCache;
use App\Models\UserDevice;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Http;

class DeviceTrackingService
{
    public function __construct(
        private DeviceFingerprintService $fingerprintService
    ) {}

    public function trackDevice(Request $request, int $userId): UserDevice
    {
        $fingerprint = $this->fingerprintService->generateFingerprint($request);
        $userAgent = $request->userAgent();
        $browser = $this->fingerprintService->parseBrowser($userAgent);
        $os = $this->fingerprintService->parseOS($userAgent);
        $deviceType = $this->fingerprintService->getDeviceType($userAgent);
        $location = $this->getGeolocation($request->ip());

        // Mark all other devices as not current
        UserDevice::where('user_id', $userId)
            ->update(['is_current' => false]);

        return UserDevice::updateOrCreate(
            [
                'user_id' => $userId,
                'session_id' => session()->getId(),
            ],
            [
                'device_fingerprint' => $fingerprint,
                'browser' => $browser['name'],
                'browser_version' => $browser['version'],
                'os' => $os['name'],
                'os_version' => $os['version'],
                'device_type' => $deviceType,
                'ip_address' => $request->ip(),
                'ip_country' => $location['country_code'] ?? null,
                'ip_city' => $location['city'] ?? null,
                'ip_latitude' => $location['latitude'] ?? null,
                'ip_longitude' => $location['longitude'] ?? null,
                'is_current' => true,
                'last_active_at' => now(),
                'login_at' => now(),
            ]
        );
    }

    public function updateActivity(int $userId): void
    {
        UserDevice::where('user_id', $userId)
            ->where('session_id', session()->getId())
            ->update(['last_active_at' => now()]);
    }

    public function revokeDevice(int $userId, int $deviceId): bool
    {
        $device = UserDevice::where('user_id', $userId)
            ->where('id', $deviceId)
            ->where('is_current', false)
            ->first();

        if (! $device) {
            return false;
        }

        // Delete the session
        \Illuminate\Support\Facades\Session::where('id', $device->session_id)->delete();

        $device->update(['revoked_at' => now()]);
        $device->delete();

        activity('device')
            ->performedOn(\App\Models\User::find($userId))
            ->withProperties([
                'device' => $device->browser . ' on ' . $device->os,
                'ip' => $device->ip_address,
            ])
            ->log('Device revoked');

        return true;
    }

    public function revokeAllDevices(int $userId, ?string $exceptSessionId = null): int
    {
        $query = UserDevice::where('user_id', $userId)
            ->where('is_current', false);

        if ($exceptSessionId) {
            $query->where('session_id', '!=', $exceptSessionId);
        }

        $devices = $query->get();
        $count = $devices->count();

        // Delete Laravel sessions
        $sessionIds = $devices->pluck('session_id')->toArray();
        \Illuminate\Support\Facades\Session::whereIn('id', $sessionIds)->delete();

        $devices->each->delete();

        activity('device')
            ->performedOn(\App\Models\User::find($userId))
            ->withProperties(['revoked_count' => $count])
            ->log('All devices revoked');

        return $count;
    }

    public function trustDevice(int $userId, int $deviceId): bool
    {
        $device = UserDevice::where('user_id', $userId)
            ->where('id', $deviceId)
            ->first();

        if (! $device) {
            return false;
        }

        $device->update(['is_trusted' => true]);

        activity('device')
            ->performedOn(\App\Models\User::find($userId))
            ->withProperties([
                'device' => $device->browser . ' on ' . $device->os,
                'ip' => $device->ip_address,
            ])
            ->log('Device trusted');

        return true;
    }

    public function untrustDevice(int $userId, int $deviceId): bool
    {
        $device = UserDevice::where('user_id', $userId)
            ->where('id', $deviceId)
            ->first();

        if (! $device) {
            return false;
        }

        $device->update(['is_trusted' => false]);

        activity('device')
            ->performedOn(\App\Models\User::find($userId))
            ->withProperties([
                'device' => $device->browser . ' on ' . $device->os,
            ])
            ->log('Device untrusted');

        return true;
    }

    public function isTrustedDevice(int $userId, string $fingerprint): bool
    {
        return UserDevice::where('user_id', $userId)
            ->where('device_fingerprint', $fingerprint)
            ->where('is_trusted', true)
            ->exists();
    }

    public function getDeviceStats(int $userId): array
    {
        $devices = UserDevice::where('user_id', $userId)->get();

        return [
            'total' => $devices->count(),
            'trusted' => $devices->where('is_trusted', true)->count(),
            'current' => $devices->where('is_current', true)->count(),
            'by_type' => [
                'desktop' => $devices->where('device_type', 'desktop')->count(),
                'mobile' => $devices->where('device_type', 'mobile')->count(),
                'tablet' => $devices->where('device_type', 'tablet')->count(),
            ],
            'by_browser' => $devices->groupBy('browser')->map->count()->toArray(),
            'by_os' => $devices->groupBy('os')->map->count()->toArray(),
            'by_country' => $devices->groupBy('ip_country')->map->count()->toArray(),
        ];
    }

    public function getDevicesByLocation(int $userId): array
    {
        return UserDevice::where('user_id', $userId)
            ->selectRaw('
                ip_country,
                ip_city,
                COUNT(*) as device_count,
                MIN(created_at) as first_seen,
                MAX(last_active_at) as last_active
            ')
            ->groupBy('ip_country', 'ip_city')
            ->orderByDesc('last_active')
            ->get()
            ->toArray();
    }

    public function getSuspiciousDevices(int $userId): array
    {
        $recentDevices = UserDevice::where('user_id', $userId)
            ->where('created_at', '>=', now()->subDays(7))
            ->get();

        $suspicious = [];

        foreach ($recentDevices as $device) {
            $reasons = [];

            // Check for unusual location
            $previousCountries = UserDevice::where('user_id', $userId)
                ->where('created_at', '<', $device->created_at)
                ->distinct('ip_country')
                ->pluck('ip_country')
                ->toArray();

            if (! empty($previousCountries) && ! in_array($device->ip_country, $previousCountries)) {
                $reasons[] = 'New country';
            }

            // Check for unusual device type
            $previousTypes = UserDevice::where('user_id', $userId)
                ->where('created_at', '<', $device->created_at)
                ->distinct('device_type')
                ->pluck('device_type')
                ->toArray();

            if (! empty($previousTypes) && ! in_array($device->device_type, $previousTypes)) {
                $reasons[] = 'New device type';
            }

            // Check for rapid IP changes
            $recentIpChanges = UserDevice::where('user_id', $userId)
                ->where('created_at', '>=', now()->subHours(1))
                ->distinct('ip_address')
                ->count('ip_address');

            if ($recentIpChanges > 3) {
                $reasons[] = 'Frequent IP changes';
            }

            if (! empty($reasons)) {
                $suspicious[] = [
                    'device' => $device,
                    'reasons' => $reasons,
                ];
            }
        }

        return $suspicious;
    }

    public function markRevoked(int $userId, int $deviceId): void
    {
        UserDevice::where('user_id', $userId)
            ->where('id', $deviceId)
            ->update(['revoked_at' => now()]);
    }

    public function cleanupExpiredDevices(int $userId, int $maxAgeDays = 90): int
    {
        return UserDevice::where('user_id', $userId)
            ->where('last_active_at', '<', now()->subDays($maxAgeDays))
            ->where('is_current', false)
            ->delete();
    }

    private function getGeolocation(string $ip): array
    {
        $cached = IpGeolocationCache::where('ip_address', $ip)->first();

        if ($cached) {
            return $cached->toArray();
        }

        $data = $this->fetchGeolocation($ip);

        if ($data) {
            IpGeolocationCache::updateOrCreate(
                ['ip_address' => $ip],
                [
                    'country_code' => $data['country_code'] ?? null,
                    'country_name' => $data['country_name'] ?? null,
                    'city' => $data['city'] ?? null,
                    'region' => $data['region'] ?? null,
                    'latitude' => $data['latitude'] ?? null,
                    'longitude' => $data['longitude'] ?? null,
                    'timezone' => $data['timezone'] ?? null,
                    'isp' => $data['isp'] ?? null,
                ]
            );

            return $data;
        }

        return [];
    }

    private function fetchGeolocation(string $ip): ?array
    {
        try {
            $response = Http::timeout(5)->get("http://ip-api.com/json/{$ip}");

            if ($response->successful()) {
                $data = $response->json();
                return [
                    'country_code' => $data['countryCode'] ?? null,
                    'country_name' => $data['country'] ?? null,
                    'city' => $data['city'] ?? null,
                    'region' => $data['regionName'] ?? null,
                    'latitude' => $data['lat'] ?? null,
                    'longitude' => $data['lon'] ?? null,
                    'timezone' => $data['timezone'] ?? null,
                    'isp' => $data['isp'] ?? null,
                ];
            }
        } catch (\Exception $e) {
            report($e);
        }

        return null;
    }
}
```

```php
// app/Livewire/DeviceManagerComponent.php

namespace App\Livewire;

use App\Services\DeviceTrackingService;
use Livewire\Component;

class DeviceManagerComponent extends Component
{
    public DeviceTrackingService $deviceService;

    public int $revokeDeviceId = 0;
    public int $trustDeviceId = 0;
    public int $untrustDeviceId = 0;
    public bool $showRevokeAllConfirm = false;

    protected $listeners = [
        'revokeDeviceConfirmed' => 'revokeDevice',
        'revokeAllDevicesConfirmed' => 'revokeAllDevices',
        'trustDeviceConfirmed' => 'trustDevice',
        'untrustDeviceConfirmed' => 'untrustDevice',
    ];

    public function boot(DeviceTrackingService $deviceService): void
    {
        $this->deviceService = $deviceService;
    }

    public function getDevicesProperty(): array
    {
        return \App\Models\UserDevice::where('user_id', auth()->id())
            ->orderByDesc('is_current')
            ->orderByDesc('last_active_at')
            ->get()
            ->map(fn ($device) => [
                'id' => $device->id,
                'browser' => $device->browser,
                'browser_version' => $device->browser_version,
                'os' => $device->os,
                'os_version' => $device->os_version,
                'device_type' => $device->device_type,
                'ip_address' => $device->ip_address,
                'location' => $device->ip_city
                    ? "{$device->ip_city}, {$device->ip_country}"
                    : ($device->ip_country ?? 'Unknown'),
                'is_trusted' => $device->is_trusted,
                'is_current' => $device->is_current,
                'last_active' => $device->last_active_at->diffForHumans(),
                'first_seen' => $device->created_at->diffForHumans(),
                'login_at' => $device->login_at?->diffForHumans(),
            ])
            ->toArray();
    }

    public function getStatsProperty(): array
    {
        return $this->deviceService->getDeviceStats(auth()->id());
    }

    public function getSuspiciousProperty(): array
    {
        return $this->deviceService->getSuspiciousDevices(auth()->id());
    }

    public function promptRevoke(int $deviceId): void
    {
        $this->revokeDeviceId = $deviceId;
        $this->dispatch('show-confirm', message: 'Revoke this device? The associated session will be terminated.');
    }

    public function revokeDevice(): void
    {
        $this->deviceService->revokeDevice(auth()->id(), $this->revokeDeviceId);
        $this->dispatch('notify', type: 'success', message: 'Device revoked');
        $this->revokeDeviceId = 0;
    }

    public function promptTrust(int $deviceId): void
    {
        $this->trustDeviceId = $deviceId;
        $this->dispatch('show-confirm', message: 'Trust this device? It will skip 2FA verification for future logins.');
    }

    public function trustDevice(): void
    {
        $this->deviceService->trustDevice(auth()->id(), $this->trustDeviceId);
        $this->dispatch('notify', type: 'success', message: 'Device trusted');
        $this->trustDeviceId = 0;
    }

    public function promptUntrust(int $deviceId): void
    {
        $this->untrustDeviceId = $deviceId;
        $this->dispatch('show-confirm', message: 'Remove trust from this device? 2FA will be required for future logins.');
    }

    public function untrustDevice(): void
    {
        $this->deviceService->untrustDevice(auth()->id(), $this->untrustDeviceId);
        $this->dispatch('notify', type: 'success', message: 'Device untrusted');
        $this->untrustDeviceId = 0;
    }

    public function promptRevokeAll(): void
    {
        $this->showRevokeAllConfirm = true;
    }
```

```php
namespace App\Http\Middleware;

use App\Services\DeviceTrackingService;
use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class DeviceTracking
{
    public function __construct(
        private DeviceTrackingService $deviceService
    ) {}

    public function handle(Request $request, Closure $next)
    {
        if (Auth::check()) {
            $device = $this->deviceService->trackDevice($request, Auth::id());

            // Check if device IP changed (potential session hijacking)
            if ($device->ip_address !== $request->ip()) {
                $device->update(['ip_address' => $request->ip()]);
            }

            // Update activity timestamp
            $this->deviceService->updateActivity(Auth::id());
        }

        return $next($request);
    }
}
```

### 2FA Skip for Trusted Devices

```php
// app/Http/Middleware/TrustedDeviceMiddleware.php

namespace App\Http\Middleware;

use App\Services\DeviceTrackingService;
use App\Services\DeviceFingerprintService;
use Closure;
use Illuminate\Http\Request;

class TrustedDeviceMiddleware
{
    public function __construct(
        private DeviceTrackingService $deviceService,
        private DeviceFingerprintService $fingerprintService
    ) {}

    public function handle(Request $request, Closure $next)
    {
        if (! auth()->check()) {
            return $next($request);
        }

        $fingerprint = $this->fingerprintService->generateFingerprint($request);

        if ($this->deviceService->isTrustedDevice(auth()->id(), $fingerprint)) {
            // Device is trusted, skip 2FA if recently verified
            if (session('2fa_verified_recently', false)) {
                return redirect()->intended('/dashboard');
            }
        }

        return $next($request);
    }
}
```

```php
namespace App\Services;

class DeviceCleanupService

    public function cleanup(): int
    {
        $maxAgeDays = config('devices.cleanup_max_age_days', 90);
        $count = 0;

        UserDevice::where('last_active_at', '<', now()->subDays($maxAgeDays))
            ->where('is_current', false)
            ->chunk(100, function ($devices) use (&$count) {
                foreach ($devices as $device) {
                    // Delete the session first
                    \Illuminate\Support\Facades\Session::where('id', $device->session_id)->delete();
                    $device->delete();
                    $count++;
                }
            });

        Log::info("Device cleanup: removed {$count} expired devices");

        return $count;
    }
}
```

## Device Activity Monitor

```php
// app/Services/DeviceActivityMonitor.php

namespace App\Services;

use App\Models\UserDevice;
use App\Notifications\SuspiciousDeviceNotification;
use Illuminate\Support\Facades\Log;

class DeviceActivityMonitor
{
    public function checkForSuspiciousActivity(int $userId): array
    {
        $alerts = [];

        // Check for rapid location changes
        $recentDevices = UserDevice::where('user_id', $userId)
            ->where('created_at', '>=', now()->subHours(24))
            ->get();

        $countries = $recentDevices->pluck('ip_country')->unique()->filter();

        if ($countries->count() > 2) {
            $alerts[] = [
                'type' => 'rapid_location_change',
                'message' => "Login from {$countries->count()} different countries in 24 hours",
                'severity' => 'high',
            ];
        }

        // Check for multiple failed device attempts
        $failedAttempts = UserDevice::where('user_id', $userId)
            ->where('created_at', '>=', now()->subHours(1))
            ->count();

        if ($failedAttempts > 5) {
            $alerts[] = [
                'type' => 'multiple_device_attempts',
                'message' => "{$failedAttempts} device sessions in the last hour",
                'severity' => 'medium',
            ];
        }

        // Check for unusual device types
        $deviceTypes = $recentDevices->pluck('device_type')->unique();
        $knownTypes = UserDevice::where('user_id', $userId)
            ->where('created_at', '<', now()->subDays(7))
            ->pluck('device_type')
            ->unique();

        $newTypes = $deviceTypes->diff($knownTypes);

        if ($newTypes->isNotEmpty()) {
            $alerts[] = [
                'type' => 'new_device_type',
                'message' => 'New device type detected: ' . $newTypes->implode(', '),
                'severity' => 'low',
            ];
        }

        // Send notifications for high severity alerts
        foreach ($alerts as $alert) {
            if ($alert['severity'] === 'high') {
                $user = \App\Models\User::find($userId);
                $latestDevice = $recentDevices->last();

                if ($user && $latestDevice) {
                    $user->notify(new SuspiciousDeviceNotification($latestDevice, $alert['message']));
                }
            }
        }

        return $alerts;
    }

    public function getDeviceActivitySummary(int $userId, int $days = 30): array
    {
        return [
            'total_logins' => UserDevice::where('user_id', $userId)
                ->where('login_at', '>=', now()->subDays($days))
                ->count(),
            'unique_devices' => UserDevice::where('user_id', $userId)
                ->where('created_at', '>=', now()->subDays($days))
                ->distinct('device_fingerprint')
                ->count('device_fingerprint'),
            'unique_countries' => UserDevice::where('user_id', $userId)
                ->where('created_at', '>=', now()->subDays($days))
                ->distinct('ip_country')
                ->count('ip_country'),
            'unique_ips' => UserDevice::where('user_id', $userId)
                ->where('created_at', '>=', now()->subDays($days))
                ->distinct('ip_address')
                ->count('ip_address'),
            'trusted_devices' => UserDevice::where('user_id', $userId)
                ->where('is_trusted', true)
                ->count(),
            'active_devices_24h' => UserDevice::where('user_id', $userId)
                ->where('last_active_at', '>=', now()->subHours(24))
                ->count(),
            'browser_breakdown' => UserDevice::where('user_id', $userId)
                ->where('created_at', '>=', now()->subDays($days))
                ->selectRaw('browser, COUNT(*) as count')
                ->groupBy('browser')
                ->pluck('count', 'browser')
                ->toArray(),
            'os_breakdown' => UserDevice::where('user_id', $userId)
                ->where('created_at', '>=', now()->subDays($days))
                ->selectRaw('os, COUNT(*) as count')
                ->groupBy('os')
                ->pluck('count', 'os')
                ->toArray(),
            'country_breakdown' => UserDevice::where('user_id', $userId)
                ->where('created_at', '>=', now()->subDays($days))
                ->selectRaw('ip_country, COUNT(*) as count')
                ->groupBy('ip_country')
                ->pluck('count', 'ip_country')
                ->toArray(),
        ];
    }
```

## Security Notes

* Device data stored with session lifecycle
* Trusted device status persists across sessions
* IP changes trigger re-verification
* Suspicious device additions alert user via email
* Device revocation revokes the associated session
* All device management actions logged
* Device fingerprints use SHA-256 hash of UA + headers + screen
* Geolocation is cached to reduce external API calls
* Cleanup job runs daily to remove expired device records
* Maximum trusted devices per user is configurable
* Device tracking middleware runs on every authenticated request
* Session and device data are deleted together on cascade
* Activity monitoring detects rapid location changes automatically
* Device type detection supports desktop, mobile, and tablet classifications
* Browser and OS parsing covers all major browsers
* Suspicious activity detection compares new countries and device types
* Device stats endpoint provides aggregated breakdowns by type, browser, OS, and country
* Trusted device check is used by 2FA skip middleware

```php
// app/Http/Controllers/Api/DeviceApiController.php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Services\DeviceTrackingService;
use App\Services\DeviceActivityMonitor;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class DeviceApiController extends Controller
{
    public function __construct(
        private DeviceTrackingService $deviceService,
        private DeviceActivityMonitor $activityMonitor
    ) {}

    public function index(Request $request): JsonResponse
    {
        $devices = $this->deviceService->getActiveSessions($request->user()->id);

        return response()->json([
            'data' => $devices,
            'meta' => [
                'total' => count($devices),
                'current' => collect($devices)->where('is_current', true)->count(),
                'trusted' => collect($devices)->where('is_trusted', true)->count(),
            ],
        ]);
    }

    public function stats(Request $request): JsonResponse
    {
        return response()->json([
            'data' => $this->deviceService->getDeviceStats($request->user()->id),
        ]);
    }

    public function suspicious(Request $request): JsonResponse
    {
        return response()->json([
            'data' => $this->activityMonitor->checkForSuspiciousActivity($request->user()->id),
        ]);
    }

    public function trust(Request $request, int $device): JsonResponse
    {
        $result = $this->deviceService->trustDevice($request->user()->id, $device);

        return $result
            ? response()->json(['message' => 'Device trusted'])
            : response()->json(['message' => 'Device not found'], 404);
    }

    public function untrust(Request $request, int $device): JsonResponse
    {
        $result = $this->deviceService->untrustDevice($request->user()->id, $device);

        return $result
            ? response()->json(['message' => 'Device untrusted'])
            : response()->json(['message' => 'Device not found'], 404);
    }

    public function revoke(Request $request, int $device): JsonResponse
    {
        $result = $this->deviceService->revokeDevice($request->user()->id, $device);

        return $result
            ? response()->json(['message' => 'Device revoked'])
            : response()->json(['message' => 'Cannot revoke current device'], 403);
    }

    public function revokeAll(Request $request): JsonResponse
    {
        $count = $this->deviceService->revokeAllDevices(
            $request->user()->id,
            session()->getId()
        );

        return response()->json([
            'message' => "{$count} devices revoked",
            'revoked_count' => $count,
        ]);
    }
}
```
