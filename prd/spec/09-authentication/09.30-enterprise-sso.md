# 09.30 — Enterprise SSO

## Purpose

Define the Enterprise Single Sign-On (SSO) architecture for GsmUi, supporting SAML 2.0, OpenID Connect, and LDAP/Active Directory.

## Status: Phase 4 (Enterprise)

Enterprise SSO is a Phase 4 feature. This document defines the architecture for implementation.

## SSO Methods

| Method | Status | Use Case |
|---|---|---|
| SAML 2.0 | Phase 4 | Enterprise SSO (Okta, Azure AD) |
| OpenID Connect | Phase 4 | Modern SSO (Google, Auth0) |
| LDAP | Phase 4 | On-premise AD |
| SCIM Provisioning | Phase 4 | Automated user management |

## SAML 2.0 Flow

```text
User visits GsmUi
    ↓
Clicks "Sign in with SSO"
    ↓
Redirect to Identity Provider (IdP)
    ↓
User authenticates at IdP
    ↓
IdP sends SAML response
    ↓
GsmUi validates SAML assertion
    ↓
User created/updated
    ↓
Session created
    ↓
Redirect to dashboard
```

## OpenID Connect Flow

```text
User visits GsmUi
    ↓
Clicks "Sign in with SSO"
    ↓
Redirect to Authorization Server
    ↓
User authenticates
    ↓
Authorization code returned
    ↓
Code exchanged for tokens
    ↓
ID token validated
    ↓
User info fetched
    ↓
User created/updated
    ↓
Session created
```

|---|---|

| PATCH | IdP updates specific attributes |

## Security Notes

* SAML assertions validated against IdP certificate
* OIDC tokens validated against IdP JWKS
* LDAP connections use TLS
* SCIM requests authenticated with bearer tokens
* All SSO events logged to audit trail
* Just-in-time (JIT) user provisioning supported
* User deactivation synced from IdP

### SAML Service Provider

```php
// app/Services/SSO/Saml/SamlServiceProvider.php

namespace App\Services\SSO\Saml;

use App\Models\SsoProvider;
use App\Models\SsoUser;
use App\Models\User;
use App\Services\SSO\SsoService;
use App\Services\SSO\Saml\SamlAuth;
use Illuminate\Support\Facades\Log;

class SamlServiceProvider
{
    public function __construct(
        protected SsoService $ssoService,
        protected SamlAuth $samlAuth
    ) {}

    public function handleLoginRequest(SsoProvider $provider): string
    {
        $this->samlAuth->login($provider);
        return $this->samlAuth->getLoginUrl();
    }

    public function handleCallback(SsoProvider $provider, array $samlResponse): User
    {
        $this->samlAuth->processResponse($provider);

        if ($this->samlAuth->getErrors()) {
            Log::channel('sso')->error('SAML response processing failed', [
                'provider_id' => $provider->id,
                'errors' => $this->samlAuth->getErrors(),
            ]);
            throw new \App\Exceptions\SSO\SamlAuthenticationException(
                'SAML authentication failed: ' . implode(', ', $this->samlAuth->getErrors())
            );
        }

        $attributes = $this->samlAuth->getAttributes();
        $nameId = $this->samlAuth->getNameId();

        return $this->ssoService->findOrCreateUser(
            provider: $provider,
            externalId: $nameId,
            attributes: $attributes
        );
    }

    public function handleLogoutRequest(SsoProvider $provider, User $user): string
    {
        $ssoUser = $user->ssoUsers()->where('sso_provider_id', $provider->id)->first();

        if ($ssoUser) {
            $this->samlAuth->logout($provider, $ssoUser->external_id);
            return $this->samlAuth->getLogoutUrl();
        }

        return '/';
    }

    public function getMetadata(SsoProvider $provider): string
    {
        return $this->samlAuth->getMetadata($provider);
    }

    public function validateConfiguration(SsoProvider $provider): array
    {
        $errors = [];

        try {
            $metadataUrl = $provider->config['metadata_url'] ?? null;
            if ($metadataUrl) {
                $metadata = $this->fetchMetadata($metadataUrl);
                if (empty($metadata['idp']['singleSignOnService']['url'])) {
                    $errors[] = 'Missing SSO service URL in metadata';
                }
                if (empty($metadata['idp']['singleLogoutService']['url'])) {
                    $errors[] = 'Missing SLO service URL in metadata';
                }
                if (empty($metadata['idp']['x509cert'])) {
                    $errors[] = 'Missing X.509 certificate in metadata';
                }
            }
        } catch (\Exception $e) {
            $errors[] = 'Failed to fetch metadata: ' . $e->getMessage();
        }

        return $errors;
    }

    protected function fetchMetadata(string $url): array
    {
        $response = \Illuminate\Support\Facades\Http::get($url);
        $xml = $response->body();
        return \LightSaml\Credential\X509Certificate::fromString($xml)->toArray();
    }
}
```

### SAML Authentication Handler

```php
// app/Services/SSO/Saml/SamlAuth.php

namespace App\Services\SSO\Saml;

use App\Models\SsoProvider;
use LightSaml\Binding\HttpPostBinding;
use LightSaml\Binding\HttpRedirectBinding;
use LightSaml\Context\Profile\DestinationContext;
use LightSaml\Credential\KeyHelper;
use LightSaml\Credential\X509Certificate;
use LightSaml\Model\Assertion\Assertion;
use LightSaml\Model\Protocol\AuthnRequest;
use LightSaml\Model\Protocol\Response;
use LightSaml\SamlConstants;

class SamlAuth
{
    protected array $errors = [];
    protected array $attributes = [];
    protected ?string $nameId = null;
    protected ?AuthnRequest $authnRequest = null;

    public function login(SsoProvider $provider): void
    {
        $this->authnRequest = $this->buildAuthnRequest($provider);
    }

    public function getLoginUrl(): string
    {
        if (!$this->authnRequest) {
            throw new \LogicException('Login not initiated');
        }

        $binding = new HttpRedirectBinding();
        $destination = new DestinationContext(
            SamlConstants::BINDING_SAML2_HTTP_REDIRECT,
            $this->authnRequest->getDestination()->getUrl()
        );

        return $binding->build($this->authnRequest, $destination);
    }

    public function processResponse(SsoProvider $provider): void
    {
        $binding = new HttpPostBinding();
        $response = $binding->receive($_POST);

        if (!$response instanceof Response) {
            $this->errors[] = 'Invalid SAML response type';
            return;
        }

        $certificate = X509Certificate::fromString($provider->config['certificate']);

        try {
            $response->validate($certificate);
        } catch (\Exception $e) {
            $this->errors[] = 'SAML response validation failed: ' . $e->getMessage();
            return;
        }

        $assertion = $response->getFirstAssertion();
        if (!$assertion) {
            $this->errors[] = 'No assertion found in SAML response';
            return;
        }

        if (!$this->validateAssertion($assertion, $provider)) {
            return;
        }

        $this->nameId = $assertion->getSubject()->getNameID()->getValue();
        $this->attributes = $this->extractAttributes($assertion);
    }

    public function getErrors(): array
    {
        return $this->errors;
    }

    public function getAttributes(): array
    {
        return $this->attributes;
    }

    public function getNameId(): ?string
    {
        return $this->nameId;
    }

    public function logout(SsoProvider $provider, string $nameId): void
    {
        // Build logout request
    }

    public function getLogoutUrl(): string
    {
        // Return logout URL
        return '/';
    }

    public function getMetadata(SsoProvider $provider): string
    {
        // Generate SP metadata XML
        return '<?xml version="1.0"?>';
    }

    protected function buildAuthnRequest(SsoProvider $provider): AuthnRequest
    {
        $request = new AuthnRequest();
        $request->setID('_' . bin2hex(random_bytes(16)));
        $request->setIssueInstant(new \DateTime());
        $request->setDestination(new DestinationContext(
            SamlConstants::BINDING_SAML2_HTTP_REDIRECT,
            $provider->config['sso_url'] ?? ''
        ));
        $request->setAssertionConsumerServiceURL(config('app.url') . '/sso/saml/callback');
        $request->setProtocolBinding(SamlConstants::BINDING_SAML2_HTTP_POST);
        $request->setNameIDPolicyFormat(SamlConstants::NAME_ID_FORMAT_EMAIL_ADDRESS);

        return $request;
    }

    protected function validateAssertion(Assertion $assertion, SsoProvider $provider): bool
    {
        $conditions = $assertion->getConditions();
        if ($conditions) {
            $notBefore = $conditions->getNotBefore();
            if ($notBefore && $notBefore->getDateTime()->isFuture()) {
                $this->errors[] = 'Assertion not yet valid';
                return false;
            }

            $notOnOrAfter = $conditions->getNotOnOrAfter();
            if ($notOnOrAfter && $notOnOrAfter->getDateTime()->isPast()) {
                $this->errors[] = 'Assertion has expired';
                return false;
            }

            $audienceRestriction = $conditions->getAudienceRestriction();
            if ($audienceRestriction) {
                $audiences = $audienceRestriction->getAudiences();
                $validAudience = false;
                foreach ($audiences as $audience) {
                    if ($audience->getUri() === config('app.url')) {
                        $validAudience = true;
                        break;
                    }
                }
                if (!$validAudience) {
                    $this->errors[] = 'Invalid audience restriction';
                    return false;
                }
            }
        }

        return true;
    }

    protected function extractAttributes(Assertion $assertion): array
    {
        $attributes = [];

        foreach ($assertion->getAttributeStatements() as $attributeStatement) {
            foreach ($attributeStatement->getAttributes() as $attribute) {
                $name = $attribute->getName();
                $values = $attribute->getValues();
                $attributes[$name] = count($values) === 1
                    ? $values[0]->getValue()
                    : array_map(fn($v) => $v->getValue(), $values);
            }
        }

        return $attributes;
    }
}
```

```php
// app/Services/SSO/Saml/AttributeMapper.php

namespace App\Services\SSO\Saml;

class AttributeMapper
{
    protected array $defaultMapping = [
        'email' => ['urn:oid:0.9.2342.19200300.100.1.3', 'email', 'mail'],
        'name' => ['urn:oid:2.5.4.3', 'name', 'displayName', 'cn'],
        'first_name' => ['urn:oid:2.5.4.42', 'givenName', 'first_name'],
        'last_name' => ['urn:oid:2.5.4.4', 'sn', 'surname', 'last_name'],
        'phone' => ['urn:oid:2.5.4.20', 'phone', 'telephoneNumber'],
        'department' => ['urn:oid:2.5.4.11', 'department', 'ou'],
        'title' => ['urn:oid:2.5.4.12', 'title', 'jobTitle'],
        'groups' => ['urn:oid:1.3.6.1.4.1.5923.1.5.1.1', 'memberOf', 'groups'],
    ];

    public function __construct(protected ?array $customMapping = null) {}

    public function map(array $samlAttributes): array
    {
        $mapping = $this->customMapping ?? $this->defaultMapping;
        $mapped = [];

        foreach ($mapping as $field => $possibleKeys) {
            foreach ($possibleKeys as $key) {
                if (isset($samlAttributes[$key])) {
                    $mapped[$field] = is_array($samlAttributes[$key])
                        ? $samlAttributes[$key]
                        : $samlAttributes[$key];
                    break;
                }
            }
        }

        return $mapped;
    }

    public function getGroups(array $samlAttributes): array
    {
        $mapped = $this->map($samlAttributes);
        return $mapped['groups'] ?? [];
    }

    public function getEmail(array $samlAttributes): ?string
    {
        $mapped = $this->map($samlAttributes);
        return $mapped['email'] ?? null;
    }
}
```

### OIDC Service Provider

```php
// app/Services/SSO/Oidc/OidcServiceProvider.php

namespace App\Services\SSO\Oidc;

use App\Models\SsoProvider;
use App\Services\SSO\SsoService;
use App\Services\SSO\Oidc\OidcClient;
use Illuminate\Support\Facades\Log;

class OidcServiceProvider
{
    public function __construct(
        protected SsoService $ssoService,
        protected OidcClient $oidcClient
    ) {}

    public function handleLoginRequest(SsoProvider $provider): string
    {
        return $this->oidcClient->getAuthorizationUrl($provider);
    }

    public function handleCallback(SsoProvider $provider, string $code): \App\Models\User
    {
        $tokenResponse = $this->oidcClient->exchangeCodeForTokens($provider, $code);

        if (isset($tokenResponse['error'])) {
            Log::channel('sso')->error('OIDC token exchange failed', [
                'provider_id' => $provider->id,
                'error' => $tokenResponse['error'],
                'error_description' => $tokenResponse['error_description'] ?? '',
            ]);
            throw new \App\Exceptions\SSO\OidcAuthenticationException(
                'OIDC authentication failed: ' . ($tokenResponse['error_description'] ?? $tokenResponse['error'])
            );
        }

        $idTokenClaims = $this->oidcClient->validateIdToken(
            $provider,
            $tokenResponse['id_token']
        );

        if (!$idTokenClaims) {
            throw new \App\Exceptions\SSO\OidcAuthenticationException(
                'Invalid ID token'
            );
        }

        $userInfo = $tokenResponse['access_token']
            ? $this->oidcClient->getUserInfo($provider, $tokenResponse['access_token'])
            : $idTokenClaims;

        return $this->ssoService->findOrCreateUser(
            provider: $provider,
            externalId: $idTokenClaims['sub'],
            attributes: $userInfo
        );
    }

    public function handleLogoutRequest(SsoProvider $provider, \App\Models\User $user): string
    {
        $endSessionUrl = $this->oidcClient->getEndSessionUrl($provider);
        return $endSessionUrl;
    }

    public function refreshTokens(SsoProvider $provider, string $refreshToken): ?array
    {
        return $this->oidcClient->refreshTokens($provider, $refreshToken);
    }

    public function validateConfiguration(SsoProvider $provider): array
    {
        $errors = [];

        try {
            $discovery = $this->oidcClient->getDiscoveryDocument($provider);

            if (empty($discovery['issuer'])) {
                $errors[] = 'Missing issuer in discovery document';
            }
            if (empty($discovery['authorization_endpoint'])) {
                $errors[] = 'Missing authorization endpoint';
            }
            if (empty($discovery['token_endpoint'])) {
                $errors[] = 'Missing token endpoint';
            }
            if (empty($discovery['jwks_uri'])) {
                $errors[] = 'Missing JWKS URI';
            }
        } catch (\Exception $e) {
            $errors[] = 'Failed to fetch discovery document: ' . $e->getMessage();
        }

        return $errors;
    }
}
```

### OIDC Client

```php
// app/Services/SSO/Oidc/OidcClient.php

namespace App\Services\SSO\Oidc;

use App\Models\SsoProvider;
use Firebase\JWT\JWK;
use Firebase\JWT\JWT;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Http;

class OidcClient
{
    public function getAuthorizationUrl(SsoProvider $provider): string
    {
        $discovery = $this->getDiscoveryDocument($provider);
        $state = bin2hex(random_bytes(16));
        $nonce = bin2hex(random_bytes(16));

        Cache::put("oidc_state_{$state}", [
            'provider_id' => $provider->id,
            'nonce' => $nonce,
            'created_at' => now()->timestamp,
        ], 600);

        $params = [
            'client_id' => $provider->config['client_id'],
            'redirect_uri' => $this->getRedirectUri($provider),
            'response_type' => 'code',
            'scope' => 'openid email profile',
            'state' => $state,
            'nonce' => $nonce,
        ];

        return $discovery['authorization_endpoint'] . '?' . http_build_query($params);
    }

    public function exchangeCodeForTokens(SsoProvider $provider, string $code): array
    {
        $discovery = $this->getDiscoveryDocument($provider);

        $response = Http::asForm()->post($discovery['token_endpoint'], [
            'grant_type' => 'authorization_code',
            'code' => $code,
            'redirect_uri' => $this->getRedirectUri($provider),
            'client_id' => $provider->config['client_id'],
            'client_secret' => $provider->config['client_secret'],
        ]);

        return $response->json();
    }

    public function validateIdToken(SsoProvider $provider, string $idToken): ?array
    {
        try {
            $discovery = $this->getDiscoveryDocument($provider);
            $jwks = $this->getJwks($discovery['jwks_uri']);
            $keys = JWK::parseKeySet($jwks);

            $decoded = JWT::decode($idToken, $keys, ['RS256']);

            if ($decoded->iss !== $discovery['issuer']) {
                return null;
            }

            if ($decoded->aud !== $provider->config['client_id']) {
                return null;
            }

            if ($decoded->exp < time()) {
                return null;
            }

            return (array) $decoded;
        } catch (\Exception $e) {
            return null;
        }
    }

    public function getUserInfo(SsoProvider $provider, string $accessToken): array
    {
        $discovery = $this->getDiscoveryDocument($provider);

        if (empty($discovery['userinfo_endpoint'])) {
            return [];
        }

        $response = Http::withToken($accessToken)
            ->get($discovery['userinfo_endpoint']);

        return $response->json();
    }

    public function getEndSessionUrl(SsoProvider $provider): string
    {
        $discovery = $this->getDiscoveryDocument($provider);

        if (empty($discovery['end_session_endpoint'])) {
            return '/';
        }

        return $discovery['end_session_endpoint'] . '?' . http_build_query([
            'client_id' => $provider->config['client_id'],
            'post_logout_redirect_uri' => config('app.url') . '/sso/oidc/logout/callback',
        ]);
    }

    public function refreshTokens(SsoProvider $provider, string $refreshToken): ?array
    {
        $discovery = $this->getDiscoveryDocument($provider);

        $response = Http::asForm()->post($discovery['token_endpoint'], [
            'grant_type' => 'refresh_token',
            'refresh_token' => $refreshToken,
            'client_id' => $provider->config['client_id'],
            'client_secret' => $provider->config['client_secret'],
        ]);

        if ($response->failed()) {
            return null;
        }

        return $response->json();
    }

    public function getDiscoveryDocument(SsoProvider $provider): array
    {
        $cacheKey = "oidc_discovery_{$provider->id}";

        return Cache::remember($cacheKey, 3600, function () use ($provider) {
            $response = Http::get($provider->config['discovery_url']);
            return $response->json();
        });
    }

    protected function getJwks(string $jwksUri): array
    {
        $cacheKey = 'oidc_jwks_' . md5($jwksUri);

        return Cache::remember($cacheKey, 3600, function () use ($jwksUri) {
            $response = Http::get($jwksUri);
            return $response->json();
        });
    }

    protected function getRedirectUri(SsoProvider $provider): string
    {
        return config('app.url') . '/sso/oidc/callback/' . $provider->id;
    }
}
```

### LDAP Service

```php
// app/Services/SSO/Ldap/LdapService.php

namespace App\Services\SSO\Ldap;

use App\Models\SsoProvider;
use App\Models\SsoUser;
use App\Models\User;
use App\Services\SSO\SsoService;
use Illuminate\Support\Facades\Log;

class LdapService
{
    public function __construct(
        protected SsoService $ssoService,
        protected LdapConnection $connection
    ) {}

    public function authenticate(SsoProvider $provider, string $username, string $password): ?User
    {
        try {
            $this->connection->connect($provider);

            $ldapUser = $this->connection->authenticate($username, $password);

            if (!$ldapUser) {
                Log::channel('sso')->warning('LDAP authentication failed', [
                    'provider_id' => $provider->id,
                    'username' => $username,
                ]);
                return null;
            }

            $attributes = $this->connection->getAttributes($ldapUser);

            return $this->ssoService->findOrCreateUser(
                provider: $provider,
                externalId: $attributes['dn'][0] ?? $username,
                attributes: $this->mapLdapAttributes($attributes)
            );
        } catch (\Exception $e) {
            Log::channel('sso')->error('LDAP authentication error', [
                'provider_id' => $provider->id,
                'error' => $e->getMessage(),
            ]);
            return null;
        } finally {
            $this->connection->disconnect();
        }
    }

    public function testConnection(SsoProvider $provider): array
    {
        try {
            $this->connection->connect($provider);
            $this->connection->bind(
                $provider->config['bind_dn'],
                $provider->config['bind_password']
            );

            $users = $this->connection->search(
                $provider->config['base_dn'],
                '(objectClass=*)',
                ['dn'],
                1
            );

            $this->connection->disconnect();

            return [
                'status' => 'success',
                'message' => 'Connection successful',
                'users_found' => count($users),
            ];
        } catch (\Exception $e) {
            return [
                'status' => 'error',
                'message' => $e->getMessage(),
            ];
        }
    }

    public function syncUsers(SsoProvider $provider): array
    {
        $this->connection->connect($provider);
        $this->connection->bind(
            $provider->config['bind_dn'],
            $provider->config['bind_password']
        );

        $users = $this->connection->search(
            $provider->config['base_dn'],
            $provider->config['user_filter'] ?? '(objectClass=*)',
            ['*']
        );

        $results = [
            'created' => 0,
            'updated' => 0,
            'skipped' => 0,
        ];

        foreach ($users as $userAttributes) {
            try {
                $externalId = $userAttributes['dn'][0] ?? $userAttributes['uid'][0] ?? null;

                if (!$externalId) {
                    $results['skipped']++;
                    continue;
                }

                $mappedAttributes = $this->mapLdapAttributes($userAttributes);

                $ssoUser = SsoUser::where('sso_provider_id', $provider->id)
                    ->where('external_id', $externalId)
                    ->first();

                if ($ssoUser) {
                    $ssoUser->update([
                        'metadata' => $mappedAttributes,
                    ]);
                    $results['updated']++;
                } else {
                    $this->ssoService->findOrCreateUser(
                        provider: $provider,
                        externalId: $externalId,
                        attributes: $mappedAttributes
                    );
                    $results['created']++;
                }
            } catch (\Exception $e) {
                Log::channel('sso')->error('LDAP user sync failed', [
                    'provider_id' => $provider->id,
                    'error' => $e->getMessage(),
                ]);
                $results['skipped']++;
            }
        }

        $this->connection->disconnect();

        return $results;
    }

    protected function mapLdapAttributes(array $ldapAttributes): array
    {
        return [
            'email' => $ldapAttributes['mail'][0] ?? $ldapAttributes['userPrincipalName'][0] ?? null,
            'name' => $ldapAttributes['displayName'][0] ?? $ldapAttributes['cn'][0] ?? null,
            'first_name' => $ldapAttributes['givenName'][0] ?? null,
            'last_name' => $ldapAttributes['sn'][0] ?? null,
            'phone' => $ldapAttributes['telephoneNumber'][0] ?? null,
            'department' => $ldapAttributes['department'][0] ?? null,
            'title' => $ldapAttributes['title'][0] ?? null,
            'groups' => $ldapAttributes['memberOf'] ?? [],
        ];
    }
}
```

```php
// app/Services/SSO/Ldap/LdapConnection.php

namespace App\Services\SSO\Ldap;

use App\Models\SsoProvider;

class LdapConnection
{
    protected ?\LDAPConnection $connection = null;

    public function connect(SsoProvider $provider): void
    {
        $host = $provider->config['host'];
        $port = $provider->config['port'] ?? 389;
        $useSsl = $provider->config['use_ssl'] ?? false;
        $useTls = $provider->config['use_tls'] ?? false;

        $this->connection = ldap_connect($host, $port);

        ldap_set_option($this->connection, LDAP_OPT_PROTOCOL_VERSION, 3);
        ldap_set_option($this->connection, LDAP_OPT_REFERRALS, 0);

        if ($useSsl) {
            ldap_start_tls($this->connection);
        } elseif ($useTls) {
            $this->startTls();
        }

        $timeout = $provider->config['timeout'] ?? 5;
        ldap_set_option($this->connection, LDAP_OPT_NETWORK_TIMEOUT, $timeout);
    }

    public function bind(string $dn, string $password): bool
    {
        return @ldap_bind($this->connection, $dn, $password);
    }

    public function authenticate(string $username, string $password): ?string
    {
        $result = @ldap_bind($this->connection, $username, $password);
        return $result ? $username : null;
    }

    public function search(string $baseDn, string $filter, array $attributes = ['*'], int $limit = 0): array
    {
        $search = @ldap_search($this->connection, $baseDn, $filter, $attributes, 0, $limit);

        if (!$search) {
            return [];
        }

        $entries = ldap_get_entries($this->connection, $search);
        return $entries['count'] ? array_slice($entries, 0, $entries['count']) : [];
    }

    public function getAttributes(string $dn): array
    {
        $search = @ldap_read($this->connection, $dn, '(objectClass=*)', ['*']);
        $entries = ldap_get_entries($this->connection, $search);
        return $entries[0] ?? [];
    }

    public function disconnect(): void
    {
        if ($this->connection) {
            @ldap_unbind($this->connection);
            $this->connection = null;
        }
    }

    protected function startTls(): void
    {
        $result = @ldap_start_tls($this->connection);
        if (!$result) {
            throw new \RuntimeException('Failed to start TLS: ' . ldap_error($this->connection));
        }
    }
}
```

```php
// app/Services/SSO/Scim/ScimService.php

namespace App\Services\SSO\Scim;

use App\Models\SsoProvider;
use App\Models\SsoUser;
use App\Models\User;
use Illuminate\Support\Facades\Log;

class ScimService
{
    public function listUsers(SsoProvider $provider, ?string $filter, int $startIndex, int $count): array
    {
        $query = SsoUser::where('sso_provider_id', $provider->id);

        if ($filter) {
            $query->whereRaw($this->parseFilter($filter));
        }

        $total = $query->count();
        $users = $query->skip($startIndex - 1)
            ->take($count)
            ->with('user')
            ->get()
            ->map(fn($ssoUser) => $this->formatScimUser($ssoUser));

        return [
            'total' => $total,
            'users' => $users->toArray(),
        ];
    }

    public function getUser(SsoProvider $provider, string $externalId): ?array
    {
        $ssoUser = SsoUser::where('sso_provider_id', $provider->id)
            ->where('external_id', $externalId)
            ->with('user')
            ->first();

        return $ssoUser ? $this->formatScimUser($ssoUser) : null;
    }

    public function createUser(SsoProvider $provider, array $data): SsoUser
    {
        $email = $data['emails'][0]['value'] ?? $data['userName'] ?? null;
        $name = $data['name']['givenName'] . ' ' . $data['name']['familyName'] ?? $data['displayName'] ?? '';

        $user = User::create([
            'name' => trim($name),
            'email' => $email,
            'password' => null,
            'email_verified_at' => now(),
        ]);

        return SsoUser::create([
            'sso_provider_id' => $provider->id,
            'user_id' => $user->id,
            'external_id' => $data['externalId'] ?? $data['id'] ?? bin2hex(random_bytes(16)),
            'email' => $email,
            'name' => trim($name),
            'metadata' => $data,
        ]);
    }

    public function updateUser(SsoProvider $provider, string $externalId, array $data): ?SsoUser
    {
        $ssoUser = SsoUser::where('sso_provider_id', $provider->id)
            ->where('external_id', $externalId)
            ->first();

        if (!$ssoUser) {
            return null;
        }

        $ssoUser->update([
            'email' => $data['emails'][0]['value'] ?? $ssoUser->email,
            'name' => $data['name']['givenName'] . ' ' . $data['name']['familyName'] ?? $ssoUser->name,
            'metadata' => array_merge($ssoUser->metadata ?? [], $data),
        ]);

        $user = $ssoUser->user;
        if ($user) {
            $user->update([
                'name' => $ssoUser->name,
                'email' => $ssoUser->email,
            ]);
        }

        return $ssoUser;
    }

    public function deleteUser(SsoProvider $provider, string $externalId): void
    {
        $ssoUser = SsoUser::where('sso_provider_id', $provider->id)
            ->where('external_id', $externalId)
            ->first();

        if ($ssoUser) {
            $ssoUser->user()->update(['active' => false]);
            $ssoUser->delete();

            Log::channel('sso')->info('SCIM user deleted', [
                'provider_id' => $provider->id,
                'external_id' => $externalId,
            ]);
        }
    }

    protected function formatScimUser(SsoUser $ssoUser): array
    {
        $user = $ssoUser->user;

        return [
            'schemas' => ['urn:ietf:params:scim:schemas:core:2.0:User'],
            'id' => $ssoUser->external_id,
            'externalId' => $ssoUser->external_id,
            'userName' => $ssoUser->email,
            'name' => [
                'givenName' => $user->name ?? '',
                'familyName' => '',
            ],
            'displayName' => $ssoUser->name ?? $user->name ?? '',
            'emails' => [
                ['value' => $ssoUser->email, 'type' => 'work', 'primary' => true],
            ],
            'active' => $user?->active ?? true,
            'meta' => [
                'resourceType' => 'User',
                'created' => $ssoUser->created_at->toIso8601String(),
                'lastModified' => $ssoUser->updated_at->toIso8601String(),
            ],
        ];
    }

    protected function parseFilter(string $filter): string
    {
        $filter = preg_replace('/userName eq "([^"]+)"/', 'email = "$1"', $filter);
        $filter = preg_replace('/externalId eq "([^"]+)"/', 'external_id = "$1"', $filter);
        return $filter;
    }
}
```

```php
// app/Services/SSO/SsoService.php

namespace App\Services\SSO;

use App\Enums\SSO\SsoProviderType;
use App\Models\SsoProvider;
use App\Models\SsoUser;
use App\Models\User;
use App\Services\SSO\Saml\SamlServiceProvider;
use App\Services\SSO\Oidc\OidcServiceProvider;
use App\Services\SSO\Ldap\LdapService;
use Illuminate\Support\Facades\Log;

class SsoService
{
    public function __construct(
        protected SamlServiceProvider $samlProvider,
        protected OidcServiceProvider $oidcProvider,
        protected LdapService $ldapService
    ) {}

    public function findOrCreateUser(SsoProvider $provider, string $externalId, array $attributes): User
    {
        $email = $attributes['email'] ?? null;

        if (!$email) {
            throw new \App\Exceptions\SSO\SsoAuthenticationException(
                'No email address provided by SSO provider'
            );
        }

        $ssoUser = SsoUser::where('sso_provider_id', $provider->id)
            ->where('external_id', $externalId)
            ->first();

        if ($ssoUser) {
            return $this->updateExistingUser($ssoUser, $attributes);
        }

        $existingUser = User::where('email', $email)->first();

        if ($existingUser) {
            return $this->linkExistingUser($existingUser, $provider, $externalId, $attributes);
        }

        if (!$provider->config['provisioning']['auto_create'] ?? true) {
            throw new \App\Exceptions\SSO\SsoAuthenticationException(
                'User not found and auto-provisioning is disabled'
            );
        }

        return $this->createNewUser($provider, $externalId, $attributes);
    }

    protected function updateExistingUser(SsoUser $ssoUser, array $attributes): User
    {
        $user = $ssoUser->user;

        $user->update([
            'name' => $attributes['name'] ?? $user->name,
            'email' => $attributes['email'] ?? $user->email,
        ]);

        $ssoUser->update([
            'email' => $attributes['email'] ?? $ssoUser->email,
            'name' => $attributes['name'] ?? $ssoUser->name,
            'metadata' => array_merge($ssoUser->metadata ?? [], $attributes),
        ]);

        Log::channel('sso')->info('SSO user updated', [
            'user_id' => $user->id,
            'provider_id' => $ssoUser->sso_provider_id,
        ]);

        return $user;
    }

    protected function linkExistingUser(User $user, SsoProvider $provider, string $externalId, array $attributes): User
    {
        SsoUser::create([
            'sso_provider_id' => $provider->id,
            'user_id' => $user->id,
            'external_id' => $externalId,
            'email' => $attributes['email'] ?? $user->email,
            'name' => $attributes['name'] ?? $user->name,
            'metadata' => $attributes,
        ]);

        Log::channel('sso')->info('Existing user linked to SSO provider', [
            'user_id' => $user->id,
            'provider_id' => $provider->id,
        ]);

        return $user;
    }

    protected function createNewUser(SsoProvider $provider, string $externalId, array $attributes): User
    {
        $user = User::create([
            'name' => $attributes['name'] ?? 'SSO User',
            'email' => $attributes['email'],
            'password' => null,
            'email_verified_at' => now(),
        ]);

        SsoUser::create([
            'sso_provider_id' => $provider->id,
            'user_id' => $user->id,
            'external_id' => $externalId,
            'email' => $attributes['email'],
            'name' => $attributes['name'],
            'metadata' => $attributes,
        ]);

        $defaultRole = $provider->config['provisioning']['default_role'] ?? 'user';
        $user->assignRole($defaultRole);

        Log::channel('sso')->info('SSO user created', [
            'user_id' => $user->id,
            'provider_id' => $provider->id,
            'email' => $attributes['email'],
        ]);

        return $user;
    }

    public function deactivateUser(User $user, SsoProvider $provider): void
    {
        $user->update(['active' => false]);

        $user->ssoUsers()
            ->where('sso_provider_id', $provider->id)
            ->update(['metadata' => ['deactivated_at' => now()->toIso8601String()]]);

        Log::channel('sso')->info('SSO user deactivated', [
            'user_id' => $user->id,
            'provider_id' => $provider->id,
        ]);
    }

    public function syncGroups(User $user, SsoProvider $provider, array $groups): void
    {
        $teamIds = [];

        foreach ($groups as $groupName) {
            $team = \App\Models\Team::firstOrCreate(
                ['name' => $groupName],
                ['personal_team' => false]
            );
            $teamIds[] = $team->id;
        }

        $user->teams()->sync($teamIds);
    }
}
```

## SSO Migration Commands

```php
// app/Console/Commands/MigrateSsoProvidersCommand.php

namespace App\Console\Commands;

use App\Models\SsoProvider;
use App\Services\SSO\Saml\SamlServiceProvider;
use App\Services\SSO\Oidc\OidcServiceProvider;
use App\Services\SSO\Ldap\LdapService;
use Illuminate\Console\Command;

class MigrateSsoProvidersCommand extends Command
{
    protected $signature = 'sso:migrate-providers {--sync-users}';
    protected $description = 'Migrate and sync SSO provider configurations';

    public function handle(): int
    {
        $this->info('Migrating SSO providers...');

        $providers = SsoProvider::all();

        foreach ($providers as $provider) {
            $this->line("Processing: {$provider->name} ({$provider->type})");

            $errors = match ($provider->type) {
                'saml' => app(SamlServiceProvider::class)->validateConfiguration($provider),
                'oidc' => app(OidcServiceProvider::class)->validateConfiguration($provider),
                default => [],
            };

            if (!empty($errors)) {
                $this->error("  Configuration errors:");
                foreach ($errors as $error) {
                    $this->error("    - {$error}");
                }
            } else {
                $this->info("  Configuration valid");
            }

            if ($this->option('sync-users') && $provider->type === 'ldap') {
                $this->info("  Syncing LDAP users...");
                $results = app(LdapService::class)->syncUsers($provider);
                $this->info("  Created: {$results['created']}, Updated: {$results['updated']}, Skipped: {$results['skipped']}");
            }
        }

        $this->info('Migration complete!');
        return Command::SUCCESS;
    }
}
```
