# 09.05 — Login

## Purpose

Define the login system for GsmUi, including credential validation, session creation, rate limiting, "remember me", and error handling.

## Login Flow

```text
User visits /login
    ↓
Enters email + password
    ↓
Client-side validation (Livewire)
    ↓
Server-side validation
    ↓
Rate limit check
    ↓
Credential verification (Fortify)
    ↓
Check email verification status
    ↓
Check account status (suspended/deactivated)
    ↓
Create session (Redis)
    ↓
Log login event
    ↓
Redirect to intended URL or dashboard
```

## Livewire Component

```php
// app/Livewire/Auth/Login.php

class Login extends Component
{
    public string $email = '';
    public string $password = '';
    public bool $remember = false;

    protected array $rules = [
        'email' => ['required', 'email'],
        'password' => ['required', 'string'],
    ];

    public function login(): void
    {
        $this->validate();

        // Rate limit: 5 attempts per IP per minute
        $throttles = RateLimiter::attempt(
            'login:' . request()->ip(),
            5,
            fn() => $this->attemptLogin()
        );

        if (! $throttles) {
            $this->addError('email', 'Too many login attempts. Please try again in 60 seconds.');
            return;
        }

        redirect()->intended('/dashboard');
    }

    protected function attemptLogin(): bool
    {
        $user = User::where('email', $this->email)->first();

        if (! $user) {
            // Run hash to prevent timing attacks
            Hash::make('dummy');
            return false;
        }

        if (! Hash::check($this->password, $user->password)) {
            $this->logFailedAttempt($user);
            return false;
        }

        if (! $user->hasVerifiedEmail()) {
            return false;
        }

        if ($user->isSuspended() || $user->isDeactivated()) {
            return false;
        }

        Auth::login($user, $this->remember);

        $this->logSuccessfulLogin($user);

        return true;
    }
}
```

## Remember Me

* Session lifetime extends to 30 days (vs. session-only)
* Uses a secure, HTTP-only cookie
* Token stored in `users.remember_token`

## Post-Login Actions

1. **Session created** in Redis
2. **Login event** dispatched
3. **Activity logged** via Spatie Activity Log
4. **Login history** recorded (IP, user agent, timestamp)
5. **Last login timestamp** updated
6. **Redirect** to intended URL or `/dashboard`

## Security Notes

* Generic error messages prevent email enumeration
* Hash dummy password on invalid email to prevent timing attacks
* All login attempts logged (successful and failed)
* IP-based rate limiting prevents brute force
* Email-based rate limiting prevents credential stuffing
* Session fixation prevented by regenerating session ID on login

## Detailed Login Architecture

### Login Request Validation

```php
// app/Actions/Fortify/AttemptAuthentication.php

class AttemptAuthentication
{
    public function __invoke(Request $request): User
    {
        $request->validate([
            'email' => ['required', 'string'],
            'password' => ['required', 'string'],
        ]);

        $user = User::where('email', $request->email)->first();

        if (! $user || ! Hash::check($request->password, $user->password)) {
            $this->logFailedAttempt($request);
            throw ValidationException::withMessages([
                'email' => ['The provided credentials do not match our records.'],
            ]);
        }

        $this->validateAccountStatus($user);

        return $user;
    }

    protected function validateAccountStatus(User $user): void
    {
        if ($user->hasVerifiedEmail() === false) {
            throw ValidationException::withMessages([
                'email' => ['Please verify your email address before signing in.'],
            ]);
        }

        if ($user->isSuspended()) {
            throw ValidationException::withMessages([
                'email' => ['Your account has been suspended. Please contact support.'],
            ]);
        }

        if ($user->isDeactivated()) {
            throw ValidationException::withMessages([
                'email' => ['Your account has been deactivated.'],
            ]);
        }
    }

    protected function logFailedAttempt(Request $request): void
    {
        activity()
            ->withProperties([
                'email' => $request->email,
                'ip' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ])
            ->log('auth.login_failed');
    }
}
```

### Login Request

```php
// app/Http/Requests/Auth/LoginRequest.php

class LoginRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    public function rules(): array
    {
        return [
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
        ];
    }

    public function authenticate(): void
    {
        $this->ensureIsNotRateLimited();

        $user = User::where('email', $this->email)->first();

        if (! $user || ! Hash::check($this->password, $user->password)) {
            RateLimiter::hit($this->throttleKey(), 60);

            throw ValidationException::withMessages([
                'email' => __('auth.failed'),
            ]);
        }

        if (! $user->hasVerifiedEmail()) {
            throw ValidationException::withMessages([
                'email' => __('auth.not_verified'),
            ]);
        }

        if ($user->isSuspended()) {
            throw ValidationException::withMessages([
                'email' => __('auth.suspended'),
            ]);
        }

        if ($user->isDeactivated()) {
            throw ValidationException::withMessages([
                'email' => __('auth.deactivated'),
            ]);
        }

        Auth::login($user, $this->boolean('remember'));

        RateLimiter::clear($this->throttleKey());

        activity()
            ->performedOn($user)
            ->withProperties([
                'ip' => $this->ip(),
                'user_agent' => $this->userAgent(),
                'remember' => $this->boolean('remember'),
            ])
            ->log('auth.login');
    }

    public function ensureIsNotRateLimited(): void
    {
        if (RateLimiter::tooManyAttempts($this->throttleKey(), 5)) {
            throw ValidationException::withMessages([
                'email' => __('auth.throttle', [
                    'seconds' => RateLimiter::availableIn($this->throttleKey()),
                ]),
            ]);
        }

        RateLimiter::hit($this->throttleKey(), 60);
    }

    public function throttleKey(): string
    {
        return Str::lower($this->input('email')) . '|' . $this->ip();
    }
}
```

### Login Service

```php
// app/Services/LoginService.php

class LoginService
{
    public function login(Request $request, User $user, bool $remember = false): void
    {
        Auth::login($user, $remember);

        $request->session()->regenerate();

        $this->storeSessionMetadata($request, $user);
        $this->logLogin($request, $user, $remember);
        $this->updateLastLogin($user);
    }

    protected function storeSessionMetadata(Request $request, User $user): void
    {
        session([
            'login_ip' => $request->ip(),
            'login_user_agent' => $request->userAgent(),
            'login_at' => now(),
            'two_factor_verified' => false,
        ]);
    }

    protected function logLogin(Request $request, User $user, bool $remember): void
    {
        activity()
            ->performedOn($user)
            ->withProperties([
                'ip' => $request->ip(),
                'user_agent' => $request->userAgent(),
                'remember' => $remember,
                'method' => 'email_password',
            ])
            ->log('auth.login');

        LoginHistory::create([
            'user_id' => $user->id,
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
            'country' => $this->getCountryFromIp($request->ip()),
            'city' => $this->getCityFromIp($request->ip()),
            'successful' => true,
        ]);
    }

    protected function updateLastLogin(User $user): void
    {
        $user->update([
            'last_login_at' => now(),
            'last_login_ip' => request()->ip(),
        ]);
    }

    public function logout(Request $request): void
    {
        $user = $request->user();

        Auth::logout();

        $request->session()->invalidate();
        $request->session()->regenerateToken();

        activity()
            ->performedOn($user)
            ->log('auth.logout');
    }
}
```

### Login Livewire Component

```php
// app/Livewire/Auth/LoginForm.php

namespace App\Livewire\Auth;

use App\Models\User;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Validation\ValidationException;
use Livewire\Component;

class LoginForm extends Component
{
    public string $email = '';
    public string $password = '';
    public bool $remember = false;
    public bool $isSubmitting = false;
    public string $status = '';
    public int $retryAfter = 0;

    protected $listeners = [
        'socialLoggedIn' => 'handleSocialLogin',
    ];

    public function updatedEmail(): void
    {
        $this->validateOnly('email', [
            'email' => ['required', 'string', 'email'],
        ]);
    }

    public function updatedPassword(): void
    {
        $this->validateOnly('password', [
            'password' => ['required', 'string'],
        ]);
    }

    public function login(): void
    {
        $this->validate([
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
        ]);

        // Rate limit check
        if (RateLimiter::tooManyAttempts($this->loginThrottleKey(), 5)) {
            $this->retryAfter = RateLimiter::availableIn($this->loginThrottleKey());

            throw ValidationException::withMessages([
                'email' => 'Too many login attempts. Please try again in ' . $this->retryAfter . ' seconds.',
            ]);
        }

        $this->isSubmitting = true;

        try {
            // Find user
            $user = User::where('email', $this->email)->first();

            if (! $user || ! Hash::check($this->password, $user->password)) {
                RateLimiter::hit($this->loginThrottleKey(), 60);

                throw ValidationException::withMessages([
                    'email' => 'The provided credentials do not match our records.',
                ]);
            }

            // Check verification
            if (! $user->hasVerifiedEmail()) {
                throw ValidationException::withMessages([
                    'email' => 'Please verify your email address before signing in.',
                ]);
            }

            // Check status
            if ($user->isSuspended()) {
                throw ValidationException::withMessages([
                    'email' => 'Your account has been suspended. Please contact support.',
                ]);
            }

            if ($user->isDeactivated()) {
                throw ValidationException::withMessages([
                    'email' => 'Your account has been deactivated.',
                ]);
            }

            // Check 2FA
            if ($user->hasTwoFactorEnabled()) {
                Auth::login($user, $this->remember);

                session(['login.id' => $user->id]);
                session(['login.remember' => $this->remember]);

                return redirect()->route('two-factor.challenge');
            }

            // Login
            Auth::login($user, $this->remember);

            // Regenerate session
            request()->session()->regenerate();

            // Update last login
            $user->update([
                'last_login_at' => now(),
                'last_login_ip' => request()->ip(),
            ]);

            // Log activity
            activity()
                ->performedOn($user)
                ->event('logged_in')
                ->withProperties([
                    'ip' => request()->ip(),
                    'user_agent' => request()->userAgent(),
                    'remember' => $this->remember,
                ])
                ->log('User logged in');

            RateLimiter::clear($this->loginThrottleKey());

            $this->status = 'success';

            return redirect()->intended(route('dashboard'));
        } catch (ValidationException $e) {
            throw $e;
        } catch (\Exception $e) {
            $this->status = 'error';
            $this->addError('login', 'Login failed. Please try again.');
        } finally {
            $this->isSubmitting = false;
        }
    }

    public function handleSocialLogin($data): void
    {
        $this->status = 'success';
    }

    public function loginThrottleKey(): string
    {
        return 'login:' . strtolower($this->email);
    }

    public function render()
    {
        return view('livewire.auth.login-form');
    }
}
```

### Login Form Request

```php
// app/Http/Requests/Auth/LoginRequest.php

class LoginRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    public function rules(): array
    {
        return [
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
            'remember' => ['boolean'],
        ];
    }

    public function authenticate(): void
    {
        $this->ensureIsNotRateLimited();

        $user = User::where('email', $this->email)->first();

        if (! $user || ! Hash::check($this->password, $user->password)) {
            RateLimiter::hit($this->throttleKey(), 60);

            throw ValidationException::withMessages([
                'email' => 'The provided credentials do not match our records.',
            ]);
        }

        if (! $user->hasVerifiedEmail()) {
            throw ValidationException::withMessages([
                'email' => 'Please verify your email address before signing in.',
            ]);
        }

        if ($user->isSuspended()) {
            throw ValidationException::withMessages([
                'email' => 'Your account has been suspended. Please contact support.',
            ]);
        }

        if ($user->isDeactivated()) {
            throw ValidationException::withMessages([
                'email' => 'Your account has been deactivated.',
            ]);
        }

        Auth::login($user, $this->boolean('remember'));

        RateLimiter::clear($this->throttleKey());
    }

    public function ensureIsNotRateLimited(): void
    {
        if (! RateLimiter::tooManyAttempts($this->throttleKey(), 5)) {
            return;
        }

        $seconds = RateLimiter::availableIn($this->throttleKey());

        throw ValidationException::withMessages([
            'email' => 'Too many login attempts. Please try again in ' . $seconds . ' seconds.',
        ]);
    }

    public function throttleKey(): string
    {
        return 'login:' . Str::lower($this->input('email'));
    }
}
```

### Login Security Deep Dive

| Security Measure | Implementation | Purpose |
|---|---|---|
| CSRF Protection | @csrf token on forms | Prevent cross-site request forgery |
| Rate Limiting | 5 per minute per email | Prevent brute force attacks |
| Session Regeneration | Session ID regenerated on login | Prevent session fixation |
| Secure Cookies | HttpOnly, Secure, SameSite=Lax | Prevent cookie theft |
| Password Hashing | Argon2id verification | Never store plain passwords |
| Timing Attack Prevention | Constant-time hash comparison | Prevent timing attacks |
| IP Logging | Log IP with login events | Audit trail |
| User Agent Logging | Browser fingerprinting | Detect suspicious activity |
| Account Lockout | Temporary lockout after 5 failures | Prevent brute force |
| Remember Me Security | Token-based, single-use | Prevent token theft |
| Session Timeout | 24-hour session expiry | Limit exposure window |
| Concurrent Session Control | Max sessions per user | Prevent session abuse |

```php
class DetectSuspiciousLogin
{
    public function handle(UserLoggedIn $event): void
    {
        $lastLogin = $event->user->last_login_at;
        $lastIp = $event->user->last_login_ip;

        $isNewLocation = $lastIp && $lastIp !== $event->ip;
        $isNewDevice = ! str_contains($event->userAgent, 'Chrome') &&
                       ! str_contains($event->userAgent, 'Firefox') &&
                       ! str_contains($event->userAgent, 'Safari');

        if ($isNewLocation || $isNewDevice) {
            $event->user->notify(new SuspiciousLoginNotification(
                ip: $event->ip,
                userAgent: $event->userAgent,
                isNewLocation: $isNewLocation,
                isNewDevice: $isNewDevice,
            ));
        }
    }
```

```php
// app/Livewire/Auth/LoginForm.php

namespace App\Livewire\Auth;

use App\Models\User;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Validation\ValidationException;
use Livewire\Component;

class LoginForm extends Component
{
    public string $email = '';
    public string $password = '';
    public bool $remember = false;
    public bool $isSubmitting = false;
    public string $status = '';
    public int $retryAfter = 0;

    protected $listeners = [
        'socialLoggedIn' => 'handleSocialLogin',
    ];

    public function updatedEmail(): void
    {
        $this->validateOnly('email', [
            'email' => ['required', 'string', 'email'],
        ]);
    }

    public function updatedPassword(): void
    {
        $this->validateOnly('password', [
            'password' => ['required', 'string'],
        ]);
    }

    public function login(): void
    {
        $this->validate([
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
        ]);

        // Rate limit check
        if (RateLimiter::tooManyAttempts($this->loginThrottleKey(), 5)) {
            $this->retryAfter = RateLimiter::availableIn($this->loginThrottleKey());

            throw ValidationException::withMessages([
                'email' => 'Too many login attempts. Please try again in ' . $this->retryAfter . ' seconds.',
            ]);
        }

        $this->isSubmitting = true;

        try {
            // Find user
            $user = User::where('email', $this->email)->first();

            if (! $user || ! Hash::check($this->password, $user->password)) {
                RateLimiter::hit($this->loginThrottleKey(), 60);

                throw ValidationException::withMessages([
                    'email' => 'The provided credentials do not match our records.',
                ]);
            }

            // Check verification
            if (! $user->hasVerifiedEmail()) {
                throw ValidationException::withMessages([
                    'email' => 'Please verify your email address before signing in.',
                ]);
            }

            // Check status
            if ($user->isSuspended()) {
                throw ValidationException::withMessages([
                    'email' => 'Your account has been suspended. Please contact support.',
                ]);
            }

            if ($user->isDeactivated()) {
                throw ValidationException::withMessages([
                    'email' => 'Your account has been deactivated.',
                ]);
            }

            // Check 2FA
            if ($user->hasTwoFactorEnabled()) {
                Auth::login($user, $this->remember);

                session(['login.id' => $user->id]);
                session(['login.remember' => $this->remember]);

                return redirect()->route('two-factor.challenge');
            }

            // Login
            Auth::login($user, $this->remember);

            // Regenerate session
            request()->session()->regenerate();

            // Update last login
            $user->update([
                'last_login_at' => now(),
                'last_login_ip' => request()->ip(),
            ]);

            // Log activity
            activity()
                ->performedOn($user)
                ->event('logged_in')
                ->withProperties([
                    'ip' => request()->ip(),
                    'user_agent' => request()->userAgent(),
                    'remember' => $this->remember,
                ])
                ->log('User logged in');

            RateLimiter::clear($this->loginThrottleKey());

            $this->status = 'success';

            return redirect()->intended(route('dashboard'));
        } catch (ValidationException $e) {
            throw $e;
        } catch (\Exception $e) {
            $this->status = 'error';
            $this->addError('login', 'Login failed. Please try again.');
        } finally {
            $this->isSubmitting = false;
        }
    }

    public function handleSocialLogin($data): void
    {
        $this->status = 'success';
    }

    public function loginThrottleKey(): string
    {
        return 'login:' . strtolower($this->email);
    }

    public function render()
    {
        return view('livewire.auth.login-form');
    }
}
```

### Login Form Request

```php
// app/Http/Requests/Auth/LoginRequest.php

class LoginRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    public function rules(): array
    {
        return [
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
            'remember' => ['boolean'],
        ];
    }

    public function authenticate(): void
    {
        $this->ensureIsNotRateLimited();

        $user = User::where('email', $this->email)->first();

        if (! $user || ! Hash::check($this->password, $user->password)) {
            RateLimiter::hit($this->throttleKey(), 60);

            throw ValidationException::withMessages([
                'email' => 'The provided credentials do not match our records.',
            ]);
        }

        if (! $user->hasVerifiedEmail()) {
            throw ValidationException::withMessages([
                'email' => 'Please verify your email address before signing in.',
            ]);
        }

        if ($user->isSuspended()) {
            throw ValidationException::withMessages([
                'email' => 'Your account has been suspended. Please contact support.',
            ]);
        }

        if ($user->isDeactivated()) {
            throw ValidationException::withMessages([
                'email' => 'Your account has been deactivated.',
            ]);
        }

        Auth::login($user, $this->boolean('remember'));

        RateLimiter::clear($this->throttleKey());
    }

    public function ensureIsNotRateLimited(): void
    {
        if (! RateLimiter::tooManyAttempts($this->throttleKey(), 5)) {
            return;
        }

        $seconds = RateLimiter::availableIn($this->throttleKey());

        throw ValidationException::withMessages([
            'email' => 'Too many login attempts. Please try again in ' . $seconds . ' seconds.',
        ]);
    }

    public function throttleKey(): string
    {
        return 'login:' . Str::lower($this->input('email'));
    }
}
```
