## Password Reset Livewire Component

```php
// app/Livewire/Auth/ForgotPassword.php

class ForgotPassword extends Component
{
    public string $email = '';

    protected array $rules = [
        'email' => ['required', 'email'],
    ];

    public function sendResetLink(): void
    {
        $this->validate();

        // Rate limit: 1 per minute, 3 per hour
        RateLimiter::attempt(
            'password-reset:' . $this->email,
            1,
            fn() => $this->dispatchResetLink()
        );

        $this->dispatch('reset-link-sent');
    }

    protected function dispatchResetLink(): void
    {
        $status = Password::sendResetLink([
            'email' => $this->email,
        ]);

        if ($status === Password::RESET_LINK_SENT) {
            $this->dispatch('reset-link-sent');
        }
    }
}
```

## Password Reset Handler

```php
// app/Livewire/Auth/ResetPassword.php

class ResetPassword extends Component
{
    public string $token = '';
    public string $email = '';
    public string $password = '';
    public string $password_confirmation = '';

    protected array $rules = [
        'email' => ['required', 'email'],
        'password' => ['required', 'string', 'min:8', 'confirmed'],
    ];

    public function resetPassword(): void
    {
        $this->validate();

        $status = Password::reset(
            [
                'email' => $this->email,
                'password' => $this->password,
                'password_confirmation' => $this->password_confirmation,
                'token' => $this->token,
            ],
            function (User $user, string $password) {
                $user->forceFill([
                    'password' => Hash::make($password),
                    'remember_token' => Str::random(60),
                ])->save();

                // Revoke all sessions except current
                Session::where('user_id', $user->id)
                    ->where('id', '!=', session()->getId())
                    ->delete();

                event(new PasswordReset($user));
            }
        );

        if ($status === Password::PASSWORD_RESET) {
            redirect()->route('login')
                ->with('status', __($status));
        }
    }
}
```

## Rate Limiting

|---|---|---|
| Request password reset | 1 | Per email per minute |
| Request password reset | 3 | Per email per hour |

## Security Notes

* Password reset tokens are single-use
* Old tokens invalidated when new one generated
* All other sessions revoked after password reset
* Password change requires current password verification
* Password confirmation required for sensitive operations
* No plain-text passwords in logs or responses
* Argon2id prevents GPU-based attacks
* Password history not enforced (future enhancement)

### Password Strength Indicator

```php
// app/Services/PasswordStrengthService.php

class PasswordStrengthService
{
    public function calculate(string $password): array
    {
        $score = 0;

        // Length
        if (strlen($password) >= 8) $score++;
        if (strlen($password) >= 12) $score++;
        if (strlen($password) >= 16) $score++;

        // Character types
        if (preg_match('/[a-z]/', $password)) $score++;
        if (preg_match('/[A-Z]/', $password)) $score++;
        if (preg_match('/[0-9]/', $password)) $score++;
        if (preg_match('/[^a-zA-Z0-9]/', $password)) $score++;

        // Patterns (reduce score for common patterns)
        if (preg_match('/^[a-z]+$/i', $password)) $score--;
        if (preg_match('/^[0-9]+$/', $password)) $score--;
        if (preg_match('/(.)\1{2,}/', $password)) $score--;

        return [
            'score' => max(0, min($score, 4)),
            'label' => $this->getLabel($score),
            'color' => $this->getColor($score),
            'suggestions' => $this->getSuggestions($password),
        ];
    }

    protected function getLabel(int $score): string
    {
        return match(true) {
            $score <= 1 => 'Weak',
            $score <= 2 => 'Fair',
            $score <= 3 => 'Good',
            default => 'Strong',
        };
    }

    protected function getColor(int $score): string
    {
        return match(true) {
            $score <= 1 => 'red',
            $score <= 2 => 'orange',
            $score <= 3 => 'yellow',
            default => 'green',
        };
    }

    protected function getSuggestions(string $password): array
    {
        $suggestions = [];

        if (strlen($password) < 12) {
            $suggestions[] = 'Use at least 12 characters';
        }

        if (! preg_match('/[A-Z]/', $password)) {
            $suggestions[] = 'Add an uppercase letter';
        }

        if (! preg_match('/[a-z]/', $password)) {
            $suggestions[] = 'Add a lowercase letter';
        }

        if (! preg_match('/[0-9]/', $password)) {
            $suggestions[] = 'Add a number';
        }

        if (! preg_match('/[^a-zA-Z0-9]/', $password)) {
            $suggestions[] = 'Add a special character';
        }

        return $suggestions;
    }
}
```

### Password Service

```php
// app/Services/PasswordService.php

class PasswordService
{
    public function requestReset(string $email): string
    {
        $status = Password::sendResetLink(['email' => $email]);

        if ($status !== Password::RESET_LINK_SENT) {
            throw new PasswordResetException('Failed to send reset link.');
        }

        $user = User::where('email', $email)->first();

        if ($user) {
            activity()
                ->performedOn($user)
                ->withProperties(['ip' => request()->ip()])
                ->log('auth.password_reset_requested');
        }

        return $status;
    }

    public function reset(array $data): string
    {
        $status = Password::reset($data, function (User $user, string $password) {
            $user->forceFill([
                'password' => Hash::make($password),
                'remember_token' => Str::random(60),
            ])->save();

            // Revoke all other sessions
            Session::where('user_id', $user->id)
                ->where('id', '!=', session()->getId())
                ->delete();

            // Revoke all API tokens
            $user->tokens()->delete();

            event(new PasswordReset($user));
        });

        if ($status !== Password::PASSWORD_RESET) {
            throw new PasswordResetException('Failed to reset password.');
        }

        return $status;
    }

    public function change(User $user, array $data): void
    {
        if (! Hash::check($data['current_password'], $user->password)) {
            throw new InvalidPasswordException('Current password is incorrect.');
        }

        if (Hash::check($data['password'], $user->password)) {
            throw new InvalidPasswordException('New password must be different from current password.');
        }

        $user->forceFill([
            'password' => Hash::make($data['password']),
        ])->save();

        // Revoke other sessions
        Session::where('user_id', $user->id)
            ->where('id', '!=', session()->getId())
            ->delete();

        event(new PasswordChanged($user));
    }

    public function confirmPassword(User $user, string $password): void
    {
        if (! Hash::check($password, $user->password)) {
            throw new InvalidPasswordException('Password is incorrect.');
        }

        session(['auth.password_confirmed_at' => time()]);
    }

    public function checkPasswordStrength(string $password): array
    {
        $service = new PasswordStrengthService();
        return $service->calculate($password);
    }
}
```

```php
    public string $token = '';

    public string $password = '';
    public string $password_confirmation = '';

    public string $status = '';
    public array $passwordStrength = [];

    {
        $this->token = $token;
        $this->email = $email;
    }

    public function updatedPassword(): void
    {
        $this->passwordStrength = $this->calculatePasswordStrength($this->password);
    }

    public function resetPassword(): void
    {
        $this->validate([
            'token' => ['required'],
            'email' => ['required', 'email'],
            'password' => ['required', 'confirmed', 'min:8'],
        ]);

        $this->isSubmitting = true;

        try {
            $status = Password::reset(
                [
                    'email' => $this->email,
                    'password' => $this->password,
                    'password_confirmation' => $this->password_confirmation,
                    'token' => $this->token,
                ],
                function (User $user, string $password) {

                        'password' => Hash::make($password),
                        'remember_token' => Str::random(60),
                    ])->save();

                    event(new PasswordReset($user));

                    // Revoke all other sessions
                    $user->sessions()->where('id', '!=', session()->getId())->delete();

            $this->isSubmitting = false;
        }

        if (preg_match('/[A-Z]/', $password)) $score++;
        if (preg_match('/[a-z]/', $password)) $score++;
        if (preg_match('/[0-9]/', $password)) $score++;
        if (preg_match('/[^A-Za-z0-9]/', $password)) $score++;

        $labels = ['Very Weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very Strong'];
        $colors = ['bg-red-500', 'bg-orange-500', 'bg-yellow-500', 'bg-lime-500', 'bg-green-500', 'bg-emerald-500'];

        return [
            'score' => min($score, 5),
            'label' => $labels[min($score, 5)],
            'color' => $colors[min($score, 5)],
            'percentage' => min($score * 20, 100),
        ];
    }
```

### Password Database Schema

```sql
-- Password reset tokens table (Laravel default)
CREATE TABLE password_reset_tokens (
    email VARCHAR(255) PRIMARY KEY,
    token VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP
);

-- Password history table (for rotation policy)
CREATE TABLE password_history (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_created (user_id, created_at DESC)
);

-- Password strength tracking (analytics)
CREATE TABLE password_strength_logs (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NULL,
    entropy_score DECIMAL(10,2) NOT NULL,
    strength_label VARCHAR(50) NOT NULL,
    is_compromised BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_user_created (user_id, created_at DESC),
    INDEX idx_strength (strength_label)
);
```

### Password Monitoring Queries

```php
// app/Services/PasswordMonitoringService.php

class PasswordMonitoringService
{
    public function getCompromisedPasswordCount(): int
    {
        return DB::table('password_strength_logs')
            ->where('is_compromised', true)
            ->count();
    }

    public function getAverageEntropy(): float
    {
        return DB::table('password_strength_logs')
            ->avg('entropy_score') ?? 0;
    }

    public function getStrengthDistribution(): array
    {
        return DB::table('password_strength_logs')
            ->select('strength_label', DB::raw('COUNT(*) as count'))
            ->groupBy('strength_label')
            ->pluck('count', 'strength_label')
            ->toArray();
    }

    public function getWeakPasswordUsers(int $limit = 100): Collection
    {
        return DB::table('password_strength_logs')
            ->join('users', 'password_strength_logs.user_id', '=', 'users.id')
            ->where('entropy_score', '<', 36)
            ->where('created_at', '>', now()->subDays(30))
            ->select('users.id', 'users.email', 'users.name', 'entropy_score', 'strength_label')
            ->orderBy('entropy_score')
            ->limit($limit)
            ->get();
    }

    public function getResetTokenUsage(int $days = 30): array
    {
        $total = DB::table('activity_log')
            ->where('log_name', 'auth')
            ->where('event', 'password_reset_requested')
            ->where('created_at', '>', now()->subDays($days))
            ->count();

        $completed = DB::table('activity_log')
            ->where('log_name', 'auth')
            ->where('event', 'password_reset')
            ->where('created_at', '>', now()->subDays($days))
            ->count();

        return [
            'total_requests' => $total,
            'completed' => $completed,
            'completion_rate' => $total > 0 ? round(($completed / $total) * 100, 2) : 0,
        ];
    }
}
```

### Password Rotation Scheduler

```php
// app/Console/Kernel.php - Schedule password rotation checks

protected function schedule(Schedule $schedule): void
{
    if (config('password-rotation.enabled')) {
        // Daily check for expiring passwords
        $schedule->job(new CheckPasswordExpirations)
            ->daily()
            ->at('02:00')
            ->onQueue('passwords');

        // Weekly cleanup of old reset tokens
        $schedule->command('password:cleanup-tokens')
            ->weekly()
            ->onQueue('maintenance');

        // Monthly password strength report
        $schedule->job(new GeneratePasswordReport)
            ->monthlyOn(1, '03:00')
            ->onQueue('reports');
    }
}

// app/Jobs/GeneratePasswordReport.php

class GeneratePasswordReport implements ShouldQueue
{
    public function handle(): void
    {
        $service = app(PasswordMonitoringService::class);
        
        $report = [
            'period' => now()->subMonth()->format('Y-m'),
            'total_users' => User::count(),
            'compromised_passwords' => $service->getCompromisedPasswordCount(),
            'average_entropy' => round($service->getAverageEntropy(), 2),
            'strength_distribution' => $service->getStrengthDistribution(),
            'weak_password_users' => $service->getWeakPasswordUsers(1000)->count(),
            'reset_token_usage' => $service->getResetTokenUsage(),
        ];

        // Store report
        DB::table('password_reports')->insert([
            'period' => $report['period'],
            'data' => json_encode($report),
            'created_at' => now(),
        ]);

        // Email to admins
        Notification::route('mail', config('admin.email'))
            ->notify(new PasswordReportNotification($report));
    }
}
```
