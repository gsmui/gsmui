# 09.31 — Team Authentication

## Purpose

Define the team authentication architecture for GsmUi, supporting team accounts with role-based access within teams.

## Status: Phase 3 (Future)

Team authentication is a Phase 3 feature.

## Team Roles

| Role | Description |
|---|---|
| Team Owner | Full team control, billing |
| Team Admin | Manage members, settings |
| Team Member | Standard access |
| Team Viewer | Read-only access |

## Team Authentication Flow

```text
User belongs to Team
    ↓
User authenticates (standard login)
    ↓
Team context resolved
    ↓
Team role determines access
    ↓
Resources scoped to team
```

```php
// app/Services/TeamCreationService.php

class TeamCreationService
{
    public function createTeam(User $owner, string $name, array $settings = []): Team
    {
        $team = DB::transaction(function () use ($owner, $name, $settings) {
            $team = Team::create([
                'name' => $name,
                'owner_id' => $owner->id,
                'personal_team' => false,
                'settings' => array_merge([
                    'allow_member_invites' => true,
                    'default_role' => 'member',
                    'require_2fa' => false,
                    'allow_sso' => false,
                    'max_members' => 50,
                    'max_projects' => 10,
                    'billing_email' => $owner->email,
                ], $settings),
            ]);

            TeamUser::create([
                'team_id' => $team->id,
                'user_id' => $owner->id,
                'role' => 'owner',
                'joined_at' => now(),
            ]);

            $owner->update(['current_team_id' => $team->id]);

            activity('teams')
                ->event('team.created')
                ->performedOn($team)
                ->causedBy($owner)
                ->withProperties(['name' => $name])
                ->log("Team '{$name}' created");

            return $team;
        });

        return $team;
    }

    public function createPersonalTeam(User $user): Team
    {
        return $this->createTeam($user, $user->name . "'s Team", [
            'personal_team' => true,
            'max_members' => 1,
        ]);
    }
}
```

```php
    }

    public function isAccepted(): bool
    {
        return $this->accepted_at !== null;
    }

    public function isValid(): bool
    {
        return ! $this->isAccepted() && ! $this->isExpired();
    }
```

```php
// app/Services/TeamInvitationService.php

class TeamInvitationService
{
    private const EXPIRY_DAYS = 7;
    private const MAX_INVITES_PER_DAY = 20;

    public function invite(Team $team, string $email, string $role, User $inviter): TeamInvitation
    {
        if (! $this->canInvite($team, $inviter)) {
            throw new UnauthorizedTeamActionException('You do not have permission to invite members.');
        }

        if ($team->users()->where('email', $email)->exists()) {
            throw new TeamMemberAlreadyExistsException('User is already a team member.');
        }

        if ($team->invitations()->where('email', $email)->whereNull('accepted_at')->exists()) {
            throw new PendingInvitationExistsException('An invitation is already pending for this email.');
        }

        if ($team->users()->count() >= $team->settings['max_members']) {
            throw new TeamMemberLimitReachedException('Team has reached its maximum member limit.');
        }

        $invitation = TeamInvitation::create([
            'team_id' => $team->id,
            'email' => $email,
            'role' => $role,
            'invited_by' => $inviter->id,
            'token' => Str::random(64),
            'expires_at' => now()->addDays(self::EXPIRY_DAYS),
        ]);

        Mail::to($email)->send(new TeamInvitationEmail($invitation, $inviter, $team));

        activity('teams')
            ->event('team.member_invited')
            ->performedOn($team)
            ->causedBy($inviter)
            ->withProperties(['email' => $email, 'role' => $role])
            ->log("Invitation sent to {$email} as {$role}");

        return $invitation;
    }

    public function accept(TeamInvitation $invitation, User $user): void
    {
        if (! $invitation->isValid()) {
            throw new InvalidInvitationException('This invitation is no longer valid.');
        }

        if ($invitation->email !== $user->email) {
            throw new EmailMismatchException('This invitation was sent to a different email address.');
        }

        DB::transaction(function () use ($invitation, $user) {
            TeamUser::create([
                'team_id' => $invitation->team_id,
                'user_id' => $user->id,
                'role' => $invitation->role,
                'joined_at' => now(),
            ]);

            $invitation->update(['accepted_at' => now()]);

            activity('teams')
                ->event('team.member_joined')
                ->performedOn($invitation->team)
                ->causedBy($user)
                ->withProperties(['role' => $invitation->role])
                ->log("{$user->email} joined the team as {$invitation->role}");
        });
    }

    public function revoke(TeamInvitation $invitation, User $revoker): void
    {
        $invitation->delete();

        activity('teams')
            ->event('team.invitation_revoked')
            ->performedOn($invitation->team)
            ->causedBy($revoker)
            ->withProperties(['email' => $invitation->email])
            ->log("Invitation to {$invitation->email} revoked");
    }

    private function canInvite(Team $team, User $inviter): bool
    {
        $membership = $team->users()->where('user_id', $inviter->id)->first();

        if (! $membership) return false;

        return in_array($membership->role, ['owner', 'admin']);
    }
}
```

### Team Middleware Implementation

```php
// app/Http/Middleware/EnsureUserBelongsToTeam.php

class EnsureUserBelongsToTeam
{
    public function handle(Request $request, Closure $next, string ...$roles): mixed
    {
        $user = $request->user();

        if (! $user) {
            return redirect()->route('login');
        }

        $teamId = $request->route('team')?->id ?? $user->current_team_id;

        if (! $teamId) {
            return redirect()->route('teams.select');
        }

        $membership = TeamUser::where('team_id', $teamId)
            ->where('user_id', $user->id)
            ->first();

        if (! $membership) {
            abort(403, 'You are not a member of this team.');
        }

        if ($membership->is_suspended) {
            abort(403, 'Your team access has been suspended.');
        }

        if (! empty($roles) && ! in_array($membership->role, $roles)) {
            abort(403, 'You do not have the required role for this action.');
        }

        $request->attributes->set('team', Team::find($teamId));
        $request->attributes->set('team_role', $membership->role);

        return $next($request);
    }
}
```

### Team Scoping for Eloquent

```php
// app/Models/Concerns/BelongsToTeam.php

trait BelongsToTeam
{
    public function bootBelongsToTeam(): void
    {
        // Auto-filter by team context
        static::addGlobalScope('team', function (Builder $builder) {
            if (auth()->check() && request()->route('team')) {
                $builder->where('team_id', request()->route('team')->id);
            }
        });
    }

    public function team(): BelongsTo
    {
        return $this->belongsTo(Team::class);
    }

    public function scopeForTeam(Builder $query, Team $team): Builder
    {
        return $query->where('team_id', $team->id);
    }
}
```

### Team API Token Scoping

```php
// app/Services/TeamTokenService.php

class TeamTokenService
{
    public function createTeamToken(
        Team $team,
        User $creator,
        string $name,
        array $scopes,
        ?Carbon $expiresAt = null,
    ): NewAccessToken {
        $token = $creator->createToken($name, $scopes);

        $token->accessToken->update([
            'team_id' => $team->id,
            'expires_at' => $expiresAt,
        ]);

        activity('teams')
            ->event('team.token_created')
            ->performedOn($team)
            ->causedBy($creator)
            ->withProperties(['token_name' => $name, 'scopes' => $scopes])
            ->log("API token '{$name}' created for team");

        return $token;
    }

    public function getTeamTokens(Team $team): Collection
    {
        return PersonalAccessToken::where('team_id', $team->id)
            ->with('tokenable')
            ->get();
    }

    public function revokeTeamTokens(Team $team, ?User $exceptUser = null): int
    {
        $query = PersonalAccessToken::where('team_id', $team->id);

        if ($exceptUser) {
            $query->where('tokenable_id', '!=', $exceptUser->id);
        }

        return $query->delete();
    }
}
```

### Team Model

```php
// app/Models/Team.php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\Relations\HasOne;

class Team extends Model
{
    use HasFactory;

    protected $fillable = [
        'name',
        'slug',
        'owner_id',
        'personal_team',
        'billing_plan',
        'billing_trial_ends_at',
        'settings',
    ];

    protected $casts = [
        'personal_team' => 'boolean',
        'settings' => 'array',
        'billing_trial_ends_at' => 'datetime',
    ];

    public function owner(): BelongsTo
    {
        return $this->belongsTo(User::class, 'owner_id');
    }

    public function users(): BelongsToMany
    {
        return $this->belongsToMany(User::class, 'team_user')
            ->withPivot(['role', 'is_suspended', 'suspended_at', 'invited_by', 'joined_at'])
            ->withTimestamps();
    }

    public function invitations(): HasMany
    {
        return $this->hasMany(TeamInvitation::class);
    }

    public function auditLog(): HasMany
    {
        return $this->hasMany(TeamAuditLog::class);
    }

    public function settings(): HasMany
    {
        return $this->hasMany(TeamSetting::class);
    }

    public function apiTokens(): HasMany
    {
        return $this->hasMany(TeamApiToken::class);
    }

    public function scopes(): HasMany
    {
        return $this->hasMany(TeamScope::class);
    }

    public function isPersonalTeam(): bool
    {
        return $this->personal_team;
    }

    public function isOwner(User $user): bool
    {
        return $this->owner_id === $user->id;
    }

    public function hasUser(User $user): bool
    {
        return $this->users()->where('user_id', $user->id)->exists();
    }

    public function getUserRole(User $user): ?string
    {
        $membership = $this->users()->where('user_id', $user->id)->first();
        return $membership ? $membership->pivot->role : null;
    }

    public function addUser(User $user, string $role = 'member', ?int $invitedBy = null): void
    {
        $this->users()->attach($user->id, [
            'role' => $role,
            'invited_by' => $invitedBy,
            'joined_at' => now(),
        ]);

        $this->logEvent('member_joined', $user, ['role' => $role]);
    }

    public function removeUser(User $user): void
    {
        $this->users()->detach($user->id);
        $this->logEvent('member_removed', $user);
    }

    public function changeUserRole(User $user, string $newRole): void
    {
        $oldRole = $this->getUserRole($user);

        $this->users()->updateExistingPivot($user->id, ['role' => $newRole]);

        $this->logEvent('role_changed', $user, [
            'old_role' => $oldRole,
            'new_role' => $newRole,
        ]);
    }

    public function suspendUser(User $user, ?string $reason = null): void
    {
        $this->users()->updateExistingPivot($user->id, [
            'is_suspended' => true,
            'suspended_at' => now(),
        ]);

        $this->logEvent('member_suspended', $user, ['reason' => $reason]);
    }

    public function unsuspendUser(User $user): void
    {
        $this->users()->updateExistingPivot($user->id, [
            'is_suspended' => false,
            'suspended_at' => null,
        ]);

        $this->logEvent('member_unsuspended', $user);
    }

    public function transferOwnership(User $newOwner): void
    {
        $oldOwner = $this->owner;

        $this->update(['owner_id' => $newOwner->id]);

        $this->changeUserRole($oldOwner, 'admin');
        $this->changeUserRole($newOwner, 'owner');

        $this->logEvent('ownership_transferred', $newOwner, [
            'old_owner_id' => $oldOwner->id,
            'new_owner_id' => $newOwner->id,
        ]);
    }

    public function getMemberCount(): int
    {
        return $this->users()->count();
    }

    public function getActiveMemberCount(): int
    {
        return $this->users()->wherePivot('is_suspended', false)->count();
    }

    public function getMaxMembers(): int
    {
        return match ($this->billing_plan) {
            'free' => 5,
            'pro' => 25,
            'business' => 100,
            'enterprise' => PHP_INT_MAX,
            default => 5,
        };
    }

    public function canAddMember(): bool
    {
        return $this->getMemberCount() < $this->getMaxMembers();
    }

    protected function logEvent(string $event, ?User $user = null, array $properties = []): void
    {
        TeamAuditLog::create([
            'team_id' => $this->id,
            'user_id' => $user?->id,
            'event' => $event,
            'description' => "Team {$event}",
```

```php
// app/Services/Team/TeamInvitationService.php

namespace App\Services\Team;

use App\Models\Team;
use App\Models\TeamInvitation;
use App\Models\User;
use App\Notifications\TeamInvitationNotification;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class TeamInvitationService
{
    public function invite(Team $team, string $email, string $role, User $invitedBy): TeamInvitation
    {
        if (!$team->canAddMember()) {
            throw new \App\Exceptions\Team\TeamFullException('Team has reached maximum member limit.');
        }

        if ($team->users()->where('email', $email)->exists()) {
            throw new \App\Exceptions\Team\UserAlreadyInTeamException('User is already a team member.');
        }

        $existingInvitation = TeamInvitation::where('team_id', $team->id)
            ->where('email', $email)
            ->whereNull('accepted_at')
            ->where('expires_at', '>', now())
            ->first();

        if ($existingInvitation) {
            throw new \App\Exceptions\Team\InvitationPendingException('An invitation is already pending for this email.');
        }

        $invitation = TeamInvitation::create([
            'team_id' => $team->id,
            'email' => $email,
            'role' => $role,
            'invited_by' => $invitedBy->id,
            'token' => Str::random(64),
            'expires_at' => now()->addDays(7),
        ]);

        $team->logEvent('member_invited', $invitedBy, [
            'email' => $email,
            'role' => $role,
        ]);

        return $invitation;
    }

    public function accept(TeamInvitation $invitation, User $user): void
    {
        if ($invitation->accepted_at) {
            throw new \App\Exceptions\Team\InvitationAlreadyAcceptedException('Invitation has already been accepted.');
        }

        if ($invitation->expires_at->isPast()) {
            throw new \App\Exceptions\Team\InvitationExpiredException('Invitation has expired.');
        }

        if ($invitation->email !== $user->email) {
            throw new \App\Exceptions\Team\InvitationEmailMismatchException('Invitation email does not match user email.');
        }

        DB::transaction(function () use ($invitation, $user) {
            $invitation->team->addUser($user, $invitation->role, $invitation->invited_by);

            $invitation->update(['accepted_at' => now()]);
        });

        $invitation->team->logEvent('member_joined', $user, [
            'role' => $invitation->role,
            'invited_by' => $invitation->invited_by,
        ]);
    }

    public function revoke(TeamInvitation $invitation, User $revokedBy): void
    {
        $invitation->delete();

        $invitation->team->logEvent('invitation_revoked', $revokedBy, [
            'email' => $invitation->email,
        ]);
    }

    public function resend(TeamInvitation $invitation): void
    {
        if ($invitation->expires_at->isPast()) {
            $invitation->update(['expires_at' => now()->addDays(7)]);
        }

        $invitation->user->notify(new TeamInvitationNotification($invitation));
    }

    public function getPendingInvitations(Team $team): \Illuminate\Database\Eloquent\Collection
    {
        return TeamInvitation::where('team_id', $team->id)
            ->whereNull('accepted_at')
            ->where('expires_at', '>', now())
            ->with('invitedBy')
            ->get();
    }

    public function cleanupExpired(): int
    {
        return TeamInvitation::where('expires_at', '<', now())
            ->whereNull('accepted_at')
            ->delete();
    }
}
```

```php
        $requiredLevel = $this->getRequiredLevel($permission);

        if ($roleLevel < $requiredLevel) {
            return false;
        }

        return in_array($permission, $this->rolePermissions[$role] ?? []);
    }

    }

    {
        $role = $team->getUserRole($user);
        return $this->rolePermissions[$role] ?? [];
    }

    public function getRoleLevel(string $role): int
    {
        return $this->roleHierarchy[$role] ?? 0;

            if (in_array($permission, $permissions)) {
                return $this->roleHierarchy[$role];
            }
        }
        return PHP_INT_MAX;
    }
```

```php
namespace App\Http\Middleware;

use App\Services\Team\TeamAuthorizationService;
use Closure;
use Illuminate\Http\Request;

class EnsureTeamMember
{
    public function __construct(
        protected TeamAuthorizationService $authService
    ) {}

    public function handle(Request $request, Closure $next, ?string $permission = null)
    {
        $team = $request->route('team');
        $user = $request->user();

        if (!$team || !$user) {
            abort(403, 'Unauthorized.');
        }

        if (!$team->hasUser($user)) {
            abort(403, 'You are not a member of this team.');
        }

        if ($team->users()->where('user_id', $user->id)->first()?->pivot->is_suspended) {
            abort(403, 'Your team membership is suspended.');
        }

        if ($permission && !$this->authService->can($user, $team, $permission)) {
            abort(403, 'You do not have permission to perform this action.');
        }

        $request->attributes->set('current_team', $team);

        return $next($request);
    }
}

use Closure;
use Illuminate\Http\Request;

class EnsureTeamApiToken
{
    public function handle(Request $request, Closure $next)
    {
        $token = $request->bearerToken();

        if (!$token) {
            return response()->json(['error' => 'Token required'], 401);
        }

        $apiToken = TeamApiToken::where('token', hash('sha256', $token))
            ->where('is_active', true)
            ->where('expires_at', '>', now())
            ->first();

        if (!$apiToken) {
            return response()->json(['error' => 'Invalid token'], 401);
        }

        $team = $apiToken->team;

        if (!$team) {
            return response()->json(['error' => 'Team not found'], 401);
        }

        $request->attributes->set('current_team', $team);
        $request->attributes->set('team_api_token', $apiToken);
        $request->setUser($team->owner);

        return $next($request);
    }
}
```

```php
namespace App\Services\Team;

use App\Models\Team;
use App\Models\TeamApiToken;
use App\Models\User;
use Illuminate\Support\Str;

class TeamApiTokenService
{
    public function create(Team $team, string $name, array $scopes, ?User $createdBy, ?\DateTimeInterface $expiresAt = null): array
    {
        $token = Str::random(64);
        $hashedToken = hash('sha256', $token);

        $apiToken = TeamApiToken::create([
            'team_id' => $team->id,
            'name' => $name,
            'token' => $hashedToken,
            'scopes' => $scopes,
            'created_by' => $createdBy?->id,
            'expires_at' => $expiresAt,
            'is_active' => true,
        ]);

        $team->logEvent('api_token_created', $createdBy, [
            'token_name' => $name,
            'scopes' => $scopes,
        ]);

        return [
            'token' => $token,
            'token_id' => $apiToken->id,
            'name' => $name,
            'scopes' => $scopes,
            'expires_at' => $expiresAt,
        ];
    }

    public function revoke(TeamApiToken $token, ?User $revokedBy): void
    {
        $token->update(['is_active' => false]);

        $token->team->logEvent('api_token_revoked', $revokedBy, [
            'token_name' => $token->name,
        ]);
    }

    public function revokeAll(Team $team, ?User $revokedBy): int
    {
        $count = $team->apiTokens()->where('is_active', true)->update(['is_active' => false]);

        $team->logEvent('all_api_tokens_revoked', $revokedBy, [
            'count' => $count,
        ]);

        return $count;
    }

    public function validate(TeamApiToken $token, string $scope): bool
    {
        if (!$token->is_active) return false;
        if ($token->expires_at && $token->expires_at->isPast()) return false;

        return in_array('*', $token->scopes) || in_array($scope, $token->scopes);
    }

    public function getActiveTokens(Team $team): \Illuminate\Database\Eloquent\Collection
    {
        return $team->apiTokens()
            ->where('is_active', true)
            ->orderByDesc('created_at')
            ->get();
    }
}
```

```php
namespace Tests\Feature\Team;

use App\Models\Team;
use App\Models\User;
use App\Services\Team\TeamAuthorizationService;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

class TeamAuthorizationTest extends TestCase
{
    use RefreshDatabase;

    protected TeamAuthorizationService $authService;

    protected function setUp(): void
    {
        parent::setUp();
        $this->authService = new TeamAuthorizationService();
    }

    public function test_owner_has_all_permissions(): void
    {
        $owner = User::factory()->create();
        $team = Team::create(['name' => 'Team', 'owner_id' => $owner->id]);
        $team->addUser($owner, 'owner');

        $this->assertTrue($this->authService->canManageTeam($owner, $team));
        $this->assertTrue($this->authService->canInviteMembers($owner, $team));
        $this->assertTrue($this->authService->canRemoveMembers($owner, $team));
        $this->assertTrue($this->authService->canManageBilling($owner, $team));
        $this->assertTrue($this->authService->canManageSettings($owner, $team));
    }

    public function test_member_has_limited_permissions(): void
    {
        $owner = User::factory()->create();
        $member = User::factory()->create();
        $team = Team::create(['name' => 'Team', 'owner_id' => $owner->id]);
        $team->addUser($owner, 'owner');
        $team->addUser($member, 'member');

        $this->assertFalse($this->authService->canManageTeam($member, $team));
        $this->assertFalse($this->authService->canInviteMembers($member, $team));
        $this->assertFalse($this->authService->canRemoveMembers($member, $team));
        $this->assertFalse($this->authService->canManageBilling($member, $team));
        $this->assertTrue($this->authService->canCreateResources($member, $team));
        $this->assertTrue($this->authService->canViewResources($member, $team));
    }

    public function test_viewer_can_only_view(): void
    {
        $owner = User::factory()->create();
        $viewer = User::factory()->create();
        $team = Team::create(['name' => 'Team', 'owner_id' => $owner->id]);
        $team->addUser($owner, 'owner');
        $team->addUser($viewer, 'viewer');

        $this->assertFalse($this->authService->canCreateResources($viewer, $team));
        $this->assertFalse($this->authService->canEditResources($viewer, $team));
        $this->assertTrue($this->authService->canViewResources($viewer, $team));
    }

    public function test_suspended_member_has_no_permissions(): void
    {
        $owner = User::factory()->create();
        $member = User::factory()->create();
        $team = Team::create(['name' => 'Team', 'owner_id' => $owner->id]);
        $team->addUser($owner, 'owner');
        $team->addUser($member, 'member');
        $team->suspendUser($member);

        $this->assertFalse($this->authService->can($member, $team, 'resources.view'));
    }

    public function test_non_member_has_no_permissions(): void
    {
        $owner = User::factory()->create();
        $outsider = User::factory()->create();
        $team = Team::create(['name' => 'Team', 'owner_id' => $owner->id]);
        $team->addUser($owner, 'owner');

        $this->assertFalse($this->authService->can($outsider, $team, 'resources.view'));
    }
}
```
