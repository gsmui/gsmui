### Plan Archive

Admins can archive plans. Archived plans:
- Are hidden from the pricing page
- Cannot be subscribed to
- Existing subscriptions continue until current period
- Renewals on archived plans are blocked — subscriber is notified to choose a new plan

## Admin Actions

### Grant Entitlement

```php
// app/Livewire/Admin/Subscriptions/GrantEntitlement.php
class GrantEntitlement extends Component
{
    public Subscription $subscription;
    public string $key = '';
    public string $value = 'true';
    public ?string $expiresAt = null;

    public function grant(): void
    {
        $this->validate([
            'key' => 'required|string',
            'value' => 'required|string',
            'expiresAt' => 'nullable|date|after:now',
        ]);

        app(GrantEntitlement::class)->grant(
            admin: auth()->user(),
            billable: $this->subscription->billable,
            key: EntitlementKey::from($this->key),
            value: $this->value,
            expiresAt: $this->expiresAt ? Carbon::parse($this->expiresAt) : null,
        );

        $this->dispatch('entitlement-granted');
        $this->reset(['key', 'value', 'expiresAt']);
    }
}
```

### Manual State Transition

Admins can force subscription state transitions for support scenarios:

| Transition | Use Case |
|---|---|
| Past Due → Active | Payment recovered outside webhook |
| Canceled → Active | Customer requested reinstatement |
| Active → Canceled | Customer request |
| Any → Past Due | Simulate payment failure for testing |
| Expired → Active | Reactivation after payment |

```php
class AdminStateTransition
{
    public function transition(Subscription $subscription, string $newStatus, string $reason): void
    {
        $stateMachine = app(SubscriptionStateMachine::class);

        try {
            $stateMachine->transition($subscription, $newStatus);
        } catch (InvalidTransitionException $e) {
            // Admin override for billing support scenarios
            $subscription->status = $newStatus;
            $subscription->save();
        }

        $subscription->statusHistories()->create([
            'from_status' => $subscription->status,
            'to_status' => $newStatus,
            'changed_by' => auth()->id(),
            'reason' => "[Admin] {$reason}",
        ]);

        app(EntitlementService::class)->synchronize($subscription);

        SubscriptionStateChanged::dispatch($subscription, $newStatus, $reason);
    }
}
```

### Issue Refund

```php
class AdminRefund
{
    public function refund(Subscription $subscription, int $amount = null, string $reason): void
    {
        $provider = app(BillingManager::class)->provider();

        $receipt = $provider->refund($subscription, $amount);

            'from_status' => $subscription->status,

            'changed_by' => auth()->id(),
            'reason' => "[Admin] Refund issued: {$reason}",
        ]);
```

```php
class PlanList extends Component
{
    public string $search = '';
    public string $statusFilter = 'all';
    public string $sortField = 'sort_order';
    public string $sortDirection = 'asc';

    public function getPlansProperty(): LengthAwarePaginator
    {
        $query = Plan::query();

        if ($this->search) {
            $query->where(function ($q) {
                $q->where('name', 'like', "%{$this->search}%")
                  ->orWhere('slug', 'like', "%{$this->search}%");
            });
        }

        if ($this->statusFilter === 'active') {
            $query->where('is_active', true);
        } elseif ($this->statusFilter === 'archived') {
            $query->where('is_active', false);
        }

        return $query->orderBy($this->sortField, $this->sortDirection)
            ->withCount('subscriptions')
            ->paginate(15);
    }

    public function archivePlan(Plan $plan): void
    {
        $this->authorize('archive', $plan);

        $plan->update(['is_active' => false]);

        PlanArchived::dispatch($plan);

        $this->dispatch('plan-archived');
        $this->dispatch('notify', type: 'success', message: 'Plan archived successfully.');
    }

    public function restorePlan(Plan $plan): void
    {
        $this->authorize('update', $plan);

        $plan->update(['is_active' => true]);

        $this->dispatch('plan-restored');
        $this->dispatch('notify', type: 'success', message: 'Plan restored successfully.');
    }

    public function render(): View
    {
        return view('admin.plans.index');
    }
}
```

## Revenue Analytics

### Revenue Analytics Service

```php
class RevenueAnalyticsService
{
    public function getMRR(?Carbon $startDate = null, ?Carbon $endDate = null): array
    {
        $startDate = $startDate ?? now()->startOfMonth()->subMonths(12);
        $endDate = $endDate ?? now();

        $monthlyRevenue = DB::table('subscriptions')
            ->where('status', 'active')
            ->whereBetween('created_at', [$startDate, $endDate])
            ->select(
                DB::raw("DATE_FORMAT(created_at, '%Y-%m') as month"),
                DB::raw('SUM(plan_price) as revenue'),
                DB::raw('COUNT(*) as subscriptions')
            )
            ->groupBy('month')
            ->orderBy('month')
            ->get();

        return [
            'data' => $monthlyRevenue,
            'current_mrr' => $monthlyRevenue->last()->revenue ?? 0,
            'growth_rate' => $this->calculateGrowthRate($monthlyRevenue),
        ];
    }

    public function getChurnRate(int $months = 12): array
    {
        $startDate = now()->subMonths($months);

        $cancellations = DB::table('subscriptions')
            ->where('status', 'canceled')
            ->where('canceled_at', '>=', $startDate)
            ->select(
                DB::raw("DATE_FORMAT(canceled_at, '%Y-%m') as month"),
                DB::raw('COUNT(*) as canceled')
            )
            ->groupBy('month')
            ->get();

        $totalStart = DB::table('subscriptions')
            ->where('created_at', '<', $startDate)
            ->count();

        return [
            'data' => $cancellations,
            'current_churn_rate' => $this->calculateChurnRate($cancellations, $totalStart),
        ];
    }

    public function getTrialConversionRate(int $weeks = 12): array
    {
        $startDate = now()->subWeeks($weeks);

        $trials = DB::table('subscriptions')
            ->where('status', 'trialing')
            ->where('created_at', '>=', $startDate)
            ->select(
                DB::raw("WEEK(created_at) as week"),
                DB::raw('COUNT(*) as total_trials'),
                DB::raw('SUM(CASE WHEN status = "active" THEN 1 ELSE 0 END) as converted')
            )
            ->groupBy('week')
            ->get();

        return [
            'data' => $trials,
            'overall_conversion_rate' => $trials->sum('converted') / max(1, $trials->sum('total_trials')) * 100,
        ];
    }

    public function getPlanDistribution(): array
    {
        return Plan::withCount('subscriptions')
            ->where('is_active', true)
            ->get()
            ->map(fn ($plan) => [
                'name' => $plan->name,
                'count' => $plan->subscriptions_count,
                'revenue' => $plan->price * $plan->subscriptions_count,
            ])
            ->toArray();
    }

    public function getUpgradeDowngradeRate(int $months = 6): array
    {
        $startDate = now()->subMonths($months);

        $changes = DB::table('subscription_status_histories')
            ->where('created_at', '>=', $startDate)
            ->where('reason', 'like', '%plan%')
            ->get();

        return [
            'upgrades' => $changes->where('reason', 'like', '%upgrade%')->count(),
            'downgrades' => $changes->where('reason', 'like', '%downgrade%')->count(),
        ];
    }

    private function calculateGrowthRate($monthlyRevenue): float
    {
        if ($monthlyRevenue->count() < 2) {
            return 0;
        }

        $current = $monthlyRevenue->last()->revenue;
        $previous = $monthlyRevenue->slice(-2, 1)->first()->revenue;

        if ($previous == 0) {
            return 100;
        }

        return round((($current - $previous) / $previous) * 100, 2);
    }

    private function calculateChurnRate($cancellations, $totalStart): float
    {
        if ($totalStart == 0) {
            return 0;
        }

        return round(($cancellations->sum('canceled') / $totalStart) * 100, 2);
    }
}
```

### System Health Service

```php
class SystemHealthService
{
    public function getHealthReport(): array
    {
        return [
            'database' => $this->checkDatabase(),
            'cache' => $this->checkCache(),
            'queue' => $this->checkQueue(),
            'storage' => $this->checkStorage(),
            'billing_provider' => $this->checkBillingProvider(),
            'webhook_health' => $this->checkWebhookHealth(),
            'job_health' => $this->checkJobHealth(),
        ];
    }

    private function checkDatabase(): array
    {
        try {
            DB::connection()->getPdo();
            return ['status' => 'healthy', 'latency_ms' => $this->measureLatency(fn () => DB::select('SELECT 1'))];
        } catch (Exception $e) {
            return ['status' => 'unhealthy', 'error' => $e->getMessage()];
        }
    }

    private function checkCache(): array
    {
        try {
            $start = microtime(true);
            Cache::put('health_check', true, 10);
            $value = Cache::get('health_check');
            $latency = (microtime(true) - $start) * 1000;

            return ['status' => $value ? 'healthy' : 'unhealthy', 'latency_ms' => round($latency, 2)];
        } catch (Exception $e) {
            return ['status' => 'unhealthy', 'error' => $e->getMessage()];
        }
    }

    private function checkQueue(): array
    {
        try {
            $stats = app(JobPerformanceService::class)->getQueueStats();

            $hasIssues = collect($stats)->contains(fn ($s) => $s['status'] === 'critical');

            return [
                'status' => $hasIssues ? 'degraded' : 'healthy',
                'queues' => $stats,
            ];
        } catch (Exception $e) {
            return ['status' => 'unhealthy', 'error' => $e->getMessage()];
        }
    }

    private function checkStorage(): array
    {
        $freeSpace = disk_free_space(storage_path());
        $totalSpace = disk_total_space(storage_path());
        $usagePercent = round((($totalSpace - $freeSpace) / $totalSpace) * 100, 2);

        return [
            'status' => $usagePercent > 90 ? 'critical' : ($usagePercent > 80 ? 'warning' : 'healthy'),
            'free_mb' => round($freeSpace / 1048576),
            'total_mb' => round($totalSpace / 1048576),
            'usage_percent' => $usagePercent,
        ];
    }

    private function checkBillingProvider(): array
    {
        try {
            $provider = app(BillingProvider::class);
            // Simple API check
            return ['status' => 'healthy', 'provider' => config('billing.default')];
        } catch (Exception $e) {
            return ['status' => 'unhealthy', 'error' => $e->getMessage()];
        }
    }

    private function checkWebhookHealth(): array
    {
        $health = app(WebhookHealthService::class)->getHealthMetrics('stripe', 1);

        return [
            'status' => $health['failure_rate'] > 10 ? 'degraded' : 'healthy',
            'metrics' => $health,
        ];
    }

    private function checkJobHealth(): array
    {
        $stats = app(JobPerformanceService::class)->getQueueStats();

        return [
            'status' => collect($stats)->contains(fn ($s) => $s['failed'] > 10) ? 'degraded' : 'healthy',
            'queues' => $stats,
        ];
    }

    private function measureLatency(callable $callback): float
    {
        $start = microtime(true);
        $callback();
        return round((microtime(true) - $start) * 1000, 2);
    }
}
```

```php
            'max_team_members' => 'required|integer|min:1',
            'max_downloads_per_month' => 'required|integer|min:0',
            'features' => 'array',
            'is_active' => 'boolean',
            'is_visible' => 'boolean',
            'sort_order' => 'required|integer|min:0',
            'stripe_price_id' => 'nullable|string',
        ]);

        $oldPrice = $plan->price;
        $plan->update($validated);

        if ($validated['price'] !== $oldPrice) {
            PlanPriceChanged::dispatch($plan, $oldPrice, $validated['price']);
        }

        PlanUpdated::dispatch($plan, $validated);

        return response()->json([
            'data' => new PlanResource($plan),
        ]);
    }

    public function archive(Plan $plan): JsonResponse
    {
        $plan->update(['is_active' => false]);

        PlanArchived::dispatch($plan);

        return response()->json(['message' => 'Plan archived.']);
    }

    public function restore(Plan $plan): JsonResponse
    {
        $plan->update(['is_active' => true]);

        return response()->json(['message' => 'Plan restored.']);
    }
}
```

```php
class AdminSubscriptionApiController extends Controller

    public function index(Request $request): JsonResponse
    {
        $query = Subscription::with(['user', 'plan']);

        if ($request->status) {
            $query->where('status', $request->status);
        }

        if ($request->search) {
            $query->whereHas('user', fn ($q) => $q->where('email', 'like', "%{$request->search}%"));
        }

        $subscriptions = $query->orderByDesc('created_at')->paginate(25);

        return response()->json([
            'data' => SubscriptionResource::collection($subscriptions),
            'meta' => [
                'current_page' => $subscriptions->currentPage(),
                'last_page' => $subscriptions->lastPage(),
                'per_page' => $subscriptions->perPage(),
                'total' => $subscriptions->total(),
            ],
        ]);
    }

    public function show(Subscription $subscription): JsonResponse
    {
        $subscription->load(['user', 'plan', 'entitlements', 'statusHistories']);

        return response()->json([
            'data' => new SubscriptionDetailResource($subscription),
        ]);
    }

    public function update(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'plan_id' => 'sometimes|exists:plans,id',
            'quantity' => 'sometimes|integer|min:1',
        ]);

        $subscription->update($validated);

        return response()->json([
            'data' => new SubscriptionResource($subscription->fresh()),
        ]);
    }

    public function transition(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'status' => 'required|string|in:active,past_due,trialing,canceled,paused,unpaid',
            'reason' => 'required|string|min:3',
        ]);

        app(AdminStateTransition::class)->transition(
            $subscription,
            $validated['status'],
            $validated['reason']
        );

        return response()->json([
            'data' => new SubscriptionResource($subscription->fresh()),
            'message' => 'Subscription status updated.',
        ]);
    }

    public function refund(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'amount' => 'nullable|integer|min:1',
            'reason' => 'required|string|min:3',
        ]);

        app(AdminRefund::class)->refund(
            $subscription,
            $validated['amount'],
            $validated['reason']
        );

        return response()->json(['message' => 'Refund processed.']);
    }

    public function cancel(Request $request, Subscription $subscription): JsonResponse
    {
        $validated = $request->validate([
            'reason' => 'required|string|min:3',
        ]);

        app(AdminStateTransition::class)->transition(
            $subscription,
            'canceled',
            $validated['reason']
        );

        return response()->json(['message' => 'Subscription canceled.']);
```
