# 12.19 — Jobs

**Project:** GsmUi
**Domain:** Subscription Management
**Version:** 1.0.0

## Purpose

Define all scheduled and queued jobs that support subscription operations — renewal processing, trial expiry, grace period management, and entitlement clean-up.

## Job Architecture

```text
Event / Scheduler
    ↓
Job Dispatched (queued or scheduled)
    ↓
Horizon Queue (redis)
    ↓
Worker Processes
    ↓
Side Effects (notifications, state changes, API calls)
```

## Scheduled Jobs

| Job | Schedule | Purpose |

## Job Specifications

### ExpireTrialJob

```php
// app/Jobs/ExpireTrialJob.php
class ExpireTrialJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public function __construct(
        public readonly Subscription $subscription,
    ) {}

    public function handle(TrialService $trialService): void
    {
        if ($this->subscription->status !== SubscriptionStatus::Trialing) {
            return; // Already converted or modified
        }

        if ($this->subscription->trial_ends_at->isFuture()) {
            // Trial not yet ended — re-dispatch
            self::dispatch($this->subscription)
                ->delay($this->subscription->trial_ends_at);
            return;
        }

        try {
            $trialService->convert($this->subscription);
        } catch (PaymentFailedException $e) {
            $this->subscription->status = SubscriptionStatus::PastDue;
            $this->subscription->save();

            app(GracePeriodRetrySchedule::class)->schedule($this->subscription);

            PaymentActionRequired::dispatch($this->subscription, $e);
        }
    }

    public function failed(\Throwable $e): void
    {
        Log::error('Trial expiry failed', [
            'subscription_id' => $this->subscription->id,
            'error' => $e->getMessage(),
        ]);

        // Notify admin
        NotifyAdminJob::dispatch('Trial expiry failed', [
            'subscription' => $this->subscription->id,
            'error' => $e->getMessage(),
        ]);
    }
}
```

```php
        );
    }
```

### RevokeEntitlementsJob

```php
class RevokeEntitlementsJob implements ShouldQueue
{
    public function handle(): void
    {
        // Only revoke if subscription is still in terminal state
        if (!in_array($this->subscription->status, ['expired', 'unpaid', 'canceled'])) {
            return; // Subscription was resumed
        }

        app(EntitlementService::class)->invalidate($this->subscription->billable);

        // Remove pro role if no other active subscription
        if (!$this->subscription->user->subscriptions()
            ->where('status', 'active')
            ->exists()
        ) {
            $this->subscription->user->removeRole('pro_user');
        }

        $this->subscription->user->notify(
            new AccessRevokedNotification($this->subscription)
        );
    }
}
```

```php
class SendRenewalReminderJob implements ShouldQueue
```

### ProcessRenewals Command

```php
class ProcessRenewalsCommand extends Command
{
    protected $signature = 'subscriptions:process-renewals';
    protected $description = 'Process subscriptions due for renewal';

    public function handle(): int
    {
        $subscriptions = Subscription::where('status', 'active')
            ->where('current_period_end', '<=', now())
            ->where('current_period_end', '>', now()->subDay())
            ->with('plan', 'user')
            ->get();

        $this->info("Found {$subscriptions->count()} subscriptions to renew.");

        $processed = 0;
        $failed = 0;

        foreach ($subscriptions as $subscription) {
            try {
                ProcessRenewalJob::dispatch($subscription);
                $processed++;
            } catch (Exception $e) {
                $failed++;
                Log::error("Failed to dispatch renewal for subscription {$subscription->id}", [
                    'error' => $e->getMessage(),
                ]);
            }
        }

        $this->info("Dispatched {$processed} renewals, {$failed} failed.");

        return Command::SUCCESS;
    }
}
```

### ProcessRenewalJob

```php
class ProcessRenewalJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $tries = 3;
    public int $timeout = 120;
    public string $queue = 'renewals';

    public function __construct(
        public readonly Subscription $subscription,
    ) {}

    public function handle(BillingProvider $billing): void
    {
        if ($this->subscription->status !== 'active') {
            return;
        }

        if ($this->subscription->current_period_end->isFuture()) {
            return;
        }

        try {
            $receipt = $billing->charge($this->subscription);

            $this->subscription->update([
                'current_period_start' => $receipt->paid_at,
                'current_period_end' => $receipt->paid_at->copy()->addMonth(),
            ]);

            $this->subscription->statusHistories()->create([
                'from_status' => 'active',
                'to_status' => 'active',
                'reason' => 'Renewal processed',
            ]);

            SubscriptionRenewed::dispatch($this->subscription, $receipt);
        } catch (PaymentFailedException $e) {
            $this->subscription->status = 'past_due';
            $this->subscription->save();

            app(GracePeriodRetrySchedule::class)->schedule($this->subscription);

            PaymentFailed::dispatch($this->subscription, $e->getCode(), $e->getMessage());
        }
    }

    public function failed(\Throwable $e): void
    {
        Log::error('Renewal processing failed', [
            'subscription_id' => $this->subscription->id,
            'error' => $e->getMessage(),
        ]);

        NotifyAdminJob::dispatch('Renewal processing failed', [
            'subscription_id' => $this->subscription->id,
            'error' => $e->getMessage(),
        ]);
    }
}
```

### ProcessScheduledDowngradeJob

```php
class ProcessScheduledDowngradeJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public function __construct(
        public readonly Subscription $subscription,
    ) {}

    public function handle(): void
    {
        if (!$this->subscription->pending_plan_id) {
            return;
        }

        $oldPlan = $this->subscription->plan;
        $newPlan = Plan::find($this->subscription->pending_plan_id);

        if (!$newPlan) {
            Log::warning("Pending plan not found for subscription {$this->subscription->id}");
            return;
        }

        $this->subscription->update([
            'plan_id' => $newPlan->id,
            'pending_plan_id' => null,
            'quantity' => 1,
        ]);

        app(EntitlementService::class)->synchronize($this->subscription);

        $this->subscription->statusHistories()->create([
            'from_status' => $this->subscription->status,
            'to_status' => $this->subscription->status,
            'reason' => "Downgraded from {$oldPlan->name} to {$newPlan->name}",
        ]);

        PlanChanged::dispatch($this->subscription, $oldPlan, $newPlan);

        $this->subscription->user->notify(
            new PlanDowngradedNotification($this->subscription, $oldPlan, $newPlan)
        );
    }
}
```

### SyncProviderStatusesCommand

```php
class SyncProviderStatusesCommand extends Command
{
    protected $signature = 'subscriptions:sync-provider-statuses';
    protected $description = 'Reconcile local subscription states with payment provider';

    public function handle(): int
    {
        $provider = app(BillingProvider::class);

        $subscriptions = Subscription::whereNotNull('stripe_id')
            ->whereIn('status', ['active', 'past_due', 'trialing'])
            ->get();

        $this->info("Syncing {$subscriptions->count()} subscriptions with provider.");

        $synced = 0;
        $discrepancies = 0;

        foreach ($subscriptions as $subscription) {
            try {
                $providerStatus = $provider->getSubscriptionStatus($subscription);

                if ($providerStatus && $providerStatus !== $subscription->status) {
                    $discrepancies++;

                    Log::warning("Subscription status discrepancy", [
                        'subscription_id' => $subscription->id,
                        'local_status' => $subscription->status,
                        'provider_status' => $providerStatus,
                    ]);

                    // Auto-correct based on provider status
                    $subscription->update(['status' => $providerStatus]);
                    $subscription->statusHistories()->create([
                        'from_status' => $subscription->status,
                        'to_status' => $providerStatus,
                        'reason' => 'Auto-synced from provider',
                    ]);
                }

                $synced++;
            } catch (Exception $e) {
                Log::error("Failed to sync subscription {$subscription->id}", [
                    'error' => $e->getMessage(),
                ]);
            }
        }

        $this->info("Synced {$synced} subscriptions, {$discrepancies} discrepancies found.");

        return Command::SUCCESS;
    }
}
```

```php
            ->where('created_at', '<', now()->subDays(30))
            ->delete();
```

```php
            })

            ->limit(100)
            ->get();

            WinBackEmailJob::dispatch($subscription);
```

```php
class JobPerformanceService
{
    public function getQueueStats(): array
    {
        $queues = ['renewals', 'subscriptions', 'entitlements', 'notifications', 'permissions'];

        $stats = [];
        foreach ($queues as $queue) {
            $jobs = DB::table('jobs')->where('queue', $queue)->get();
            $failedJobs = DB::table('failed_jobs')->where('queue', $queue)->get();

            $stats[$queue] = [
                'pending' => $jobs->count(),
                'failed' => $failedJobs->count(),
                'avg_processing_time' => $jobs->avg('reserved_at') ? $this->calculateAvgTime($jobs) : 0,
                'oldest_job' => $jobs->min('created_at'),
            ];
        }

        return $stats;
    }

    public function getFailedJobs(string $queue = null, int $limit = 50): Collection
    {
        $query = DB::table('failed_jobs');

        if ($queue) {
            $query->where('queue', $queue);
        }

        return $query->orderByDesc('failed_at')->limit($limit)->get();
    }

    public function retryFailedJob(int $failedJobId): bool
    {
        $failedJob = DB::table('failed_jobs')->find($failedJobId);

        if (!$failedJob) {
            return false;
        }

        $job = unserialize($failedJob->payload);
        dispatch($job);

        DB::table('failed_jobs')->where('id', $failedJobId)->delete();

        return true;
    }

    private function calculateAvgTime($jobs): float
    {
        return $jobs->filter(fn ($j) => $j->reserved_at)
            ->map(fn ($j) => $j->reserved_at - $j->created_at)
            ->avg();
    }
}
```

```php
class JobAlertService
{
    public function checkAlerts(): void
    {
        $stats = app(JobPerformanceService::class)->getQueueStats();

        foreach ($stats as $queue => $stat) {
            // Alert on high pending count
            if ($stat['pending'] > 100) {
                $this->sendAlert(
                    "job_queue_backlog",
                    "Queue '{$queue}' has {$stat['pending']} pending jobs"
                );
            }

            // Alert on high failure rate
            if ($stat['failed'] > 10) {
                $this->sendAlert(
                    "job_queue_failures",
                    "Queue '{$queue}' has {$stat['failed']} failed jobs"
                );
            }

            // Alert on stuck jobs
            if ($stat['oldest_job'] && $stat['oldest_job'] < now()->subHour()) {
                $this->sendAlert(
                    "job_queue_stuck",
                    "Queue '{$queue}' has jobs older than 1 hour"
                );
            }
        }
    }

    private function sendAlert(string $type, string $message): void
    {
        Log::critical("Job Alert: {$type}", ['message' => $message]);

        NotifyAdminJob::dispatch($type, ['message' => $message]);
    }
}
```

### Job Batches

For parallel operations with a shared completion callback:

```php
class EntitlementSyncBatch
{
    public static function create(Collection $subscriptions): PendingBatch
    {
        $jobs = $subscriptions->map(
            fn ($sub) => new SynchronizeEntitlementsJob($sub)
        );

        return Bus::batch($jobs)
            ->then(function (Batch $batch) {
                Log::info("Entitlement sync batch completed", [
                    'total' => $batch->totalJobs,
                    'failed' => $batch->failedJobs,
                ]);
            })
            ->finally(function (Batch $batch) {
                NotifyAdminJob::dispatch('Entitlement sync batch finished', [
                    'batch_id' => $batch->id,
                    'total' => $batch->totalJobs,
                    'failed' => $batch->failedJobs,
                ]);
            });
    }
}
```

```php
                    "job_queue_failures",
                    "Queue '{$queue}' has {$stat['failed']} failed jobs"
                );
            }

            if ($stat['oldest_job'] && $stat['oldest_job'] < now()->subHour()) {
                $this->sendAlert(
                    "job_queue_stuck",
                    "Queue '{$queue}' has jobs older than 1 hour"
                );
            }
        }
    }

    private function sendAlert(string $type, string $message): void
    {
        Log::critical("Job Alert: {$type}", ['message' => $message]);
        NotifyAdminJob::dispatch($type, ['message' => $message]);
    }
}
```
