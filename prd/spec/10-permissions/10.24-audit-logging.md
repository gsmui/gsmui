# 10.24 - Audit Logging

**Project:** GsmUi
**Domain:** Authorization & RBAC
**Version:** 1.0.0

---

## Purpose

This document defines the **audit logging system** for all authorization events in GsmUi. Every authentication event, permission change, role assignment, access denial, and resource modification is recorded for security, compliance, and debugging. The audit log uses Spatie Activity Log and provides a complete, immutable record of who did what, when, and on which resource.

Audit logs answer four questions:
1. **Who** performed the action?
2. **What** was done?
3. **When** did it happen?
4. **What changed** (old vs. new values)?

```sql
CREATE TABLE activity_log (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    log_name VARCHAR(255) NULL,
    description TEXT NOT NULL,
    subject_type VARCHAR(255) NULL,
    subject_id BIGINT UNSIGNED NULL,
    event VARCHAR(255) NULL,
    causer_type VARCHAR(255) NULL,
    causer_id BIGINT UNSIGNED NULL,

    INDEX activity_log_subject_type_subject_id_index (subject_type, subject_id),

    INDEX activity_log_event_index (event),
```

## Logging Implementation

### Activity Logger Service

```php
// app/Services/AuditLogService.php

namespace App\Services;

use App\Models\User;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Http\Request;
use Spatie\Activitylog\Facades\Activity;
use Spatie\Activitylog\Models\Activity as ActivityModel;

class AuditLogService
{
    protected ?Request $request;

    public function __construct(?Request $request = null)
    {
        $this->request = $request;
    }

    /**
     * Log an authentication event.
     */
    public function logAuth(
        string $event,
        ?User $user = null,
        array $properties = [],
    ): ActivityModel {
        $properties = array_merge($properties, [
            'ip' => $this->request?->ip(),
            'user_agent' => $this->request?->userAgent(),
        ]);

        return $this->log(
            description: "Auth: {$event}",
            subject: $user,
            event: $event,
            properties: $properties,
            logName: 'auth',
        );
    }

    /**
     * Log a failed login attempt.
     */
    public function logLoginFailed(string $email): ActivityModel
    {
        return $this->logAuth('login_failed', null, [
            'email' => $email,
            'reason' => 'invalid_credentials',
        ]);
    }

    /**
     * Log a role change.
     */
    public function logRoleChange(
        string $event,
        User $user,
        string $role,
        ?User $changedBy = null,
    ): ActivityModel {
        return $this->log(
            description: "Role {$event}: {$role}",
            subject: $user,
            event: $event,
            properties: [
                'role' => $role,
                'changed_by' => $changedBy?->id,
            ],
            causer: $changedBy,
            logName: 'authorization',
        );
    }

    /**
     * Log a permission change.
     */
    public function logPermissionChange(
        string $event,
        User $user,
        string $permission,
        ?User $changedBy = null,
    ): ActivityModel {
        return $this->log(
            description: "Permission {$event}: {$permission}",
            subject: $user,
            event: $event,
            properties: [
                'permission' => $permission,
                'changed_by' => $changedBy?->id,
            ],
            causer: $changedBy,
            logName: 'authorization',
        );
    }

    /**
     * Log an access denial.
     */
    public function logAccessDenied(
        string $permission,
        ?User $user = null,
        ?string $resourceType = null,
        ?int $resourceId = null,
    ): ActivityModel {
        $properties = [
            'permission' => $permission,
            'ip' => $this->request?->ip(),
            'user_agent' => $this->request?->userAgent(),
        ];

        if ($resourceType) {
            $properties['resource_type'] = $resourceType;
            $properties['resource_id'] = $resourceId;
        }

        return $this->log(
            description: "Access denied: {$permission}",
            subject: $user,
            event: 'access_denied',
            properties: $properties,
            logName: 'security',
        );
    }

    /**
     * Log a resource event.
     */
    public function logResource(
        string $event,
        Model $resource,
        ?array $oldValues = null,
        ?array $newValues = null,
        ?User $causedBy = null,
    ): ActivityModel {
        $properties = [];

        if ($oldValues) {
            $properties['old'] = $oldValues;
        }
        if ($newValues) {
            $properties['new'] = $newValues;
        }

        $classParts = explode('\\', get_class($resource));
        $shortName = strtolower(end($classParts));

        return $this->log(
            description: "{$shortName}.{$event}",
            subject: $resource,
            event: $event,
            properties: $properties,
            causer: $causedBy,
            logName: 'resource',
        );
    }

    /**
     * Log a system event.
     */
    public function logSystem(
        string $event,
        ?array $properties = null,
        ?User $causedBy = null,
    ): ActivityModel {
        return $this->log(
            description: "System: {$event}",
            event: $event,
            properties: $properties ?? [],
            causer: $causedBy,
            logName: 'system',
        );
    }

    /**
     * Core logging method.
     */
    protected function log(
        string $description,
        ?Model $subject = null,
        ?string $event = null,
        array $properties = [],
        ?User $causer = null,
        string $logName = 'default',
    ): ActivityModel {
        $activity = Activity::log($description);

        if ($subject) {
            $activity->performedOn($subject);
        }

        if ($event) {
            $activity->event($event);
        }

        if (! empty($properties)) {
            $activity->withProperties($properties);
        }

        $causer = $causer ?? auth()->user();

        if ($causer) {
            $activity->causedBy($causer);
        }

        $activity->useLogName($logName);

        return $activity->log($description);
    }
}
```

## Audit Log Retention

| Event Type | Retention | Rationale |
|---|---|---|
| Authentication | 1 year | Security incident investigation |
| Authorization | 2 years | Compliance and access review |
| Resource changes | 1 year | Content disputes and rollback |
| System changes | 3 years | Infrastructure audit |
| Financial (billing) | 7 years | Tax and legal requirements |
| Security (access denied) | 2 years | Threat analysis |

### Retention Implementation

```php
// app/Console/Jobs/CleanupAuditLogs.php

namespace App\Console\Jobs;

use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Support\Facades\DB;
use Carbon\Carbon;

class CleanupAuditLogs implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable;

    public function handle(): void
    {
        $retentions = [
            'auth' => now()->subYear(),
            'authorization' => now()->subYears(2),
            'resource' => now()->subYear(),
            'system' => now()->subYears(3),
            'security' => now()->subYears(2),
            'financial' => now()->subYears(7),
        ];

        foreach ($retentions as $logName => $cutoffDate) {
            $deleted = DB::table('activity_log')
                ->where('log_name', $logName)
                ->where('created_at', '<', $cutoffDate)
                ->delete();

            if ($deleted > 0) {
                \Log::info("Cleaned up {$deleted} audit logs for '{$logName}' older than {$cutoffDate->toDateString()}");
            }
        }
    }
}
```

### Audit Log Service Queries

```php
// app/Services/AuditLogQueryService.php

namespace App\Services;

use App\Models\User;
use Carbon\Carbon;
use Spatie\Activitylog\Models\Activity;

class AuditLogQueryService
{
    /**
     * Get user activity summary.
     */
    public function getUserSummary(User $user, int $days = 30): array
    {
        $cutoff = Carbon::now()->subDays($days);

        return [
            'total_activities' => Activity::causedBy($user)
                ->where('created_at', '>=', $cutoff)
                ->count(),
            'auth_events' => Activity::causedBy($user)
                ->inLog('auth')
                ->where('created_at', '>=', $cutoff)
                ->count(),
            'authorization_events' => Activity::causedBy($user)
                ->inLog('authorization')
                ->where('created_at', '>=', $cutoff)
                ->count(),
            'resource_changes' => Activity::causedBy($user)
                ->inLog('resource')
                ->where('created_at', '>=', $cutoff)
                ->count(),
            'access_denials' => Activity::causedBy($user)
                ->where('event', 'access_denied')
                ->where('created_at', '>=', $cutoff)
                ->count(),
            'last_activity' => Activity::causedBy($user)
                ->latest()
                ->first()?->created_at,
        ];
    }

    /**
     * Get resource change history.
     */
    public function getResourceHistory($resource, int $limit = 50): array
    {
        return Activity::performedOn($resource)
            ->with('causer')
            ->orderByDesc('created_at')
            ->limit($limit)
            ->get()
            ->map(fn ($activity) => [
                'event' => $activity->event,
                'description' => $activity->description,
                'causer' => $activity->causer?->name ?? 'System',
                'properties' => $activity->properties->toArray(),
                'created_at' => $activity->created_at->toIso8601String(),
            ])
            ->toArray();
    }

    /**
     * Get security events (failed access, suspicious activity).
     */
    public function getSecurityEvents(int $hours = 24): array
    {
        $cutoff = Carbon::now()->subHours($hours);

        return [
            'failed_logins' => Activity::where('event', 'login_failed')
                ->where('created_at', '>=', $cutoff)
                ->orderByDesc('created_at')
                ->get(),
            'access_denials' => Activity::where('event', 'access_denied')
                ->where('created_at', '>=', $cutoff)
                ->orderByDesc('created_at')
                ->get(),
            'permission_changes' => Activity::inLog('authorization')
                ->whereIn('event', ['role_assigned', 'role_removed', 'permission_granted', 'permission_revoked'])
                ->where('created_at', '>=', $cutoff)
                ->orderByDesc('created_at')
                ->get(),
        ];
    }

    /**
     * Get compliance report for a date range.
     */
    public function getComplianceReport(
        Carbon $start,
        Carbon $end,
    ): array {
        return [
            'period' => [
                'start' => $start->toDateString(),
                'end' => $end->toDateString(),
            ],
            'summary' => [
                'total_events' => Activity::whereBetween('created_at', [$start, $end])->count(),
                'auth_events' => Activity::inLog('auth')->whereBetween('created_at', [$start, $end])->count(),
                'authorization_events' => Activity::inLog('authorization')->whereBetween('created_at', [$start, $end])->count(),
                'resource_events' => Activity::inLog('resource')->whereBetween('created_at', [$start, $end])->count(),
                'system_events' => Activity::inLog('system')->whereBetween('created_at', [$start, $end])->count(),
                'security_events' => Activity::inLog('security')->whereBetween('created_at', [$start, $end])->count(),
            ],
            'top_users' => Activity::whereBetween('created_at', [$start, $end])
                ->select('causer_id', DB::raw('count(*) as activity_count'))
                ->groupBy('causer_id')
                ->orderByDesc('activity_count')
                ->limit(10)
                ->get()
                ->map(fn ($item) => [
                    'user_id' => $item->causer_id,
                    'activity_count' => $item->activity_count,
                    'user_name' => User::find($item->causer_id)?->name ?? 'Unknown',
                ]),
            'access_denials' => Activity::where('event', 'access_denied')
                ->whereBetween('created_at', [$start, $end])
                ->count(),
        ];
    }
}
```

## Compliance Requirements

### Requirements Summary

| Requirement | Description | Implementation |
|---|---|---|
| Financial logging | All billing transactions logged | `billing.*` events |
| Admin action logging | All admin actions logged | `causedBy` tracking |
| Access denial logging | All 403s logged | `access_denied` events |
| Immutability | Logs cannot be modified or deleted | Read-only in production |
| Searchability | Logs must be searchable | Spatie query builder |
| Retention | Logs retained per policy | Scheduled cleanup job |
| Export | Logs exportable for compliance | CSV/JSON export endpoint |

### Immutability Enforcement

```php
// app/Providers/AuditLogServiceProvider.php

namespace App\Providers;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\DB;

class AuditLogServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        // In production, prevent updates/deletes on activity_log
        if ($this->app->environment('production')) {
            DB::listen(function ($query) {
                $sql = strtolower($query->sql);

                if (str_contains($sql, 'update activity_log') ||
                    str_contains($sql, 'delete from activity_log')) {
                    \Log::critical('Attempted to modify audit log in production', [
                        'sql' => $query->sql,
                        'user_id' => auth()->id(),
                    ]);

                    throw new \RuntimeException(
                        'Audit logs cannot be modified in production'
                    );
                }
            });
        }
    }
}
```

### Export Endpoint

```php
// app/Http/Controllers/Admin/AuditLogExportController.php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Services\AuditLogQueryService;
use Carbon\Carbon;
use Illuminate\Http\Request;
use Illuminate\Http\StreamedResponse;

class AuditLogExportController extends Controller
{
    public function export(
        Request $request,
        AuditLogQueryService $queryService,
    ): StreamedResponse {
        $this->authorize('system.logs');

        $start = Carbon::parse($request->input('start', now()->subDays(30)));
        $end = Carbon::parse($request->input('end', now()));
        $format = $request->input('format', 'csv');

        $activities = \Spatie\Activitylog\Models\Activity::whereBetween('created_at', [$start, $end])
            ->with('causer', 'subject')
            ->orderByDesc('created_at')
            ->cursor();

        $filename = "audit-log-{$start->toDateString()}-to-{$end->toDateString()}.{$format}";

        return response()->streamDownload(function () use ($activities, $format) {
            if ($format === 'csv') {
                $handle = fopen('php://output', 'w');

                fputcsv($handle, [
                    'ID',
                    'Log Name',
                    'Event',
                    'Description',
                    'Causer ID',
                    'Causer Name',
                    'Subject Type',
                    'Subject ID',
                    'Properties',
                    'Created At',
                ]);

                foreach ($activities as $activity) {
                    fputcsv($handle, [
                        $activity->id,
                        $activity->log_name,
                        $activity->event,
                        $activity->description,
                        $activity->causer_id,
                        $activity->causer?->name ?? 'System',
                        $activity->subject_type,
                        $activity->subject_id,
                        json_encode($activity->properties->toArray()),
                        $activity->created_at->toIso8601String(),
                    ]);
                }

                fclose($handle);
            }
        }, $filename, [
            'Content-Type' => $format === 'csv' ? 'text/csv' : 'application/json',
        ]);
    }
}
```

```php
        $token = $request->bearerToken();

        if (! $token) {

        }

        $hashedToken = hash('sha256', $token);

        $apiToken = \App\Models\ApiToken::where('token_hash', $hashedToken)->first();

        return $apiToken?->id ? (string) $apiToken->id : null;
    }
```

### Retention Management Service

```php
// app/Services/AuditRetentionService.php

namespace App\Services;

use App\Models\AuditLogEntry;
use App\Models\LegalHold;
use App\Models\Organization;
use Carbon\Carbon;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;

class AuditRetentionService
{
    /**
     * Archive old audit logs based on retention policy.
     */
    public function archiveOldLogs(?Organization $organization = null): array
    {
        $retentionConfig = $this->getRetentionConfig($organization);
        $results = ['archived' => 0, 'deleted' => 0, 'errors' => []];

        foreach ($retentionConfig as $logName => $config) {
            try {
                $cutoffDate = Carbon::now()->subDays($config['hot_storage']);

                // Export to archive before deletion
                $archivedCount = $this->archiveLogs($logName, $cutoffDate, $organization);
                $results['archived'] += $archivedCount;

                // Delete archived logs from hot storage
                $deleted = AuditLogEntry::where('log_name', $logName)
                    ->where('created_at', '<', $cutoffDate)
                    ->when($organization, fn ($q) => $q->where('organization_id', $organization->id))
                    ->whereNotIn('id', $this->getLegalHoldIds($organization))
                    ->delete();

                $results['deleted'] += $deleted;
            } catch (\Exception $e) {
                $results['errors'][] = [
                    'log_name' => $logName,
                    'error' => $e->getMessage(),
                ];
            }
        }

        return $results;
    }

    /**
     * Export logs to cold storage archive.
     */
    protected function archiveLogs(
        string $logName,
        Carbon $cutoffDate,
        ?Organization $organization,
    ): int {
        $logs = AuditLogEntry::where('log_name', $logName)
            ->where('created_at', '<', $cutoffDate)
            ->when($organization, fn ($q) => $q->where('organization_id', $organization->id))
            ->orderBy('created_at')
            ->cursor();

        $buffer = [];
        $batchNumber = 0;

        foreach ($logs as $log) {
            $buffer[] = $log->toArray();

            if (count($buffer) >= 10000) {
                $this->writeArchiveBatch($logName, $batchNumber, $buffer, $organization);
                $buffer = [];
                $batchNumber++;
            }
        }

        if (! empty($buffer)) {
            $this->writeArchiveBatch($logName, $batchNumber, $buffer, $organization);
        }

        return $batchNumber * 10000 + count($buffer);
    }

    protected function writeArchiveBatch(
        string $logName,
        int $batchNumber,
        array $data,
        ?Organization $organization,
    ): void {
        $date = now()->format('Y/m/d');
        $orgPrefix = $organization ? "org/{$organization->id}/" : '';
        $path = config('audit-retention.archive.path')
            . "/{$orgPrefix}{$logName}/{$date}/batch_{$batchNumber}.json.gz";

        $json = json_encode($data);
        $compressed = gzencode($json, 9);

        $disk = Storage::disk(config('audit-retention.archive.disk'));
        $disk->put($path, $compressed);
    }

    /**
     * Apply legal hold to specific logs.
     */
    public function applyLegalHold(
        Organization $organization,
        string $reason,
        ?string $criteria = null,
        ?Carbon $expiresAt = null,
    ): LegalHold {
        return LegalHold::create([
            'organization_id' => $organization->id,
            'reason' => $reason,
            'criteria' => $criteria,
            'expires_at' => $expiresAt,
            'applied_by' => auth()->id(),
        ]);
    }

    /**
     * Check if a log entry is under legal hold.
     */
    protected function getLegalHoldIds(?Organization $organization): array
    {
        if (! config('audit-retention.legal_hold_enabled')) {
            return [];
        }

        $holds = LegalHold::where(function ($q) use ($organization) {
            $q->whereNull('expires_at')
                ->orWhere('expires_at', '>', now());
        });

        if ($organization) {
            $holds->where('organization_id', $organization->id);
        }

        // In practice, legal holds would mark specific log entry IDs
        // For simplicity, we return empty — actual implementation depends on criteria
        return [];
    }

    protected function getRetentionConfig(?Organization $organization): array
    {
        $defaults = config('audit-retention.default_retention');

        if ($organization && isset($organization->settings['audit_retention'])) {
            return array_merge($defaults, $organization->settings['audit_retention']);
        }

        return $defaults;
    }
}
```

```php
namespace App\Services;

use App\Models\User;

class GdprAuditService

    /**
     * Handle right to erasure (Article 17) for audit logs.
     * Personal data in audit logs must be anonymized or deleted.
     */
    public function handleErasureRequest(User $user, ?string $reason = null): array
    {
        $results = ['anonymized' => 0, 'deleted' => 0, 'retained' => 0];

        // Anonymize personal data in logs where possible
        $anonymized = DB::table('activity_log')
            ->where('causer_id', $user->id)
            ->update([
                'causer_id' => null,
                'causer_type' => null,
                'properties' => DB::raw("JSON_SET(COALESCE(properties, '{}'), '$.anonymized', true)"),
            ]);

        $results['anonymized'] = $anonymized;

        // Delete logs that have no retention requirement
        $deletable = DB::table('activity_log')
            ->where('subject_id', $user->id)
            ->where('subject_type', User::class)
            ->whereNotIn('log_name', ['financial', 'security'])
            ->delete();

        $results['deleted'] = $deletable;

        // Count logs retained for legal compliance
        $results['retained'] = DB::table('activity_log')
            ->where(function ($q) use ($user) {
                $q->where('causer_id', $user->id)
                    ->orWhere(function ($q2) use ($user) {
                        $q2->where('subject_id', $user->id)
                            ->where('subject_type', User::class);
                    });
            })
            ->where('log_name', 'financial')
            ->count();

        // Log the erasure action
        activity()
            ->withProperties([
                'erased_user_id' => $user->id,
                'reason' => $reason,
                'results' => $results,
            ])
            ->event('gdpr_erasure_executed')
            ->log('GDPR erasure request executed');

        return $results;
    }

    /**
     * Generate data portability export (Article 20).
     */
    public function handlePortabilityRequest(User $user): array
    {
        $activities = DB::table('activity_log')
            ->where('causer_id', $user->id)
            ->orderBy('created_at', 'desc')
            ->get()
            ->map(fn ($activity) => [
                'event' => $activity->event,
                'description' => $activity->description,
                'subject_type' => $activity->subject_type,
                'subject_id' => $activity->subject_id,
                'properties' => json_decode($activity->properties, true),
                'created_at' => $activity->created_at,
            ])
```

```php
            ->where('created_at', '>=', $cutoff)
            ->whereJsonContains('properties->unusual_country', true)
            ->get();

        // Brute force attempts
        $bruteForce = DB::table('activity_log')
            ->where('event', 'login_failed')
            ->where('created_at', '>=', $cutoff)
            ->select('ip_address', DB::raw('COUNT(*) as attempts'))
            ->groupBy('ip_address')
            ->having('attempts', '>', 10)
            ->get();

        return [

            ->select(
                DB::raw('DAYOFWEEK(created_at) as day'),
                DB::raw('HOUR(created_at) as hour'),
                DB::raw('COUNT(*) as count')
            )
            ->groupBy('day', 'hour')
            ->get();

        $heatmap = [];
        for ($day = 1; $day <= 7; $day++) {
            for ($hour = 0; $hour < 24; $hour++) {
                $record = $activities->first(fn ($a) => $a->day === $day && $a->hour === $hour);
                $heatmap[$day][$hour] = $record?->count ?? 0;
            }
        }

        return $heatmap;
```

### Alert Evaluation Service

```php
// app/Services/AuditAlertService.php

namespace App\Services;

use App\Models\AuditAlertRule;
use App\Models\Organization;
use Carbon\Carbon;

class AuditAlertService
{
    public function evaluateRules(?Organization $organization = null): array
    {
        $rules = AuditAlertRule::where('is_active', true)
            ->when($organization, fn ($q) => $q->where('organization_id', $organization->id))
            ->get();

        $triggered = [];

        foreach ($rules as $rule) {
            if ($this->isInCooldown($rule)) {
                continue;
            }

            $result = $this->evaluateRule($rule);

            if ($result['triggered']) {
                $this->sendNotifications($rule, $result);
                $rule->update(['last_triggered_at' => now()]);

                $triggered[] = [
                    'rule_id' => $rule->id,
                    'rule_name' => $rule->name,
                    'result' => $result,
                ];
            }
        }

        return $triggered;
    }

    protected function evaluateRule(AuditAlertRule $rule): array
    {
        return match ($rule->rule_type) {
            'threshold' => $this->evaluateThreshold($rule),
            'anomaly' => $this->evaluateAnomaly($rule),
            'pattern' => $this->evaluatePattern($rule),
            default => ['triggered' => false],
        };
    }

    protected function evaluateThreshold(AuditAlertRule $rule): array
    {
        $config = $rule->rule_config;
        $windowMinutes = $config['window_minutes'] ?? 60;
        $threshold = $config['threshold'] ?? 10;
        $event = $config['event'] ?? null;

        $cutoff = Carbon::now()->subMinutes($windowMinutes);

        $query = \Spatie\Activitylog\Models\Activity::where('created_at', '>=', $cutoff);

        if ($event) {
            $query->where('event', $event);
        }

        if ($config['log_name'] ?? null) {
            $query->where('log_name', $config['log_name']);
        }

        $count = $query->count();

        return [
            'triggered' => $count >= $threshold,
            'count' => $count,
            'threshold' => $threshold,
            'window_minutes' => $windowMinutes,
        ];
    }

    protected function evaluateAnomaly(AuditAlertRule $rule): array
    {
        $config = $rule->rule_config;
        $metric = $config['metric'] ?? 'failed_logins';

        $analyzer = app(AuditLogAnalyzer::class);

        $anomalies = match ($metric) {
            'failed_logins' => $analyzer->detectAnomalousLogins(1),
            'multi_ip' => $analyzer->detectAnomalousLogins(1)['multi_ip_logins']->count(),
            default => 0,
        };

        $threshold = $config['threshold'] ?? 5;

        return [
            'triggered' => is_array($anomalies)
                ? ($anomalies['total_anomalies'] ?? 0) >= $threshold
                : $anomalies >= $threshold,
            'anomalies' => $anomalies,
            'threshold' => $threshold,
        ];
    }

    protected function evaluatePattern(AuditAlertRule $rule): array
    {
        $config = $rule->rule_config;
        $pattern = $config['pattern'] ?? '';
        $windowMinutes = $config['window_minutes'] ?? 60;

        $cutoff = Carbon::now()->subMinutes($windowMinutes);

        $count = \Spatie\Activitylog\Models\Activity::where('created_at', '>=', $cutoff)
            ->where('event', 'like', $pattern)
            ->count();

        $threshold = $config['threshold'] ?? 1;

        return [
            'triggered' => $count >= $threshold,
            'count' => $count,
            'threshold' => $threshold,
        ];
    }

    protected function isInCooldown(AuditAlertRule $rule): bool
    {
        if (! $rule->last_triggered_at) {
            return false;
        }

        return $rule->last_triggered_at->addMinutes($rule->cooldown_minutes)->isFuture();
    }

    protected function sendNotifications(AuditAlertRule $rule, array $result): void
    {
        $channels = $rule->notification_channels ?? ['email'];

        $data = [
            'rule_name' => $rule->name,
            'description' => $rule->description,
            'result' => $result,
            'triggered_at' => now()->toIso8601String(),
            'organization_id' => $rule->organization_id,
        ];

        foreach ($channels as $channel) {
            match ($channel) {
                'email' => $this->sendEmailNotification($rule, $data),
                'slack' => $this->sendSlackNotification($rule, $data),
                'webhook' => $this->sendWebhookNotification($rule, $data),
            };
        }
    }

    protected function sendEmailNotification(AuditAlertRule $rule, array $data): void
    {
        $recipients = $rule->notification_config['email_recipients'] ?? [];

        foreach ($recipients as $email) {
            \Mail::to($email)->send(new \App\Mail\AuditAlertMail($data));
        }
    }

    protected function sendSlackNotification(AuditAlertRule $rule, array $data): void
    {
        $webhookUrl = $rule->notification_config['slack_webhook_url'] ?? null;

        if ($webhookUrl) {
            \Http::post($webhookUrl, [
                'text' => "Audit Alert: {$rule->name}\n" . json_encode($data['result'], JSON_PRETTY_PRINT),
            ]);
        }
    }

    protected function sendWebhookNotification(AuditAlertRule $rule, array $data): void
    {
        $url = $rule->notification_config['webhook_url'] ?? null;

        if ($url) {
            \Http::withHeaders([
                'X-Audit-Alert-Signature' => hash_hmac('sha256', json_encode($data), $rule->notification_config['webhook_secret'] ?? ''),
            ])->post($url, $data);
        }
    }
}
```

## Tamper-Proof Logging

Tamper-proof logging ensures that audit log entries cannot be modified, deleted, or corrupted after creation. This uses cryptographic hash chaining similar to blockchain.

### Hash Chain Implementation

```php
// app/Services/TamperProofLogService.php

namespace App\Services;

use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;

class TamperProofLogService
{
    /**
     * Append an entry with a hash chain.
     */
    public function appendWithChain(
        int $activityId,
        int $organizationId,
        array $data,
    ): array {
        $lastEntry = DB::table('tamper_proof_audit_logs')
            ->where('organization_id', $organizationId)
            ->orderByDesc('sequence_number')
            ->first();

        $sequenceNumber = $lastEntry ? $lastEntry->sequence_number + 1 : 1;
        $previousHash = $lastEntry?->current_hash ?? '0';

        $blockContent = json_encode([
            'previous_hash' => $previousHash,
            'activity_id' => $activityId,
            'sequence' => $sequenceNumber,
            'data' => $data,
            'timestamp' => now()->toIso8601String(),
        ]);

        $currentHash = hash('sha256', $blockContent);

        DB::table('tamper_proof_audit_logs')->insert([
            'audit_entry_id' => $activityId,
            'organization_id' => $organizationId,
            'previous_hash' => $previousHash,
            'current_hash' => $currentHash,
            'checksum' => hash('crc32b', $currentHash),
            'sequence_number' => $sequenceNumber,
            'block_data' => json_encode($data),
            'created_at' => now(),
            'updated_at' => now(),
        ]);

        return [
            'sequence' => $sequenceNumber,
            'hash' => $currentHash,
        ];
    }

    /**
     * Verify the entire chain integrity for an organization.
     */
    public function verifyChain(int $organizationId): array
    {
        $entries = DB::table('tamper_proof_audit_logs')
            ->where('organization_id', $organizationId)
            ->orderBy('sequence_number')
            ->get();

        $issues = [];
        $previousHash = null;

        foreach ($entries as $entry) {
            // Check chain continuity
            if ($previousHash !== null && $entry->previous_hash !== $previousHash) {
                $issues[] = [
                    'type' => 'chain_break',
                    'sequence' => $entry->sequence_number,
                    'expected_previous' => $previousHash,
                    'actual_previous' => $entry->previous_hash,
                ];
            }

            // Verify hash integrity
            $blockContent = json_encode([
                'previous_hash' => $entry->previous_hash,
                'activity_id' => $entry->audit_entry_id,
                'sequence' => $entry->sequence_number,
                'data' => json_decode($entry->block_data, true),
                'timestamp' => $entry->created_at,
            ]);

            $expectedHash = hash('sha256', $blockContent);

            if (! hash_equals($expectedHash, $entry->current_hash)) {
                $issues[] = [
                    'type' => 'hash_mismatch',
                    'sequence' => $entry->sequence_number,
                    'expected' => $expectedHash,
                    'actual' => $entry->current_hash,
                ];
            }

            $previousHash = $entry->current_hash;
        }

        return [
            'organization_id' => $organizationId,
            'total_entries' => $entries->count(),
            'is_valid' => empty($issues),
            'issues' => $issues,
            'verified_at' => now()->toIso8601String(),
        ];
    }
}
```

```php
                    $activity->description,
                    $activity->causer_id,
                    $this->getCauserName($activity->causer_id),
                    $activity->subject_type,
                    $activity->subject_id,
                    $activity->ip_address ?? '',
                    $activity->user_agent ?? '',
                    $activity->severity ?? 'info',
                    $activity->category ?? '',
                    $activity->properties ?? '{}',
                    $activity->created_at,
                ]);
            }

            fclose($handle);
        }, $filename, [
            'Content-Type' => 'text/csv',
            'Content-Disposition' => "attachment; filename=\"{$filename}\"",
        ]);
    }

    /**
     * Export audit logs in JSON format.
     */
    public function exportJson(
        Carbon $start,
        Carbon $end,
        ?int $organizationId = null,
        ?string $logName = null,
    ): StreamedResponse {
        $filename = "audit-logs-{$start->toDateString()}-{$end->toDateString()}.json";

        return response()->streamDownload(function () use ($start, $end, $organizationId, $logName) {
            $query = DB::table('activity_log')
                ->whereBetween('created_at', [$start, $end])
                ->orderBy('created_at');

            if ($organizationId) {
                $query->where('organization_id', $organizationId);
            }

            if ($logName) {
                $query->where('log_name', $logName);
            }

            echo "[\n";
            $first = true;

            foreach ($query->cursor() as $activity) {
                if (! $first) {
                    echo ",\n";
                }
                $first = false;

                echo json_encode([
                    'id' => $activity->id,
                    'log_name' => $activity->log_name,
                    'event' => $activity->event,
                    'description' => $activity->description,
                    'causer_id' => $activity->causer_id,
                    'subject_type' => $activity->subject_type,
                    'subject_id' => $activity->subject_id,
                    'ip_address' => $activity->ip_address,
                    'user_agent' => $activity->user_agent,
                    'severity' => $activity->severity,
                    'category' => $activity->category,
                    'properties' => json_decode($activity->properties, true),
                    'created_at' => $activity->created_at,
                ], JSON_PRETTY_PRINT);
            }

            echo "\n]";
        }, $filename, [
            'Content-Type' => 'application/json',
            'Content-Disposition' => "attachment; filename=\"{$filename}\"",
        ]);
    }

    /**
     * Export audit logs in SIEM-compatible format (CEF).
     */
    public function exportCef(
        Carbon $start,
        Carbon $end,
        ?int $organizationId = null,
    ): StreamedResponse {
        $filename = "audit-logs-{$start->toDateString()}-{$end->toDateString()}.cef";

        return response()->streamDownload(function () use ($start, $end, $organizationId) {
            $query = DB::table('activity_log')
                ->whereBetween('created_at', [$start, $end])
                ->orderBy('created_at');

            if ($organizationId) {
                $query->where('organization_id', $organizationId);
            }

            foreach ($query->cursor() as $activity) {
                $severity = match ($activity->severity ?? 'info') {
                    'critical' => 10,
                    'high' => 7,
                    'medium' => 5,
                    'low' => 3,
                    default => 1,
                };

                $timestamp = Carbon::parse($activity->created_at)->timestamp;

                echo sprintf(
                    "CEF:0|GsmUi|AuditLog|1.0|%s|%s|%d|src=%s suser=%s spt=%s\n",
                    $activity->event ?? 'unknown',
                    $activity->description ?? '',
                    $severity,
                    $activity->ip_address ?? '0.0.0.0',
                    $this->getCauserName($activity->causer_id),
                    $activity->created_at,
                );
            }
        }, $filename, [
            'Content-Type' => 'text/plain',
        ]);
    }

    /**
     * Export audit logs for Splunk (key=value format).
     */
    public function exportSplunk(
        Carbon $start,
        Carbon $end,
        ?int $organizationId = null,
    ): StreamedResponse {
        $filename = "audit-logs-{$start->toDateString()}-{$end->toDateString()}.splunk";

        return response()->streamDownload(function () use ($start, $end, $organizationId) {
            $query = DB::table('activity_log')
                ->whereBetween('created_at', [$start, $end])
                ->orderBy('created_at');

            if ($organizationId) {
                $query->where('organization_id', $organizationId);
            }

            foreach ($query->cursor() as $activity) {
                $line = sprintf(
                    'timestamp=%s event=%s log_name=%s description="%s" causer_id=%s ip=%s severity=%s',
                    $activity->created_at,
                    $activity->event ?? 'unknown',
                    $activity->log_name ?? '',
                    addslashes($activity->description ?? ''),
                    $activity->causer_id ?? '',
                    $activity->ip_address ?? '',
                    $activity->severity ?? 'info',
                );

                echo $line . "\n";
            }
        }, $filename, [
            'Content-Type' => 'text/plain',
        ]);
    }

    protected function getCauserName(?int $causerId): string
    {
        if (! $causerId) {
            return 'system';
        }

        return cache()->remember("user_name:{$causerId}", 3600, function () use ($causerId) {
            return DB::table('users')->where('id', $causerId)->value('name') ?? 'unknown';
        });
    }
}
```

```php
// tests/Feature/AuditLogging/ExtendedAuditLogTest.php

use App\Models\User;
```
