# 10.10 - Marketplace Permissions

**Project:** GsmUi
**Domain:** Authorization & RBAC
**Version:** 1.0.0

---

## Purpose

Define the **complete authorization rules for the marketplace domain**: browsing, purchasing, selling, refunds, listing management, and payouts. The marketplace is the revenue engine of GsmUi — every transaction must be authorized correctly to protect buyers, sellers, and the platform.

This document covers the full lifecycle of a marketplace transaction from browse to refund, including all permission checks, ownership validations, and business rule enforcement.

## Permission Definitions

## Purchase Authorization Flow

```text
User clicks "Buy" on listing
    ↓
Gate check: marketplace.purchase
    ↓
Verify user has payment method
    ↓
Check for duplicate purchase
    ↓
Calculate total (base price - coupon + tax)
    ↓
Create Order (status: pending)
    ↓
Reserve inventory / license slot
    ↓
Process payment via billing provider
    ↓
  ┌─── Payment Success ──→ Order status: completed
  │                           ↓
  │                       Issue License to buyer
  │                           ↓
  │                       Notify creator of sale
  │                           ↓
  │                       Update analytics
  │                           ↓
  │                       Update creator earnings
  │
  └─── Payment Failed ───→ Order status: failed
                              ↓
                          Notify buyer of failure
                              ↓
                          Release inventory / slot
```

### Implementation

```php
// app/Services/Marketplace/PurchaseService.php

namespace App\Services\Marketplace;

use App\Events\OrderCreated;
use App\Events\OrderCompleted;
use App\Events\PaymentFailed;
use App\Exceptions\DuplicatePurchaseException;
use App\Exceptions\PaymentRequiredException;
use App\Models\MarketplaceListing;
use App\Models\Order;
use App\Models\User;
use Illuminate\Auth\Access\AuthorizationException;

class PurchaseService
{
    public function __construct(
        protected PaymentService $paymentService,
        protected LicenseService $licenseService,
        protected NotificationService $notificationService,
        protected AnalyticsService $analyticsService,
    ) {}

    public function purchase(
        User $buyer,
        MarketplaceListing $listing,
        ?string $couponCode = null
    ): Order {
        // 1. Permission check
        if (! $buyer->can('marketplace.purchase')) {
            throw new AuthorizationException(
                'You do not have permission to make purchases.'
            );
        }

        // 2. Payment method check
        if (! $buyer->hasPaymentMethod() && ! $buyer->hasRole(['admin', 'super_admin'])) {
            throw new PaymentRequiredException(
                'Please add a payment method before purchasing.'
            );
        }

        // 3. Duplicate purchase check
        if ($this->hasPurchased($buyer, $listing)) {
            throw new DuplicatePurchaseException(
                'You have already purchased this item.'
            );
        }

        // 4. Listing availability check
        if ($listing->status !== 'published') {
            throw new \Exception('This listing is no longer available.');
        }

        // 5. Calculate total
        $total = $this->calculateTotal($listing, $couponCode, $buyer);

        // 6. Create order
        $order = Order::create([
            'user_id' => $buyer->id,
            'listing_id' => $listing->id,
            'amount' => $total['amount'],
            'discount' => $total['discount'],
            'tax' => $total['tax'],
            'coupon_code' => $couponCode,
            'status' => 'pending',
        ]);

        event(new OrderCreated($order));

        // 7. Process payment
        try {
            $this->paymentService->charge($buyer, $order);
        } catch (\Exception $e) {
            $order->update(['status' => 'failed']);
            event(new PaymentFailed($order, $e));
            throw $e;
        }

        // 8. Complete order
        $order->update(['status' => 'completed']);

        // 9. Issue license
        $this->licenseService->issue($buyer, $listing, $order);

        // 10. Notify creator
        $this->notificationService->notifyCreatorOfSale($listing->user, $order);

        // 11. Update analytics
        $this->analyticsService->recordSale($listing, $order);

        // 12. Update creator earnings
        $this->updateCreatorEarnings($listing->user, $order);

        event(new OrderCompleted($order));

        return $order;
    }

    public function gift(
        User $giver,
        User $recipient,
        MarketplaceListing $listing
    ): Order {
        if (! $giver->can('marketplace.gift')) {
            throw new AuthorizationException(
                'You do not have permission to gift purchases.'
            );
        }

        // Same flow as purchase but license issued to recipient
        $order = $this->purchase($giver, $listing);
        $this->licenseService->issue($recipient, $listing, $order);

        return $order;
    }

    protected function hasPurchased(User $buyer, MarketplaceListing $listing): bool
    {
        return Order::where('user_id', $buyer->id)
            ->where('listing_id', $listing->id)
            ->where('status', 'completed')
            ->exists();
    }

    protected function calculateTotal(
        MarketplaceListing $listing,
        ?string $couponCode,
        User $buyer
    ): array {
        $amount = $listing->price;
        $discount = 0;
        $tax = 0;

        if ($couponCode) {
            $discount = $this->applyCoupon($couponCode, $amount);
        }

        $taxableAmount = $amount - $discount;
        $tax = $this->calculateTax($taxableAmount, $buyer);

        return [
            'amount' => $amount,
            'discount' => $discount,
            'tax' => $tax,
        ];
    }

    protected function updateCreatorEarnings(User $creator, Order $order): void
    {
        $profile = $creator->creatorProfile;
        $commission = $profile->commission_rate ?? 30;
        $earnings = $order->amount * (1 - $commission / 100);

        $profile->increment('available_balance', $earnings);
        $profile->increment('total_earnings', $earnings);

        CreatorEarning::create([
            'user_id' => $creator->id,
            'order_id' => $order->id,
            'amount' => $earnings,
            'commission_rate' => $commission,
            'status' => 'available',
        ]);
    }
}
```

## Refund Authorization Flow

```text
Refund request received
    ↓
Gate check: marketplace.refund (Support or Admin only)
    ↓
Verify order exists and is refundable
    ↓
Check refund policy window (30 days standard)
    ↓
  ┌─── Within 30 days ──→ Admin can process directly
  │
  └─── Over 30 days ────→ Requires Super Admin approval
    ↓
Calculate refund amount (full or partial)
    ↓
Process refund via billing provider
    ↓
Revoke license from buyer
    ↓
Notify buyer of refund
    ↓
Notify creator of refund
    ↓
Update creator earnings (deduct)
    ↓
Update analytics
    ↓
Log audit event
```

### Implementation

```php
// app/Services/Marketplace/RefundService.php

namespace App\Services\Marketplace;

use App\Events\RefundProcessed;
use App\Models\Order;
use App\Models\Refund;
use App\Models\User;
use Illuminate\Auth\Access\AuthorizationException;

class RefundService
{
    public function processRefund(
        Order $order,
        User $admin,
        float $amount,
        ?string $reason = null
    ): Refund {
        // 1. Permission check
        if (! $admin->can('marketplace.refund')) {
            throw new AuthorizationException(
                'Insufficient permissions to process refunds.'
            );
        }

        // 2. Refundability check
        if (! $order->isRefundable()) {
            throw new \Exception('This order is not eligible for a refund.');
        }

        // 3. Policy window check
        $daysSincePurchase = $order->created_at->diffInDays(now());

        if ($daysSincePurchase > 30 && ! $admin->hasRole('super_admin')) {
            throw new AuthorizationException(
                'Refunds after 30 days require Super Admin approval.'
            );
        }

        // 4. Amount validation
        if ($amount > $order->total) {
            throw new \Exception('Refund amount cannot exceed order total.');
        }

        // 5. Create refund record
        $refund = Refund::create([
            'order_id' => $order->id,
            'admin_id' => $admin->id,
            'amount' => $amount,
            'reason' => $reason,
            'status' => 'processing',
        ]);

        // 6. Process via billing provider
        $this->paymentService->refund($order, $amount);

        // 7. Revoke license
        $this->licenseService->revokeByOrder($order);

        // 8. Update refund status
        $refund->update(['status' => 'completed']);

        // 9. Notify buyer
        $order->user->notify(new RefundProcessed($refund));

        // 10. Notify creator
        $order->listing->user->notify(new CreatorSaleRefunded($refund));

        // 11. Deduct from creator earnings
        $this->deductCreatorEarnings($order, $amount);

        // 12. Audit log
        activity()
            ->causedBy($admin)
            ->performedOn($order)
            ->withProperties([
                'refund_id' => $refund->id,
                'amount' => $amount,
                'reason' => $reason,
            ])
            ->log('marketplace.refund_processed');

        event(new RefundProcessed($refund));

        return $refund;
    }

    protected function deductCreatorEarnings(Order $order, float $refundAmount): void
    {
        $creator = $order->listing->user;
        $profile = $creator->creatorProfile;
        $commission = $profile->commission_rate ?? 30;
        $deduction = $refundAmount * (1 - $commission / 100);

        $profile->decrement('available_balance', $deduction);

        CreatorEarning::where('order_id', $order->id)
            ->update(['status' => 'refunded']);
    }
}
```

## Listing Approval Flow

```text
Creator submits listing (themes.publish creates pending status)
    ↓
Automated quality checks
    ↓
Listing enters moderation queue
    ↓
Moderator reviews
    ↓
  ┌─── Approve ──→ Listing becomes public
  │                   ↓
  │               Notify creator
  │                   ↓
  │               Index in search
  │
  └─── Reject ───→ Creator notified with reasons
                      ↓
                  Creator revises
                      ↓
                  Resubmits for review
```

### Implementation

```php
// app/Services/Marketplace/ListingApprovalService.php

namespace App\Services\Marketplace;

use App\Events\ListingApproved;
use App\Events\ListingRejected;
use App\Models\MarketplaceListing;
use App\Models\User;

class ListingApprovalService
{
    public function approve(
        MarketplaceListing $listing,
        User $moderator
    ): void {
        if (! $moderator->can('marketplace.approve_listing')) {
            throw new AuthorizationException(
                'Insufficient permissions to approve listings.'
            );
        }

        $listing->update([
            'status' => 'published',
            'approved_by' => $moderator->id,
            'approved_at' => now(),
        ]);

        // Index in search engine
        $this->searchService->index($listing);

        // Notify creator
        $listing->user->notify(new ListingApproved($listing));

        // Audit log
        activity()
            ->causedBy($moderator)
            ->performedOn($listing)
            ->log('marketplace.listing_approved');

        event(new ListingApproved($listing));
    }

    public function reject(
        MarketplaceListing $listing,
        User $moderator,
        string $reason
    ): void {
        if (! $moderator->can('marketplace.reject_listing')) {
            throw new AuthorizationException(
                'Insufficient permissions to reject listings.'
            );
        }

        $listing->update([
            'status' => 'rejected',
            'rejection_reason' => $reason,
            'rejected_by' => $moderator->id,
            'rejected_at' => now(),
        ]);

        // Notify creator
        $listing->user->notify(new ListingRejected($listing, $reason));

        activity()
            ->causedBy($moderator)
            ->performedOn($listing)
            ->withProperties(['reason' => $reason])
            ->log('marketplace.listing_rejected');
    }
}
```

## Listing Lifecycle States

```text
draft → pending_review → published
                          ↓
                       featured
                          ↓
                       suspended (by admin)
                          ↓
                       banned (by admin)
                          ↓
                       archived (by creator or admin)
```

| State | Who Can Set | Permission |
|---|---|---|
| `draft` | Creator (owner) | `marketplace.manage_own` |
| `pending_review` | Creator (owner) | `marketplace.sell` |
| `published` | Moderator, Admin | `marketplace.approve_listing` |
| `featured` | Admin | `marketplace.feature` |
| `suspended` | Admin | `marketplace.manage` |
| `banned` | Admin | `marketplace.ban_listing` |
| `archived` | Creator (owner), Admin | `marketplace.manage_own`, `marketplace.manage` |

## Review Permissions

| Action | Who | Permission | Constraints |
|---|---|---|---|
| Write review | Buyer | `marketplace.write_review` | Must have purchased item |
| Edit review | Reviewer | `marketplace.edit_review` | Own review only |
| Delete review | Reviewer | `marketplace.delete_review` | Own review only |
| Flag review | Anyone | `marketplace.flag_review` | Authenticated users |
| Remove flagged | Moderator | `marketplace.manage` | Flagged reviews only |
| Respond to review | Creator | `marketplace.respond_review` | Own product only |

```php
// app/Policies/ReviewPolicy.php

class ReviewPolicy
{
    public function create(User $user, MarketplaceListing $listing): bool
    {
        if (! $user->can('marketplace.write_review')) {
            return false;
        }

        // Must have purchased the item
        return Order::where('user_id', $user->id)
            ->where('listing_id', $listing->id)
            ->where('status', 'completed')
            ->exists();
    }

    public function update(User $user, Review $review): bool
    {
        return $user->id === $review->user_id
            && $user->can('marketplace.edit_review');
    }

    public function delete(User $user, Review $review): bool
    {
        return $user->id === $review->user_id
            && $user->can('marketplace.delete_review');
    }

    public function respond(User $user, Review $review): bool
    {
        return $user->id === $review->listing->user_id
            && $user->can('marketplace.respond_review');
    }
}
```

```php
        $this->authorize('marketplace.manage_own');
    }

    {
        if ($listing->user_id !== auth()->id()) {

        $this->dispatch('notify', type: 'success', message: 'Listing archived.');

            $this->dispatch('notify', type: 'error', message: 'Cannot set pricing.');

        }

            return;
        }

        $listing->update(['price' => $price]);

        $this->dispatch('notify', type: 'success', message: 'Price updated.');
    }
```

## Marketplace Permission Caching

### Listing Permission Cache

Marketplace listing permissions are cached to avoid repeated database queries during browsing:

```php
// app/Services/MarketplacePermissionCache.php

class MarketplacePermissionCache
{
    private CacheRepository $cache;
    private int $ttl = 300;

    public function __construct(CacheRepository $cache)
    {
        $this->cache = $cache;
    }

    public function canUserPurchase(User $user, Listing $listing): bool
    {
        $key = "marketplace:purchase:{$user->id}:{$listing->id}";

        return $this->cache->remember($key, $this->ttl, function () use ($user, $listing) {
            return $this->evaluatePurchasePermission($user, $listing);
        });
    }

    public function canUserViewListing(User $user, Listing $listing): bool
    {
        $key = "marketplace:view:{$user->id}:{$listing->id}";

        return $this->cache->remember($key, $this->ttl, function () use ($user, $listing) {
            return $this->evaluateViewPermission($user, $listing);
        });
    }

    public function canUserReviewListing(User $user, Listing $listing): bool
    {
        $key = "marketplace:review:{$user->id}:{$listing->id}";

        return $this->cache->remember($key, $this->ttl, function () use ($user, $listing) {
            return $this->evaluateReviewPermission($user, $listing);
        });
    }

    protected function evaluatePurchasePermission(User $user, Listing $listing): bool
    {
        if (! $user->hasPermissionTo('marketplace.purchase')) {
            return false;
        }

        if ($listing->user_id === $user->id) {
            return false;
        }

        if ($listing->status !== 'published') {
            return false;
        }

        $alreadyPurchased = Order::where('user_id', $user->id)
            ->where('listing_id', $listing->id)
            ->exists();

        if ($alreadyPurchased && ! $listing->allow_repurchase) {
            return false;
        }

        return true;
    }

    protected function evaluateViewPermission(User $user, Listing $listing): bool
    {
        if ($listing->status === 'published') {
            return true;
        }

        if ($listing->user_id === $user->id) {
            return true;
        }

        return $user->hasAnyRole(['admin', 'moderator']);
    }

    protected function evaluateReviewPermission(User $user, Listing $listing): bool
    {
        if (! $user->hasPermissionTo('marketplace.review')) {
            return false;
        }
```

## Marketplace Rate Limiting

### Purchase Rate Limits

Prevent abuse of marketplace purchase functionality:

```php
// app/Http/Marketplace/PurchaseRateLimiter.php

class PurchaseRateLimiter
{
    private const LIMITS = [
        'guest' => [
            'max_purchases_per_hour' => 3,
            'max_purchases_per_day' => 5,
        ],
        'user' => [
            'max_purchases_per_hour' => 10,
            'max_purchases_per_day' => 50,
        ],
        'pro_user' => [
            'max_purchases_per_hour' => 20,
            'max_purchases_per_day' => 200,
        ],
    ];

    public function attempt(User $user): bool
    {
        $tier = $this->getUserTier($user);
        $limits = self::LIMITS[$tier] ?? self::LIMITS['user'];

        $hourlyKey = "marketplace:purchases:hourly:{$user->id}";
        $dailyKey = "marketplace:purchases:daily:{$user->id}";

        $hourlyCount = Cache::get($hourlyKey, 0);
        $dailyCount = Cache::get($dailyKey, 0);

        if ($hourlyCount >= $limits['max_purchases_per_hour']) {
            Log::warning('Purchase rate limit exceeded (hourly)', [
                'user_id' => $user->id,
                'count' => $hourlyCount,
            ]);
            return false;
        }

        if ($dailyCount >= $limits['max_purchases_per_day']) {
            Log::warning('Purchase rate limit exceeded (daily)', [
                'user_id' => $user->id,
                'count' => $dailyCount,
            ]);
            return false;
        }

        Cache::put($hourlyKey, $hourlyCount + 1, 3600);
        Cache::put($dailyKey, $dailyCount + 1, 86400);

        return true;
    }

    protected function getUserTier(User $user): string
    {
        if ($user->hasRole('pro_user')) return 'pro_user';
        if ($user->hasRole('user')) return 'user';
        return 'guest';
    }
}
```

```php
    private const CREATION_LIMITS = [
```

## Marketplace Purchase Flow Permissions

### Multi-Step Purchase Authorization

Marketplace purchases require multiple authorization checks at different stages:

```php
// app/Services/MarketplacePurchaseFlow.php

class MarketplacePurchaseFlow
{
    public function initiatePurchase(User $user, Listing $listing): PurchaseSession
    {
        // Step 1: Basic permission check
        if (! $user->hasPermissionTo('marketplace.purchase')) {
            throw new AuthorizationException('You do not have permission to make purchases.');
        }

        // Step 2: Ownership check
        if ($listing->user_id === $user->id) {
            throw new AuthorizationException('You cannot purchase your own listing.');
        }

        // Step 3: Listing status check
        if ($listing->status !== 'published') {
            throw new ListingUnavailableException('This listing is not available for purchase.');
        }

        // Step 4: Duplicate purchase check
        if ($this->hasAlreadyPurchased($user, $listing)) {
            if (! $listing->allow_repurchase) {
                throw new DuplicatePurchaseException('You have already purchased this listing.');
            }
        }

        // Step 5: Rate limit check
        $rateLimiter = app(PurchaseRateLimiter::class);
        if (! $rateLimiter->attempt($user)) {
            throw new RateLimitException('Purchase rate limit exceeded. Please try again later.');
        }

        // Step 6: Fraud check
        if ($this->isFraudulentPurchase($user, $listing)) {
            throw new FraudDetectedException('This purchase has been flagged for review.');
        }

        return PurchaseSession::create([
            'user_id' => $user->id,
            'listing_id' => $listing->id,
            'amount' => $listing->price,
            'status' => 'initiated',
            'expires_at' => now()->addMinutes(15),
        ]);
    }

    public function completePurchase(PurchaseSession $session, PaymentMethod $payment): Order
    {
        // Step 7: Session validity check
        if ($session->expires_at->isPast()) {
            throw new SessionExpiredException('Purchase session has expired.');
        }

        // Step 8: Payment authorization
        if (! $payment->isAuthorized()) {
            throw new PaymentNotAuthorizedException('Payment method is not authorized.');
        }

        // Step 9: Final permission re-check (in case permissions changed)
        $user = User::find($session->user_id);
        $listing = Listing::find($session->listing_id);

        if (! $user->hasPermissionTo('marketplace.purchase')) {
            throw new AuthorizationException('Purchase permission was revoked.');
        }

        // Create order
        return DB::transaction(function () use ($session, $payment, $user, $listing) {
            $order = Order::create([
                'user_id' => $user->id,
                'listing_id' => $listing->id,
                'payment_method_id' => $payment->id,
                'total' => $session->amount,
                'status' => 'completed',
            ]);

            $session->update(['status' => 'completed']);

            ActivityLog::purchaseCompleted($user, $listing, $order);

            return $order;
        });
    }

    protected function hasAlreadyPurchased(User $user, Listing $listing): bool
    {
        return Order::where('user_id', $user->id)
            ->where('listing_id', $listing->id)
            ->exists();
    }

    protected function isFraudulentPurchase(User $user, Listing $listing): bool
    {
        // Check for rapid purchase patterns
        $recentPurchases = Order::where('user_id', $user->id)
            ->where('created_at', '>=', now()->subMinutes(5))
            ->count();

        if ($recentPurchases >= 5) {
            return true;
        }

        // Check for self-purchase via different accounts
        $sameEmailOrders = Order::where('listing_id', $listing->id)
            ->whereHas('user', fn ($q) => $q->where('email', $user->email))
            ->exists();

        return $sameEmailOrders;
    }
}
```

## Marketplace Refund Permissions

### Refund Eligibility Rules

```php
// app/Services/RefundEligibilityChecker.php

class RefundEligibilityChecker
{
    private const REFUND_WINDOW_DAYS = 30;
    private const MAX_REFUNDS_PER_USER_PER_MONTH = 5;

    public function isEligible(User $user, Order $order): array
    {
        $reasons = [];

        // Check: order exists and belongs to user
        if ($order->user_id !== $user->id) {
            return ['eligible' => false, 'reasons' => ['Order does not belong to this user.']];
        }

        // Check: order is completed
        if ($order->status !== 'completed') {
            $reasons[] = 'Order is not in completed status.';
        }

        // Check: within refund window
        if ($order->created_at->diffInDays(now()) > self::REFUND_WINDOW_DAYS) {
            $reasons[] = 'Order is outside the ' . self::REFUND_WINDOW_DAYS . '-day refund window.';
        }

        // Check: not already refunded
        if ($order->refund_status === 'refunded') {
            $reasons[] = 'Order has already been refunded.';
        }

        // Check: refund rate limit
        $monthlyRefunds = Order::where('user_id', $user->id)
            ->where('refund_status', 'refunded')
            ->whereMonth('refunded_at', now()->month)
            ->count();

        if ($monthlyRefunds >= self::MAX_REFUNDS_PER_USER_PER_MONTH) {
            $reasons[] = 'Monthly refund limit reached.';
        }

        return [
            'eligible' => empty($reasons),
            'reasons' => $reasons,
        ];
    }

    public function processRefund(User $user, Order $order, string $reason): Refund
    {
        $eligibility = $this->isEligible($user, $order);

        if (! $eligibility['eligible']) {
            throw new RefundNotEligibleException(implode(' ', $eligibility['reasons']));
        }

        return DB::transaction(function () use ($user, $order, $reason) {
            $refund = Refund::create([
                'order_id' => $order->id,
                'user_id' => $user->id,
                'amount' => $order->total,
                'reason' => $reason,
                'status' => 'pending',
            ]);

            $order->update([
                'refund_status' => 'pending',
                'refunded_at' => now(),
            ]);

            ActivityLog::refundRequested($user, $order, $refund);

            return $refund;
        });
    }
}
```

```php
// app/Policies/ListingPolicy.php

class ListingPolicy
{
    public function viewAny(User $user): bool
    {
        return $user->hasPermissionTo('marketplace.browse');
    }

    public function view(User $user, Listing $listing): bool
    {
        // Published listings are public
        if ($listing->status === 'published') {
            return true;
        }

        // Owners can see their own drafts
        if ($listing->user_id === $user->id) {
            return true;
        }

        // Moderators and admins can see all
        return $user->hasAnyRole(['admin', 'moderator']);
    }

    public function create(User $user): bool
    {
        if (! $user->hasPermissionTo('marketplace.sell')) {
            return false;
        }

        // Must have approved creator profile
        $creatorProfile = $user->creatorProfile;
        if (! $creatorProfile || $creatorProfile->status !== 'approved') {
            return false;
        }

        // Check subscription limits
        return app(CreatorSubscriptionPermissions::class)->canCreateListing($user);
    }

    public function update(User $user, Listing $listing): bool
    {
        if ($listing->user_id !== $user->id) {
            return $user->hasPermissionTo('marketplace.manage');
        }

        // Owners can only update published or draft listings
        return in_array($listing->status, ['draft', 'published', 'revision_requested']);
    }

    public function delete(User $user, Listing $listing): bool
    {
        if ($listing->user_id !== $user->id) {
            return $user->hasPermissionTo('marketplace.manage');
        }

        // Cannot delete listings with active orders
        $activeOrders = Order::where('listing_id', $listing->id)
            ->where('status', 'completed')
            ->where('refund_status', '!=', 'refunded')
            ->count();

        return $activeOrders === 0;
    }

    public function feature(User $user, Listing $listing): bool
    {
        return $user->hasPermissionTo('marketplace.feature');
    }
}
```

```php
            'requires_mfa' => true,

        ],

            'description' => 'Export payment data',
            'allowed_roles' => ['admin', 'super_admin'],
            'requires_mfa' => true,
        ],
    ];

    public function canPerformAction(User $user, string $action): bool
    {
        $config = self::SCOPED_PERMISSIONS[$action] ?? null;

        if (! $config) {
            return false;
        }

        // Check role
        if (! $user->hasAnyRole($config['allowed_roles'])) {
            return false;
        }

        if ($config['requires_mfa'] && ! $user->hasVerifiedTwoFactorAuthentication()) {
            return false;
        }

        return true;
    }

    {
        $config = self::SCOPED_PERMISSIONS['payment.refund'];

        if ($user->hasRole('admin')) {
            return PHP_FLOAT_MAX;
        }

        return $config['max_amount_without_approval'];
    }
}
```

```php
        ],
    ];

    public function analyze(User $buyer, Listing $listing): FraudResult
    {
        $score = 0;
        $flags = [];

        // Self-purchase check
        if ($listing->user_id === $buyer->id) {
            return new FraudResult(100, ['self_purchase'], 'block');
        }

        // Rapid purchase check
        $recentCount = Order::where('user_id', $buyer->id)
            ->where('created_at', '>=', now()->subHour())
            ->count();

        if ($recentCount >= self::RULES['rapid_purchase']['max_per_hour']) {
            $score += 75;
            $flags[] = 'rapid_purchases';
        }

        // IP analysis
        $ipAccounts = User::where('last_ip', $buyer->last_ip)
            ->where('created_at', '>=', now()->subDay())
            ->count();

        if ($ipAccounts >= self::RULES['ip']['max_accounts_per_ip_per_day']) {
            $score += 80;
            $flags[] = 'suspicious_ip';
        }

        return new FraudResult(
            score: min($score, 100),
            flags: $flags,
            decision: $score >= 100 ? 'block' : ($score >= 50 ? 'review' : 'allow')
        );
    }
}
```

### Fraud Decision Handling

```php
<?php

namespace App\Services\Marketplace;

class FraudDecisionHandler
{
    public function handle(FraudResult $result, User $buyer, Listing $listing): void
    {
        match ($result->decision) {
            'block' => $this->blockPurchase($buyer, $listing, $result),
            'review' => $this->flagForReview($buyer, $listing, $result),
            'allow' => null, // No action needed
        };
    }

    protected function blockPurchase(User $buyer, Listing $listing, FraudResult $result): void
    {
        Log::warning('Marketplace purchase blocked', [
            'user_id' => $buyer->id,
            'listing_id' => $listing->id,
            'score' => $result->score,
            'flags' => $result->flags,
        ]);

        // Notify buyer
        $buyer->notify(new PurchaseBlockedNotification($result));

        // Notify admins
        User::role('admin')->notify(new FraudAlertNotification($buyer, $listing, $result));
    }

    protected function flagForReview(User $buyer, Listing $listing, FraudResult $result): void
    {
        FraudReview::create([
            'user_id' => $buyer->id,
            'listing_id' => $listing->id,
            'score' => $result->score,
            'flags' => $result->flags,
            'status' => 'pending',
        ]);
    }
}
```

```php
            if ($listing->user_id === $user?->id) {
                return true;

            if ($user && $user->hasAnyRole(['admin', 'moderator'])) {
                return true;

            // Premium listings require subscription
            if ($listing->is_premium && !$user?->hasRole(['pro_user', 'creator'])) {
                return false;
            }

            return false;
        });
    }
```

```php
namespace App\Services\Marketplace;

class MarketplaceExportService

    public function exportOrders(User $admin, array $filters = []): string

        if (!$admin->can('marketplace.manage')) {

        $csv = "Order ID,Buyer,Listing,Amount,Status,Refund Status,Created\n";

            $csv .= sprintf(
                "%d,%s,%s,%.2f,%s,%s,%s\n",

                $order->listing->title,
                $order->total,
                $order->status,
                $order->refund_status,
                $order->created_at->format('Y-m-d')
            );
        }

        // Log the export
        activity()
            ->causedBy($admin)
            ->withProperties(['record_count' => $orders->count()])
            ->log('marketplace.orders_exported');

        return $csv;
    }
```

```php
<?php

namespace App\Services\Marketplace;

class MarketplaceFeatureFlags
{
    private const FLAGS = [
        'marketplace_beta_search' => [
            'description' => 'Enable new search algorithm',
            'default' => false,
            'roles' => ['admin'],
        ],
        'marketplace_ai_recommendations' => [
            'description' => 'AI-powered listing recommendations',
            'default' => false,
            'roles' => ['pro_user', 'creator'],
        ],
        'marketplace_social_proof' => [
            'description' => 'Show purchase counts and activity',
            'default' => true,
            'roles' => [],
        ],
        'marketplace_discount_codes' => [
            'description' => 'Enable coupon/discount system',
            'default' => true,
            'roles' => ['admin'],
        ],
    ];

    public function isEnabled(string $flag, User $user): bool
    {
        $config = self::FLAGS[$flag] ?? null;

        if (!$config) {
            return false;
        }

        // Check role-based feature flag
        if (!empty($config['roles']) && !$user->hasAnyRole($config['roles'])) {
            return false;
        }

        // Check database override
        $dbValue = DB::table('feature_flags')
            ->where('name', $flag)
            ->value('enabled');

        return $dbValue ?? $config['default'];
    }

    public function getEnabledFlags(User $user): array
    {
        return array_filter(array_keys(self::FLAGS), fn($flag) => $this->isEnabled($flag, $user));
    }
}
```

### Marketplace Webhook System

```php
<?php

namespace App\Services\Marketplace;

class MarketplaceWebhookService
{
    private const WEBHOOK_EVENTS = [
        'purchase.completed',
        'purchase.failed',
        'refund.processed',
        'listing.approved',
        'listing.rejected',
        'payout.completed',
    ];

    public function dispatch(string $event, array $payload): void
    {
        if (!in_array($event, self::WEBHOOK_EVENTS)) {
            return;
        }

        $subscriptions = WebhookSubscription::where('event', $event)
            ->orWhere('event', '*')
            ->get();

        foreach ($subscriptions as $subscription) {
            WebhookJob::dispatch($subscription, $event, $payload)
                ->onQueue('webhooks');
        }
    }
}
```
