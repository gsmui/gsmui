# 10.09 - Creator Permissions

**Project:** GsmUi
**Domain:** Authorization & RBAC
**Version:** 1.0.0

---

## Purpose

Define the **complete permission set for Creators** — users who publish themes, components, or templates on the marketplace. Creators inherit all [Pro User permissions](./10.08-user-permissions.md) plus content creation and marketplace listing capabilities. The Creator role is the economic engine of the platform: creators build the products that drive marketplace revenue.

## Creator Role Definition

A Creator is a user who has:

1. An active Pro User subscription
2. An approved `creator_profile` record
3. Accepted the Creator Terms of Service
4. Completed identity verification (if required by jurisdiction)

The Creator role does **not** grant admin, moderator, or any elevated platform permissions. Creators operate within a sandbox of their own content.

| `templates.delete` | ✅ (own) | Delete own templates |
| `templates.publish` | ✅ (own) | Submit templates for review |

**Ownership enforcement:** Every content operation checks that the authenticated user is the `user_id` of the resource. Creators cannot view, edit, or delete other creators' unpublished content.

```php
// app/Policies/ThemePolicy.php

namespace App\Policies;

use App\Models\User;
use App\Models\Theme;

class ThemePolicy
{
    public function viewAny(User $user): bool
    {
        return $user->hasRole(['creator', 'admin', 'super_admin']);
    }

    public function view(User $user, Theme $theme): bool
    {
        // Public themes are viewable by anyone
        if ($theme->status === 'published') {
            return true;
        }

        // Unpublished themes only visible to owner and admins
        return $user->id === $theme->user_id
            || $user->hasRole(['admin', 'super_admin']);
    }

    public function create(User $user): bool
    {
        return $user->hasRole(['creator', 'admin', 'super_admin']);
    }

    public function update(User $user, Theme $theme): bool
    {
        return $user->id === $theme->user_id
            || $user->hasRole(['admin', 'super_admin']);
    }

    public function delete(User $user, Theme $theme): bool
    {
        return $user->id === $theme->user_id
            || $user->hasRole(['admin', 'super_admin']);
    }

    public function publish(User $user, Theme $theme): bool
    {
        // Creators can submit for review, admins can publish directly
        if ($user->id === $theme->user_id && $theme->status === 'draft') {
            return true;
        }

        return $user->hasRole(['admin', 'super_admin']);
    }

    public function export(User $user, Theme $theme): bool
    {
        return $user->id === $theme->user_id
            || $user->hasRole(['admin', 'super_admin']);
    }
}
```

```php
                fputcsv($handle, $row);

            fclose($handle);

        return auth()->user()
            ->themes()
            ->withCount(['views' => fn($q) => $q->where('created_at', '>=', now()->subDays($period === '30d' ? 30 : 90))])
            ->get();
    }
```

### Flow Diagram

```text
User clicks "Become a Creator"
    ↓
System verifies:
  - Email verified

  - Accepts Creator ToS
    ↓
creator_profile created (status: pending)
```

### Livewire Component

```php
// app/Livewire/Creator/Apply.php

class Apply extends Component
{
    public string $displayName = '';
    public string $bio = '';
    public ?string $portfolioUrl = null;
    public bool $acceptsTerms = false;

    protected array $rules = [
        'displayName' => ['required', 'string', 'max:100'],
        'bio' => ['required', 'string', 'max:500'],
        'portfolioUrl' => ['nullable', 'url'],
        'acceptsTerms' => ['accepted'],
    ];

    public function submit(): void
    {
        $this->validate();

        $user = auth()->user();

        // Pre-flight checks
        if (! $user->hasVerifiedEmail()) {
            $this->addError('displayName', 'Please verify your email first.');
            return;
        }

        if (! $user->hasRole('pro_user')) {
            $this->addError('displayName', 'An active Pro subscription is required.');
            return;
        }

        if ($user->creatorProfile) {
            $this->addError('displayName', 'You already have a creator profile.');
            return;
        }

        CreatorProfile::create([
            'user_id' => $user->id,
            'status' => 'pending',
            'display_name' => $this->displayName,
            'bio' => $this->bio,
            'portfolio_url' => $this->portfolioUrl,
        ]);

        // Notify admins of new application
        Notification::send(
            User::role('admin')->get(),
            new CreatorApplicationSubmitted($user)
        );

        $this->dispatch('notify', type: 'success',
            message: 'Application submitted! We will review it within 48 hours.');

        redirect()->route('creator.dashboard');
    }
}
```

### Admin Review

```php
// app/Livewire/Admin/CreatorReview.php

class CreatorReview extends Component
{
    public function approve(CreatorProfile $profile): void
    {
        $this->authorize('roles.assign');

        $profile->update([
            'status' => 'approved',
            'approved_at' => now(),
        ]);

        $profile->user->assignRole('creator');

        // Notify creator
        $profile->user->notify(new CreatorApplicationApproved());

        activity()
            ->causedBy(auth()->user())
            ->performedOn($profile->user)
            ->withProperties(['action' => 'creator_approved'])
            ->log('creator_approved');
    }

    public function reject(CreatorProfile $profile, string $reason): void
    {
        $this->authorize('roles.assign');

        $profile->update(['status' => 'rejected']);

        $profile->user->notify(new CreatorApplicationRejected($reason));

        activity()
            ->causedBy(auth()->user())
            ->performedOn($profile->user)
            ->withProperties(['action' => 'creator_rejected', 'reason' => $reason])
            ->log('creator_rejected');
    }
}
```

## Content Submission & Review

### Submission States

```text
draft → pending_review → approved → published
                           ↓
                        rejected (with reasons)
                           ↓
                        revision_needed → draft
```

### State Transitions

| From | To | Trigger | Allowed By |
|---|---|---|---|
| `draft` | `pending_review` | Creator submits | Owner |
| `pending_review` | `approved` | Admin approves | Admin, Super Admin |
| `pending_review` | `rejected` | Admin rejects | Admin, Super Admin |
| `pending_review` | `revision_needed` | Admin requests changes | Admin, Super Admin |
| `revision_needed` | `draft` | Creator revises | Owner |
| `approved` | `published` | Auto-publish (scheduled) | System |
| `published` | `archived` | Admin archives | Admin, Super Admin |
| `published` | `draft` | Creator unpublishes | Owner |

```php
// app/Services/ContentSubmissionService.php

class ContentSubmissionService
{
    public function submitForReview(User $creator, Theme $theme): void
    {
        if ($creator->id !== $theme->user_id) {
            throw new AuthorizationException('Not the owner of this theme.');
        }

        if ($theme->status !== 'draft') {
            throw new InvalidStateException('Only draft themes can be submitted for review.');
        }

        // Run automated quality checks
        $this->runQualityChecks($theme);

        $theme->update(['status' => 'pending_review']);

        // Notify moderators
        Notification::send(
            User::role(['moderator', 'admin'])->get(),
            new ContentPendingReview($theme)
        );
    }

    protected function runQualityChecks(Theme $theme): void
    {
        // Check file structure, manifest validity, screenshot presence
        // Log warnings but don't block submission
    }
}
```

```php
namespace App\Scopes;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Scope;

class OwnableScope implements Scope
{
    public function apply(Builder $builder, Model $model): void
    {
        $builder->where('user_id', auth()->id());
    }
}

// Usage in Creator model
class Creator extends Model
{
    protected static function booted(): void
    {
        static::addGlobalScope(new OwnableScope);
    }
}
```

### Middleware Alternative

```php
// app/Http/Middleware/EnsureOwnership.php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;

class EnsureOwnership
{
    public function handle(Request $request, Closure $next, string $parameter = 'id', string $column = 'user_id')
    {
        $model = $request->route($parameter);

        if ($model && $model->{$column} !== auth()->id()) {
            abort(403, 'You do not own this resource.');
        }

        return $next($request);
    }
}
```

### Payout Flow

```text
Creator earns revenue from sales
    ↓
Revenue accumulates in creator_profile.available_balance
    ↓
Creator requests payout (min $50)
    ↓
Admin reviews payout request
    ↓
Payout processed via payment provider
    ↓
Balance deducted, earnings logged
```

```php
// app/Services/PayoutService.php

public function requestPayout(User $creator, float $amount): Payout
{
    if (! $creator->can('marketplace.request_payout')) {
        throw new AuthorizationException('Cannot request payout.');
    }

    $profile = $creator->creatorProfile;

    if ($amount < 50) {
        throw new ValidationException('Minimum payout is $50.');
    }

    if ($amount > $profile->available_balance) {
        throw new ValidationException('Insufficient balance.');
    }

    $payout = Payout::create([
        'user_id' => $creator->id,
        'amount' => $amount,
        'status' => 'pending',
        'method' => $profile->payout_method,
    ]);

    // Notify admins for review
    Notification::send(
        User::role('admin')->get(),
        new PayoutRequested($payout)
    );

    return $payout;
}
```

## Security Notes

* Creators cannot access other creators' unpublished content.
* Content is soft-deleted with a 30-day recovery window.
* All content modifications are audit-logged.
* Revenue data is only visible to the creator and admins.
* Payout requests require admin approval before processing.
* Creator role is automatically revoked if the Pro subscription lapses.

```php
// app/Livewire/Creator/ThemeManager.php

class ThemeManager extends Component
{
    public ?Theme $editingTheme = null;
    public string $title = '';
    public string $description = '';
    public string $price = '0';
    public ?string $previewImage = null;

    protected array $rules = [
        'title' => ['required', 'string', 'max:255'],
        'description' => ['required', 'string', 'max:2000'],
        'price' => ['required', 'numeric', 'min:0'],
    ];

    public function createTheme(): void
    {
        $this->authorize('themes.create');
        $this->validate();

        $theme = Theme::create([
            'user_id' => auth()->id(),
            'title' => $this->title,
            'description' => $this->description,
            'price' => $this->price,
            'status' => 'draft',
        ]);

        activity()
            ->causedBy(auth()->user())
            ->performedOn($theme)
            ->log('theme_created');

        $this->dispatch('notify', type: 'success', message: 'Theme created.');
        $this->resetForm();
    }

    public function updateTheme(): void
    {
        $this->authorize('themes.update', $this->editingTheme);
        $this->validate();

        $this->editingTheme->update([
            'title' => $this->title,
            'description' => $this->description,
            'price' => $this->price,
        ]);

        activity()
            ->causedBy(auth()->user())
            ->performedOn($this->editingTheme)
            ->log('theme_updated');

        $this->dispatch('notify', type: 'success', message: 'Theme updated.');
        $this->editingTheme = null;
    }

    public function deleteTheme(Theme $theme): void
    {
        $this->authorize('themes.delete', $theme);

        $theme->delete();

        activity()
            ->causedBy(auth()->user())
            ->performedOn($theme)
            ->log('theme_deleted');

        $this->dispatch('notify', type: 'success', message: 'Theme deleted.');
    }

    public function publishTheme(Theme $theme): void
    {
        $this->authorize('themes.publish', $theme);

        // Run quality checks
        $this->runQualityChecks($theme);

        $theme->update(['status' => 'pending_review']);
```

```php
// app/Livewire/Creator/AnalyticsDashboard.php

class AnalyticsDashboard extends Component
{
    public string $period = '30d';
    public ?string $selectedProduct = null;

    public function mount(): void
    {
        $this->authorize('analytics.view');
    }

    public function render()
    {

            'overview' => $this->getOverview(),
            'salesData' => $this->getSalesData(),
            'topProducts' => $this->getTopProducts(),
            'conversionRates' => $this->getConversionRates(),
        ]);
    }

    protected function getOverview(): array
    {
        $user = auth()->user();
        $days = $this->getDaysFromPeriod();
        
        return [
            'total_views' => $user->themes()->sum('views'),
            'total_downloads' => $user->themes()->sum('downloads'),
            'total_revenue' => $user->orders()->where('created_at', '>=', now()->subDays($days))->sum('total'),
            'avg_order_value' => $user->orders()->where('created_at', '>=', now()->subDays($days))->avg('total') ?? 0,
            'conversion_rate' => $this->calculateConversionRate(),
        ];
    }

    protected function getSalesData(): Collection
    {
        $days = $this->getDaysFromPeriod();
        
        return auth()->user()->orders()
            ->selectRaw('DATE(created_at) as date, COUNT(*) as count, SUM(total) as revenue')
            ->where('created_at', '>=', now()->subDays($days))
            ->groupBy('date')
            ->orderBy('date')
            ->get();
    }

    protected function getTopProducts(): Collection
    {
        return auth()->user()->themes()
            ->withCount('orders')
            ->withSum('orders', 'total')
            ->orderByDesc('orders_count')
            ->take(10)
            ->get();
    }

    protected function getConversionRates(): array
    {
        $themes = auth()->user()->themes()->where('status', 'published')->get();
        
        return $themes->map(function ($theme) {
            return [
                'id' => $theme->id,
                'title' => $theme->title,
                'views' => $theme->views,
                'downloads' => $theme->downloads,
                'conversion' => $theme->views > 0 
                    ? round(($theme->downloads / $theme->views) * 100, 2) 
                    : 0,
            ];
        })->toArray();
    }

    public function exportAnalytics(): StreamedResponse
    {
        $this->authorize('analytics.export');

        $data = $this->getSalesData();

        return StreamedResponse::create(function () use ($data) {
            $handle = fopen('php://output', 'w');
            fputcsv($handle, ['Date', 'Orders', 'Revenue']);

            foreach ($data as $row) {
                fputcsv($handle, [$row->date, $row->count, $row->revenue]);
            }

            fclose($handle);
        }, 200, [
            'Content-Type' => 'text/csv',
            'Content-Disposition' => 'attachment; filename="analytics-' . now()->format('Y-m-d') . '.csv"',
        ]);
    }

    protected function getDaysFromPeriod(): int
    {
        return match($this->period) {
            '7d' => 7,
            '30d' => 30,
            '90d' => 90,
            '1y' => 365,
            default => 30,
        };
    }

    protected function calculateConversionRate(): float
    {
        $totalViews = auth()->user()->themes()->sum('views');
        $totalDownloads = auth()->user()->themes()->sum('downloads');
        
        return $totalViews > 0 ? round(($totalDownloads / $totalViews) * 100, 2) : 0;
    }
```

```php
class RequestPayout extends Component
{
    public float $amount = 0;
    public string $method = 'paypal';
    public array $methodDetails = [];

    protected array $rules = [
        'amount' => ['required', 'numeric', 'min:50', 'max:available_balance'],
        'method' => ['required', 'in:paypal,stripe,bank_transfer'],
    ];

    public function mount(): void
    {
        $this->authorize('marketplace.request_payout');
    }

    public function requestPayout(): void
    {
        $this->validate();

        $user = auth()->user();
        $profile = $user->creatorProfile;

        if ($this->amount > $profile->available_balance) {
            $this->addError('amount', 'Insufficient balance.');
            return;
        }

        $payout = Payout::create([
            'user_id' => $user->id,
            'amount' => $this->amount,
            'method' => $this->method,
            'method_details' => $this->methodDetails,
            'status' => 'pending',
        ]);

        // Deduct from available balance
        $profile->update([
            'available_balance' => $profile->available_balance - $this->amount,
        ]);

        // Notify admins
        Notification::send(
            User::role('admin')->get(),
            new PayoutRequested($payout)
        );

        activity()
            ->causedBy($user)
            ->performedOn($payout)
            ->log('payout_requested');

        $this->dispatch('notify', type: 'success', message: 'Payout requested. It will be processed within 3-5 business days.');
        $this->amount = 0;
    }
}
```

## Creator Content States

### State Machine

```php
// Content state transitions
class ContentStateMachine
{
    private array $transitions = [
        'draft' => ['pending_review'],
        'pending_review' => ['approved', 'rejected', 'revision_needed'],
        'revision_needed' => ['draft'],
        'approved' => ['published'],
        'published' => ['archived', 'draft'],
        'archived' => ['published'],
    ];

    public function canTransition(string $from, string $to): bool
    {
        return in_array($to, $this->transitions[$from] ?? []);
    }

    public function getValidTransitions(string $state): array
    {
        return $this->transitions[$state] ?? [];
    }
}
```

### State Validation

```php
// Validate state transitions
class ContentStateValidator
{
    public function validateTransition(User $user, Theme $theme, string $newStatus): bool
    {
        $machine = new ContentStateMachine();
        
        if (! $machine->canTransition($theme->status, $newStatus)) {
            return false;
        }

        // Check permissions for specific transitions
        return match($newStatus) {
            'pending_review' => $user->id === $theme->user_id,
            'approved' => $user->hasRole(['admin', 'super_admin']),
            'rejected' => $user->hasRole(['admin', 'super_admin']),
            'published' => $user->hasRole(['admin', 'super_admin']),
            'archived' => $user->hasRole(['admin', 'super_admin']),
            'draft' => $user->id === $theme->user_id || $user->hasRole(['admin', 'super_admin']),
            default => false,
        };
    }
}
```

```php
class CreatorSubmissionWorkflow
{
    private const STATUSES = [
        'draft',
        'pending_review',
        'revision_requested',
        'approved',
        'published',
        'rejected',
        'suspended',
    ];

    public function submitForReview(User $creator, Theme $theme): void
    {
        if ($theme->user_id !== $creator->id) {
            throw new AuthorizationException('Not the owner of this theme.');
        }

        if ($theme->status !== 'draft') {
            throw new InvalidStateException('Only draft themes can be submitted for review.');
        }

        $qualityChecker = app(ContentQualityChecker::class);
        $issues = $qualityChecker->checkTheme($theme);

        if (! empty($issues)) {
            throw new ValidationException(implode(', ', $issues));
        }

        $theme->update(['status' => 'pending_review']);

        Notification::send(
            User::role(['moderator', 'admin'])->get(),
            new ContentPendingReview($theme)
        );

        ActivityLog::submittedForReview($creator, $theme);
    }

    public function requestRevision(Moderator $moderator, Theme $theme, string $reason): void
    {
        $theme->update(['status' => 'revision_requested']);

        Notification::send(
            $theme->user,
            new RevisionRequested($theme, $reason)
        );

        ActivityLog::revisionRequested($moderator, $theme, $reason);
    }

    public function approve(Moderator $moderator, Theme $theme): void
    {
        $theme->update(['status' => 'approved']);

        Notification::send(
            $theme->user,
            new ContentApproved($theme)
        );

        ActivityLog::contentApproved($moderator, $theme);
    }

    public function publish(Moderator $moderator, Theme $theme): void
    {
        if ($theme->status !== 'approved') {
            throw new InvalidStateException('Only approved themes can be published.');
        }

        $theme->update([
            'status' => 'published',
            'published_at' => now(),
        ]);

        ActivityLog::contentPublished($moderator, $theme);
    }
}
```

```php
class CreatorTokenManager
{
    public function createToken(User $creator, string $name, array $scopes): NewAccessToken
    {
        $allowedScopes = $this->filterAllowedScopes($creator, $scopes);

        return $creator->createToken($name, $allowedScopes);

            $allowed[] = 'storefront.manage';
        }

        if ($config['analytics_retention_days'] > 0) {
            $allowed[] = 'analytics.read';
        }

        return array_intersect($scopes, $allowed);
    }
}
```

### Creator Scorecard

```php
// app/Services/CreatorScorecard.php

class CreatorScorecard
{
    public function getScorecard(User $creator): array
    {
        return [
            'overall_score' => $this->calculateOverallScore($creator),
            'metrics' => [
                'quality_score' => $this->getQualityScore($creator),
                'sales_volume' => $this->getSalesVolumeScore($creator),
                'customer_satisfaction' => $this->getCustomerSatisfactionScore($creator),
                'response_time' => $this->getResponseTimeScore($creator),
                'consistency' => $this->getConsistencyScore($creator),
            ],
            'badges' => $this->getBadges($creator),
            'tiers_unlocked' => $this->getUnlockedTiers($creator),
        ];
    }

    protected function calculateOverallScore(User $creator): float
    {
        $weights = [
            'quality_score' => 0.30,
            'sales_volume' => 0.25,
            'customer_satisfaction' => 0.25,
            'response_time' => 0.10,
            'consistency' => 0.10,
        ];

        $score = 0;
        foreach ($weights as $metric => $weight) {
            $score += $this->getMetricValue($creator, $metric) * $weight;
        }

        return round($score, 2);
    }

    protected function getQualityScore(User $creator): float
    {
        $avgRating = $creator->themes()->avg('average_rating') ?? 0;
        return min($avgRating / 5.0, 1.0);
    }

    protected function getSalesVolumeScore(User $creator): float
    {
        $totalSales = $creator->themes()->sum('sales_count');
        $maxSales = Theme::max('sales_count') ?: 1;
        return min($totalSales / ($maxSales * 0.5), 1.0);
    }

    protected function getCustomerSatisfactionScore(User $creator): float
    {
        $positiveReviews = Review::whereHas('listing', fn ($q) => $q->where('user_id', $creator->id))
            ->where('rating', '>=', 4)
            ->count();

        $totalReviews = Review::whereHas('listing', fn ($q) => $q->where('user_id', $creator->id))
            ->count();

        return $totalReviews > 0 ? $positiveReviews / $totalReviews : 0;
    }

    protected function getResponseTimeScore(User $creator): float
    {
        $avgResponseHours = SupportTicket::where('creator_id', $creator->id)
            ->whereNotNull('first_response_at')
            ->selectRaw('AVG(TIMESTAMPDIFF(HOUR, created_at, first_response_at))')
            ->value('avg_hours') ?? 72;

        return max(0, 1 - ($avgResponseHours / 72));
    }

    protected function getConsistencyScore(User $creator): float
    {
        $monthsActive = $creator->created_at->diffInMonths(now());
        $monthsWithSales = Order::whereHas('listing', fn ($q) => $q->where('user_id', $creator->id))
            ->selectRaw('COUNT(DISTINCT DATE_FORMAT(created_at, "%Y-%m"))')
            ->value('count') ?? 0;

        return $monthsActive > 0 ? min($monthsWithSales / $monthsActive, 1.0) : 0;
    }

    protected function getBadges(User $creator): array
    {
        $badges = [];
        $score = $this->calculateOverallScore($creator);

        if ($score >= 0.9) $badges[] = 'Elite Creator';
        if ($score >= 0.7) $badges[] = 'Top Creator';
        if ($creator->themes()->sum('sales_count') >= 100) $badges[] = '100+ Sales';
        if ($this->getCustomerSatisfactionScore($creator) >= 0.95) $badges[] = '5-Star Rated';

        return $badges;
    }
}
```
